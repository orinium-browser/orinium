6/// Trait for custom/replaced elements that produce their own draw commands.
7///
8/// # Coordinate System
9/// Commands must be emitted in the content-box coordinate space:
10/// `(0, 0)` = top-left of the content box. The parent's transform/clip
11/// stack handles positioning.
12///
13/// # Lifecycle (Phase 1)
14/// - `draw()` is called every frame during `generate_draw_commands`.
15/// - Event handling is deferred to a later phase.
16pub trait CustomNode: std::fmt::Debug + 'static {
17    /// Emit draw commands into `cmd_buf`.
18    ///
19    /// `text_style` carries the inherited CSS text properties (color,
20    /// font-size, font-weight, etc.) resolved for this element.
21    ///
22    /// Called once per frame for each visible custom node.
23    fn draw(&self, cmd_buf: &mut Vec<DrawCommand>, text_style: &TextStyle);
24
25    /// Optional background color override.
26    ///
27    /// When `Some`, the returned color replaces the CSS `background-color`
28    /// for this element's box-model background.
29    fn background_color(&self) -> Option<Color> {
30        None
31    }
32
33    /// Intrinsic (content-box) size in pixels `(width, height)`.
34    ///
35    /// The layout engine uses this to size the element when no explicit
36    /// width/height is set via CSS.
37    fn intrinsic_size(&self) -> (f32, f32);
38}
