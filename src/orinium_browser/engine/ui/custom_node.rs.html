8/// Trait for custom/replaced elements that produce their own draw commands.
9///
10/// # Coordinate System
11/// Commands must be emitted in the content-box coordinate space:
12/// `(0, 0)` = top-left of the content box. The parent's transform/clip
13/// stack handles positioning.
14///
15/// # Lifecycle (Phase 1)
16/// - `draw()` is called every frame during `generate_draw_commands`.
17/// - Event handling is deferred to a later phase.
18pub trait CustomNode: std::fmt::Debug + 'static {
19    /// Emit draw commands into `cmd_buf`.
20    ///
21    /// `text_style` carries the inherited CSS text properties (color,
22    /// font-size, font-weight, etc.) resolved for this element.
23    ///
24    /// Called once per frame for each visible custom node.
25    fn draw(&self, cmd_buf: &mut Vec<DrawCommand>, text_style: &TextStyle);
26
27    /// Emit draw commands fitted to the resolved content-box `size`.
28    ///
29    /// Replaced elements that support CSS sizing can override this method.
30    /// The default preserves the original intrinsic-size drawing behavior.
31    fn draw_sized(
32        &self,
33        cmd_buf: &mut Vec<DrawCommand>,
34        text_style: &TextStyle,
35        _size: (f32, f32),
36    ) {
37        self.draw(cmd_buf, text_style);
38    }
39
40    /// Optional background color override.
41    ///
42    /// When `Some`, the returned color replaces the CSS `background-color`
43    /// for this element's box-model background.
44    fn background_color(&self) -> Option<Color> {
45        None
46    }
47
48    /// Intrinsic (content-box) size in pixels `(width, height)`.
49    ///
50    /// The layout engine uses this to size the element when no explicit
51    /// width/height is set via CSS.
52    fn intrinsic_size(&self) -> (f32, f32);
53
54    /// Whether one resolved dimension should scale the other dimension using
55    /// the node's intrinsic aspect ratio.
56    fn preserves_intrinsic_aspect_ratio(&self) -> bool {
57        false
58    }
59
60    /// Whether this node can receive keyboard and IME text input.
61    fn accepts_text_input(&self) -> bool {
62        false
63    }
64
65    /// Updates keyboard focus for this node.
66    fn set_focused(&self, _focused: bool) {}
67
68    /// Returns whether this node currently owns keyboard focus.
69    fn is_focused(&self) -> bool {
70        false
71    }
72
73    /// Applies a platform-neutral text editing event.
74    fn handle_text_input(&self, _event: TextInputEvent) -> bool {
75        false
76    }
77
78    /// Returns whether an IME preedit string is active.
79    fn is_composing(&self) -> bool {
80        false
81    }
82}
