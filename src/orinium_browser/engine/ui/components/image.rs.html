1//! Replaced-element component for decoded HTML images.
2
3use crate::engine::layouter::types::TextStyle;
4use crate::engine::renderer_model::{Brush, DrawCommand, FillRule, Image, Paint, rect_path};
5use crate::engine::ui::custom_node::CustomNode;
6
7/// A decoded image participating in layout as a replaced element.
8#[derive(Debug)]
9pub struct ImageComponent {
10    pub image: Option<Image>,
11}
12
13impl CustomNode for ImageComponent {
14    fn draw(&self, cmd_buf: &mut Vec<DrawCommand>, text_style: &TextStyle) {
15        let size = self.intrinsic_size();
16        self.draw_sized(cmd_buf, text_style, size);
17    }
18
19    fn draw_sized(
20        &self,
21        cmd_buf: &mut Vec<DrawCommand>,
22        _text_style: &TextStyle,
23        size: (f32, f32),
24    ) {
25        let Some(image) = &self.image else {
26            return;
27        };
28        cmd_buf.push(DrawCommand::Fill {
29            path: rect_path(0.0, 0.0, size.0, size.1),
30            paint: Paint {
31                brush: Brush::Image(image.clone()),
32                opacity: 1.0,
33            },
34            rule: FillRule::NonZero,
35        });
36    }
37
38    fn intrinsic_size(&self) -> (f32, f32) {
39        self.image.as_ref().map_or((0.0, 0.0), |image| {
40            (image.width() as f32, image.height() as f32)
41        })
42    }
43
44    fn preserves_intrinsic_aspect_ratio(&self) -> bool {
45        true
46    }
47}
48
49#[cfg(test)]
50mod tests {
51    use super::*;
52
53    #[test]
54    fn image_component_uses_intrinsic_size_and_draw_target() {
55        let image = Image::from_rgba(2, 3, vec![255; 24]).unwrap();
56        let component = ImageComponent { image: Some(image) };
57        assert_eq!(component.intrinsic_size(), (2.0, 3.0));
58
59        let mut commands = Vec::new();
60        component.draw_sized(&mut commands, &TextStyle::default(), (20.0, 30.0));
61        let DrawCommand::Fill { path, paint, .. } = &commands[0] else {
62            panic!("expected image fill");
63        };
64        assert!(matches!(&paint.brush, Brush::Image(_)));
65        let points = path.subpaths().remove(0);
66        assert!(points.iter().any(|point| *point == (20.0, 30.0)));
67    }
68}
