1//! Bridge between the layout engine's traits and
2//! the engine's [`CustomNode`] rendering interface.
3//!
4//! [`BlockLayouter`] (from `ui_layout`) handles block-level layout positioning:
5//! it reports the node's border-box rect to the layout engine.
6//!
7//! [`FlowLayouter`] (from `ui_layout`) handles inline-level layout:
8//! it reports line spans for participation in an inline formatting context.
9//!
10//! The corresponding [`CustomNode`](super::super::custom_node::CustomNode)
11//! (stored in [`NodeKind::Custom`](super::super::layouter::types::NodeKind))
12//! handles draw-command generation.
13

28/// A [`BlockLayouter`] implementation for custom nodes.
29///
30/// Wraps a [`ContainerStyle`] so the layout engine can account for
31/// borders and padding when sizing the element.  The actual content
32/// rendering is delegated to a separate [`CustomNode`](super::super::custom_node::CustomNode).
33#[derive(Debug)]
34pub struct CustomLayoutBridge {
35    style: ContainerStyle,
36    layout_style: Style,
37    node: Rc<dyn CustomNode>,
38}
39
40impl CustomLayoutBridge {
41    pub fn new(style: ContainerStyle, layout_style: Style, node: Rc<dyn CustomNode>) -> Self {
42        Self {
43            style,
44            layout_style,
45            node,
46        }
47    }
48
49    pub fn style(&self) -> &ContainerStyle {
50        &self.style
51    }
52}
53
54impl BlockLayouter for CustomLayoutBridge {
55    fn layout(&mut self, ctx: &LayoutContext) -> Rect {
56        let (intrinsic_w, intrinsic_h) = self.node.intrinsic_size();
57
58        let width = match &self.layout_style.size.width {
59            LengthOrAuto::Length(len) => len
60                .resolve_with(
61                    ctx.containing_block_width,
62                    ctx.containing_block_width.unwrap_or(0.0),
63                    ctx.containing_block_height.unwrap_or(0.0),
64                )
65                .unwrap_or(intrinsic_w),
66            LengthOrAuto::Auto => intrinsic_w,
67        };
68
69        let height = match &self.layout_style.size.height {
70            LengthOrAuto::Length(len) => len
71                .resolve_with(
72                    ctx.containing_block_height,
73                    ctx.containing_block_width.unwrap_or(0.0),
74                    ctx.containing_block_height.unwrap_or(0.0),
75                )
76                .unwrap_or(intrinsic_h),
77            LengthOrAuto::Auto => intrinsic_h,
78        };
79
80        Rect {
81            x: 0.0,
82            y: 0.0,
83            width,
84            height,
85        }
86    }
87
88    fn write_debug(&self, f: &mut std::fmt::Formatter<'_>) -> std::fmt::Result {
89        write!(f, "CustomLayoutBridge")
90    }
91}

97thread_local! {
98    static CUSTOM_INLINE_RESULTS: RefCell<HashMap<usize, CustomInlineResult>> =
99        RefCell::new(HashMap::new());
100}

102static NEXT_CUSTOM_INLINE_ID: AtomicUsize = AtomicUsize::new(1);

104#[derive(Debug, Clone)]
105pub struct CustomInlineResult {
106    pub spans: Vec<LineSpan>,
107    pub width: f32,
108    pub height: f32,
109    pub border_top: f32,
110    pub border_right: f32,
111    pub border_bottom: f32,
112    pub border_left: f32,
113    pub padding_top: f32,
114    pub padding_right: f32,
115    pub padding_bottom: f32,
116    pub padding_left: f32,
117}

119/// An inline-level bridge for custom nodes implementing [`FlowLayouter`].
120///
121/// During layout, the engine calls [`layout`](Self::layout) which stores
122/// the result in a thread-local cache keyed by [`id`](Self::id).  The
123/// rendering layer later retrieves it via
124/// [`get_custom_inline_result`].
125#[derive(Debug)]
126pub struct CustomInlineBridge {
127    pub id: usize,
128    node: Rc<dyn CustomNode>,
129    layout_style: Style,
130    width: f32,
131    height: f32,
132    style: ContainerStyle,
133}
134
135impl CustomInlineBridge {
136    pub fn new(node: Rc<dyn CustomNode>, layout_style: Style, style: ContainerStyle) -> Self {
137        let (w, h) = node.intrinsic_size();
138        Self {
139            id: NEXT_CUSTOM_INLINE_ID.fetch_add(1, Ordering::Relaxed),
140            node,
141            layout_style,
142            width: w,
143            height: h,
144            style,
145        }
146    }
147
148    pub fn node(&self) -> &Rc<dyn CustomNode> {
149        &self.node
150    }
151
152    pub fn style(&self) -> &ContainerStyle {
153        &self.style
154    }
155}

157impl FlowLayouter for CustomInlineBridge {
158    fn layout(&self, ctx: &FlowLayoutContext) -> Vec<LineSpan> {
159        let x = ctx.start_pos.0;
160        let y = ctx.start_pos.1;
161
162        // If measure() was called first, use the CSS-resolved size;
163        // otherwise fall back to intrinsic size.
164        let (use_width, use_height) = CUSTOM_INLINE_RESULTS.with(|m| {
165            m.borrow()
166                .get(&self.id)
167                .map_or((self.width, self.height), |r| (r.width, r.height))
168        });
169
170        let spans = vec![LineSpan {
171            x_range: x..(x + use_width),
172            line_pos: (x, y),
173            line_index: 0,
174        }];
175
176        // Update the cache entry with spans (measure() may have pre-filled it)
177        CUSTOM_INLINE_RESULTS.with(|m| {
178            if let Some(entry) = m.borrow_mut().get_mut(&self.id) {
179                entry.spans = spans.clone();
180            } else {
181                m.borrow_mut().insert(
182                    self.id,
183                    CustomInlineResult {
184                        spans: spans.clone(),
185                        width: use_width,
186                        height: use_height,
187                        border_top: 0.0,
188                        border_right: 0.0,
189                        border_bottom: 0.0,
190                        border_left: 0.0,
191                        padding_top: 0.0,
192                        padding_right: 0.0,
193                        padding_bottom: 0.0,
194                        padding_left: 0.0,
195                    },
196                );
197            }
198        });
199
200        spans
201    }
202
203    fn measure(&self, ctx: &LayoutContext) -> MeasureResult {
204        let vw = ctx.containing_block_width.unwrap_or(0.0);
205        let vh = ctx.containing_block_height.unwrap_or(0.0);
206
207        let css_w = self
208            .layout_style
209            .size
210            .width
211            .resolve_with(ctx.containing_block_width, vw, vh)
212            .unwrap_or(self.width);
213        let css_h = self
214            .layout_style
215            .size
216            .height
217            .resolve_with(ctx.containing_block_height, vw, vh)
218            .unwrap_or(self.height);
219
220        let sp = &self.layout_style.spacing;
221        let b_top = sp
222            .border_top
223            .resolve_with(ctx.containing_block_width, vw, vh)
224            .unwrap_or(0.0);
225        let b_right = sp
226            .border_right
227            .resolve_with(ctx.containing_block_width, vw, vh)
228            .unwrap_or(0.0);
229        let b_bottom = sp
230            .border_bottom
231            .resolve_with(ctx.containing_block_width, vw, vh)
232            .unwrap_or(0.0);
233        let b_left = sp
234            .border_left
235            .resolve_with(ctx.containing_block_width, vw, vh)
236            .unwrap_or(0.0);
237        let p_top = sp
238            .padding_top
239            .resolve_with(ctx.containing_block_width, vw, vh)
240            .unwrap_or(0.0);
241        let p_right = sp
242            .padding_right
243            .resolve_with(ctx.containing_block_width, vw, vh)
244            .unwrap_or(0.0);
245        let p_bottom = sp
246            .padding_bottom
247            .resolve_with(ctx.containing_block_width, vw, vh)
248            .unwrap_or(0.0);
249        let p_left = sp
250            .padding_left
251            .resolve_with(ctx.containing_block_width, vw, vh)
252            .unwrap_or(0.0);
253
254        CUSTOM_INLINE_RESULTS.with(|m| {
255            m.borrow_mut().insert(
256                self.id,
257                CustomInlineResult {
258                    spans: Vec::new(),
259                    width: css_w,
260                    height: css_h,
261                    border_top: b_top,
262                    border_right: b_right,
263                    border_bottom: b_bottom,
264                    border_left: b_left,
265                    padding_top: p_top,
266                    padding_right: p_right,
267                    padding_bottom: p_bottom,
268                    padding_left: p_left,
269                },
270            );
271        });
272
273        MeasureResult {
274            width: css_w,
275            height: css_h,
276        }
277    }
278
279    fn write_debug(&self, f: &mut std::fmt::Formatter<'_>) -> std::fmt::Result {
280        write!(f, "CustomInlineBridge(id={})", self.id)
281    }
282}

284/// Retrieve the cached layout result for a custom inline element.
285pub fn get_custom_inline_result(id: usize) -> Option<CustomInlineResult> {
286    CUSTOM_INLINE_RESULTS.with(|m| m.borrow().get(&id).cloned())
287}
