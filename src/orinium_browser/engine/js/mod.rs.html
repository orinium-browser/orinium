1//! Minimal JS runtime backed by `pixi_byte`.
2//!
3//! Hosts the JS engine on the UI thread and installs a small set of DOM
4//! bindings (`console`, `document.getElementById`, element properties).
5//! The engine never imports `platform`; DOM access goes through the shared
6//! host slot that `JsRuntime` registers on the VM.
7
8use crate::engine::html::{DomTree, HtmlNodeType};
9use crate::engine::tree::NodeRef;
10use pixi_byte::value::jsobject::{JSObject, Property};
11use pixi_byte::vm::VM;
12use pixi_byte::{JSResult, JSValue};
13use std::any::Any;
14use std::cell::{Cell, RefCell};
15use std::collections::HashMap;
16use std::rc::Rc;

18/// State shared between the JS natives and the browser side.
19///
20/// The JS-facing `u64` counter (`__orinium_dom_id`) maps to a live DOM node so
21/// element handles survive relayouts: `Rc` handles on the DOM nodes are stable,
22/// while snapshot ids are not.
23pub struct JsHost {
24    dom: Rc<DomTree>,
25    refs: HashMap<
26        u64,
27        std::rc::Weak<std::cell::RefCell<crate::engine::tree::TreeNode<HtmlNodeType>>>,
28    >,
29    /// Element JS objects per DOM id, kept alive so `onclick` handlers
30    /// registered on them survive and can be invoked on user clicks.
31    objects: HashMap<u64, Rc<RefCell<JSObject>>>,
32    next_id: u64,
33    needs_redraw: Rc<Cell<bool>>,
34}
35
36impl JsHost {
37    /// Finds the JS-facing DOM id registered for a live DOM node, if any.
38    fn dom_id_for_node(&self, node: &NodeRef<HtmlNodeType>) -> Option<u64> {
39        self.refs.iter().find_map(|(&dom_id, weak)| {
40            if std::ptr::eq(weak.as_ptr(), Rc::as_ptr(node)) {
41                Some(dom_id)
42            } else {
43                None
44            }
45        })
46    }
47}

49/// A JS engine instance with DOM bindings installed.
50pub struct JsRuntime {
51    engine: pixi_byte::JSEngine,
52    needs_redraw: Rc<Cell<bool>>,
53}
54
55impl JsRuntime {
56    /// Creates a runtime sharing the given DOM tree with the browser side.
57    pub fn new(dom: Rc<DomTree>) -> Self {
58        let needs_redraw = Rc::new(Cell::new(false));
59        let host = Rc::new(RefCell::new(JsHost {
60            dom,
61            refs: HashMap::new(),
62            objects: HashMap::new(),
63            next_id: 0,
64            needs_redraw: Rc::clone(&needs_redraw),
65        }));
66
67        let mut engine = pixi_byte::JSEngine::new();
68        engine.set_host(host);
69
70        install_console(&mut engine);
71        install_document(&mut engine);
72
73        Self {
74            engine,
75            needs_redraw,
76        }
77    }
78
79    /// Evaluates a script, logging JS errors instead of crashing the page.
80    pub fn run_script(&mut self, source: &str) {
81        match self.engine.eval(source) {
82            Ok(_) => {}
83            Err(err) => log::info!("JS error: {}", err),
84        }
85    }
86
87    /// Returns whether a script mutated the DOM and a relayout is needed.
88    pub fn needs_redraw(&self) -> bool {
89        self.needs_redraw.get()
90    }
91
92    /// Clears and returns the redraw flag.
93    pub fn take_needs_redraw(&self) -> bool {
94        self.needs_redraw.replace(false)
95    }
96
97    /// Dispatches a click to the `onclick` handler of `node`, if one was
98    /// registered on its JS element object.
99    ///
100    /// Returns whether the handler ran (regardless of DOM mutation).
101    pub fn click(&mut self, node: &NodeRef<HtmlNodeType>) -> bool {
102        let Some(dom_id) = with_host(self.engine.vm(), |host| host.dom_id_for_node(node)).flatten()
103        else {
104            return false;
105        };
106        let Some(obj) =
107            with_host(self.engine.vm(), |host| host.objects.get(&dom_id).cloned()).flatten()
108        else {
109            return false;
110        };
111
112        let onclick = obj.borrow().get("onclick");
113        if !matches!(
114            onclick,
115            JSValue::Function(..) | JSValue::NativeFunction(_) | JSValue::BoundFunction(..)
116        ) {
117            return false;
118        }
119        if let Err(err) = self.engine.call(onclick, JSValue::Object(obj), Vec::new()) {
120            log::info!("JS error on click: {}", err);
121        }
122        true
123    }
124}

128/// Runs `f` with an immutable borrow of the host data, if set and downcastable.
129fn with_host<R>(vm: &VM, f: impl FnOnce(&JsHost) -> R) -> Option<R> {
130    let host = vm.host.as_ref()?;
131    let host_ref = host.borrow();
132    let js_host = (&*host_ref as &dyn Any).downcast_ref::<JsHost>()?;
133    Some(f(js_host))
134}
135
136/// Runs `f` with a mutable borrow of the host data, if set and downcastable.
137fn with_host_mut<R>(vm: &VM, f: impl FnOnce(&mut JsHost) -> R) -> Option<R> {
138    let host = vm.host.as_ref()?;
139    let mut host_ref = host.borrow_mut();
140    let js_host = (&mut *host_ref as &mut dyn Any).downcast_mut::<JsHost>()?;
141    Some(f(js_host))
142}
143
144/// Records a DOM mutation: bumps the tree version and flags a relayout.
145fn mark_dom_dirty(vm: &VM) {
146    if let Some(host) = vm.host.as_ref() {
147        let host_ref = host.borrow();
148        if let Some(js_host) = (&*host_ref as &dyn Any).downcast_ref::<JsHost>() {
149            js_host.dom.mark_dirty();
150            js_host.needs_redraw.set(true);
151        }
152    }
153}

155/// Extracts the hidden DOM id counter from an element `this` object.
156fn element_dom_id(this: &JSValue) -> Option<u64> {
157    let JSValue::Object(obj) = this else {
158        return None;
159    };
160    let JSValue::Number(n) = obj.borrow().get("__orinium_dom_id") else {
161        return None;
162    };
163    Some(n as u64)
164}
165
166/// Resolves the `this` element back to a live DOM node (dead node -> None).
167fn element_node(vm: &VM, this: &JSValue) -> Option<NodeRef<HtmlNodeType>> {
168    let dom_id = element_dom_id(this)?;
169    with_host(vm, |host| host.refs.get(&dom_id).and_then(|w| w.upgrade())).flatten()
170}

174fn install_console(engine: &mut pixi_byte::JSEngine) {
175    let console_obj = Rc::new(RefCell::new(JSObject::new()));
176    {
177        let mut console = console_obj.borrow_mut();
178        console.set("log".to_string(), JSValue::NativeFunction(console_log));
179        console.set("warn".to_string(), JSValue::NativeFunction(console_warn));
180        console.set("error".to_string(), JSValue::NativeFunction(console_error));
181    }
182    engine
183        .global_mut()
184        .borrow_mut()
185        .set("console".to_string(), JSValue::Object(console_obj));
186}
187
188fn console_message(vm: &mut VM, args: Vec<JSValue>, level: log::Level) -> JSResult<JSValue> {
189    let message: Vec<String> = args.iter().skip(1).map(|v| v.to_console_string()).collect();
190    log::log!(level, "{}", message.join(" "));
191    let _ = vm;
192    Ok(JSValue::Undefined)
193}
194
195fn console_log(vm: &mut VM, args: Vec<JSValue>) -> JSResult<JSValue> {
196    console_message(vm, args, log::Level::Info)
197}
198
199fn console_warn(vm: &mut VM, args: Vec<JSValue>) -> JSResult<JSValue> {
200    console_message(vm, args, log::Level::Warn)
201}
202
203fn console_error(vm: &mut VM, args: Vec<JSValue>) -> JSResult<JSValue> {
204    console_message(vm, args, log::Level::Error)
205}

209fn install_document(engine: &mut pixi_byte::JSEngine) {
210    let document_obj = Rc::new(RefCell::new(JSObject::new()));
211    {
212        let mut document = document_obj.borrow_mut();
213        document.set(
214            "getElementById".to_string(),
215            JSValue::NativeFunction(get_element_by_id),
216        );
217    }
218    engine
219        .global_mut()
220        .borrow_mut()
221        .set("document".to_string(), JSValue::Object(document_obj));
222}
223
224fn get_element_by_id(vm: &mut VM, args: Vec<JSValue>) -> JSResult<JSValue> {
225    let Some(JSValue::String(id)) = args.get(1) else {
226        return Ok(JSValue::Null);
227    };
228
229    // Look up the node and capture its static attributes first.
230    let Some((node, tag_name, attr_id)) = with_host(vm, |host| {
231        host.dom.get_element_by_id(id).map(|node| {
232            let (tag_name, attr_id) = {
233                let n = node.borrow();
234                (
235                    n.value.tag_name().unwrap_or("").to_string(),
236                    n.value.get_attr("id").unwrap_or("").to_string(),
237                )
238            };
239            (node, tag_name, attr_id)
240        })
241    })
242    .flatten() else {
243        return Ok(JSValue::Null);
244    };
245
246    // Register the live node so later property access can resolve it. Reuse
247    // the existing id and element object when this node was already exposed.
248    let Some(dom_id) = with_host_mut(vm, |host| {
249        if let Some(dom_id) = host.dom_id_for_node(&node) {
250            return Some(dom_id);
251        }
252        host.next_id += 1;
253        let dom_id = host.next_id;
254        host.refs.insert(dom_id, Rc::downgrade(&node));
255        Some(dom_id)
256    })
257    .flatten() else {
258        return Ok(JSValue::Null);
259    };
260
261    let obj = with_host_mut(vm, |host| {
262        if let Some(existing) = host.objects.get(&dom_id) {
263            return Rc::clone(existing);
264        }
265        let obj = make_element(tag_name, attr_id, dom_id);
266        host.objects.insert(dom_id, Rc::clone(&obj));
267        obj
268    })
269    .expect("host must be present");
270
271    Ok(JSValue::Object(obj))
272}

276fn make_element(tag_name: String, attr_id: String, dom_id: u64) -> Rc<RefCell<JSObject>> {
277    let mut obj = JSObject::new();
278    obj.define_property(
279        "__orinium_dom_id".to_string(),
280        Property {
281            value: JSValue::Number(dom_id as f64),
282            enumerable: false,
283            writable: false,
284            configurable: false,
285            getter: None,
286            setter: None,
287        },
288    );
289    obj.define_property(
290        "tagName".to_string(),
291        Property::read_only(JSValue::String(tag_name)),
292    );
293    obj.define_property(
294        "id".to_string(),
295        Property::read_only(JSValue::String(attr_id)),
296    );
297    obj.define_property(
298        "textContent".to_string(),
299        accessor_property(get_text_content, set_text_content),
300    );
301    obj.define_property(
302        "innerText".to_string(),
303        accessor_property(get_inner_text, set_inner_text),
304    );
305    obj.set(
306        "getAttribute".to_string(),
307        JSValue::NativeFunction(get_attribute),
308    );
309    obj.set(
310        "setAttribute".to_string(),
311        JSValue::NativeFunction(set_attribute),
312    );
313    Rc::new(RefCell::new(obj))
314}
315
316fn accessor_property(
317    getter: pixi_byte::NativeFunctionType,
318    setter: pixi_byte::NativeFunctionType,
319) -> Property {
320    Property {
321        value: JSValue::Undefined,
322        enumerable: true,
323        writable: false,
324        configurable: false,
325        getter: Some(JSValue::NativeFunction(getter)),
326        setter: Some(JSValue::NativeFunction(setter)),
327    }
328}

330fn get_text_content(vm: &mut VM, args: Vec<JSValue>) -> JSResult<JSValue> {
331    let Some(node) = element_node(vm, args.first().unwrap_or(&JSValue::Undefined)) else {
332        return Ok(JSValue::Null);
333    };
334    Ok(JSValue::String(DomTree::inner_text(&node)))
335}
336
337fn set_text_content(vm: &mut VM, args: Vec<JSValue>) -> JSResult<JSValue> {
338    let Some(node) = element_node(vm, args.first().unwrap_or(&JSValue::Undefined)) else {
339        return Ok(JSValue::Undefined);
340    };
341    let new_text = args
342        .get(1)
343        .map(|v| v.to_console_string())
344        .unwrap_or_default();
345    DomTree::set_text_content(&node, &new_text);
346    mark_dom_dirty(vm);
347    Ok(JSValue::Undefined)
348}
349
350fn get_inner_text(vm: &mut VM, args: Vec<JSValue>) -> JSResult<JSValue> {
351    get_text_content(vm, args)
352}
353
354fn set_inner_text(vm: &mut VM, args: Vec<JSValue>) -> JSResult<JSValue> {
355    set_text_content(vm, args)
356}
357
358fn get_attribute(vm: &mut VM, args: Vec<JSValue>) -> JSResult<JSValue> {
359    let Some(node) = element_node(vm, args.first().unwrap_or(&JSValue::Undefined)) else {
360        return Ok(JSValue::Null);
361    };
362    let Some(JSValue::String(name)) = args.get(1) else {
363        return Ok(JSValue::Undefined);
364    };
365    match node.borrow().value.get_attr(name) {
366        Some(value) => Ok(JSValue::String(value.to_string())),
367        None => Ok(JSValue::Null),
368    }
369}
370
371fn set_attribute(vm: &mut VM, args: Vec<JSValue>) -> JSResult<JSValue> {
372    let Some(node) = element_node(vm, args.first().unwrap_or(&JSValue::Undefined)) else {
373        return Ok(JSValue::Undefined);
374    };
375    let Some(JSValue::String(name)) = args.get(1) else {
376        return Ok(JSValue::Undefined);
377    };
378    let value = args
379        .get(2)
380        .map(|v| v.to_console_string())
381        .unwrap_or_default();
382    node.borrow_mut().value.set_attr(name, value);
383    mark_dom_dirty(vm);
384    Ok(JSValue::Undefined)
385}

392    fn runtime_from_html(html: &str) -> (JsRuntime, Rc<DomTree>) {
393        let mut parser = HtmlParser::new(html);
394        let dom = Rc::new(parser.parse());
395        let runtime = JsRuntime::new(Rc::clone(&dom));
396        (runtime, dom)
397    }

399    #[test]
400    fn set_text_content_mutates_dom_and_marks_dirty() {
401        let (mut runtime, dom) = runtime_from_html(r#"<div id="hello">before</div>"#);
402        runtime.run_script(
403            r#"const el = document.getElementById("hello"); el.textContent = "hello from js";"#,
404        );
405        assert!(runtime.needs_redraw());
406        assert!(runtime.take_needs_redraw());
407
408        let node = dom.get_element_by_id("hello").unwrap();
409        assert_eq!(DomTree::inner_text(&node), "hello from js");
410    }

414        let (mut runtime, dom) = runtime_from_html(r#"<div id="hello"></div>"#);

417        );
