1//! Minimal JS runtime backed by `pixi_byte`.
2//!
3//! Installs a small set of DOM bindings (`console`, `document.getElementById`,
4//! element properties). The engine never imports `platform`; DOM access goes
5//! through the shared host slot that `JsRuntime` registers on the VM. The
6//! runtime normally lives on a background thread (see [`processor`]), owning a
7//! private mirror of the DOM that is synced with the UI thread via
8//! [`DomSnapshot`] commits. It can also be used directly on any thread.

443    /// Updates the language preferences exposed through `navigator`.
444    pub fn set_language(&mut self, language: &str) {
445        let language = language.trim();
446        if language.is_empty() {
447            return;
448        }
449        let mut languages = vec![JSValue::from_string(language.to_string())];
450        if let Some(base) = language.split('-').next()
451            && !base.eq_ignore_ascii_case(language)
452        {
453            languages.push(JSValue::from_string(base.to_string()));
454        }
455        if !language.eq_ignore_ascii_case("en-US") {
456            languages.push(JSValue::from_string("en-US".to_string()));
457        }
458
459        let global = self.engine.global_mut().borrow_mut();
460        let Some(navigator) = global.get("navigator").as_object() else {
461            return;
462        };
463        drop(global);
464        let mut navigator = navigator.borrow_mut();
465        navigator.define_property(
466            "language".to_string(),
467            host_read_only_property(JSValue::from_string(language.to_string())),
468        );
469        navigator.define_property(
470            "languages".to_string(),
471            host_read_only_property(self.engine.vm().array_from_values(languages)),
472        );
473    }

475    /// Evaluates a script, logging JS errors instead of crashing the page.
476    pub fn run_script(&mut self, source: &str) {
477        match self.engine.eval(source) {
478            Ok(_) => {}
479            Err(err) => {
480                if let JSError::Thrown(value) = &err
481                    && let Some(object) = value.as_object()
482                {
483                    let object = object.borrow();
484                    let details = object
485                        .keys()
486                        .into_iter()
487                        .map(|key| format!("{key}={}", object.get(&key).to_console_string()))
488                        .collect::<Vec<_>>()
489                        .join(", ");
490                    log::info!("JS error: uncaught object ({details})");
491                }
492                log::info!(
493                    "JS error: {} (source {:?}, pc={:?}, fn={:?}, stack={:?})",
494                    err,
495                    source.chars().take(60).collect::<String>(),
496                    self.engine.last_error_pc(),
497                    self.engine.last_error_fn(),
498                    self.engine
499                        .last_error_stack()
500                        .iter()
501                        .take(10)
502                        .collect::<Vec<_>>(),
503                );
504            }
505        }
506        self.perform_microtask_checkpoint();
507    }

509    /// Evaluates a script, returning the first JS error (parse or compile
510    /// failure, or an uncaught thrown value). Diagnostic harnesses use this to
511    /// distinguish "script ran clean" from "never parsed"; runtime exceptions
512    /// that the page itself catches still surface as `Ok`.
513    pub fn try_run_script(&mut self, source: &str) -> Result<(), String> {
514        let result = self.engine.eval(source);
515        self.perform_microtask_checkpoint();
516        match result {
517            Ok(_) => Ok(()),
518            Err(err) => Err(err.to_string()),
519        }

554            if host.dom_content_loaded_fired {
555                return None;
556            }
557
558            host.dom_content_loaded_fired = true;

592    /// Dispatches the window `load` event once the page has finished loading.
593    ///
594    /// Supports the body `onload` inline handler, the `window.onload` property,
595    /// and `addEventListener("load", ...)` registrations in that order. Returns
596    /// `true` only for the first dispatch attempt.
597    pub fn dispatch_window_load(&mut self) -> bool {
598        let state = with_host_mut(self.engine.vm(), |host| {
599            if host.window_load_fired {
600                return None;
601            }
602            host.window_load_fired = true;
603            Some((
604                host.window_inline_event_handlers
605                    .get("load")
606                    .cloned()
607                    .unwrap_or_default(),
608                host.document_event_listeners
609                    .get("load")
610                    .cloned()
611                    .unwrap_or_default(),
612            ))
613        });
614        let Some((body_onload, listeners)) = state.flatten() else {
615            return false;
616        };
617
618        let window_object = Rc::clone(self.engine.global_mut());
619
620        // The body `onload` is a window-level inline event handler that fires
621        // first, as if it had been registered first by the parser.
622        if !body_onload.trim().is_empty() {
623            self.call_inline_load_handler(&window_object, &body_onload, "body onload");
624        }

626        let onload = window_object.borrow().get("onload");
627        if is_callable(&onload) {
628            self.call_load_handler(&window_object, onload, "window.onload");
629        }
630
631        for listener in listeners {
632            self.call_load_handler(&window_object, listener, "window load listener");
633        }
634
635        self.perform_microtask_checkpoint();
636        true
637    }

677    /// Runs timer callbacks whose deadlines have elapsed.
678    ///
679    /// Returns whether at least one callback was invoked. Repeating timers are
680    /// rescheduled before invocation so they can cancel themselves.
681    pub fn run_due_timers(&mut self) -> bool {
682        let invocations = with_host_mut(self.engine.vm(), |host| {
683            let now = Instant::now();
684            let mut invocations = Vec::new();
685            let mut index = 0;
686            while index < host.timers.len() {
687                if host.timers[index].deadline > now {
688                    index += 1;
689                    continue;
690                }
691
692                let callback = host.timers[index].callback.clone();
693                let arguments = host.timers[index].arguments.clone();
694                if let Some(interval) = host.timers[index].interval {
695                    host.timers[index].deadline = now + interval;
696                    index += 1;
697                } else {
698                    host.timers.remove(index);
699                }
700                invocations.push((callback, arguments));
701            }
702            invocations
703        })
704        .unwrap_or_default();
705
706        let ran_callback = !invocations.is_empty();
707        for (callback, arguments) in invocations {
708            if let Err(err) = self.engine.call(callback, JSValue::undefined(), arguments) {
709                log::info!("JS error in timer callback: {}", err);
710            }
711            self.perform_microtask_checkpoint();
712        }
713        ran_callback
714    }

802                        .value

907        resolve_xml_http_request(&mut self.engine, xhr, response);

985    /// Replaces the mirror DOM with a snapshot produced by the browser side.
986    ///
987    /// The mirror is rebuilt from `snapshot` and node references are re-registered
988    /// so existing JS element handles keep resolving. JS-created (detached) nodes
989    /// are preserved; they are not part of the committed DOM but may still be
990    /// referenced from scripts.
991    pub fn apply_dom(&mut self, snapshot: &DomSnapshot) {
992        let (tree, dom_ids) = snapshot.into_tree();
993        with_host_mut(self.engine.vm(), |host| {
994            host.dom = Rc::new(tree);
995            let mut refs = std::mem::take(&mut host.refs);
996            host.dom.traverse(|node| {
997                if let Some(&dom_id) = dom_ids.get(&(Rc::as_ptr(node) as usize)) {
998                    refs.insert(dom_id, Rc::downgrade(node));
999                }
1000            });
1001            for (&dom_id, node) in &host.detached_nodes {
1002                refs.entry(dom_id).or_insert_with(|| Rc::downgrade(node));
1003            }
1004            host.refs = refs;
1005            if let Some(max_id) = dom_ids.values().max() {
1006                host.next_id = host.next_id.max(*max_id);
1007            }
1008            Self::register_window_event_handlers(host);
1009        });
1010    }

1040    /// Dispatches a click to the handlers registered on `node`.
1041    ///
1042    /// Both the `onclick` property and `addEventListener("click", ...)` are
1043    /// supported. The event bubbles through exposed ancestor elements so
1044    /// delegated listeners such as React's root listener receive it.
1045    /// Returns whether at least one handler ran.
1046    pub fn click(&mut self, node: &NodeRef<HtmlNodeType>) -> bool {
1047        let mut path = Vec::new();
1048        let mut current = Some(Rc::clone(node));
1049        while let Some(node) = current {
1050            current = node.borrow().parent();
1051            if let Some(object) = expose_node(self.engine.vm(), node).and_then(|v| v.as_object()) {
1052                path.push(object);
1053            }
1054        }
1055        let Some(target) = path.first().cloned() else {
1056            return false;
1057        };
1058
1059        let mut ran_handler = false;
1060        for current_target in path {
1061            let Some(dom_id) = node_dom_id(&JSValue::from_object(Rc::clone(&current_target)))
1062            else {
1063                continue;
1064            };
1065            let onclick = current_target.borrow().get("onclick");
1066            let listeners = with_host(self.engine.vm(), |host| {
1067                host.element_event_listeners
1068                    .get(&dom_id)
1069                    .and_then(|events| events.get("click"))
1070                    .cloned()
1071                    .unwrap_or_default()
1072            })
1073            .unwrap_or_default();
1074            let has_onclick = is_callable(&onclick);
1075            if !has_onclick && listeners.is_empty() {
1076                continue;
1077            }
1078
1079            ran_handler = true;
1080            let event = make_event("click", Rc::clone(&target), Rc::clone(&current_target));
1081            if has_onclick
1082                && let Err(err) = self.engine.call(
1083                    onclick,
1084                    JSValue::from_object(Rc::clone(&current_target)),
1085                    vec![JSValue::from_object(Rc::clone(&event))],
1086                )
1087            {
1088                log::info!("JS error in onclick: {}", err);
1089            }
1090            if !event_flag(&event, "__orinium_immediate_propagation_stopped") {
1091                for listener in listeners {
1092                    if let Err(err) = self.engine.call(
1093                        listener.0,
1094                        JSValue::from_object(Rc::clone(&current_target)),
1095                        vec![JSValue::from_object(Rc::clone(&event))],
1096                    ) {
1097                        log::info!("JS error in click listener: {}", err);
1098                    }
1099                    if event_flag(&event, "__orinium_immediate_propagation_stopped") {
1100                        break;
1101                    }
1102                }
1103            }
1104            if event_flag(&event, "cancelBubble") {
1105                break;
1106            }
1107        }
1108        if ran_handler {
1109            self.perform_microtask_checkpoint();
1110        }
1111        ran_handler
1112    }

1207    ///
1208    /// Framework schedulers (React's `MessageChannel`-based pump) queue their
1209    /// work as jobs rather than timers, so harnesses need this to settle them.

1211        self.perform_microtask_checkpoint();
1212    }

1214    /// Drains queued microtasks in FIFO order, including jobs queued by jobs.
1215    fn perform_microtask_checkpoint(&mut self) {
1216        while let Err(err) = self.engine.run_jobs() {
1217            if let JSError::Thrown(value) = &err
1218                && let Some(object) = value.as_object()
1219            {
1220                let object = object.borrow();
1221                let details = object
1222                    .keys()
1223                    .into_iter()
1224                    .map(|key| format!("{key}={}", object.get(&key).to_console_string()))
1225                    .collect::<Vec<_>>()
1226                    .join(", ");
1227                log::info!("JS error in microtask: {} ({details})", err);
1228            } else {
1229                log::info!("JS error in microtask: {}", err);
1230            }
1231        }
1232    }
