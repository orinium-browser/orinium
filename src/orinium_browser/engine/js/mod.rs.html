1//! Minimal JS runtime backed by `pixi_byte`.
2//!
3//! Installs a small set of DOM bindings (`console`, `document.getElementById`,
4//! element properties). The engine never imports `platform`; DOM access goes
5//! through the shared host slot that `JsRuntime` registers on the VM. The
6//! runtime normally lives on a background thread (see [`processor`]), owning a
7//! private mirror of the DOM that is synced with the UI thread via
8//! [`DomSnapshot`] commits. It can also be used directly on any thread.

302    /// Updates the language preferences exposed through `navigator`.
303    pub fn set_language(&mut self, language: &str) {
304        let language = language.trim();
305        if language.is_empty() {
306            return;
307        }
308        let mut languages = vec![JSValue::String(language.to_string())];
309        if let Some(base) = language.split('-').next()
310            && !base.eq_ignore_ascii_case(language)
311        {
312            languages.push(JSValue::String(base.to_string()));
313        }
314        if !language.eq_ignore_ascii_case("en-US") {
315            languages.push(JSValue::String("en-US".to_string()));
316        }
317
318        let global = self.engine.global_mut().borrow_mut();
319        let JSValue::Object(navigator) = global.get("navigator") else {
320            return;
321        };
322        drop(global);
323        let mut navigator = navigator.borrow_mut();
324        navigator.define_property(
325            "language".to_string(),
326            host_read_only_property(JSValue::String(language.to_string())),
327        );
328        navigator.define_property(
329            "languages".to_string(),
330            host_read_only_property(JSArray::from_vec(languages).to_object()),
331        );
332    }

334    /// Evaluates a script, logging JS errors instead of crashing the page.
335    pub fn run_script(&mut self, source: &str) {
336        match self.engine.eval(source) {
337            Ok(_) => {}
338            Err(JSError::Thrown(JSValue::Object(object))) => {
339                let object = object.borrow();
340                let details = object
341                    .keys()
342                    .into_iter()
343                    .map(|key| format!("{key}={}", object.get(&key).to_console_string()))
344                    .collect::<Vec<_>>()
345                    .join(", ");
346                log::info!("JS error: uncaught object ({details})");
347            }
348            Err(err) => log::info!("JS error: {}", err),
349        }
350        self.perform_microtask_checkpoint();
351    }

360    /// Dispatches `DOMContentLoaded` to document listeners once.
361    ///
362    /// Returns `true` only for the first dispatch attempt. Listener errors are
363    /// logged and do not prevent the remaining listeners from running.
364    pub fn dispatch_dom_content_loaded(&mut self) -> bool {
365        let Some((document, listeners)) = with_host_mut(self.engine.vm(), |host| {
366            if host.dom_content_loaded_fired {
367                return None;
368            }
369
370            host.dom_content_loaded_fired = true;
371            Some((
372                host.document.as_ref().cloned(),
373                host.document_event_listeners
374                    .get("DOMContentLoaded")
375                    .cloned()
376                    .unwrap_or_default(),
377            ))
378        })
379        .flatten() else {
380            return false;
381        };
382
383        let Some(document) = document else {
384            return true;
385        };
386        for listener in listeners {
387            let event = make_event(
388                "DOMContentLoaded",
389                Rc::clone(&document),
390                Rc::clone(&document),
391            );
392            if let Err(err) = self.engine.call(
393                listener,
394                JSValue::Object(Rc::clone(&document)),
395                vec![JSValue::Object(event)],
396            ) {
397                log::info!("JS error on DOMContentLoaded: {}", err);
398            }
399        }
400        self.perform_microtask_checkpoint();
401        true
402    }

404    /// Runs timer callbacks whose deadlines have elapsed.
405    ///
406    /// Returns whether at least one callback was invoked. Repeating timers are
407    /// rescheduled before invocation so they can cancel themselves.
408    pub fn run_due_timers(&mut self) -> bool {
409        let invocations = with_host_mut(self.engine.vm(), |host| {
410            let now = Instant::now();
411            let mut invocations = Vec::new();
412            let mut index = 0;
413            while index < host.timers.len() {
414                if host.timers[index].deadline > now {
415                    index += 1;
416                    continue;
417                }
418
419                let callback = host.timers[index].callback.clone();
420                let arguments = host.timers[index].arguments.clone();
421                if let Some(interval) = host.timers[index].interval {
422                    host.timers[index].deadline = now + interval;
423                    index += 1;
424                } else {
425                    host.timers.remove(index);
426                }
427                invocations.push((callback, arguments));
428            }
429            invocations
430        })
431        .unwrap_or_default();
432
433        let ran_callback = !invocations.is_empty();
434        for (callback, arguments) in invocations {
435            if let Err(err) = self.engine.call(callback, JSValue::Undefined, arguments) {
436                log::info!("JS error in timer callback: {}", err);
437            }
438            self.perform_microtask_checkpoint();
439        }
440        ran_callback
441    }

482    /// Dispatches a non-bubbling event to a dynamically inserted element.
483    pub(crate) fn dispatch_element_event(&mut self, node_id: u64, event_type: &str) {
484        let Some((target, listeners)) = with_host(self.engine.vm(), |host| {
485            let target = host.objects.get(&node_id).cloned()?;
486            let listeners = host
487                .element_event_listeners
488                .get(&node_id)
489                .and_then(|events| events.get(event_type))
490                .cloned()
491                .unwrap_or_default();
492            Some((target, listeners))
493        })
494        .flatten() else {
495            return;
496        };
497        let handler = target.borrow().get(&format!("on{event_type}"));
498        let event = make_event(event_type, Rc::clone(&target), Rc::clone(&target));
499        if is_callable(&handler)
500            && let Err(error) = self.engine.call(
501                handler,
502                JSValue::Object(Rc::clone(&target)),
503                vec![JSValue::Object(Rc::clone(&event))],
504            )
505        {
506            log::info!("JS error in on{event_type}: {error}");
507        }
508        for listener in listeners {
509            if event_flag(&event, "__orinium_immediate_propagation_stopped") {
510                break;
511            }
512            if let Err(error) = self.engine.call(
513                listener,
514                JSValue::Object(Rc::clone(&target)),
515                vec![JSValue::Object(Rc::clone(&event))],
516            ) {
517                log::info!("JS error in {event_type} listener: {error}");
518            }
519        }
520    }

522    /// Resolves a pending JavaScript fetch and runs its microtask checkpoint.
523    pub(crate) fn resolve_fetch(&mut self, id: u64, response: JsFetchResponse) {
524        let capability =
525            with_host_mut(self.engine.vm(), |host| host.fetch_capabilities.remove(&id)).flatten();
526        if let Some(capability) = capability {
527            let response = make_fetch_response(response);
528            if let Err(err) = self.engine.call(
529                capability.resolve,
530                JSValue::Undefined,
531                vec![JSValue::Object(response)],
532            ) {
533                log::info!("JS error while resolving fetch: {}", err);
534            }
535            self.perform_microtask_checkpoint();
536            return;
537        }
538        let xhr = with_host_mut(self.engine.vm(), |host| host.xhr_requests.remove(&id)).flatten();
539        let Some(xhr) = xhr else { return };
540        resolve_xml_http_request(&mut self.engine, xhr, response);
541        self.perform_microtask_checkpoint();
542    }

544    /// Rejects a pending JavaScript fetch and runs its microtask checkpoint.
545    pub(crate) fn reject_fetch(&mut self, id: u64, reason: String) {
546        let capability =
547            with_host_mut(self.engine.vm(), |host| host.fetch_capabilities.remove(&id)).flatten();
548        if let Some(capability) = capability {
549            if let Err(err) = self.engine.call(
550                capability.reject,
551                JSValue::Undefined,
552                vec![JSValue::String(reason)],
553            ) {
554                log::info!("JS error while rejecting fetch: {}", err);
555            }
556            self.perform_microtask_checkpoint();
557            return;
558        }
559        let xhr = with_host_mut(self.engine.vm(), |host| host.xhr_requests.remove(&id)).flatten();
560        let Some(xhr) = xhr else { return };
561        let handler = xhr.borrow().get("onerror");
562        if is_callable(&handler) {
563            let _ = self.engine.call(
564                handler,
565                JSValue::Object(Rc::clone(&xhr)),
566                vec![JSValue::String(reason)],
567            );
568        }
569        self.perform_microtask_checkpoint();
570    }

572    /// Serializes the current mirror DOM for the browser side.
573    ///
574    /// Nodes exposed to scripts keep their stable `dom_id` so the UI thread can
575    /// rebuild the tree and re-register references on commit.
576    pub fn snapshot(&self) -> DomSnapshot {
577        let Some((root, dom_ids)) = with_host(self.engine.vm(), |host| {
578            let mut reverse = HashMap::new();
579            for (dom_id, weak) in &host.refs {
580                if let Some(node) = weak.upgrade() {
581                    reverse.insert(Rc::as_ptr(&node) as usize, *dom_id);
582                }
583            }
584            (Rc::clone(&host.dom.root), reverse)
585        }) else {
586            return DomSnapshot::default();
587        };
588        DomSnapshot::from_mirror(&root, &dom_ids)
589    }

591    /// Replaces the mirror DOM with a snapshot produced by the browser side.
592    ///
593    /// The mirror is rebuilt from `snapshot` and node references are re-registered
594    /// so existing JS element handles keep resolving. JS-created (detached) nodes
595    /// are preserved; they are not part of the committed DOM but may still be
596    /// referenced from scripts.
597    pub fn apply_dom(&mut self, snapshot: &DomSnapshot) {
598        let (tree, dom_ids) = snapshot.into_tree();
599        with_host_mut(self.engine.vm(), |host| {
600            host.dom = Rc::new(tree);
601            let mut refs = std::mem::take(&mut host.refs);
602            host.dom.traverse(|node| {
603                if let Some(&dom_id) = dom_ids.get(&(Rc::as_ptr(node) as usize)) {
604                    refs.insert(dom_id, Rc::downgrade(node));
605                }
606            });
607            for (&dom_id, node) in &host.detached_nodes {
608                refs.entry(dom_id).or_insert_with(|| Rc::downgrade(node));
609            }
610            host.refs = refs;
611            if let Some(max_id) = dom_ids.values().max() {
612                host.next_id = host.next_id.max(*max_id);
613            }
614        });
615    }

617    /// Dispatches a click to the handlers registered on the element with the
618    /// given JS-facing dom id. Returns whether at least one handler ran.
619    pub fn click_dom_id(&mut self, dom_id: u64) -> bool {
620        let Some(node) = with_host(self.engine.vm(), |host| {
621            host.refs.get(&dom_id).and_then(|w| w.upgrade())
622        })
623        .flatten() else {
624            return false;
625        };
626        self.click(&node)
627    }

629    /// Dispatches a click to the handlers registered on `node`.
630    ///
631    /// Both the `onclick` property and `addEventListener("click", ...)` are
632    /// supported. The event bubbles through exposed ancestor elements so
633    /// delegated listeners such as React's root listener receive it.
634    /// Returns whether at least one handler ran.
635    pub fn click(&mut self, node: &NodeRef<HtmlNodeType>) -> bool {
636        let mut path = Vec::new();
637        let mut current = Some(Rc::clone(node));
638        while let Some(node) = current {
639            current = node.borrow().parent();
640            if let Some(JSValue::Object(object)) = expose_node(self.engine.vm(), node) {
641                path.push(object);
642            }
643        }
644        let Some(target) = path.first().cloned() else {
645            return false;
646        };
647
648        let mut ran_handler = false;
649        for current_target in path {
650            let Some(dom_id) = node_dom_id(&JSValue::Object(Rc::clone(&current_target))) else {
651                continue;
652            };
653            let onclick = current_target.borrow().get("onclick");
654            let listeners = with_host(self.engine.vm(), |host| {
655                host.element_event_listeners
656                    .get(&dom_id)
657                    .and_then(|events| events.get("click"))
658                    .cloned()
659                    .unwrap_or_default()
660            })
661            .unwrap_or_default();
662            let has_onclick = is_callable(&onclick);
663            if !has_onclick && listeners.is_empty() {
664                continue;
665            }
666
667            ran_handler = true;
668            let event = make_event("click", Rc::clone(&target), Rc::clone(&current_target));
669            if has_onclick
670                && let Err(err) = self.engine.call(
671                    onclick,
672                    JSValue::Object(Rc::clone(&current_target)),
673                    vec![JSValue::Object(Rc::clone(&event))],
674                )
675            {
676                log::info!("JS error in onclick: {}", err);
677            }
678            if !event_flag(&event, "__orinium_immediate_propagation_stopped") {
679                for listener in listeners {
680                    if let Err(err) = self.engine.call(
681                        listener,
682                        JSValue::Object(Rc::clone(&current_target)),
683                        vec![JSValue::Object(Rc::clone(&event))],
684                    ) {
685                        log::info!("JS error in click listener: {}", err);
686                    }
687                    if event_flag(&event, "__orinium_immediate_propagation_stopped") {
688                        break;
689                    }
690                }
691            }
692            if event_flag(&event, "cancelBubble") {
693                break;
694            }
695        }
696        if ran_handler {
697            self.perform_microtask_checkpoint();
698        }
699        ran_handler
700    }
701
702    /// Drains queued microtasks in FIFO order, including jobs queued by jobs.
703    fn perform_microtask_checkpoint(&mut self) {
704        while let Err(err) = self.engine.run_jobs() {
705            if let JSError::Thrown(JSValue::Object(object)) = &err {
706                let object = object.borrow();
707                let details = object
708                    .keys()
709                    .into_iter()
710                    .map(|key| format!("{key}={}", object.get(&key).to_console_string()))
711                    .collect::<Vec<_>>()
712                    .join(", ");
713                log::info!("JS error in microtask: {} ({details})", err);
714            } else {
715                log::info!("JS error in microtask: {}", err);
716            }
717        }
718    }
