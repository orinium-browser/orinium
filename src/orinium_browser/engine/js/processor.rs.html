1//! Runs the page's JavaScript runtime on a dedicated background thread.
2//!
3//! Script evaluation, DOM events, timers, and fetch settlement are processed
4//! off the UI thread. The executor maintains a private DOM mirror and sends
5//! [`DomSnapshot`]s back when the DOM is mutated.
6//!
7//! This is not a Web Worker: scripts have full `window`/`document` access.
8//! Tasks are processed FIFO, with coalescable timer wakeups.
9

24/// What the background JS thread should do next.
25#[derive(Debug)]
26pub enum JsTask {
27    /// Set the URL exposed through `document` and related browser APIs.
28    SetDocumentUrl { url: String },
29    /// Update CSS-pixel dimensions exposed through the Window API.
30    SetViewport { width: f32, height: f32 },
31    /// Update the language preferences exposed through `navigator`.
32    SetLanguage { language: String },
33    /// Replace DOM geometry exposed by measurement APIs.
34    SetLayoutMetrics {
35        metrics: HashMap<u64, JsLayoutMetrics>,
36    },
37    /// Execute a classic (blocking or deferred) script.
38    RunScript { source: String },
39    /// Dispatch `DOMContentLoaded` to document listeners.
40    DispatchDomContentLoaded,
41    /// Run timer callbacks whose deadlines have elapsed (coalescable).
42    RunTimers,
43    /// Dispatch a click on the element with the given JS-facing dom id.
44    Click { dom_id: u64 },
45    /// Dispatch an event on the element with the given JS-facing dom id.
46    DispatchElementEvent { dom_id: u64, event_type: String },
47    /// Resolve a pending JavaScript `fetch()` with a network response.
48    ResolveFetch { id: u64, response: JsFetchResponse },
49    /// Reject a pending JavaScript `fetch()` after a network failure.
50    RejectFetch { id: u64, reason: String },
51    /// Replace the JS thread's mirror DOM with the UI's tree (write-backs).
52    UpdateDom { snapshot: DomSnapshot },
53}
54
55/// A task stamped with its position in the ordered stream.
56#[derive(Debug)]
57enum JsCommand {
58    Task { task: JsTask, version: u64 },
59}
60
61/// The outcome of a JS task, ready to be applied on the UI thread.
62#[derive(Debug)]
63pub struct JsTaskResult {
64    /// The JS thread's mirror DOM, present only when a script mutated it.
65    pub dom: Option<DomSnapshot>,
66    /// Whether the DOM changed and the UI needs to relayout and redraw.
67    pub needs_redraw: bool,
68    /// `fetch()` requests queued by scripts while running this task.
69    pub fetch_requests: Vec<JsFetchRequest>,
70    /// Dynamically inserted script elements discovered while running this task.
71    pub(crate) dynamic_script_requests: Vec<JsDynamicScriptRequest>,
72    /// Dynamically inserted stylesheet links discovered while running this task.
73    pub(crate) dynamic_style_requests: Vec<JsDynamicStyleRequest>,
74    /// Images created or populated while running this task.
75    pub(crate) dynamic_image_requests: Vec<JsDynamicImageRequest>,
76    /// The sequence number of the task that produced this result.
77    pub version: u64,
78}
79
80/// A processor that accepts [`JsTask`]s and returns the results produced by the
81/// background JS thread.
82pub struct JsProcessor {
83    cmd_tx: mpsc::Sender<JsCommand>,
84    result_rx: mpsc::Receiver<JsTaskResult>,
85    /// Latest task sequence number; shared with the background thread so it can
86    /// detect and skip superseded timer pokes.
87    latest: Arc<AtomicU64>,
88}
89
90impl std::fmt::Debug for JsProcessor {
91    fn fmt(&self, f: &mut std::fmt::Formatter<'_>) -> std::fmt::Result {
92        f.write_str("JsProcessor")
93    }
94}
95
96impl JsProcessor {
97    /// Starts the background JS thread initialized with a snapshot of the
98    /// parsed document.
99    pub fn new(initial: DomSnapshot) -> Self {
100        let (cmd_tx, cmd_rx) = mpsc::channel::<JsCommand>();
101        let (result_tx, result_rx) = mpsc::channel::<JsTaskResult>();
102
103        let latest = Arc::new(AtomicU64::new(0));
104        let thread_latest = Arc::clone(&latest);
105
106        thread::spawn(move || {
107            perf_scope!(init);
108            let (tree, _dom_ids) = initial.into_tree();
109            let mut runtime = JsRuntime::new(Rc::new(tree));
110            perf_scope!(apply_dom);
111            runtime.apply_dom(&initial);
112            profile_log!(
113                target: "JsProc",
114                log::Level::Info,
115                "[JsMetrics] runtime_init: {:?} | initial_apply_dom: {:?}",
116                init.elapsed(),
117                apply_dom.elapsed(),
118            );
119
120            for cmd in cmd_rx {
121                let JsCommand::Task { task, version } = cmd;
122                // A newer task queued after this timer poke supersedes it.
123                if matches!(task, JsTask::RunTimers)
124                    && version < thread_latest.load(Ordering::SeqCst)
125                {
126                    continue;
127                }
128
129                perf_scope!(total);
130                perf_scope!(run);
131                #[cfg_attr(not(feature = "profile"), allow(unused_variables))]
132                let did_work = run_task(&mut runtime, task);
133                #[cfg(any(feature = "profile", debug_assertions))]
134                let run_time = run.elapsed();
135
136                perf_scope!(collect);
137                let needs_redraw = runtime.take_needs_redraw();
138                let fetch_requests = runtime.take_fetch_requests();
139                let dynamic_script_requests = runtime.take_dynamic_script_requests();
140                let dynamic_style_requests = runtime.take_dynamic_style_requests();
141                let dynamic_image_requests = runtime.take_dynamic_image_requests();
142                let dom = if needs_redraw {
143                    Some(runtime.snapshot())
144                } else {
145                    None
146                };
147                #[cfg(any(feature = "profile", debug_assertions))]
148                let collect_time = collect.elapsed();
149
150                let _ = result_tx.send(JsTaskResult {
151                    dom,
152                    needs_redraw,
153                    fetch_requests,
154                    dynamic_script_requests,
155                    dynamic_style_requests,
156                    dynamic_image_requests,
157                    version,
158                });
159                profile_log!(
160                    target: "JsProc",
161                    if did_work {
162                        log::Level::Info
163                    } else {
164                        log::Level::Debug
165                    },
166                    "[JsMetrics] total: {:?} | run_task: {:?} | collect: {:?}",
167                    total.elapsed(),
168                    run_time,
169                    collect_time,
170                );
171            }
172        });
173
174        Self {
175            cmd_tx,
176            result_rx,
177            latest,
178        }
179    }
180
181    /// Sends a task to the background thread, stamped with a fresh sequence
182    /// number, and returns that sequence number so the caller can track when
183    /// the task's result has been applied.
184    pub fn send(&self, task: JsTask) -> u64 {
185        let version = self.latest.fetch_add(1, Ordering::SeqCst) + 1;
186        let _ = self.cmd_tx.send(JsCommand::Task { task, version });
187        version
188    }
189
190    /// Returns a completed task result, or `None` if none is ready yet.
191    pub fn try_receive(&self) -> Option<JsTaskResult> {
192        self.result_rx.try_recv().ok()
193    }
194}

196/// Executes a task on the runtime, reporting whether page JavaScript ran.
197///
198/// Pure state-sync tasks (`SetViewport`, `SetLayoutMetrics`, `UpdateDom`, …)
199/// return `false`; they dominate the task stream even on script-less pages.
200fn run_task(runtime: &mut JsRuntime, task: JsTask) -> bool {
201    match task {
202        JsTask::SetDocumentUrl { url } => {
203            runtime.set_document_url(&url);
204            false
205        }
206        JsTask::SetViewport { width, height } => {
207            runtime.set_viewport(width, height);
208            false
209        }
210        JsTask::SetLanguage { language } => {
211            runtime.set_language(&language);
212            false
213        }
214        JsTask::SetLayoutMetrics { metrics } => {
215            runtime.set_layout_metrics_by_dom_id(metrics);
216            false
217        }
218        JsTask::RunScript { source } => {
219            runtime.run_script(&source);
220            true
221        }
222        JsTask::DispatchDomContentLoaded => runtime.dispatch_dom_content_loaded(),
223        JsTask::RunTimers => runtime.run_due_timers(),
224        JsTask::Click { dom_id } => runtime.click_dom_id(dom_id),
225        JsTask::DispatchElementEvent { dom_id, event_type } => {
226            runtime.dispatch_element_event(dom_id, &event_type);
227            true
228        }
229        JsTask::ResolveFetch { id, response } => {
230            runtime.resolve_fetch(id, response);
231            true
232        }
233        JsTask::RejectFetch { id, reason } => {
234            runtime.reject_fetch(id, reason);
235            true
236        }
237        JsTask::UpdateDom { snapshot } => {
238            runtime.apply_dom(&snapshot);
239            false
240        }
241    }
242}

252    fn snapshot_of(html: &str) -> DomSnapshot {
253        let dom = HtmlParser::new(html).parse();
254        let (snapshot, _) = DomSnapshot::from_tree(&dom.root);
255        snapshot
256    }
257
258    fn wait_for_result(processor: &JsProcessor) -> JsTaskResult {
259        let deadline = Instant::now() + Duration::from_secs(5);
260        loop {
261            if let Some(result) = processor.try_receive() {
262                return result;
263            }
264            assert!(
265                Instant::now() < deadline,
266                "JS result did not arrive before the timeout"
267            );
268            std::thread::sleep(Duration::from_millis(1));
269        }
270    }
271
272    fn find_element<'i>(snapshot: &'i DomSnapshot, id: NodeId, tag: &str) -> Option<NodeId> {
273        if snapshot.node(id).kind.tag_name() == Some(tag) {
274            return Some(id);
275        }
276        snapshot
277            .children(id)
278            .iter()
279            .find_map(|&c| find_element(snapshot, c, tag))
280    }
281
282    fn get_attr(snapshot: &DomSnapshot, id: NodeId, name: &str) -> Option<String> {
283        snapshot.node(id).kind.get_attr(name).map(str::to_string)
284    }

286    #[test]
287    fn script_dom_changes_are_reported_in_the_result_snapshot() {
288        let processor =
289            JsProcessor::new(snapshot_of("<html><body><div id='x'></div></body></html>"));
290        processor.send(JsTask::RunScript {
291            source: r#"document.getElementById("x").setAttribute("data-a", "1");"#.to_string(),
292        });
293
294        let result = wait_for_result(&processor);
295        assert!(result.needs_redraw);
296        let snapshot = result.dom.expect("mutating script must commit a snapshot");
297        let root = snapshot.roots()[0];
298        let div = find_element(&snapshot, root, "div").unwrap();
299        assert_eq!(get_attr(&snapshot, div, "data-a").as_deref(), Some("1"));
300    }

302    #[test]
303    fn ordered_scripts_run_in_send_order() {
304        let processor =
305            JsProcessor::new(snapshot_of("<html><body><div id='x'></div></body></html>"));
306        processor.send(JsTask::RunScript {
307            source: r#"globalThis.__s = "a";"#.to_string(),
308        });
309        processor.send(JsTask::RunScript {
310            source: r#"
311                globalThis.__s += "b";
312                document.getElementById("x").setAttribute("data-s", globalThis.__s);
313            "#
314            .to_string(),
315        });
316
317        // Only the last script mutates the DOM; its snapshot must reflect both.
318        let mut result = None;
319        for _ in 0..2 {
320            let received = wait_for_result(&processor);
321            if received.needs_redraw {
322                result = Some(received);
323            }
324        }
325        let snapshot = result
326            .expect("the second script must commit a snapshot")
327            .dom
328            .unwrap();
329        let root = snapshot.roots()[0];
330        let div = find_element(&snapshot, root, "div").unwrap();
331        assert_eq!(get_attr(&snapshot, div, "data-s").as_deref(), Some("ab"));
332    }

334    #[test]
335    fn update_dom_replaces_the_mirror_before_later_tasks() {
336        let processor =
337            JsProcessor::new(snapshot_of("<html><body><div id='x'></div></body></html>"));
338        // The UI changes the value of #x and pushes the real tree over.
339        processor.send(JsTask::UpdateDom {
340            snapshot: snapshot_of("<html><body><div id='x' data-v='ui'></div></body></html>"),
341        });
342        processor.send(JsTask::RunScript {
343            source: r#"
344                const el = document.getElementById("x");
345                el.setAttribute("data-read", el.getAttribute("data-v"));
346            "#
347            .to_string(),
348        });
349
350        // UpdateDom applies first (FIFO), so the script reads the new value.
351        let update = wait_for_result(&processor);
352        assert!(!update.needs_redraw);
353        assert!(update.dom.is_none());
354
355        let result = wait_for_result(&processor);
356        assert!(result.needs_redraw);
357        let snapshot = result.dom.unwrap();
358        let root = snapshot.roots()[0];
359        let div = find_element(&snapshot, root, "div").unwrap();
360        assert_eq!(get_attr(&snapshot, div, "data-read").as_deref(), Some("ui"));
361    }
