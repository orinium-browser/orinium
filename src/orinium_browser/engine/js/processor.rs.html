1//! Runs the page's JavaScript runtime on a dedicated background thread.
2//!
3//! Script evaluation, DOM events, timers, and fetch settlement are processed
4//! off the UI thread. The executor maintains a private DOM mirror and sends
5//! [`DomSnapshot`]s back when the DOM is mutated.
6//!
7//! This is not a Web Worker: scripts have full `window`/`document` access.
8//! Tasks are processed FIFO, with coalescable timer wakeups.
9

24/// What the background JS thread should do next.
25#[derive(Debug)]
26pub enum JsTask {
27    /// Set the URL exposed through `document` and related browser APIs.
28    SetDocumentUrl { url: String },
29    /// Set the serialized origin exposed through `window`/`location`/`document`.
30    SetOrigin { origin: String },
31    /// Update CSS-pixel dimensions exposed through the Window API.
32    SetViewport { width: f32, height: f32 },
33    /// Update the language preferences exposed through `navigator`.
34    SetLanguage { language: String },
35    /// Replace DOM geometry exposed by measurement APIs.
36    SetLayoutMetrics {
37        metrics: HashMap<u64, JsLayoutMetrics>,
38    },
39    /// Execute a classic (blocking or deferred) script.
40    RunScript { source: String },
41    /// Dispatch `DOMContentLoaded` to document listeners.
42    DispatchDomContentLoaded,
43    /// Dispatch the window `load` event (fires `window.onload`).
44    DispatchWindowLoad,
45    /// Run timer callbacks whose deadlines have elapsed (coalescable).
46    RunTimers,
47    /// Dispatch a click on the element with the given JS-facing dom id.
48    Click { dom_id: u64 },
49    /// Dispatch an event on the element with the given JS-facing dom id.
50    DispatchElementEvent { dom_id: u64, event_type: String },
51    /// Resolve a pending JavaScript `fetch()` with a network response.
52    ResolveFetch { id: u64, response: JsFetchResponse },
53    /// Reject a pending JavaScript `fetch()` after a network failure.
54    RejectFetch { id: u64, reason: String },
55    /// Settle a pending DevTools inspection request with its JSON envelope.
56    ResolveDevTools { id: u64, result: String },
57    /// Replace the JS thread's mirror DOM with the UI's tree (write-backs).
58    UpdateDom { snapshot: DomSnapshot },
59}
60
61/// A task stamped with its position in the ordered stream.
62#[derive(Debug)]
63enum JsCommand {
64    Task { task: JsTask, version: u64 },
65}
66
67/// The outcome of a JS task, ready to be applied on the UI thread.
68#[derive(Debug)]
69pub struct JsTaskResult {
70    /// The JS thread's mirror DOM, present only when a script mutated it.
71    pub dom: Option<DomSnapshot>,
72    /// Whether the DOM changed and the UI needs to relayout and redraw.
73    pub needs_redraw: bool,
74    /// `fetch()` requests queued by scripts while running this task.
75    pub fetch_requests: Vec<JsFetchRequest>,
76    /// DevTools inspection requests queued by scripts while running this task.
77    pub devtools_requests: Vec<JsDevToolsRequest>,
78    /// Dynamically inserted script elements discovered while running this task.
79    pub(crate) dynamic_script_requests: Vec<JsDynamicScriptRequest>,
80    /// Dynamically inserted stylesheet links discovered while running this task.
81    pub(crate) dynamic_style_requests: Vec<JsDynamicStyleRequest>,
82    /// Images created or populated while running this task.
83    pub(crate) dynamic_image_requests: Vec<JsDynamicImageRequest>,
84    /// The sequence number of the task that produced this result.
85    pub version: u64,
86}
87
88/// A processor that accepts [`JsTask`]s and returns the results produced by the
89/// background JS thread.
90pub struct JsProcessor {
91    cmd_tx: mpsc::Sender<JsCommand>,
92    result_rx: mpsc::Receiver<JsTaskResult>,
93    /// Latest task sequence number; shared with the background thread so it can
94    /// detect and skip superseded timer pokes.
95    latest: Arc<AtomicU64>,
96}
97
98impl std::fmt::Debug for JsProcessor {
99    fn fmt(&self, f: &mut std::fmt::Formatter<'_>) -> std::fmt::Result {
100        f.write_str("JsProcessor")
101    }
102}
103
104impl JsProcessor {
105    /// Starts the background JS thread initialized with a snapshot of the
106    /// parsed document.
107    pub fn new(initial: DomSnapshot) -> Self {
108        let (cmd_tx, cmd_rx) = mpsc::channel::<JsCommand>();
109        let (result_tx, result_rx) = mpsc::channel::<JsTaskResult>();
110
111        let latest = Arc::new(AtomicU64::new(0));
112        let thread_latest = Arc::clone(&latest);
113
114        thread::spawn(move || {
115            perf_scope!(init);
116            let (tree, _dom_ids) = initial.into_tree();
117            let mut runtime = JsRuntime::new(Rc::new(tree));
118            perf_scope!(apply_dom);
119            runtime.apply_dom(&initial);
120            profile_log!(
121                target: "JsProc",
122                log::Level::Info,
123                "[JsMetrics] runtime_init: {:?} | initial_apply_dom: {:?}",
124                init.elapsed(),
125                apply_dom.elapsed(),
126            );
127
128            for cmd in cmd_rx {
129                let JsCommand::Task { task, version } = cmd;
130                // A newer task queued after this timer poke supersedes it.
131                if matches!(task, JsTask::RunTimers)
132                    && version < thread_latest.load(Ordering::SeqCst)
133                {
134                    continue;
135                }
136
137                perf_scope!(total);
138                perf_scope!(run);
139                #[cfg_attr(not(feature = "profile"), allow(unused_variables))]
140                let did_work = run_task(&mut runtime, task);
141                #[cfg(any(feature = "profile", debug_assertions))]
142                let run_time = run.elapsed();
143
144                perf_scope!(collect);
145                let needs_redraw = runtime.take_needs_redraw();
146                let fetch_requests = runtime.take_fetch_requests();
147                let devtools_requests = runtime.take_devtools_requests();
148                let dynamic_script_requests = runtime.take_dynamic_script_requests();
149                let dynamic_style_requests = runtime.take_dynamic_style_requests();
150                let dynamic_image_requests = runtime.take_dynamic_image_requests();
151                let dom = if needs_redraw {
152                    Some(runtime.snapshot())
153                } else {
154                    None
155                };
156                #[cfg(any(feature = "profile", debug_assertions))]
157                let collect_time = collect.elapsed();
158
159                let _ = result_tx.send(JsTaskResult {
160                    dom,
161                    needs_redraw,
162                    fetch_requests,
163                    devtools_requests,
164                    dynamic_script_requests,
165                    dynamic_style_requests,
166                    dynamic_image_requests,
167                    version,
168                });
169                profile_log!(
170                    target: "JsProc",
171                    if did_work {
172                        log::Level::Info
173                    } else {
174                        log::Level::Debug
175                    },
176                    "[JsMetrics] total: {:?} | run_task: {:?} | collect: {:?}",
177                    total.elapsed(),
178                    run_time,
179                    collect_time,
180                );
181            }
182        });
183
184        Self {
185            cmd_tx,
186            result_rx,
187            latest,
188        }
189    }
190
191    /// Sends a task to the background thread, stamped with a fresh sequence
192    /// number, and returns that sequence number so the caller can track when
193    /// the task's result has been applied.
194    pub fn send(&self, task: JsTask) -> u64 {
195        let version = self.latest.fetch_add(1, Ordering::SeqCst) + 1;
196        let _ = self.cmd_tx.send(JsCommand::Task { task, version });
197        version
198    }
199
200    /// Returns a completed task result, or `None` if none is ready yet.
201    pub fn try_receive(&self) -> Option<JsTaskResult> {
202        self.result_rx.try_recv().ok()
203    }
204}

206/// Executes a task on the runtime, reporting whether page JavaScript ran.
207///
208/// Pure state-sync tasks (`SetViewport`, `SetLayoutMetrics`, `UpdateDom`, …)
209/// return `false`; they dominate the task stream even on script-less pages.
210fn run_task(runtime: &mut JsRuntime, task: JsTask) -> bool {
211    match task {
212        JsTask::SetDocumentUrl { url } => {
213            runtime.set_document_url(&url);
214            false
215        }
216        JsTask::SetOrigin { origin } => {
217            runtime.set_page_origin(&origin);
218            false
219        }
220        JsTask::SetViewport { width, height } => {
221            runtime.set_viewport(width, height);
222            false
223        }
224        JsTask::SetLanguage { language } => {
225            runtime.set_language(&language);
226            false
227        }
228        JsTask::SetLayoutMetrics { metrics } => {
229            runtime.set_layout_metrics_by_dom_id(metrics);
230            false
231        }
232        JsTask::RunScript { source } => {
233            runtime.run_script(&source);
234            true
235        }
236        JsTask::DispatchDomContentLoaded => runtime.dispatch_dom_content_loaded(),
237        JsTask::DispatchWindowLoad => runtime.dispatch_window_load(),
238        JsTask::RunTimers => runtime.run_due_timers(),
239        JsTask::Click { dom_id } => runtime.click_dom_id(dom_id),
240        JsTask::DispatchElementEvent { dom_id, event_type } => {
241            runtime.dispatch_element_event(dom_id, &event_type);
242            true
243        }
244        JsTask::ResolveFetch { id, response } => {
245            runtime.resolve_fetch(id, response);
246            true
247        }
248        JsTask::RejectFetch { id, reason } => {
249            runtime.reject_fetch(id, reason);
250            true
251        }
252        JsTask::ResolveDevTools { id, result } => {
253            runtime.resolve_devtools(id, result);
254            true
255        }
256        JsTask::UpdateDom { snapshot } => {
257            runtime.apply_dom(&snapshot);
258            false
259        }
260    }
261}

265    use std::time::{Duration, Instant};

267    use super::*;
268    use crate::engine::html::parser::Parser as HtmlParser;

271    fn snapshot_of(html: &str) -> DomSnapshot {
272        let dom = HtmlParser::new(html).parse();
273        let (snapshot, _) = DomSnapshot::from_tree(&dom.root);
274        snapshot
275    }
276
277    fn wait_for_result(processor: &JsProcessor) -> JsTaskResult {
278        let deadline = Instant::now() + Duration::from_secs(5);
279        loop {
280            if let Some(result) = processor.try_receive() {
281                return result;
282            }
283            assert!(
284                Instant::now() < deadline,
285                "JS result did not arrive before the timeout"
286            );
287            std::thread::sleep(Duration::from_millis(1));
288        }
289    }
290
291    fn find_element<'i>(snapshot: &'i DomSnapshot, id: NodeId, tag: &str) -> Option<NodeId> {
292        if snapshot.node(id).kind.tag_name() == Some(tag) {
293            return Some(id);
294        }
295        snapshot
296            .children(id)
297            .iter()
298            .find_map(|&c| find_element(snapshot, c, tag))
299    }
300
301    fn get_attr(snapshot: &DomSnapshot, id: NodeId, name: &str) -> Option<String> {
302        snapshot.node(id).kind.get_attr(name).map(str::to_string)
303    }

305    #[test]
306    fn script_dom_changes_are_reported_in_the_result_snapshot() {
307        let processor =
308            JsProcessor::new(snapshot_of("<html><body><div id='x'></div></body></html>"));
309        processor.send(JsTask::RunScript {
310            source: r#"document.getElementById("x").setAttribute("data-a", "1");"#.to_string(),
311        });
312
313        let result = wait_for_result(&processor);
314        assert!(result.needs_redraw);
315        let snapshot = result.dom.expect("mutating script must commit a snapshot");
316        let root = snapshot.roots()[0];
317        let div = find_element(&snapshot, root, "div").unwrap();
318        assert_eq!(get_attr(&snapshot, div, "data-a").as_deref(), Some("1"));
319    }

321    #[test]
322    fn ordered_scripts_run_in_send_order() {
323        let processor =
324            JsProcessor::new(snapshot_of("<html><body><div id='x'></div></body></html>"));
325        processor.send(JsTask::RunScript {
326            source: r#"globalThis.__s = "a";"#.to_string(),
327        });
328        processor.send(JsTask::RunScript {
329            source: r#"
330                globalThis.__s += "b";
331                document.getElementById("x").setAttribute("data-s", globalThis.__s);
332            "#
333            .to_string(),
334        });
335
336        // Only the last script mutates the DOM; its snapshot must reflect both.
337        let mut result = None;
338        for _ in 0..2 {
339            let received = wait_for_result(&processor);
340            if received.needs_redraw {
341                result = Some(received);
342            }
343        }
344        let snapshot = result
345            .expect("the second script must commit a snapshot")
346            .dom
347            .unwrap();
348        let root = snapshot.roots()[0];
349        let div = find_element(&snapshot, root, "div").unwrap();
350        assert_eq!(get_attr(&snapshot, div, "data-s").as_deref(), Some("ab"));
351    }

353    #[test]
354    fn update_dom_replaces_the_mirror_before_later_tasks() {
355        let processor =
356            JsProcessor::new(snapshot_of("<html><body><div id='x'></div></body></html>"));
357        // The UI changes the value of #x and pushes the real tree over.
358        processor.send(JsTask::UpdateDom {
359            snapshot: snapshot_of("<html><body><div id='x' data-v='ui'></div></body></html>"),
360        });
361        processor.send(JsTask::RunScript {
362            source: r#"
363                const el = document.getElementById("x");
364                el.setAttribute("data-read", el.getAttribute("data-v"));
365            "#
366            .to_string(),
367        });
368
369        // UpdateDom applies first (FIFO), so the script reads the new value.
370        let update = wait_for_result(&processor);
371        assert!(!update.needs_redraw);
372        assert!(update.dom.is_none());
373
374        let result = wait_for_result(&processor);
375        assert!(result.needs_redraw);
376        let snapshot = result.dom.unwrap();
377        let root = snapshot.roots()[0];
378        let div = find_element(&snapshot, root, "div").unwrap();
379        assert_eq!(get_attr(&snapshot, div, "data-read").as_deref(), Some("ui"));
380    }
