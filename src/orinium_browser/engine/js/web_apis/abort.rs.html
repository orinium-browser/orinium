1//! `AbortController` / `AbortSignal` — cancellation plumbing for scripts.
2//!
3//! `AbortSignal` instances are ordinary `JSObject`s carrying an internal
4//! `aborted` flag plus registration lists. `AbortController.abort(reason)`
5//! flips the flag, stores the reason, queues `abort` event listeners, and
6//! rejects every pending fetch that registered the signal with
7//! `fetch(url, { signal })`. Timing follows the platform: `abort()` marks
8//! state synchronously and listeners run at the next microtask checkpoint.
9
10use crate::engine::js::common::{is_callable, noop, with_host_mut};

12use pixi_byte::value::jsvalue::{BoundFunctionData, JsValueKind};
13use pixi_byte::vm::VM;

15use std::cell::RefCell;
16use std::rc::Rc;

18/// Internal flag marking an aborted signal.
19const ABORTED: &str = "__orinium_aborted";
20/// Internal storage for the abort reason.
21const ABORT_REASON: &str = "__orinium_abort_reason";
22/// Internal list of `abort` event listeners queued for the next checkpoint.
23const ABORT_LISTENERS: &str = "__orinium_abort_listeners";
24/// Marks the singleton `AbortSignal.abort()` reason object.
25const ABORT_SIGNAL_MARKER: &str = "__orinium_abort_signal";
26/// Marks the per-controller signal object.
27const SIGNAL_CONTROLLER: &str = "__orinium_signal_controller";

29/// Installs `AbortController` and `AbortSignal` on the global object.
30pub(crate) fn install_abort_apis(engine: &mut pixi_byte::JSEngine) {
31    let mut controller = JSObject::new();
32    controller.set(
33        "__call__".to_string(),
34        JSValue::from_native_function(abort_controller_constructor),
35    );
36    controller.set(
37        "__construct__".to_string(),
38        JSValue::from_native_function(abort_controller_constructor),
39    );
40    let controller_prototype = Rc::new(RefCell::new(JSObject::new()));
41    controller_prototype.borrow_mut().set(
42        "abort".to_string(),
43        JSValue::from_native_function(controller_abort),
44    );
45    controller_prototype.borrow_mut().define_property(
46        "signal".to_string(),
47        Property {
48            value: JSValue::undefined(),
49            enumerable: true,
50            writable: false,
51            configurable: true,
52            getter: Some(JSValue::from_native_function(controller_get_signal)),
53            setter: None,
54        },
55    );
56    controller.define_property(
57        "prototype".to_string(),
58        Property::read_only(JSValue::from_object(Rc::clone(&controller_prototype))),
59    );
60
61    let mut signal_constructor = JSObject::new();
62    signal_constructor.set(
63        "__call__".to_string(),
64        JSValue::from_native_function(signal_construct_error),
65    );
66    signal_constructor.set(
67        "__construct__".to_string(),
68        JSValue::from_native_function(signal_construct_error),
69    );
70    signal_constructor.set(
71        "abort".to_string(),
72        JSValue::from_native_function(signal_static_abort),
73    );
74    signal_constructor.set(
75        "timeout".to_string(),
76        JSValue::from_native_function(signal_static_timeout),
77    );
78    signal_constructor.set(
79        "any".to_string(),
80        JSValue::from_native_function(signal_static_any),
81    );
82
83    let signal_prototype = Rc::new(RefCell::new(JSObject::new()));
84    signal_prototype.borrow_mut().set(
85        "throwIfAborted".to_string(),
86        JSValue::from_native_function(signal_throw_if_aborted),
87    );
88    signal_prototype.borrow_mut().set(
89        "addEventListener".to_string(),
90        JSValue::from_native_function(signal_add_event_listener),
91    );
92    signal_prototype.borrow_mut().set(
93        "removeEventListener".to_string(),
94        JSValue::from_native_function(signal_remove_event_listener),
95    );
96    signal_prototype.borrow_mut().set(
97        "dispatchEvent".to_string(),
98        JSValue::from_native_function(noop),
99    );
100    signal_prototype.borrow_mut().define_property(
101        "aborted".to_string(),
102        Property {
103            value: JSValue::undefined(),
104            enumerable: true,
105            writable: false,
106            configurable: true,
107            getter: Some(JSValue::from_native_function(signal_get_aborted)),
108            setter: None,
109        },
110    );
111    signal_prototype.borrow_mut().define_property(
112        "reason".to_string(),
113        Property {
114            value: JSValue::undefined(),
115            enumerable: true,
116            writable: false,
117            configurable: true,
118            getter: Some(JSValue::from_native_function(signal_get_reason)),
119            setter: None,
120        },
121    );
122    signal_constructor.define_property(
123        "prototype".to_string(),
124        Property::read_only(JSValue::from_object(Rc::clone(&signal_prototype))),
125    );
126
127    let mut global = engine.global_mut().borrow_mut();
128    global.set(
129        "AbortController".to_string(),
130        JSValue::from_object(Rc::new(RefCell::new(controller))),
131    );
132    global.set(
133        "AbortSignal".to_string(),
134        JSValue::from_object(Rc::new(RefCell::new(signal_constructor))),
135    );
136}

138fn controller_get_signal(_vm: &mut VM, args: Vec<JSValue>) -> JSResult<JSValue> {
139    let Some(controller) = args.first().and_then(JSValue::as_object) else {
140        return Err(JSError::TypeError(
141            "AbortController.signal getter called on incompatible receiver".to_string(),
142        ));
143    };
144    Ok(controller.borrow().get("signal"))
145}

147/// `new AbortController()` — creates `{ signal, abort }`.
148fn abort_controller_constructor(vm: &mut VM, _args: Vec<JSValue>) -> JSResult<JSValue> {
149    let signal = make_signal(vm, false, JSValue::undefined());
150    let prototype = vm
151        .global_object
152        .borrow()
153        .get("AbortController")
154        .as_object()
155        .and_then(|ctor| ctor.borrow().get("prototype").as_object());
156    let mut controller = match prototype {
157        Some(proto) => JSObject::with_prototype(Some(proto)),
158        None => JSObject::new(),
159    };
160    controller.define_property(
161        "signal".to_string(),
162        Property::read_only(JSValue::from_object(Rc::clone(&signal))),
163    );
164    controller.set(
165        "abort".to_string(),
166        JSValue::from_native_function(controller_abort),
167    );
168    signal
169        .borrow_mut()
170        .set(SIGNAL_CONTROLLER.to_string(), JSValue::undefined());
171    Ok(JSValue::from_object(Rc::new(RefCell::new(controller))))
172}

176    Err(JSError::TypeError(

181/// Builds a signal object with the given aborted state and reason.
182fn make_signal(vm: &mut VM, aborted: bool, reason: JSValue) -> Rc<RefCell<JSObject>> {
183    let prototype = vm
184        .global_object
185        .borrow()
186        .get("AbortSignal")
187        .as_object()
188        .and_then(|ctor| ctor.borrow().get("prototype").as_object());
189    let mut signal = match prototype {
190        Some(proto) => JSObject::with_prototype(Some(proto)),
191        None => JSObject::new(),
192    };
193    signal.set(ABORTED.to_string(), JSValue::from_bool(aborted));
194    signal.set(ABORT_REASON.to_string(), reason);
195    signal.set(ABORT_LISTENERS.to_string(), JSValue::undefined());
196    signal.set(ABORT_SIGNAL_MARKER.to_string(), JSValue::from_bool(true));
197    signal.define_property(
198        "aborted".to_string(),
199        Property {
200            value: JSValue::undefined(),
201            enumerable: true,
202            writable: false,
203            configurable: true,
204            getter: Some(JSValue::from_native_function(signal_get_aborted)),
205            setter: None,
206        },
207    );
208    signal.define_property(
209        "reason".to_string(),
210        Property {
211            value: JSValue::undefined(),
212            enumerable: true,
213            writable: false,
214            configurable: true,
215            getter: Some(JSValue::from_native_function(signal_get_reason)),
216            setter: None,
217        },
218    );
219    signal.set(
220        "throwIfAborted".to_string(),
221        JSValue::from_native_function(signal_throw_if_aborted),
222    );
223    signal.set(
224        "addEventListener".to_string(),
225        JSValue::from_native_function(signal_add_event_listener),
226    );
227    signal.set(
228        "removeEventListener".to_string(),
229        JSValue::from_native_function(signal_remove_event_listener),
230    );
231    signal.set(
232        "dispatchEvent".to_string(),
233        JSValue::from_native_function(noop),
234    );
235    Rc::new(RefCell::new(signal))
236}

238fn signal_receiver(args: &[JSValue]) -> JSResult<Rc<RefCell<JSObject>>> {
239    let signal = args.first().and_then(JSValue::as_object).ok_or_else(|| {
240        JSError::TypeError("AbortSignal method called on incompatible receiver".to_string())
241    })?;
242    if signal.borrow().get(ABORT_SIGNAL_MARKER).as_boolean() != Some(true) {
243        return Err(JSError::TypeError(
244            "AbortSignal method called on incompatible receiver".to_string(),
245        ));
246    }
247    Ok(signal)
248}

250fn signal_get_aborted(_vm: &mut VM, args: Vec<JSValue>) -> JSResult<JSValue> {
251    let signal = signal_receiver(&args)?;
252    Ok(JSValue::from_bool(
253        signal.borrow().get(ABORTED).as_boolean() == Some(true),
254    ))
255}

257fn signal_get_reason(_vm: &mut VM, args: Vec<JSValue>) -> JSResult<JSValue> {
258    let signal = signal_receiver(&args)?;
259    Ok(signal.borrow().get(ABORT_REASON))

262/// `signal.throwIfAborted()` — throws the abort reason when aborted.
263fn signal_throw_if_aborted(_vm: &mut VM, args: Vec<JSValue>) -> JSResult<JSValue> {
264    let signal = signal_receiver(&args)?;
265    let (aborted, reason) = {
266        let borrowed = signal.borrow();
267        (
268            borrowed.get(ABORTED).as_boolean() == Some(true),
269            borrowed.get(ABORT_REASON),
270        )
271    };
272    if aborted {
273        return Err(JSError::Thrown(reason));
274    }
275    Ok(JSValue::undefined())
276}

278/// `signal.addEventListener("abort", listener)` — records the listener so
279/// `abort()` can invoke it synchronously when the signal transitions.
280fn signal_add_event_listener(_vm: &mut VM, args: Vec<JSValue>) -> JSResult<JSValue> {
281    let signal = signal_receiver(&args)?;
282    let event_type = args
283        .get(1)
284        .map(JSValue::to_console_string)
285        .unwrap_or_default();
286    let listener = args.get(2).cloned().unwrap_or(JSValue::undefined());
287    if event_type != "abort" || !is_callable(&listener) {
288        return Ok(JSValue::undefined());
289    }
290    let mut borrowed = signal.borrow_mut();
291    let existing = borrowed.get(ABORT_LISTENERS);
292    let listeners = match existing.as_object() {
293        Some(object) => object,
294        None => {
295            let object = Rc::new(RefCell::new(JSObject::new()));
296            borrowed.set(
297                ABORT_LISTENERS.to_string(),
298                JSValue::from_object(Rc::clone(&object)),
299            );
300            object
301        }
302    };
303    drop(borrowed);
304    let next_key = listeners.borrow().keys().len().to_string();
305    listeners.borrow_mut().set(next_key, listener);
306    Ok(JSValue::undefined())
307}

309/// `signal.removeEventListener("abort", listener)` — drops a previously
310/// registered listener (matched by strict identity).
311fn signal_remove_event_listener(_vm: &mut VM, args: Vec<JSValue>) -> JSResult<JSValue> {
312    let signal = signal_receiver(&args)?;
313    let event_type = args
314        .get(1)
315        .map(JSValue::to_console_string)
316        .unwrap_or_default();
317    let listener = args.get(2).cloned().unwrap_or(JSValue::undefined());
318    if event_type != "abort" {
319        return Ok(JSValue::undefined());
320    }
321    let listeners = match signal.borrow().get(ABORT_LISTENERS).as_object() {
322        Some(object) => object,
323        None => return Ok(JSValue::undefined()),
324    };
325    let matched: Vec<String> = listeners
326        .borrow()
327        .keys()
328        .into_iter()
329        .filter(|key| listeners.borrow().get(key).strict_equals(&listener))
330        .collect();
331    for key in matched {
332        listeners.borrow_mut().delete(&key);
333    }
334    Ok(JSValue::undefined())
335}

337/// `controller.abort(reason?)` — marks the signal aborted, stores the reason
338/// (defaulting to a fresh `AbortError` DOMException-shaped object), queues the
339/// registered `abort` listeners as microtasks, and rejects pending fetches.
340fn controller_abort(vm: &mut VM, args: Vec<JSValue>) -> JSResult<JSValue> {
341    let Some(controller) = args.first().and_then(JSValue::as_object) else {
342        return Err(JSError::TypeError(
343            "AbortController.abort called on incompatible receiver".to_string(),
344        ));
345    };
346    let signal = match controller.borrow().get("signal").as_object() {
347        Some(signal) => signal,
348        None => {
349            return Err(JSError::TypeError(
350                "AbortController.abort called on incompatible receiver".to_string(),
351            ));
352        }
353    };
354    let reason = match args.get(1) {
355        Some(value) if value.kind() != JsValueKind::Undefined => value.clone(),
356        _ => abort_error_value(),
357    };
358    abort_signal_now(vm, &signal, reason);
359    Ok(JSValue::undefined())
360}

362/// Builds the default `AbortError` reason (a DOMException-shaped object).
363fn abort_error_value() -> JSValue {
364    let mut error = JSObject::new();
365    error.define_property(
366        "name".to_string(),
367        Property::read_only(JSValue::from_string("AbortError".to_string())),
368    );
369    error.define_property(
370        "message".to_string(),
371        Property::read_only(JSValue::from_string(
372            "The operation was aborted.".to_string(),

375    error.define_property(
376        "code".to_string(),
377        Property::read_only(JSValue::from_number(20.0)),
378    );
379    JSValue::from_object(Rc::new(RefCell::new(error)))

382/// Builds an `abort` Event object carrying the abort reason as `detail`.
383fn make_abort_event(reason: &JSValue) -> JSValue {
384    let mut event = JSObject::new();
385    event.define_property(
386        "type".to_string(),
387        Property::read_only(JSValue::from_string("abort".to_string())),

391        Property::read_only(JSValue::from_bool(false)),

393    event.define_property(

395        Property::read_only(JSValue::from_bool(false)),
396    );
397    event.set("detail".to_string(), reason.clone());
398    JSValue::from_object(Rc::new(RefCell::new(event)))
399}

401/// Rejects all fetch capabilities whose request registered `signal`, removing
402/// them from the pending queue so the network layer skips them.
403///
404/// Both the queue and the signal registry live on the JS thread, so the match
405/// is done by request id here.
406fn abort_registered_fetches(vm: &mut VM, signal: &Rc<RefCell<JSObject>>) {
407    let reason = signal.borrow().get(ABORT_REASON);
408    let rejections: Vec<JSValue> = with_host_mut(vm, |host| {
409        let mut rejections = Vec::new();
410        let mut remaining = Vec::new();
411        let requests = std::mem::take(&mut host.fetch_requests);
412        for request in requests {
413            let registered = host
414                .fetch_signals
415                .get(&request.id)
416                .is_some_and(|candidate| Rc::ptr_eq(candidate, signal));
417            if registered {
418                host.fetch_signals.remove(&request.id);
419                if let Some(capability) = host.fetch_capabilities.remove(&request.id) {
420                    rejections.push(capability.reject);
421                }
422            } else {
423                remaining.push(request);
424            }
425        }
426        host.fetch_requests = remaining;
427        rejections
428    })
429    .unwrap_or_default();
430    for reject in rejections {
431        if is_callable(&reject) {
432            let _ = vm.call(reject, JSValue::undefined(), vec![reason.clone()]);
433        }
434    }
435}

437/// `AbortSignal.abort(reason?)` — returns an already-aborted signal.
438fn signal_static_abort(vm: &mut VM, args: Vec<JSValue>) -> JSResult<JSValue> {
439    let reason = match args.get(1) {
440        Some(value) if !value.is_undefined() => value.clone(),
441        _ => abort_error_value(),
442    };
443    Ok(JSValue::from_object(make_signal(vm, true, reason)))
444}

446/// `AbortSignal.timeout(ms)` — returns a signal that aborts with a
447/// `TimeoutError` after the delay (implemented as a plain timer; the timer
448/// callback aborts the signal object held in a closure-captured slot).
449fn signal_static_timeout(vm: &mut VM, args: Vec<JSValue>) -> JSResult<JSValue> {
450    let milliseconds = args.get(1).map(JSValue::to_number).unwrap_or(0.0);
451    let signal = make_signal(vm, false, JSValue::undefined());
452    let signal_value = JSValue::from_object(Rc::clone(&signal));
453    let callback = JSValue::from_native_function(timeout_abort_callback);
454    let bound =
455        JSValue::from_bound_function(BoundFunctionData::new(callback, signal_value, Vec::new()));
456    let set_timeout = vm.global_object.borrow().get("setTimeout");
457    let _ = vm.call(
458        set_timeout,
459        JSValue::undefined(),
460        vec![bound, JSValue::from_number(milliseconds)],
461    );
462    Ok(JSValue::from_object(signal))
463}

465/// Timer callback for `AbortSignal.timeout`: aborts the bound signal with a
466/// `TimeoutError` reason.
467fn timeout_abort_callback(vm: &mut VM, args: Vec<JSValue>) -> JSResult<JSValue> {
468    let signal = match args.first().and_then(JSValue::as_object) {
469        Some(signal) if signal.borrow().get(ABORT_SIGNAL_MARKER).as_boolean() == Some(true) => {
470            signal
471        }
472        _ => return Ok(JSValue::undefined()),
473    };
474    let mut error = JSObject::new();
475    error.define_property(
476        "name".to_string(),
477        Property::read_only(JSValue::from_string("TimeoutError".to_string())),
478    );
479    error.define_property(
480        "message".to_string(),
481        Property::read_only(JSValue::from_string("The operation timed out.".to_string())),
482    );
483    let reason = JSValue::from_object(Rc::new(RefCell::new(error)));
484    abort_signal_now(vm, &signal, reason);
485    Ok(JSValue::undefined())
486}

488/// `AbortSignal.any(signals)` — a signal that aborts when any input aborts.
489/// Inputs are inspected for their current state; live propagation across
490/// independent signals is registered through their listener lists.
491fn signal_static_any(vm: &mut VM, args: Vec<JSValue>) -> JSResult<JSValue> {
492    let inputs = args.get(1).and_then(JSValue::as_object);
493    let length = inputs
494        .as_ref()
495        .map(|object| object.borrow().get("length").to_number().max(0.0) as usize)
496        .unwrap_or(0);
497    let combined = make_signal(vm, false, JSValue::undefined());
498    for index in 0..length {
499        let value = inputs
500            .as_ref()
501            .map(|object| object.borrow().get(&index.to_string()))
502            .unwrap_or(JSValue::undefined());
503        let Ok(input) = signal_receiver(&[value]) else {
504            continue;
505        };
506        let aborted = input.borrow().get(ABORTED).as_boolean() == Some(true);
507        if aborted {
508            let reason = input.borrow().get(ABORT_REASON);
509            combined
510                .borrow_mut()
511                .set(ABORTED.to_string(), JSValue::from_bool(true));
512            combined.borrow_mut().set(ABORT_REASON.to_string(), reason);
513            return Ok(JSValue::from_object(combined));
514        }
515        // Register a forwarding listener on the input signal.
516        let forward = JSValue::from_native_function(any_forward_callback);
517        let bound = JSValue::from_bound_function(BoundFunctionData::new(
518            forward,
519            JSValue::from_object(Rc::clone(&combined)),
520            Vec::new(),
521        ));
522        let _ = vm.call(
523            JSValue::from_native_function(signal_add_event_listener),
524            JSValue::from_object(input),
525            vec![
526                JSValue::undefined(),
527                JSValue::from_string("abort".to_string()),
528                bound,
529            ],
530        );
531    }
532    Ok(JSValue::from_object(combined))
533}

535/// Forwarding listener for `AbortSignal.any`: copies the abort state onto the
536/// bound combined signal and dispatches its listeners.
537fn any_forward_callback(vm: &mut VM, args: Vec<JSValue>) -> JSResult<JSValue> {
538    let combined = match args.first().and_then(JSValue::as_object) {
539        Some(combined) => combined,
540        None => return Ok(JSValue::undefined()),
541    };
542    let event = args.get(2).cloned().unwrap_or(JSValue::undefined());
543    let reason = event
544        .as_object()
545        .map(|object| object.borrow().get("detail"))
546        .filter(|value| value.kind() != JsValueKind::Undefined)
547        .unwrap_or_else(abort_error_value);
548    abort_signal_now(vm, &combined, reason);
549    Ok(JSValue::undefined())
550}

552/// Returns `true` when `signal` is an aborted `AbortSignal`.
553pub(crate) fn signal_is_aborted(signal: &JSValue) -> bool {
554    signal
555        .as_object()
556        .map(|object| {
557            object.borrow().get(ABORT_SIGNAL_MARKER).as_boolean() == Some(true)
558                && object.borrow().get(ABORTED).as_boolean() == Some(true)
559        })
560        .unwrap_or(false)
561}

563/// The reason an aborted signal carries, for fetch rejections.
564pub(crate) fn signal_abort_reason(signal: &JSValue) -> JSValue {
565    signal
566        .as_object()
567        .map(|object| object.borrow().get(ABORT_REASON))
568        .unwrap_or_else(abort_error_value)
569}

571/// Marks `signal` as aborted with `reason`, dispatches its `abort` listeners
572/// synchronously, and rejects pending fetches. Shared by `controller.abort()`
573/// and timers.
574fn abort_signal_now(vm: &mut VM, signal: &Rc<RefCell<JSObject>>, reason: JSValue) {
575    if signal.borrow().get(ABORTED).as_boolean() == Some(true) {
576        return;
577    }
578    signal
579        .borrow_mut()
580        .set(ABORTED.to_string(), JSValue::from_bool(true));
581    signal
582        .borrow_mut()
583        .set(ABORT_REASON.to_string(), reason.clone());
584    let listeners = signal.borrow().get(ABORT_LISTENERS);
585    if let Some(listeners) = listeners.as_object() {
586        let keys = listeners.borrow().keys();
587        for key in keys {
588            let listener = listeners.borrow().get(&key);
589            if is_callable(&listener) {
590                let event = make_abort_event(&reason);
591                let _ = vm.call(listener, JSValue::undefined(), vec![event]);
592            }
593        }
594    }
595    abort_registered_fetches(vm, signal);
596}
