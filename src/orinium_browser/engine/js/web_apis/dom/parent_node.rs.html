1//! DOM Standard `ParentNode` mixin.
2//!
3//! Collects the `append()`, `prepend()`, and `replaceChildren()` API shared by
4//! `Element`, `Document`, `DocumentFragment` and (through `DocumentFragment`
5//! inheritance) `ShadowRoot`.
6//!
7//! `Node` (`appendChild`, `removeChild`, `firstChild`, …) is a separate
8//! responsibility and intentionally lives elsewhere.
9
10use crate::engine::html::HtmlNodeType;
11use crate::engine::js::common::{UNDEFINED, dom_node, mark_dom_dirty, with_host, with_host_mut};
12use crate::engine::js::web_apis::dom::custom_elements::fire_disconnected_callback;
13use crate::engine::js::web_apis::dom::document::{create_text_node, expose_node};
14use crate::engine::js::web_apis::dom::element::{
15    append_child, insert_before, read_only_accessor_property,
16};
17use crate::engine::tree::TreeNode;
18use pixi_byte::value::jsobject::JSObject;
19use pixi_byte::vm::VM;
20use pixi_byte::{JSResult, JSValue};
21use std::rc::Rc;

23/// `parentNode.append(...nodes)` — appends nodes or strings at the end.
24pub(crate) fn parent_node_append(vm: &mut VM, args: Vec<JSValue>) -> JSResult<JSValue> {
25    let receiver = args.first().cloned().unwrap_or(JSValue::undefined());
26    for value in args.into_iter().skip(1) {
27        let child = if dom_node(vm, &value).is_some() {
28            value
29        } else {
30            create_text_node(vm, vec![JSValue::undefined(), value])?
31        };
32        let _ = append_child(vm, vec![receiver.clone(), child]);
33    }
34    Ok(JSValue::undefined())
35}
36
37/// `parentNode.prepend(...nodes)` — inserts nodes or strings before the first
38/// child, in argument order.
39pub(crate) fn parent_node_prepend(vm: &mut VM, args: Vec<JSValue>) -> JSResult<JSValue> {
40    let receiver = args.first().cloned().unwrap_or(JSValue::undefined());
41    let Some(parent) = dom_node(vm, &receiver) else {
42        return Ok(JSValue::undefined());
43    };
44    let first_child = parent.borrow().children().first().cloned();
45    let reference = first_child
46        .as_ref()
47        .and_then(|fc| expose_node(vm, Rc::clone(fc)));
48    for value in args.into_iter().skip(1) {
49        let child = if dom_node(vm, &value).is_some() {
50            value
51        } else {
52            create_text_node(vm, vec![JSValue::undefined(), value])?
53        };
54        let mut call_args = vec![receiver.clone(), child];
55        if let Some(ref_js) = &reference {
56            call_args.push(ref_js.clone());
57        }
58        let _ = insert_before(vm, call_args);
59    }
60    Ok(JSValue::undefined())
61}

63/// `parentNode.replaceChildren(...nodes)` — replaces all children with the
64/// given nodes or strings, in argument order.
65pub(crate) fn parent_node_replace_children(vm: &mut VM, args: Vec<JSValue>) -> JSResult<JSValue> {
66    let receiver = args.first().cloned().unwrap_or(JSValue::undefined());
67    let Some(parent) = dom_node(vm, &receiver) else {
68        return Ok(JSValue::undefined());
69    };
70    // Remove all existing children, firing disconnected callbacks.
71    let children: Vec<_> = parent.borrow().children().to_vec();
72    for child in children {
73        if TreeNode::remove_child(&parent, &child).is_some() {
74            mark_dom_dirty(vm);
75            if let Some(dom_id) = with_host(vm, |host| host.dom_id_for_node(&child)).flatten() {
76                fire_disconnected_callback(vm, dom_id);
77                let _ = with_host_mut(vm, |host| {
78                    host.detached_nodes.insert(dom_id, Rc::clone(&child));
79                });
80            }
81        }
82    }
83    // Append the new nodes (strings become text nodes), in argument order.
84    for value in args.into_iter().skip(1) {
85        let child = if dom_node(vm, &value).is_some() {
86            value
87        } else {
88            create_text_node(vm, vec![JSValue::undefined(), value])?
89        };
90        let _ = append_child(vm, vec![receiver.clone(), child]);
91    }
92    Ok(JSValue::undefined())
93}

95/// `parentNode.firstElementChild` — first Element child, or null.
96pub(crate) fn get_first_element_child(vm: &mut VM, args: Vec<JSValue>) -> JSResult<JSValue> {
97    let Some(node) = dom_node(vm, args.first().unwrap_or(&UNDEFINED)) else {
98        return Ok(JSValue::null());
99    };
100    let child = node
101        .borrow()
102        .children()
103        .iter()
104        .find(|child| matches!(child.borrow().value, HtmlNodeType::Element { .. }))
105        .cloned();
106    Ok(child
107        .and_then(|child| expose_node(vm, child))
108        .unwrap_or(JSValue::null()))
109}

111/// `parentNode.lastElementChild` — last Element child, or null.
112pub(crate) fn get_last_element_child(vm: &mut VM, args: Vec<JSValue>) -> JSResult<JSValue> {
113    let Some(node) = dom_node(vm, args.first().unwrap_or(&UNDEFINED)) else {
114        return Ok(JSValue::null());
115    };
116    let child = node
117        .borrow()
118        .children()
119        .iter()
120        .rev()
121        .find(|child| matches!(child.borrow().value, HtmlNodeType::Element { .. }))
122        .cloned();
123    Ok(child
124        .and_then(|child| expose_node(vm, child))
125        .unwrap_or(JSValue::null()))
126}

128/// `parentNode.childElementCount` — number of Element children.
129pub(crate) fn get_child_element_count(vm: &mut VM, args: Vec<JSValue>) -> JSResult<JSValue> {
130    let Some(node) = dom_node(vm, args.first().unwrap_or(&UNDEFINED)) else {
131        return Ok(JSValue::from_number(0.0));
132    };
133    let count = node
134        .borrow()
135        .children()
136        .iter()
137        .filter(|child| matches!(child.borrow().value, HtmlNodeType::Element { .. }))
138        .count();
139    Ok(JSValue::from_number(count as f64))
140}

142/// Registers the `ParentNode` mixin methods on a given interface prototype.
143///
144/// `children` is intentionally not installed here because it is already wired
145/// directly on `Element`, `DocumentFragment`, and `ShadowRoot` via
146/// `get_element_children`; installing it again from the mixin would create
147/// duplicate accessor definitions.
148pub(crate) fn install_parent_node_methods(prototype: &mut JSObject) {
149    prototype.set(
150        "append".to_string(),
151        JSValue::from_native_function(parent_node_append),
152    );
153    prototype.set(
154        "prepend".to_string(),
155        JSValue::from_native_function(parent_node_prepend),
156    );
157    prototype.set(
158        "replaceChildren".to_string(),
159        JSValue::from_native_function(parent_node_replace_children),
160    );
161    prototype.define_property(
162        "firstElementChild".to_string(),
163        read_only_accessor_property(get_first_element_child),
164    );
165    prototype.define_property(
166        "lastElementChild".to_string(),
167        read_only_accessor_property(get_last_element_child),
168    );
169    prototype.define_property(
170        "childElementCount".to_string(),
171        read_only_accessor_property(get_child_element_count),
172    );
