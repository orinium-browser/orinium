7/// Static constants for DOMException error codes.
8const INDEX_SIZE_ERR: f64 = 1.0;
9const DOMSTRING_SIZE_ERR: f64 = 2.0;
10const HIERARCHY_REQUEST_ERR: f64 = 3.0;
11const WRONG_DOCUMENT_ERR: f64 = 4.0;
12const INVALID_CHARACTER_ERR: f64 = 5.0;
13const NOT_FOUND_ERR: f64 = 8.0;
14const NOT_SUPPORTED_ERR: f64 = 9.0;
15const INVALID_STATE_ERR: f64 = 11.0;
16const SYNTAX_ERR: f64 = 12.0;
17const INVALID_MODIFICATION_ERR: f64 = 13.0;
18const NAMESPACE_ERR: f64 = 14.0;
19const INVALID_ACCESS_ERR: f64 = 15.0;
20const TYPE_MISMATCH_ERR: f64 = 17.0;

22/// Maps a DOMException name string to its numeric code.
23fn name_to_code(name: &str) -> f64 {
24    match name {
25        "IndexSizeError" => INDEX_SIZE_ERR,
26        "DOMStringSizeError" => DOMSTRING_SIZE_ERR,
27        "HierarchyRequestError" => HIERARCHY_REQUEST_ERR,
28        "WrongDocumentError" => WRONG_DOCUMENT_ERR,
29        "InvalidCharacterError" => INVALID_CHARACTER_ERR,
30        "NotFoundError" => NOT_FOUND_ERR,
31        "NotSupportedError" => NOT_SUPPORTED_ERR,
32        "InvalidStateError" => INVALID_STATE_ERR,
33        "SyntaxError" => SYNTAX_ERR,
34        "InvalidModificationError" => INVALID_MODIFICATION_ERR,
35        "NamespaceError" => NAMESPACE_ERR,
36        "TypeMismatchError" => TYPE_MISMATCH_ERR,
37        _ => 0.0,
38    }
39}

41/// Creates a DOMException constructor object and returns it.
42///
43/// The prototype chain is set up so that `DOMException.prototype` inherits
44/// from `Error.prototype`, enabling `instanceof Error` checks.
45pub(crate) fn make_dom_exception_constructor() -> Rc<RefCell<JSObject>> {
46    let mut prototype = JSObject::new();
47    prototype.set(
48        "toString".to_string(),
49        JSValue::from_native_function(dom_exception_to_string),
50    );
51    let prototype = Rc::new(RefCell::new(prototype));
52
53    let mut constructor = JSObject::new();
54    constructor.set(
55        "__construct__".to_string(),
56        JSValue::from_native_function(dom_exception_construct),
57    );
58    constructor.set(
59        "prototype".to_string(),
60        JSValue::from_object(Rc::clone(&prototype)),
61    );
62
63    // Static constants on the constructor itself
64    define_exception_constants(&mut constructor);
65
66    Rc::new(RefCell::new(constructor))
67}

69/// Defines the legacy `DOMException` numeric code constants on an object so
70/// they are reachable from any thrown exception instance.
71fn define_exception_constants(obj: &mut JSObject) {
72    for (name, value) in [
73        ("INDEX_SIZE_ERR", INDEX_SIZE_ERR),
74        ("DOMSTRING_SIZE_ERR", DOMSTRING_SIZE_ERR),
75        ("HIERARCHY_REQUEST_ERR", HIERARCHY_REQUEST_ERR),
76        ("WRONG_DOCUMENT_ERR", WRONG_DOCUMENT_ERR),
77        ("INVALID_CHARACTER_ERR", INVALID_CHARACTER_ERR),
78        ("NO_DATA_ALLOWED_ERR", 6.0),
79        ("NO_MODIFICATION_ALLOWED_ERR", 7.0),
80        ("NOT_FOUND_ERR", NOT_FOUND_ERR),
81        ("NOT_SUPPORTED_ERR", NOT_SUPPORTED_ERR),
82        ("INUSE_ATTRIBUTE_ERR", 10.0),
83        ("INVALID_STATE_ERR", INVALID_STATE_ERR),
84        ("SYNTAX_ERR", SYNTAX_ERR),
85        ("INVALID_MODIFICATION_ERR", INVALID_MODIFICATION_ERR),
86        ("NAMESPACE_ERR", NAMESPACE_ERR),
87        ("INVALID_ACCESS_ERR", INVALID_ACCESS_ERR),
88        ("VALIDATION_ERR", 16.0),
89        ("TYPE_MISMATCH_ERR", TYPE_MISMATCH_ERR),
90    ] {
91        obj.define_property(
92            name.to_string(),
93            Property::read_only(JSValue::from_number(value)),
94        );
95    }
96}

98/// `new DOMException(message?, name?)`
99fn dom_exception_construct(_vm: &mut VM, args: Vec<JSValue>) -> JSResult<JSValue> {
100    let this = args.first().cloned().unwrap_or(JSValue::undefined());
101    let Some(obj) = this.as_object() else {
102        return Ok(JSValue::undefined());
103    };
104
105    let message = args
106        .get(1)
107        .map(JSValue::to_console_string)
108        .unwrap_or_default();
109    let name = args
110        .get(2)
111        .map(JSValue::to_console_string)
112        .unwrap_or_else(|| "Error".to_string());
113
114    let code = name_to_code(&name);
115
116    let mut obj = obj.borrow_mut();
117    define_exception_constants(&mut obj);
118    obj.define_property(
119        "name".to_string(),
120        Property::read_only(JSValue::from_string(name)),
121    );
122    obj.define_property(
123        "message".to_string(),
124        Property::read_only(JSValue::from_string(message)),
125    );
126    obj.define_property(
127        "code".to_string(),
128        Property::read_only(JSValue::from_number(code)),
129    );
130
131    Ok(this)
132}

134/// `DOMException.prototype.toString()`
135fn dom_exception_to_string(_vm: &mut VM, args: Vec<JSValue>) -> JSResult<JSValue> {
136    let this = args.first().cloned().unwrap_or(JSValue::undefined());
137    let Some(obj) = this.as_object() else {
138        return Ok(JSValue::from_string("Error".to_string()));
139    };
140    let name = obj
141        .borrow()
142        .get("name")
143        .as_string_owned()
144        .unwrap_or_else(|| "Error".to_string());
145    let message = obj
146        .borrow()
147        .get("message")
148        .as_string_owned()
149        .unwrap_or_default();
150    if message.is_empty() {
151        Ok(JSValue::from_string(name))
152    } else {
153        Ok(JSValue::from_string(format!("{name}: {message}")))
154    }
155}

157/// Builds a DOMException error object and returns a `JSError::Thrown` for use
158/// in native function implementations that need to throw a DOMException.
159///
160/// This constructs the object directly (without calling the JS constructor) so
161/// it can be used from Rust code without a VM reference for the constructor
162/// lookup.
163pub(crate) fn throw_dom_exception(message: &str, name: &str) -> JSError {
164    let mut obj = JSObject::new();
165    define_exception_constants(&mut obj);
166    obj.define_property(
167        "name".to_string(),
168        Property::read_only(JSValue::from_string(name.to_string())),
169    );
170    obj.define_property(
171        "message".to_string(),
172        Property::read_only(JSValue::from_string(message.to_string())),
173    );
174    obj.define_property(
175        "code".to_string(),
176        Property::read_only(JSValue::from_number(name_to_code(name))),
177    );
178    JSError::Thrown(JSValue::from_object(Rc::new(RefCell::new(obj))))
179}
