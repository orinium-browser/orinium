615/// `instanceof ShadowRoot` hook: matches exposed shadow-root wrappers, which
616/// carry the `mode` property and `#shadow-root` node name.
617fn shadow_root_has_instance(_vm: &mut VM, args: Vec<JSValue>) -> JSResult<JSValue> {
618    let matches = args
619        .get(1)
620        .and_then(JSValue::as_object)

811fn create_element_ns(vm: &mut VM, args: Vec<JSValue>) -> JSResult<JSValue> {
812    let namespace = match args.get(1) {
813        Some(value) if value.is_string() => value.as_string().unwrap_or("").to_string(),
814        Some(value) if value.is_null() || value.is_undefined() => String::new(),
815        Some(value) => value.to_console_string(),
816        None => String::new(),
817    };
818    let Some(qualified_name) = args.get(2).and_then(JSValue::as_string) else {
819        return Err(throw_dom_exception(
820            "Must provide a qualified name",
821            "SyntaxError",
822        ));
823    };
824    // The qualified name is preserved exactly (case and prefix) so it can be
825    // reported through tagName/nodeName/localName/prefix per the DOM spec.
826    let qualified_name = qualified_name.trim().to_string();
827    let (prefix, local_name) = match validate_qualified_name(&qualified_name) {
828        Ok(parts) => parts,
829        Err(code) => {
830            return Err(throw_dom_exception(
831                "The qualified name provided is not a valid name",
832                exception_name_for_code(code),
833            ));
834        }
835    };
836    validate_namespace(&prefix, &local_name, &namespace).map_err(|code| {
837        throw_dom_exception(
838            "The qualified name and namespace provided are not valid",
839            exception_name_for_code(code),
840        )
841    })?;
842
843    let node = TreeNode::new(HtmlNodeType::Element {
844        tag_name: qualified_name,
845        attributes: Vec::new(),
846    });
847    adopt_owner_document(args.first().unwrap_or(&JSValue::undefined()), &node);
848    let value = expose_detached_node(vm, node).unwrap_or(JSValue::null());
849    if let Some(dom_id) = node_dom_id(&value) {
850        let _ = with_host_mut(vm, |host| {
851            host.namespaces.insert(dom_id, namespace);
852        });
853    }
854    Ok(value)
855}

1036/// Concatenates the text of a node and its descendants.
1037fn text_content_of(node: &NodeRef<HtmlNodeType>) -> String {
1038    let mut out = String::new();
1039    fn visit(node: &NodeRef<HtmlNodeType>, out: &mut String) {
1040        match &node.borrow().value {
1041            HtmlNodeType::Text(text) => out.push_str(text),
1042            HtmlNodeType::Comment(_) | HtmlNodeType::Doctype { .. } => {}
1043            HtmlNodeType::Element { .. }
1044            | HtmlNodeType::Document
1045            | HtmlNodeType::DocumentFragment => {
1046                for child in node.borrow().children() {
1047                    visit(child, out);
1048                }
1049            }
1050            _ => {}
1051        }
1052    }
1053    visit(node, &mut out);
1054    out
1055}

1161/// creates a new, empty document with its own tree, so nothing created through
1162/// it can reach the top-level document's nodes.
1163fn dom_implementation_create_document(vm: &mut VM, args: Vec<JSValue>) -> JSResult<JSValue> {
1164    let namespace = args
1165        .get(1)
1166        .filter(|value| value.is_string())
1167        .map(|value| value.as_string().unwrap_or("").to_string())
1168        .unwrap_or_default();

1170        .get(2)

1174    // The doctype is the third argument; `createDocumentType` leaves it without
1175    // an owner until a document adopts it.
1176    let doctype = dom_node(vm, args.get(3).unwrap_or(&JSValue::undefined()));
1177
1178    let node = TreeNode::new(HtmlNodeType::Document);
1179    let doc = expose_detached_node(vm, Rc::clone(&node)).unwrap_or(JSValue::null());
1180    let doc_dom_id = node_dom_id(&doc);
1181
1182    // A doctype passed in becomes the document's first child and, from that
1183    // moment on, reports this document as its owner.
1184    if let (Some(doctype), Some(doc_dom_id)) = (doctype, doc_dom_id) {
1185        doctype.borrow().set_owner_document(doc_dom_id);
1186        TreeNode::insert_child_at(&node, 0, doctype);
1187    }
1188
1189    if !qualified_name.is_empty() {
1190        let (prefix, local_name) = validate_qualified_name(&qualified_name).map_err(|code| {
1191            throw_dom_exception(
1192                "The qualified name provided is not a valid name",
1193                exception_name_for_code(code),
1194            )
1195        })?;
1196        validate_namespace(&prefix, &local_name, &namespace).map_err(|code| {
1197            throw_dom_exception(
1198                "The qualified name and namespace provided are not valid",
1199                exception_name_for_code(code),
1200            )
1201        })?;
1202        let element = TreeNode::new(HtmlNodeType::Element {
1203            tag_name: qualified_name,
1204            attributes: Vec::new(),
1205        });
1206        if let Some(doc_dom_id) = doc_dom_id {
1207            element.borrow().set_owner_document(doc_dom_id);
1208        }
1209        let value = expose_detached_node(vm, Rc::clone(&element)).unwrap_or(JSValue::null());
1210        if let Some(dom_id) = node_dom_id(&value) {
1211            let _ = with_host_mut(vm, |host| {
1212                host.namespaces.insert(dom_id, namespace);
1213            });
1214        }
1215        TreeNode::append_child(&node, element);
1216    }
1217    mark_dom_dirty(vm);
1218    Ok(doc)
1219}

1354fn set_document_cookie(vm: &mut VM, args: Vec<JSValue>) -> JSResult<JSValue> {
1355    let cookie = args

1358        .to_console_string();

1360        return Ok(JSValue::undefined());
1361    };

1363        return Ok(JSValue::undefined());
1364    };

1366    if name.is_empty() {
1367        return Ok(JSValue::undefined());

1371        let attribute = attribute.trim();

1373            || attribute
1374                .strip_prefix("Max-Age=")
1375                .and_then(|value| value.trim().parse::<i64>().ok())
1376                .is_some_and(|max_age| max_age <= 0)
1377    });
1378    let _ = with_host_mut(vm, |host| {
1379        if should_remove {
1380            host.document_cookies.remove(name);
1381        } else {
1382            host.document_cookies
1383                .insert(name.to_string(), value.trim().to_string());
1384        }
1385    });
1386    Ok(JSValue::undefined())

1397/// Resolves the exposed element whose layout rect contains (`x`, `y`),
1398/// preferring the innermost (last in document order) match like browsers do.
1399fn element_at_layout_point(vm: &VM, x: f64, y: f64) -> Option<NodeRef<HtmlNodeType>> {
1400    let dom_id = with_host(vm, |host| {
1401        let mut found: Option<u64> = None;
1402        for (&candidate_id, metrics) in &host.layout_metrics_by_dom_id {
1403            let contains = x >= metrics.rect_left
1404                && x < metrics.rect_left + metrics.rect_width
1405                && y >= metrics.rect_top
1406                && y < metrics.rect_top + metrics.rect_height;
1407            if contains && candidate_id > found.unwrap_or(0) {
1408                found = Some(candidate_id);
1409            }
1410        }
1411        found
1412    })
1413    .flatten()?;
1414    with_host(vm, |host| {
1415        host.refs.get(&dom_id).and_then(|weak| weak.upgrade())
1416    })
1417    .flatten()
1418}

1420/// `document.elementFromPoint(x, y)` — the topmost element whose layout rect
1421/// contains the viewport point, or `null`. Layout metrics are keyed by DOM id,
1422/// so this is only meaningful after a committed layout.
1423fn document_element_from_point(vm: &mut VM, args: Vec<JSValue>) -> JSResult<JSValue> {
1424    let x = args.get(1).map(JSValue::to_number).unwrap_or(f64::NAN);
1425    let y = args.get(2).map(JSValue::to_number).unwrap_or(f64::NAN);
1426    if !x.is_finite() || !y.is_finite() {
1427        return Err(throw_dom_exception(
1428            "elementFromPoint: coordinates must be finite numbers",
1429            "TypeError",
1430        ));
1431    }
1432    let found = element_at_layout_point(vm, x, y).and_then(|node| expose_node(vm, node));
1433    Ok(found.unwrap_or(JSValue::null()))
1434}

1436/// `document.elementsFromPoint(x, y)` — every element whose layout rect
1437/// contains the point, innermost first. The engine keeps one rect per element,
1438/// so the stack is the containing elements in reverse registration order.
1439fn document_elements_from_point(vm: &mut VM, args: Vec<JSValue>) -> JSResult<JSValue> {
1440    let x = args.get(1).map(JSValue::to_number).unwrap_or(f64::NAN);
1441    let y = args.get(2).map(JSValue::to_number).unwrap_or(f64::NAN);
1442    if !x.is_finite() || !y.is_finite() {
1443        return Err(throw_dom_exception(
1444            "elementsFromPoint: coordinates must be finite numbers",
1445            "TypeError",
1446        ));
1447    }
1448    let ids = with_host(vm, |host| {
1449        let mut hits: Vec<u64> = host
1450            .layout_metrics_by_dom_id
1451            .iter()
1452            .filter(|(_, metrics)| {
1453                x >= metrics.rect_left
1454                    && x < metrics.rect_left + metrics.rect_width
1455                    && y >= metrics.rect_top
1456                    && y < metrics.rect_top + metrics.rect_height
1457            })
1458            .map(|(&id, _)| id)
1459            .collect();
1460        hits.sort_unstable_by(|a, b| b.cmp(a));
1461        hits
1462    })
1463    .unwrap_or_default();
1464    let values = ids
1465        .into_iter()
1466        .filter_map(|id| {
1467            with_host(vm, |host| {
1468                host.refs.get(&id).and_then(|weak| weak.upgrade())
1469            })
1470            .flatten()
1471            .and_then(|node| expose_node(vm, node))
1472        })
1473        .collect();
1474    Ok(vm.array_from_values(values))
1475}

1570// Expose a DOM node as a JavaScript object.
1571pub(crate) fn expose_node(vm: &VM, node: NodeRef<HtmlNodeType>) -> Option<JSValue> {
1572    let node_kind = {
1573        let borrowed = node.borrow();
1574        match &borrowed.value {
1575            HtmlNodeType::Element { tag_name, .. } => NodeKind::Element {
1576                tag_name: tag_name.clone(),
1577                id: borrowed.value.get_attr("id").unwrap_or("").to_string(),
1578            },
1579            HtmlNodeType::Text(_) => NodeKind::Text,
1580            HtmlNodeType::Comment(_) => NodeKind::Comment,
1581            HtmlNodeType::ProcessingInstruction { .. } => NodeKind::ProcessingInstruction,
1582            HtmlNodeType::DocumentFragment => NodeKind::Fragment,
1583            HtmlNodeType::Doctype {
1584                name,
1585                public_id: _,
1586                system_id: _,
1587            } => NodeKind::Doctype { name: name.clone() },
1588            HtmlNodeType::Document => NodeKind::Document,
1589            _ => return None,
1590        }
1591    };
1592
1593    expose_node_inner(vm, node, node_kind)
1594}
1595
1596enum NodeKind {
1597    Element { tag_name: String, id: String },
1598    Text,
1599    Comment,
1600    ProcessingInstruction,
1601    Fragment,
1602    Doctype { name: Option<String> },
1603    Document,
1604}
1605
1606fn expose_node_inner(vm: &VM, node: NodeRef<HtmlNodeType>, kind: NodeKind) -> Option<JSValue> {
1607    let dom_id = with_host_mut(vm, |host| {
1608        if let Some(dom_id) = host.dom_id_for_node(&node) {
1609            return dom_id;
1610        }
1611        host.next_id += 1;
1612        let dom_id = host.next_id;
1613        host.refs.insert(dom_id, Rc::downgrade(&node));
1614        dom_id
1615    })?;
1616
1617    let obj = with_host_mut(vm, |host| {
1618        if let Some(existing) = host.objects.get(&dom_id) {
1619            return Rc::clone(existing);
1620        }
1621
1622        let is_element = matches!(kind, NodeKind::Element { .. });
1623
1624        let obj = match &kind {
1625            NodeKind::Element { tag_name, id } => make_element(
1626                tag_name.clone(),
1627                id.clone(),
1628                dom_id,
1629                Rc::clone(&host.element_prototype),
1630                Rc::clone(&host.element_constructor),
1631            ),
1632            NodeKind::Text => make_text_node(dom_id),
1633            NodeKind::Comment => make_comment_node(dom_id),
1634            NodeKind::ProcessingInstruction => make_processing_instruction_node(dom_id),
1635            NodeKind::Fragment => make_document_fragment(dom_id),
1636            NodeKind::Doctype { name } => make_doctype_node(dom_id, name.clone()),
1637            // The main document keeps its long-lived object; a document made by
1638            // `DOMImplementation.createDocument` gets its own interface, which
1639            // is what makes the two trees independent.
1640            NodeKind::Document => match host.document.as_ref().cloned() {
1641                Some(document) if host.main_document_dom_id == Some(dom_id) => document,
1642                _ => make_secondary_document_object(dom_id),
1643            },
1644        };
1645
1646        // Custom elements inherit the class prototype so their methods are
1647        // reachable from `this` inside lifecycle callbacks and event handlers.
1648        if let NodeKind::Element { tag_name, .. } = &kind
1649            && let Some(definition) = host.custom_elements.get(tag_name)
1650            && let Some(class_proto) = link_custom_element_prototype(&obj, definition)
1651        {
1652            obj.borrow_mut().set_prototype(Some(class_proto));
1653        }
1654
1655        // Non-element nodes are created without a prototype; wire them to
1656        // their interface prototype so they inherit node methods (getRootNode,
1657        // contains, dispatchEvent, …). Fragments chain the dedicated
1658        // DocumentFragment prototype; other nodes chain Node.prototype.
1659        if !is_element {
1660            let proto = match &kind {
1661                NodeKind::Fragment => host.fragment_prototype.clone(),
1662                _ => host.node_prototype.clone(),
1663            };
1664            if let Some(proto) = proto {
1665                obj.borrow_mut().set_prototype(Some(proto));
1666            }
1667        }
1668
1669        host.objects.insert(dom_id, Rc::clone(&obj));
1670        obj
1671    })?;
1672
1673    Some(JSValue::from_object(obj))
1674}

1846fn document_write(vm: &mut VM, args: Vec<JSValue>) -> JSResult<JSValue> {
1847    let text = args
1848        .get(1)
1849        .map(JSValue::to_console_string)
1850        .unwrap_or_default();
1851    if text.is_empty() {
1852        return Ok(JSValue::undefined());
1853    }
1854
1855    // Reuse existing HTML parser
1856    let parsed = HtmlParser::new(&text).parse();
1857
1858    // Find insertion target: prefer existing <body>
1859    let target = with_host(vm, |host| host.dom.query_selector("body")).flatten();
1860
1861    let Some(target) = target else {
1862        // No <body> yet — nothing to insert into
1863        return Ok(JSValue::undefined());
1864    };
1865
1866    // Extract children from the parsed body, not the root.
1867    // Parser::new(text).parse() produces Document -> html -> body -> content,
1868    // so we must drill into the parsed body to avoid inserting a nested <html>.
1869    let source = parsed
1870        .query_selector("body")
1871        .unwrap_or_else(|| Rc::clone(&parsed.root));
1872
1873    let children: Vec<_> = source.borrow().children().to_vec();
1874    for child in children {
1875        TreeNode::append_child(&target, Rc::clone(&child));
1876        mark_dom_dirty(vm);
1877    }
1878
1879    Ok(JSValue::undefined())
1880}

1882fn document_writeln(vm: &mut VM, args: Vec<JSValue>) -> JSResult<JSValue> {

1886        .unwrap_or_default();
1887    let mut new_args = args;
1888    new_args.push(JSValue::from_string(format!("{text}\n")));

2313/// - If a colon is present, there must be exactly one, and the prefix must
2314///   follow the same rules as the local part.
2315fn is_valid_element_name(name: &str) -> bool {
2316    if name.is_empty() {
2317        return false;
2318    }
2319    let mut parts = name.split(':');
2320    let local = parts.next().unwrap_or("");
2321    // At most one colon (prefix:local)
2322    if parts.next().is_some() && parts.next().is_some() {
2323        return false;
2324    }
2325    // Validate the local part
2326    local_part_valid(local)
2327}
2328
2329fn local_part_valid(name: &str) -> bool {
2330    let mut chars = name.chars();
2331    let Some(first) = chars.next() else {
2332        return false;
2333    };
2334    if !first.is_ascii_alphabetic() {
2335        return false;
2336    }
2337    chars.all(|c| c.is_ascii_alphanumeric() || matches!(c, '_' | '-' | '.'))
2338}

2340/// Validates a qualified name per the XML `QName` production and extracts its
2341/// prefix (if any) and local name. A single `prefix:local` split is allowed.
2342///
2343/// Returns `Err(code)` where `code` is the legacy DOMException code to throw:
2344/// `5` (INVALID_CHARACTER_ERR) for a malformed name, `14` (NAMESPACE_ERR) when
2345/// the colon is present but the prefix is empty.
2346fn validate_qualified_name(name: &str) -> Result<(Option<String>, String), f64> {
2347    if name.is_empty() {
2348        return Err(5.0);
2349    }
2350    let mut parts = name.split(':');
2351    let first = parts.next().unwrap_or("");
2352    let second = parts.next();
2353    let third = parts.next();
2354
2355    match (first, second, third) {
2356        // No colon: plain local name.
2357        (first, None, None) => {
2358            if local_part_valid(first) {
2359                Ok((None, first.to_string()))
2360            } else {
2361                Err(5.0)
2362            }
2363        }
2364        // One colon: prefix:local. An empty prefix is a namespace error.
2365        (prefix, Some(local), None) => {
2366            if prefix.is_empty() {
2367                return Err(14.0);
2368            }
2369            if !local_part_valid(prefix) || !local_part_valid(local) {
2370                return Err(5.0);
2371            }
2372            Ok((Some(prefix.to_string()), local.to_string()))
2373        }
2374        // More than one colon.
2375        _ => Err(5.0),
2376    }
2377}

2379/// Applies the DOM namespace rules for `createElementNS`/`createAttributeNS`.
2380///
2381/// Returns `Err(code)` with `14` (NAMESPACE_ERR) on violation.
2382fn validate_namespace(
2383    prefix: &Option<String>,
2384    _local_name: &str,
2385    namespace: &str,
2386) -> Result<(), f64> {
2387    let namespace_is_null_or_empty = namespace.is_empty();
2388    match prefix {
2389        Some(prefix) if namespace_is_null_or_empty => Err(14.0),
2390        Some(prefix) if prefix == "xml" && namespace != XML_NAMESPACE => Err(14.0),
2391        Some(prefix) if prefix == "xmlns" && namespace != XMLNS_NAMESPACE => Err(14.0),
2392        Some(prefix) if namespace == XMLNS_NAMESPACE && prefix != "xmlns" => Err(14.0),
2393        Some(_) => Ok(()),
2394        None if namespace == XMLNS_NAMESPACE => Err(14.0),
2395        None => Ok(()),
2396    }
2397}
