744pub(crate) fn add_document_event_listener(vm: &mut VM, args: Vec<JSValue>) -> JSResult<JSValue> {
745    let Some(event_type) = args.get(1).and_then(JSValue::as_string_owned) else {
746        return Ok(JSValue::undefined());
747    };
748    let Some(listener) = args.get(2).filter(|value| is_callable(value)).cloned() else {
749        return Ok(JSValue::undefined());
750    };
751
752    let options = args.get(3).and_then(JSValue::as_object);
753    let once = options
754        .as_ref()
755        .map(|object| object.borrow().get("once").to_boolean())
756        .unwrap_or(false);
757    let signal = options.as_ref().map(|object| object.borrow().get("signal"));
758
759    let signal_aborted = signal
760        .as_ref()
761        .map(|signal| signal_is_aborted(signal))
762        .unwrap_or(false);
763
764    let _ = with_host_mut(vm, |host| {
765        if signal_aborted {
766            return;
767        }
768        let listeners = host
769            .document_event_listeners
770            .entry(event_type.clone())
771            .or_default();
772        if !listeners
773            .iter()
774            .any(|candidate| candidate.listener.strict_equals(&listener))
775        {
776            listeners.push(DocumentEventListener {
777                listener: listener.clone(),
778                once,
779            });
780        }
781    });
782
783    // A listener bound to an already-aborted signal is never registered.
784    if signal_aborted {
785        return Ok(JSValue::undefined());
786    }
787
788    // When the signal aborts, drop this listener so it is never invoked.
789    if let Some(signal_rc) = signal.as_ref().and_then(JSValue::as_object) {
790        let handler = JSValue::from_bound_function(BoundFunctionData::new(
791            JSValue::from_native_function(remove_document_event_listener),
792            JSValue::undefined(),
793            vec![JSValue::from_string(event_type.clone()), listener.clone()],
794        ));
795        let add_listener = signal_rc.borrow().get("addEventListener");
796        if is_callable(&add_listener) {
797            let _ = vm.call(
798                add_listener,
799                JSValue::from_object(Rc::clone(&signal_rc)),
800                vec![JSValue::from_string("abort".to_string()), handler],
801            );
802        }
803    }
804
805    Ok(JSValue::undefined())
806}

1074fn create_element_ns(vm: &mut VM, args: Vec<JSValue>) -> JSResult<JSValue> {
1075    let namespace = match args.get(1) {
1076        Some(value) if value.is_string() => value.as_string().unwrap_or("").to_string(),
1077        Some(value) if value.is_null() || value.is_undefined() => String::new(),
1078        Some(value) => value.to_console_string(),
1079        None => String::new(),
1080    };
1081    let Some(qualified_name) = args.get(2).and_then(JSValue::as_string) else {
1082        return Err(throw_dom_exception(
1083            "Must provide a qualified name",
1084            "SyntaxError",
1085        ));
1086    };
1087    // The qualified name is preserved exactly (case and prefix) so it can be
1088    // reported through tagName/nodeName/localName/prefix per the DOM spec.
1089    let qualified_name = qualified_name.trim().to_string();
1090    let (prefix, local_name) = match validate_qualified_name(&qualified_name) {
1091        Ok(parts) => parts,
1092        Err(code) => {
1093            return Err(throw_dom_exception(
1094                "The qualified name provided is not a valid name",
1095                exception_name_for_code(code),
1096            ));
1097        }
1098    };
1099    validate_namespace(&prefix, &local_name, &namespace).map_err(|code| {
1100        throw_dom_exception(
1101            "The qualified name and namespace provided are not valid",
1102            exception_name_for_code(code),
1103        )
1104    })?;
1105
1106    let node = TreeNode::new(HtmlNodeType::Element {
1107        tag_name: qualified_name,
1108        attributes: Vec::new(),
1109    });
1110    adopt_owner_document(args.first().unwrap_or(&JSValue::undefined()), &node);
1111    let value = expose_detached_node(vm, node).unwrap_or(JSValue::null());
1112    if let Some(dom_id) = node_dom_id(&value) {
1113        let _ = with_host_mut(vm, |host| {
1114            host.namespaces.insert(dom_id, namespace);
1115        });
1116    }
1117    Ok(value)
1118}

1299/// Concatenates the text of a node and its descendants.
1300fn text_content_of(node: &NodeRef<HtmlNodeType>) -> String {
1301    let mut out = String::new();
1302    fn visit(node: &NodeRef<HtmlNodeType>, out: &mut String) {
1303        match &node.borrow().value {
1304            HtmlNodeType::Text(text) => out.push_str(text),
1305            HtmlNodeType::Comment(_) | HtmlNodeType::Doctype { .. } => {}
1306            HtmlNodeType::Element { .. }
1307            | HtmlNodeType::Document
1308            | HtmlNodeType::DocumentFragment => {
1309                for child in node.borrow().children() {
1310                    visit(child, out);
1311                }
1312            }
1313            _ => {}
1314        }
1315    }
1316    visit(node, &mut out);
1317    out
1318}

1423/// `document.implementation.createDocument(namespace, qualifiedName, doctype)`:
1424/// creates a new, empty document with its own tree, so nothing created through
1425/// it can reach the top-level document's nodes.
1426fn dom_implementation_create_document(vm: &mut VM, args: Vec<JSValue>) -> JSResult<JSValue> {
1427    let namespace = args
1428        .get(1)
1429        .filter(|value| value.is_string())
1430        .map(|value| value.as_string().unwrap_or("").to_string())
1431        .unwrap_or_default();
1432    let qualified_name = args
1433        .get(2)
1434        .and_then(JSValue::as_string)
1435        .map(|name| name.trim().to_string())
1436        .unwrap_or_default();
1437    // The doctype is the third argument; `createDocumentType` leaves it without
1438    // an owner until a document adopts it.
1439    let doctype = dom_node(vm, args.get(3).unwrap_or(&JSValue::undefined()));
1440
1441    let node = TreeNode::new(HtmlNodeType::Document);
1442    let doc = expose_detached_node(vm, Rc::clone(&node)).unwrap_or(JSValue::null());
1443    let doc_dom_id = node_dom_id(&doc);
1444
1445    // A doctype passed in becomes the document's first child and, from that
1446    // moment on, reports this document as its owner.
1447    if let (Some(doctype), Some(doc_dom_id)) = (doctype, doc_dom_id) {
1448        doctype.borrow().set_owner_document(doc_dom_id);
1449        TreeNode::insert_child_at(&node, 0, doctype);
1450    }
1451
1452    if !qualified_name.is_empty() {
1453        let (prefix, local_name) = validate_qualified_name(&qualified_name).map_err(|code| {
1454            throw_dom_exception(
1455                "The qualified name provided is not a valid name",
1456                exception_name_for_code(code),
1457            )
1458        })?;
1459        validate_namespace(&prefix, &local_name, &namespace).map_err(|code| {
1460            throw_dom_exception(
1461                "The qualified name and namespace provided are not valid",
1462                exception_name_for_code(code),
1463            )
1464        })?;
1465        let element = TreeNode::new(HtmlNodeType::Element {
1466            tag_name: qualified_name,
1467            attributes: Vec::new(),
1468        });
1469        if let Some(doc_dom_id) = doc_dom_id {
1470            element.borrow().set_owner_document(doc_dom_id);
1471        }
1472        let value = expose_detached_node(vm, Rc::clone(&element)).unwrap_or(JSValue::null());
1473        if let Some(dom_id) = node_dom_id(&value) {
1474            let _ = with_host_mut(vm, |host| {
1475                host.namespaces.insert(dom_id, namespace);
1476            });
1477        }
1478        TreeNode::append_child(&node, element);
1479    }
1480    mark_dom_dirty(vm);
1481    Ok(doc)
1482}

1660/// Resolves the exposed element whose layout rect contains (`x`, `y`),
1661/// preferring the innermost (last in document order) match like browsers do.
1662fn element_at_layout_point(vm: &VM, x: f64, y: f64) -> Option<NodeRef<HtmlNodeType>> {
1663    let dom_id = with_host(vm, |host| {
1664        let mut found: Option<u64> = None;
1665        for (&candidate_id, metrics) in &host.layout_metrics_by_dom_id {
1666            let contains = x >= metrics.rect_left
1667                && x < metrics.rect_left + metrics.rect_width
1668                && y >= metrics.rect_top
1669                && y < metrics.rect_top + metrics.rect_height;
1670            if contains && candidate_id > found.unwrap_or(0) {
1671                found = Some(candidate_id);
1672            }
1673        }
1674        found
1675    })
1676    .flatten()?;
1677    with_host(vm, |host| {
1678        host.refs.get(&dom_id).and_then(|weak| weak.upgrade())
1679    })
1680    .flatten()
1681}

1685/// so this is only meaningful after a committed layout.
1686fn document_element_from_point(vm: &mut VM, args: Vec<JSValue>) -> JSResult<JSValue> {
1687    let x = args.get(1).map(JSValue::to_number).unwrap_or(f64::NAN);

1692            "TypeError",

1699/// `document.elementsFromPoint(x, y)` — every element whose layout rect
1700/// contains the point, innermost first. The engine keeps one rect per element,
1701/// so the stack is the containing elements in reverse registration order.
1702fn document_elements_from_point(vm: &mut VM, args: Vec<JSValue>) -> JSResult<JSValue> {
1703    let x = args.get(1).map(JSValue::to_number).unwrap_or(f64::NAN);
1704    let y = args.get(2).map(JSValue::to_number).unwrap_or(f64::NAN);
1705    if !x.is_finite() || !y.is_finite() {
1706        return Err(throw_dom_exception(
1707            "elementsFromPoint: coordinates must be finite numbers",
1708            "TypeError",
1709        ));
1710    }
1711    let ids = with_host(vm, |host| {
1712        let mut hits: Vec<u64> = host
1713            .layout_metrics_by_dom_id
1714            .iter()
1715            .filter(|(_, metrics)| {
1716                x >= metrics.rect_left
1717                    && x < metrics.rect_left + metrics.rect_width
1718                    && y >= metrics.rect_top
1719                    && y < metrics.rect_top + metrics.rect_height
1720            })
1721            .map(|(&id, _)| id)
1722            .collect();
1723        hits.sort_unstable_by(|a, b| b.cmp(a));
1724        hits
1725    })
1726    .unwrap_or_default();

1834pub(crate) fn expose_node(vm: &VM, node: NodeRef<HtmlNodeType>) -> Option<JSValue> {
1835    let node_kind = {
1836        let borrowed = node.borrow();
1837        match &borrowed.value {
1838            HtmlNodeType::Element { tag_name, .. } => NodeKind::Element {
1839                tag_name: tag_name.clone(),
1840                id: borrowed.value.get_attr("id").unwrap_or("").to_string(),
1841            },
1842            HtmlNodeType::Text(_) => NodeKind::Text,
1843            HtmlNodeType::Comment(_) => NodeKind::Comment,
1844            HtmlNodeType::ProcessingInstruction { .. } => NodeKind::ProcessingInstruction,
1845            HtmlNodeType::DocumentFragment => NodeKind::Fragment,
1846            HtmlNodeType::Doctype {
1847                name,
1848                public_id: _,
1849                system_id: _,
1850            } => NodeKind::Doctype { name: name.clone() },
1851            HtmlNodeType::Document => NodeKind::Document,
1852            _ => return None,
1853        }
1854    };
1855
1856    expose_node_inner(vm, node, node_kind)
1857}

1859enum NodeKind {
1860    Element { tag_name: String, id: String },
1861    Text,
1862    Comment,
1863    ProcessingInstruction,
1864    Fragment,
1865    Doctype { name: Option<String> },
1866    Document,
1867}

1869fn expose_node_inner(vm: &VM, node: NodeRef<HtmlNodeType>, kind: NodeKind) -> Option<JSValue> {
1870    let dom_id = with_host_mut(vm, |host| {
1871        if let Some(dom_id) = host.dom_id_for_node(&node) {
1872            return dom_id;
1873        }
1874        host.next_id += 1;
1875        let dom_id = host.next_id;
1876        host.refs.insert(dom_id, Rc::downgrade(&node));
1877        dom_id
1878    })?;
1879
1880    let obj = with_host_mut(vm, |host| {
1881        if let Some(existing) = host.objects.get(&dom_id) {
1882            return Rc::clone(existing);
1883        }
1884
1885        let is_element = matches!(kind, NodeKind::Element { .. });
1886
1887        let obj = match &kind {
1888            NodeKind::Element { tag_name, id } => {
1889                // `<slot>` elements use the HTMLSlotElement prototype so that
1890                // `slot instanceof HTMLSlotElement` holds and the `name`
1891                // accessor property is available.
1892                let (proto, ctor) = if tag_name.eq_ignore_ascii_case("slot") {
1893                    let proto = host
1894                        .slot_prototype
1895                        .clone()
1896                        .unwrap_or_else(|| Rc::clone(&host.element_prototype));
1897                    let ctor = vm
1898                        .global_object
1899                        .borrow()
1900                        .get("HTMLSlotElement")
1901                        .as_object()
1902                        .map(|o| Rc::clone(&o))
1903                        .unwrap_or_else(|| Rc::clone(&host.element_constructor));
1904                    (proto, ctor)
1905                } else {
1906                    (
1907                        Rc::clone(&host.element_prototype),
1908                        Rc::clone(&host.element_constructor),
1909                    )
1910                };
1911                make_element(tag_name.clone(), id.clone(), dom_id, proto, ctor)
1912            }
1913            NodeKind::Text => make_text_node(dom_id),
1914            NodeKind::Comment => make_comment_node(dom_id),
1915            NodeKind::ProcessingInstruction => make_processing_instruction_node(dom_id),
1916            NodeKind::Fragment => make_document_fragment(dom_id),
1917            NodeKind::Doctype { name } => make_doctype_node(dom_id, name.clone()),
1918            // The main document keeps its long-lived object; a document made by
1919            // `DOMImplementation.createDocument` gets its own interface, which
1920            // is what makes the two trees independent.
1921            NodeKind::Document => match host.document.as_ref().cloned() {
1922                Some(document) if host.main_document_dom_id == Some(dom_id) => document,
1923                _ => make_secondary_document_object(dom_id),
1924            },
1925        };
1926
1927        // Custom elements inherit the class prototype so their methods are
1928        // reachable from `this` inside lifecycle callbacks and event handlers.
1929        if let NodeKind::Element { tag_name, .. } = &kind
1930            && let Some(definition) = host.custom_elements.get(tag_name)
1931            && let Some(class_proto) = link_custom_element_prototype(&obj, definition)
1932        {
1933            obj.borrow_mut().set_prototype(Some(class_proto));
1934        }
1935
1936        // Non-element nodes are created without a prototype; wire them to
1937        // their interface prototype so they inherit node methods (getRootNode,
1938        // contains, dispatchEvent, …). Fragments chain the dedicated
1939        // DocumentFragment prototype; other nodes chain Node.prototype.
1940        if !is_element {
1941            let proto = match &kind {
1942                NodeKind::Fragment => host.fragment_prototype.clone(),
1943                _ => host.node_prototype.clone(),
1944            };
1945            if let Some(proto) = proto {
1946                obj.borrow_mut().set_prototype(Some(proto));
1947            }
1948        }
1949
1950        host.objects.insert(dom_id, Rc::clone(&obj));
1951        obj
1952    })?;
1953
1954    Some(JSValue::from_object(obj))
1955}

2080        }

2083    })
2084}

2189    let text = args
2190        .get(1)
2191        .map(JSValue::to_console_string)
2192        .unwrap_or_default();
2193    if text.is_empty() {
2194        return Ok(JSValue::undefined());
2195    }
2196

2198    let parsed = HtmlParser::new(&text).parse();
2199
2200    // Find insertion target: prefer existing <body>
2201    let target = with_host(vm, |host| host.dom.query_selector("body")).flatten();
2202
2203    let Some(target) = target else {
2204        // No <body> yet — nothing to insert into
2205        return Ok(JSValue::undefined());
2206    };

2209    // Parser::new(text).parse() produces Document -> html -> body -> content,
2210    // so we must drill into the parsed body to avoid inserting a nested <html>.
2211    let source = parsed
2212        .query_selector("body")
2213        .unwrap_or_else(|| Rc::clone(&parsed.root));
2214
2215    let children: Vec<_> = source.borrow().children().to_vec();
2216    for child in children {
2217        TreeNode::append_child(&target, Rc::clone(&child));
2218        mark_dom_dirty(vm);
2219    }
2220
2221    Ok(JSValue::undefined())
2222}

2698/// Validates a qualified element name per a simplified ASCII subset of the
2699/// XML `Name` production.
2700///
2701/// - Must not be empty.
2702/// - Local part: starts with `[a-zA-Z]`, followed by `[a-zA-Z0-9\-_\.]`.
2703/// - If a colon is present, there must be exactly one, and the prefix must
2704///   follow the same rules as the local part.
2705fn is_valid_element_name(name: &str) -> bool {
2706    if name.is_empty() {
2707        return false;
2708    }
2709    let mut parts = name.split(':');
2710    let local = parts.next().unwrap_or("");
2711    // At most one colon (prefix:local)
2712    if parts.next().is_some() && parts.next().is_some() {
2713        return false;
2714    }
2715    // Validate the local part
2716    local_part_valid(local)
2717}
2718
2719fn local_part_valid(name: &str) -> bool {
2720    let mut chars = name.chars();
2721    let Some(first) = chars.next() else {
2722        return false;
2723    };
2724    if !first.is_ascii_alphabetic() {
2725        return false;
2726    }
2727    chars.all(|c| c.is_ascii_alphanumeric() || matches!(c, '_' | '-' | '.'))
2728}
2729
2730/// Validates a qualified name per the XML `QName` production and extracts its
2731/// prefix (if any) and local name. A single `prefix:local` split is allowed.
2732///
2733/// Returns `Err(code)` where `code` is the legacy DOMException code to throw:
2734/// `5` (INVALID_CHARACTER_ERR) for a malformed name, `14` (NAMESPACE_ERR) when
2735/// the colon is present but the prefix is empty.
2736fn validate_qualified_name(name: &str) -> Result<(Option<String>, String), f64> {
2737    if name.is_empty() {
2738        return Err(5.0);
2739    }
2740    let mut parts = name.split(':');
2741    let first = parts.next().unwrap_or("");
2742    let second = parts.next();
2743    let third = parts.next();
2744
2745    match (first, second, third) {
2746        // No colon: plain local name.
2747        (first, None, None) => {
2748            if local_part_valid(first) {
2749                Ok((None, first.to_string()))
2750            } else {
2751                Err(5.0)
2752            }
2753        }
2754        // One colon: prefix:local. An empty prefix is a namespace error.
2755        (prefix, Some(local), None) => {
2756            if prefix.is_empty() {
2757                return Err(14.0);
2758            }
2759            if !local_part_valid(prefix) || !local_part_valid(local) {
2760                return Err(5.0);
2761            }
2762            Ok((Some(prefix.to_string()), local.to_string()))
2763        }
2764        // More than one colon.
2765        _ => Err(5.0),
2766    }
2767}

2769/// Applies the DOM namespace rules for `createElementNS`/`createAttributeNS`.
2770///
2771/// Returns `Err(code)` with `14` (NAMESPACE_ERR) on violation.
2772fn validate_namespace(
2773    prefix: &Option<String>,
2774    _local_name: &str,
2775    namespace: &str,
2776) -> Result<(), f64> {
2777    let namespace_is_null_or_empty = namespace.is_empty();
2778    match prefix {
2779        Some(prefix) if namespace_is_null_or_empty => Err(14.0),
2780        Some(prefix) if prefix == "xml" && namespace != XML_NAMESPACE => Err(14.0),
2781        Some(prefix) if prefix == "xmlns" && namespace != XMLNS_NAMESPACE => Err(14.0),
2782        Some(prefix) if namespace == XMLNS_NAMESPACE && prefix != "xmlns" => Err(14.0),
2783        Some(_) => Ok(()),
2784        None if namespace == XMLNS_NAMESPACE => Err(14.0),
2785        None => Ok(()),
2786    }
2787}

2813/// `instanceof HTMLSlotElement` hook: true for exposed `<slot>` wrappers.
2814fn slot_has_instance(vm: &mut VM, args: Vec<JSValue>) -> JSResult<JSValue> {
2815    let matches = args
2816        .get(1)
2817        .and_then(JSValue::as_object)
2818        .map(|object| {
2819            // Fast path: walk the prototype chain looking for the slot interface prototype.
2820            let constructor = vm.global_object.borrow().get("HTMLSlotElement");
2821            let proto = constructor
2822                .as_object()
2823                .and_then(|c| c.borrow().get("prototype").as_object());
2824            if let Some(proto) = proto {
2825                let mut current = Some(object.clone());
2826                while let Some(obj) = current {
2827                    if Rc::ptr_eq(&obj, &proto) {
2828                        return true;
2829                    }
2830                    current = obj.borrow().get_prototype();
2831                }
2832            }
2833            // Fallback: check whether the wrapped DOM node is a `<slot>` element.
2834            // This covers cases where the prototype chain was not wired through
2835            // `slot_prototype` (e.g. objects exposed before `install_document`).
2836            if let Some(node) =
2837                crate::engine::js::common::dom_node(vm, &JSValue::from_object(object))
2838            {
2839                if let crate::engine::html::HtmlNodeType::Element { tag_name, .. } =
2840                    &node.borrow().value
2841                {
2842                    return tag_name.eq_ignore_ascii_case("slot");
2843                }
2844            }
2845            false
2846        })
2847        .unwrap_or(false);
2848    Ok(JSValue::from_bool(matches))
2849}
