1//! `customElements` Web API — registers and instantiates custom element classes.
2//!
3//! Supports `define`, `get`, `upgrade`, and `whenDefined`.  Lifecycle callbacks
4//! (`connectedCallback`, `disconnectedCallback`, `attributeChangedCallback`) are
5//! dispatched when elements are added/removed from the DOM or have observed
6//! attributes changed.
7
8use crate::engine::js::CustomElementDefinition;
9use crate::engine::js::common::{is_callable, with_host, with_host_mut};
10use crate::engine::js::web_apis::dom::dom_exception::throw_dom_exception;
11use pixi_byte::value::jsobject::JSObject;
12use pixi_byte::vm::VM;
13use pixi_byte::{JSError, JSResult, JSValue};
14use std::cell::RefCell;
15use std::rc::Rc;

17// Cached JS helper for reading prototype callbacks from class constructors.
18// pixi_byte class constructors are `Function` values (not `JSObject`s), so
19// their `.prototype` is not accessible from native code.  The helper does
20// `c.prototype[n]` — the VM resolves property access through its internal
21// `user_function_object` map.
22thread_local! {
23    static CE_READ_CB: RefCell<Option<JSValue>> = const { RefCell::new(None) };
24}
25
26pub(crate) fn install_custom_elements(engine: &mut pixi_byte::JSEngine) {
27    // Define the JS helper in the global scope.  Must run after
28    // `install_global_aliases` so `globalThis` exists.
29    let _ = engine.eval(
30        "var __ce_read_cb = function(c, n) { \
31            if (c && c[n] !== undefined) return c[n]; \
32            var p = c && c.prototype; \
33            return (p && p[n] !== undefined) ? p[n] : undefined; \
34        }",
35    );
36    if let Ok(val) = engine.eval("__ce_read_cb") {
37        CE_READ_CB.with(|cell| *cell.borrow_mut() = Some(val));
38    }
39
40    // `CustomElementRegistry` interface: the singleton `customElements`
41    // instance carries a `__host_has_instance__` hook so
42    // `x instanceof CustomElementRegistry` works for scripts that probe the
43    // interface before using the registry.
44    let registry_prototype = Rc::new(RefCell::new(JSObject::new()));
45    let mut registry_constructor = JSObject::new();
46    // `__call__` makes `typeof CustomElementRegistry` report "function".
47    registry_constructor.set(
48        "__call__".to_string(),
49        JSValue::from_native_function(custom_elements_construct_error),
50    );
51    registry_constructor.set(
52        "__construct__".to_string(),
53        JSValue::from_native_function(custom_elements_construct_error),
54    );
55    registry_constructor.define_property(
56        "prototype".to_string(),
57        pixi_byte::value::jsobject::Property::read_only(JSValue::from_object(Rc::clone(
58            &registry_prototype,
59        ))),
60    );
61    registry_constructor.set(
62        "__host_has_instance__".to_string(),
63        JSValue::from_native_function(custom_elements_has_instance),
64    );
65
66    let mut custom_elements = JSObject::with_prototype(Some(registry_prototype));
67    custom_elements.set(
68        "define".to_string(),
69        JSValue::from_native_function(custom_elements_define),
70    );
71    custom_elements.set(
72        "get".to_string(),
73        JSValue::from_native_function(custom_elements_get),
74    );
75    custom_elements.set(
76        "upgrade".to_string(),
77        JSValue::from_native_function(custom_elements_upgrade),
78    );
79    custom_elements.set(
80        "whenDefined".to_string(),
81        JSValue::from_native_function(custom_elements_when_defined),
82    );
83    let mut global = engine.global_mut().borrow_mut();
84    global.set(
85        "CustomElementRegistry".to_string(),
86        JSValue::from_object(Rc::new(RefCell::new(registry_constructor))),
87    );
88    global.set(
89        "customElements".to_string(),
90        JSValue::from_object(Rc::new(RefCell::new(custom_elements))),
91    );
92}

94/// `new CustomElementRegistry()` is forbidden: the platform exposes exactly
95/// one registry per window.
96fn custom_elements_construct_error(_vm: &mut VM, _args: Vec<JSValue>) -> JSResult<JSValue> {
97    Err(JSError::TypeError(
98        "Illegal constructor: CustomElementRegistry cannot be constructed directly".to_string(),
99    ))
100}

102/// `instanceof CustomElementRegistry` hook: true for the singleton registry.
103fn custom_elements_has_instance(vm: &mut VM, args: Vec<JSValue>) -> JSResult<JSValue> {
104    let matches = args
105        .get(1)
106        .and_then(JSValue::as_object)
107        .map(|object| {
108            let registry = vm.global_object.borrow().get("customElements");
109            registry
110                .as_object()
111                .is_some_and(|registry| std::rc::Rc::ptr_eq(&object, &registry))
112        })
113        .unwrap_or(false);
114    Ok(JSValue::from_bool(matches))
115}

117// ---------------------------------------------------------------------------
118// `customElements.define(name, constructor, options?)`
119// ---------------------------------------------------------------------------
120
121fn custom_elements_define(vm: &mut VM, args: Vec<JSValue>) -> JSResult<JSValue> {
122    log::info!(
123        target: "ytprobe",
124        "customElements.define called: args.len={}, arg1={:?}, arg1_kind={:?}, arg2_kind={:?}, arg2_name={:?}, arg2_is={:?}",
125        args.len(),
126        args.get(1).map(JSValue::to_string),
127        args.get(1).map(JSValue::kind),
128        args.get(2).map(JSValue::kind),
129        args.get(2).and_then(JSValue::as_object).map(|o| o.borrow().get("name").to_string()),
130        args.get(2).and_then(JSValue::as_object).map(|o| o.borrow().get("is").to_string()),
131    );
132    // WebIDL DOMString conversion: apply ToString coercion to the name
133    // argument so callers passing a non-string value (e.g. a number or
134    // string-like object) are accepted rather than rejected with a TypeError.
135    let name_arg = args.get(1).cloned().unwrap_or(JSValue::undefined());
136    let name = name_arg.to_console_string().trim().to_ascii_lowercase();
137    if name.is_empty() || !is_valid_custom_element_name(&name) {
138        let msg = if name.is_empty() {
139            "customElements.define requires a name string".to_string()
140        } else {
141            format!("The element name '{name}' is not a valid custom element name")
142        };
143        return if name.is_empty() {
144            Err(JSError::TypeError(msg))
145        } else {
146            Err(throw_dom_exception(&msg, "NotSupportedError"))
147        };
148    }
149    let Some(constructor) = args.get(2).cloned() else {
150        return Err(JSError::TypeError(
151            "customElements.define requires a constructor".to_string(),
152        ));
153    };
154    if !is_callable(&constructor) {
155        return Err(JSError::TypeError(
156            "customElements.define constructor must be callable".to_string(),
157        ));
158    }
159
160    // Duplicate registration: same name already defined -> NotSupportedError.
161    let already_defined =
162        with_host(vm, |host| host.custom_elements.contains_key(&name)).unwrap_or(false);
163    if already_defined {
164        return Err(throw_dom_exception(
165            &format!(
166                "Failed to execute 'define' on 'CustomElementRegistry': the name '{name}' has already been used with this registry"
167            ),
168            "NotSupportedError",
169        ));
170    }
171    // Same constructor registered under a different name -> NotSupportedError.
172    let constructor_used = with_host(vm, |host| {
173        let target_obj = constructor.as_object();
174        host.custom_elements.values().any(|d| {
175            let d_obj = d.constructor.as_object();
176            match (target_obj.clone(), d_obj) {
177                (Some(t), Some(d)) => Rc::ptr_eq(&t, &d),
178                _ => false,
179            }
180        })
181    })
182    .unwrap_or(false);
183    if constructor_used {
184        return Err(throw_dom_exception(
185            "Failed to execute 'define' on 'CustomElementRegistry': this constructor has already been used with this registry",
186            "NotSupportedError",
187        ));
188    }
189
190    let connected_callback = get_prototype_callback(vm, &constructor, "connectedCallback");
191    let disconnected_callback = get_prototype_callback(vm, &constructor, "disconnectedCallback");
192    let attribute_changed_callback =
193        get_prototype_callback(vm, &constructor, "attributeChangedCallback");
194    let observed_attributes = get_observed_attributes(vm, &constructor);
195
196    let definition = CustomElementDefinition {
197        constructor: constructor.clone(),
198        connected_callback,
199        disconnected_callback,
200        attribute_changed_callback,
201        observed_attributes,
202        when_defined_resolvers: Vec::new(),
203    };
204
205    with_host_mut(vm, |host| {
206        host.custom_elements
207            .insert(name.clone(), definition.clone());
208    });
209
210    // Upgrade existing elements with this tag name: link the class prototype
211    // and run the custom element constructor per the Custom Elements spec.
212    let matching_nodes: Vec<crate::engine::tree::NodeRef<crate::engine::html::HtmlNodeType>> =
213        with_host(vm, |host| {
214            let mut nodes = host.dom.find_all(|n| {
215                n.tag_name()
216                    .map(|t| t.eq_ignore_ascii_case(&name))
217                    .unwrap_or(false)
218            });
219            for node in host.detached_nodes.values() {
220                let matches = node
221                    .borrow()
222                    .value
223                    .tag_name()
224                    .map(|t| t.eq_ignore_ascii_case(&name))
225                    .unwrap_or(false);
226                if matches && !nodes.iter().any(|existing| Rc::ptr_eq(existing, node)) {
227                    nodes.push(Rc::clone(node));
228                }
229            }
230            for weak in host.refs.values() {
231                if let Some(node) = weak.upgrade() {
232                    let matches = node
233                        .borrow()
234                        .value
235                        .tag_name()
236                        .map(|t| t.eq_ignore_ascii_case(&name))
237                        .unwrap_or(false);
238                    if matches && !nodes.iter().any(|existing| Rc::ptr_eq(existing, &node)) {
239                        nodes.push(node);
240                    }
241                }
242            }
243            nodes
244        })
245        .unwrap_or_default();
246
247    for node in matching_nodes {
248        upgrade_custom_element(vm, &node, &definition);
249    }
250
251    // Resolve pending `whenDefined()` promises.
252    let resolvers: Vec<JSValue> = with_host_mut(vm, |host| {
253        if let Some(def) = host.custom_elements.get_mut(&name) {
254            std::mem::take(&mut def.when_defined_resolvers)
255        } else {
256            Vec::new()
257        }
258    })
259    .unwrap_or_default();
260
261    for resolver in resolvers {
262        let _ = vm.call(resolver, JSValue::undefined(), vec![]);
263    }
264
265    Ok(JSValue::undefined())
266}

272fn custom_elements_get(vm: &mut VM, args: Vec<JSValue>) -> JSResult<JSValue> {
273    let name_arg = args.get(1).cloned().unwrap_or(JSValue::undefined());
274    let name = name_arg.to_console_string().trim().to_ascii_lowercase();
275    if name.is_empty() {
276        return Ok(JSValue::undefined());
277    }
278    with_host(vm, |host| {
279        host.custom_elements
280            .get(&name)
281            .map(|def| def.constructor.clone())
282    })
283    .flatten()
284    .ok_or_else(|| JSError::TypeError("customElements.get: definition not found".to_string()))
285    .or(Ok(JSValue::undefined()))
286}

288// ---------------------------------------------------------------------------
289// `customElements.upgrade(root)`
290// ---------------------------------------------------------------------------
291
292fn custom_elements_upgrade(vm: &mut VM, args: Vec<JSValue>) -> JSResult<JSValue> {
293    let root_value = args.get(1).cloned().unwrap_or(JSValue::undefined());
294
295    let matching_nodes: Vec<(
296        crate::engine::tree::NodeRef<crate::engine::html::HtmlNodeType>,
297        CustomElementDefinition,
298    )> = with_host(vm, |host| {
299        let root_dom_id = crate::engine::js::common::node_dom_id(&root_value);
300        let mut candidates = host.dom.find_all(|n| {
301            n.tag_name()
302                .is_some_and(|t| host.custom_elements.contains_key(&t.to_ascii_lowercase()))
303        });
304        for node in host.detached_nodes.values() {
305            if node
306                .borrow()
307                .value
308                .tag_name()
309                .is_some_and(|t| host.custom_elements.contains_key(&t.to_ascii_lowercase()))
310                && !candidates.iter().any(|existing| Rc::ptr_eq(existing, node))
311            {
312                candidates.push(Rc::clone(node));
313            }
314        }
315        for weak in host.refs.values() {
316            if let Some(node) = weak.upgrade()
317                && node
318                    .borrow()
319                    .value
320                    .tag_name()
321                    .is_some_and(|t| host.custom_elements.contains_key(&t.to_ascii_lowercase()))
322                && !candidates
323                    .iter()
324                    .any(|existing| Rc::ptr_eq(existing, &node))
325            {
326                candidates.push(node);
327            }
328        }
329
330        candidates
331            .into_iter()
332            .filter_map(|node| {
333                let tag = node.borrow().value.tag_name()?.to_ascii_lowercase();
334                let def = host.custom_elements.get(&tag)?.clone();
335                if let Some(root_id) = root_dom_id {
336                    let node_id = host.dom_id_for_node(&node)?;
337                    if node_id == root_id {
338                        return Some((node, def));
339                    }
340                    let mut current = Some(Rc::clone(&node));
341                    let mut found = false;
342                    while let Some(n) = current {
343                        if let Some(n_id) = host.dom_id_for_node(&n)
344                            && n_id == root_id
345                        {
346                            found = true;
347                            break;
348                        }
349                        current = n.borrow().parent();
350                    }
351                    if found { Some((node, def)) } else { None }
352                } else {
353                    Some((node, def))
354                }
355            })
356            .collect()
357    })
358    .unwrap_or_default();
359
360    for (node, def) in matching_nodes {
361        upgrade_custom_element(vm, &node, &def);
362    }
363
364    Ok(JSValue::undefined())
365}

367/// Upgrades an element instance to a custom element per the Custom Elements spec.
368pub(crate) fn upgrade_custom_element(
369    vm: &mut VM,
370    node: &crate::engine::tree::NodeRef<crate::engine::html::HtmlNodeType>,
371    definition: &CustomElementDefinition,
372) {
373    let Some(element_val) = super::document::expose_node(vm, Rc::clone(node)) else {
374        return;
375    };
376    let Some(dom_id) = with_host(vm, |host| host.dom_id_for_node(node)).flatten() else {
377        return;
378    };
379
380    if let Some(class_proto) = custom_element_class_prototype(definition) {
381        if let Some(element) = with_host(vm, |host| host.objects.get(&dom_id).cloned()).flatten() {
382            element
383                .borrow_mut()
384                .set_prototype(Some(Rc::clone(&class_proto)));
385        }
386    }
387
388    let should_construct = element_val
389        .as_object()
390        .map(|obj| {
391            if obj.borrow().get("__orinium_ce_constructed").to_boolean() {
392                false
393            } else {
394                obj.borrow_mut().set(
395                    "__orinium_ce_constructed".to_string(),
396                    JSValue::from_bool(true),
397                );
398                true
399            }
400        })
401        .unwrap_or(false);
402
403    if should_construct {
404        if let Err(err) = vm.call(definition.constructor.clone(), element_val, vec![]) {
405            log::info!("JS error in custom element constructor: {}", err);
406        }
407    }
408
409    fire_lifecycle_callback(vm, dom_id, |d| d.connected_callback.clone());
410}

414// ---------------------------------------------------------------------------
415
416fn custom_elements_when_defined(vm: &mut VM, args: Vec<JSValue>) -> JSResult<JSValue> {
417    let name_arg = args.get(1).cloned().unwrap_or(JSValue::undefined());
418    let name = name_arg.to_console_string().trim().to_ascii_lowercase();
419    if name.is_empty() {
420        return Err(JSError::TypeError(
421            "customElements.whenDefined requires a name string".to_string(),
422        ));
423    }
424    let is_defined =
425        with_host(vm, |host| host.custom_elements.contains_key(&name)).unwrap_or(false);
426
427    if is_defined {
428        return Ok(make_resolved_promise_stub());
429    }
430
431    // Create a thenable promise.  The `then` method stores the
432    // onFulfilled callback in the definition's pending queue so that
433    // `define()` can invoke it when the name is registered.
434    //
435    // We store the element name as a property on the promise object so
436    // the native `then` function can read it from `this`.
437    let mut promise = JSObject::new();
438    promise.set("__ce_name__".to_string(), JSValue::from_string(name));
439    promise.set(
440        "then".to_string(),
441        JSValue::from_native_function(when_defined_then),
442    );
443
444    Ok(JSValue::from_object(Rc::new(RefCell::new(promise))))
445}
446
447fn make_resolved_promise_stub() -> JSValue {
448    let mut promise = JSObject::new();
449    promise.set(
450        "then".to_string(),
451        JSValue::from_native_function(then_invoke),
452    );
453    JSValue::from_object(Rc::new(RefCell::new(promise)))
454}

456/// Resolved-promise `then`: immediately invokes the onFulfilled callback.
457fn then_invoke(vm: &mut VM, args: Vec<JSValue>) -> JSResult<JSValue> {
458    let callback = args.get(1).cloned().unwrap_or(JSValue::undefined());
459    if is_callable(&callback) {
460        let _ = vm.call(callback, JSValue::undefined(), vec![]);
461    }
462    Ok(JSValue::undefined())
463}

465/// Native function for `whenDefined().then(callback)`.  Reads the element
466/// name from `this.__ce_name__` and stores the callback in the definition's
467/// pending queue.
468fn when_defined_then(vm: &mut VM, args: Vec<JSValue>) -> JSResult<JSValue> {
469    let this = args.first().cloned().unwrap_or(JSValue::undefined());
470    let callback = args.get(1).cloned().unwrap_or(JSValue::undefined());
471    if !is_callable(&callback) {
472        return Ok(JSValue::undefined());
473    }
474    let name = if let Some(obj) = this.as_object() {
475        obj.borrow()
476            .get("__ce_name__")
477            .as_string()
478            .unwrap_or("")
479            .to_string()
480    } else {
481        return Ok(JSValue::undefined());
482    };
483    if name.is_empty() {
484        return Ok(JSValue::undefined());
485    }
486    // If the name is already defined, invoke the callback immediately
487    // (mimicking a resolved promise).
488    let already_defined =
489        with_host(vm, |host| host.custom_elements.contains_key(&name)).unwrap_or(false);
490    if already_defined {
491        let _ = vm.call(callback, JSValue::undefined(), vec![]);
492        return Ok(JSValue::undefined());
493    }
494    // Otherwise, store for later resolution by define().
495    with_host_mut(vm, |host| {
496        if let Some(def) = host.custom_elements.get_mut(&name) {
497            def.when_defined_resolvers.push(callback);
498        }
499    });
500    Ok(JSValue::undefined())
501}

507/// Reads a lifecycle callback from the constructor's `prototype` property.
508///
509/// For JSObject constructors (e.g. `HTMLElement`), reads directly.
510/// For `class` constructors (Function values), uses the cached JS helper
511/// `__ce_read_cb(constructor, propName)` which the VM resolves through
512/// its internal `user_function_object` mechanism.
513fn get_prototype_callback(vm: &mut VM, constructor: &JSValue, prop_name: &str) -> Option<JSValue> {
514    // Fast path: JSObject constructors (e.g. HTMLElement).
515    if let Some(obj) = constructor.as_object() {
516        let prototype_value = obj.borrow().get("prototype");
517        if let Some(prototype) = prototype_value.as_object() {
518            let value = prototype.borrow().get(prop_name);
519            if is_callable(&value) {
520                return Some(value);
521            }
522        }
523    }
524    // Slow path: class constructors (Function values).
525    let helper_fn = CE_READ_CB.with(|cell| cell.borrow().clone())?;
526    let result = vm.call(
527        helper_fn,
528        JSValue::undefined(),
529        vec![
530            constructor.clone(),
531            JSValue::from_string(prop_name.to_string()),
532        ],
533    );
534    match result {
535        Ok(value) if is_callable(&value) => Some(value),
536        _ => None,
537    }
538}

540/// Reads the `observedAttributes` static property from a constructor.
541/// For class constructors, this is a `static` property on the constructor
542/// itself (not on the prototype).
543fn get_observed_attributes(vm: &mut VM, constructor: &JSValue) -> Vec<String> {
544    // Fast path: JSObject constructors — read directly.
545    if let Some(obj) = constructor.as_object() {
546        let attrs_value = obj.borrow().get("observedAttributes");
547        let result = extract_string_array(&attrs_value);
548        if !result.is_empty() {
549            return result;
550        }
551    }
552    // Slow path: class constructors (Function values).
553    // The JS helper reads `c[n]` first (static properties), then
554    // `c.prototype[n]` (instance methods).
555    let helper_fn = match CE_READ_CB.with(|cell| cell.borrow().clone()) {
556        Some(f) => f,
557        None => return Vec::new(),
558    };
559    let result = vm.call(
560        helper_fn,
561        JSValue::undefined(),
562        vec![
563            constructor.clone(),
564            JSValue::from_string("observedAttributes".to_string()),
565        ],
566    );
567    match result {
568        Ok(value) => extract_string_array(&value),
569        _ => Vec::new(),
570    }
571}
572
573fn extract_string_array(value: &JSValue) -> Vec<String> {
574    let Some(obj) = value.as_object() else {
575        return Vec::new();
576    };
577    let obj_ref = obj.borrow();
578    let len = obj_ref.get("length").as_number().unwrap_or(0.0) as usize;
579    (0..len)
580        .filter_map(|i| {
581            let key = i.to_string();
582            let val = obj_ref.get(&key);
583            val.as_string().map(str::to_string)
584        })
585        .collect()
586}

592/// Returns the class `prototype` object of a custom element constructor so the
593/// element instance can inherit its methods.
594pub(crate) fn custom_element_class_prototype(
595    definition: &CustomElementDefinition,
596) -> Option<Rc<RefCell<JSObject>>> {
597    let constructor = &definition.constructor;
598    let prototype = constructor.as_object()?.borrow().get("prototype");
599    prototype.as_object()
600}

602/// Links an element wrapper to its custom element class prototype, returning
603/// the class prototype on success.
604pub(crate) fn link_custom_element_prototype(
605    _element: &Rc<RefCell<JSObject>>,
606    definition: &CustomElementDefinition,
607) -> Option<Rc<RefCell<JSObject>>> {
608    custom_element_class_prototype(definition)
609}

621/// Generic lifecycle callback dispatcher using stored callbacks from the
622/// custom element definition.
623fn fire_lifecycle_callback(
624    vm: &mut VM,
625    dom_id: u64,
626    extract: impl FnOnce(&CustomElementDefinition) -> Option<JSValue>,
627) {
628    let tag_name = with_host(vm, |host| {
629        host.refs
630            .get(&dom_id)
631            .and_then(|w| w.upgrade())
632            .and_then(|node| node.borrow().value.tag_name().map(str::to_ascii_lowercase))
633    })
634    .flatten();
635
636    let Some(tag_name) = tag_name else { return };
637
638    let callback_and_obj = with_host(vm, |host| {
639        let definition = host.custom_elements.get(&tag_name)?;
640        let callback = extract(definition)?;
641        let element_obj = host.objects.get(&dom_id).cloned()?;
642        Some((callback, element_obj))
643    })
644    .flatten();
645
646    let Some((callback, element_obj)) = callback_and_obj else {
647        return;
648    };
649
650    if let Err(err) = vm.call(
651        callback,
652        JSValue::from_object(Rc::clone(&element_obj)),
653        vec![],
654    ) {
655        log::info!("JS error in lifecycle callback: {}", err);
656    }
657}

659/// Dispatches `attributeChangedCallback` on a custom element when an observed
660/// attribute is set or removed.  Per spec, `oldValue` is `null` when the
661/// attribute is set for the first time.
662pub(crate) fn fire_attribute_changed_callback(
663    vm: &mut VM,
664    dom_id: u64,
665    attribute_name: &str,
666    old_value: Option<&str>,
667    new_value: Option<&str>,
668) {
669    let tag_name = with_host(vm, |host| {
670        host.refs
671            .get(&dom_id)
672            .and_then(|w| w.upgrade())
673            .and_then(|node| node.borrow().value.tag_name().map(str::to_ascii_lowercase))
674    })
675    .flatten();
676
677    let Some(tag_name) = tag_name else { return };
678    let definition = with_host(vm, |host| host.custom_elements.get(&tag_name).cloned()).flatten();
679    let Some(definition) = definition else {
680        return;
681    };
682
683    // Only fire if the attribute is in observedAttributes.
684    if !definition
685        .observed_attributes
686        .iter()
687        .any(|a| a == attribute_name)
688    {
689        return;
690    }
691
692    let Some(callback) = definition.attribute_changed_callback else {
693        return;
694    };
695
696    let element_obj = with_host(vm, |host| host.objects.get(&dom_id).cloned()).flatten();
697    let Some(element_obj) = element_obj else {
698        return;
699    };
700
701    // Per spec: oldValue is null (not empty string) when the attribute didn't
702    // exist before the call.
703    let old_js = match old_value {
704        Some(v) => JSValue::from_string(v.to_string()),
705        None => JSValue::null(),
706    };
707    let new_js = match new_value {
708        Some(v) => JSValue::from_string(v.to_string()),
709        None => JSValue::null(),
710    };
711
712    vm.enqueue_job(
713        callback,
714        JSValue::from_object(Rc::clone(&element_obj)),
715        vec![
716            JSValue::from_string(attribute_name.to_string()),
717            old_js,
718            new_js,
719        ],
720    );
721}

723/// Reserved custom element names per the HTML spec (MathML + SVG elements that
724// must not be redefined as custom elements).
725const RESERVED_CUSTOM_ELEMENT_NAMES: &[&str] = &[
726    "annotation-xml",
727    "color-profile",
728    "font-face",
729    "font-face-src",
730    "font-face-uri",
731    "font-face-format",
732    "font-face-name",
733    "missing-glyph",
734];
735
736/// Validates a custom element name per the HTML spec.
737///
738/// A valid custom element name must:
739/// - contain an ASCII hyphen (`-`),
740/// - consist only of ASCII lowercase alphanumerics and hyphens,
741/// - start with a lowercase letter,
742/// - not be a reserved name (MathML/SVG element names).
743fn is_valid_custom_element_name(name: &str) -> bool {
744    if name.is_empty() || !name.contains('-') {
745        return false;
746    }
747    if RESERVED_CUSTOM_ELEMENT_NAMES.contains(&name) {
748        return false;
749    }
750    let mut chars = name.chars();
751    // Must start with a lowercase ASCII letter.
752    match chars.next() {
753        Some(c) if c.is_ascii_lowercase() => {}
754        _ => return false,
755    }
756    // Remaining chars must be ASCII alphanumeric or hyphen.
757    chars.all(|c| c.is_ascii_alphanumeric() || c == '-')
758}
