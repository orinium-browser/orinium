1039pub(crate) fn append_child(vm: &mut VM, args: Vec<JSValue>) -> JSResult<JSValue> {
1040    let Some(parent) = dom_node(vm, args.first().unwrap_or(&UNDEFINED)) else {
1041        return Ok(JSValue::null());
1042    };
1043    let is_document = matches!(parent.borrow().value, HtmlNodeType::Document);
1044    if !is_document
1045        && !matches!(
1046            parent.borrow().value,
1047            HtmlNodeType::Element { .. }
1048                | HtmlNodeType::DocumentFragment
1049                | HtmlNodeType::ShadowRoot { .. }
1050        )
1051    {
1052        return Ok(JSValue::null());
1053    }
1054    let Some(child_value) = args.get(1).cloned() else {
1055        return Ok(JSValue::null());
1056    };
1057    let Some(child) = dom_node(vm, &child_value) else {
1058        return Ok(JSValue::null());
1059    };
1060
1061    // Ensure pre-insertion validity: appending a node that is an inclusive
1062    // ancestor of the parent would create a cycle and raises
1063    // HIERARCHY_REQUEST_ERR (e.g. document.body.appendChild(documentElement)).
1064    if TreeNode::is_inclusive_ancestor(&child, &parent) {
1065        return Err(throw_dom_exception(
1066            "The new child element is an ancestor of the parent element",
1067            "HierarchyRequestError",
1068        ));
1069    }
1070
1071    // Per DOM spec, a document may hold an element or a doctype, but never a
1072    // fragment's contents spliced in directly.
1073    if is_document && matches!(child.borrow().value, HtmlNodeType::DocumentFragment) {
1074        return Err(throw_dom_exception(
1075            "Cannot append a DocumentFragment to a Document",
1076            "HierarchyRequestError",
1077        ));
1078    }
1079
1080    // Per DOM spec, appending a DocumentFragment moves its children.
1081    if matches!(child.borrow().value, HtmlNodeType::DocumentFragment) {
1082        let fragment_children: Vec<_> = child.borrow().children().to_vec();
1083        for fragment_child in fragment_children {
1084            let _ = TreeNode::append_child(&parent, Rc::clone(&fragment_child));
1085            mark_dom_dirty(vm);
1086        }
1087        return Ok(child_value);
1088    }
1089
1090    if !TreeNode::append_child(&parent, child) {
1091        return Ok(JSValue::null());
1092    }
1093
1094    // Inserting into a document adopts the node, so `ownerDocument` follows the
1095    // node instead of the document that happened to create it.
1096    if is_document
1097        && let (Some(document_dom_id), Some(child_dom_id)) = (
1098            node_dom_id(args.first().unwrap_or(&UNDEFINED)),
1099            node_dom_id(&child_value),
1100        )
1101        && let Some(adopted) = with_host(vm, |host| host.refs.get(&child_dom_id).cloned()).flatten()
1102        && let Some(node) = adopted.upgrade()
1103    {
1104        node.borrow().set_owner_document(document_dom_id);
1105    }
1106
1107    if let Some(dom_id) = node_dom_id(&child_value) {
1108        let _ = with_host_mut(vm, |host| {
1109            host.detached_nodes.remove(&dom_id);
1110        });
1111        fire_connected_callback(vm, dom_id);
1112    }
1113    queue_dynamic_script(vm, &child_value);
1114    queue_dynamic_stylesheet(vm, &child_value);
1115    queue_dynamic_image(vm, &child_value);
1116    mark_dom_dirty(vm);
1117    Ok(child_value)
1118}

1133pub(crate) fn insert_before(vm: &mut VM, args: Vec<JSValue>) -> JSResult<JSValue> {
1134    let Some(parent) = dom_node(vm, args.first().unwrap_or(&UNDEFINED)) else {
1135        return Ok(JSValue::null());
1136    };
1137    let Some(child_value) = args.get(1).cloned() else {
1138        return Ok(JSValue::null());
1139    };
1140    let Some(child) = dom_node(vm, &child_value) else {
1141        return Ok(JSValue::null());
1142    };
1143
1144    let reference_value = args.get(2).cloned();
1145    let reference = match &reference_value {
1146        Some(val) if val.is_null() || val.is_undefined() => None,
1147        Some(val) => dom_node(vm, val),
1148        None => None,
1149    };
1150
1151    // Per DOM spec, inserting a DocumentFragment moves its children.
1152    if matches!(child.borrow().value, HtmlNodeType::DocumentFragment) {
1153        let fragment_children: Vec<_> = child.borrow().children().to_vec();
1154        for fragment_child in fragment_children {
1155            let inserted = match &reference {
1156                Some(r) => TreeNode::insert_before(&parent, Rc::clone(&fragment_child), r),
1157                None => TreeNode::append_child(&parent, Rc::clone(&fragment_child)),
1158            };
1159            if inserted {
1160                mark_dom_dirty(vm);
1161            }
1162        }
1163        return Ok(child_value);
1164    }
1165
1166    let inserted = match &reference {
1167        Some(r) => TreeNode::insert_before(&parent, child, r),
1168        None => TreeNode::append_child(&parent, child),
1169    };
1170    if !inserted {
1171        return Ok(JSValue::null());
1172    }
1173    if let Some(dom_id) = node_dom_id(&child_value) {
1174        let _ = with_host_mut(vm, |host| {
1175            host.detached_nodes.remove(&dom_id);
1176        });
1177        fire_connected_callback(vm, dom_id);
1178    }
1179    queue_dynamic_script(vm, &child_value);
1180    queue_dynamic_stylesheet(vm, &child_value);
1181    queue_dynamic_image(vm, &child_value);
1182    mark_dom_dirty(vm);
1183    Ok(child_value)
1184}

1186fn queue_dynamic_script(vm: &mut VM, value: &JSValue) {
1187    let Some(node_id) = node_dom_id(value) else {
1188        return;
1189    };
1190    let Some(node) = dom_node(vm, value) else {
1191        return;
1192    };
1193    let source = {
1194        let node_ref = node.borrow();
1195        if node_ref.value.tag_name() != Some("script") {
1196            return;
1197        }
1198        let script_type = node_ref.value.get_attr("type").unwrap_or("").trim();
1199        if !script_type.is_empty()
1200            && !matches!(
1201                script_type.to_ascii_lowercase().as_str(),
1202                "text/javascript"
1203                    | "application/javascript"
1204                    | "text/ecmascript"
1205                    | "application/ecmascript"
1206                    | "application/x-javascript"
1207            )
1208        {
1209            return;
1210        }
1211        match node_ref.value.get_attr("src").map(str::trim) {
1212            Some(src) if !src.is_empty() => JsDynamicScriptSource::External(src.to_string()),
1213            Some(_) => return,
1214            None => JsDynamicScriptSource::Inline(DomTree::inner_text(&node)),
1215        }
1216    };
1217    let _ = with_host_mut(vm, |host| {
1218        if host.queued_dynamic_scripts.insert(node_id) {
1219            host.dynamic_script_requests
1220                .push(JsDynamicScriptRequest { node_id, source });
1221        }
1222    });
1223}

1301fn queue_iframe_source_if_needed(vm: &mut VM, value: &JSValue) {
1302    let Some(node_id) = node_dom_id(value) else {
1303        return;
1304    };
1305    let Some(node) = dom_node(vm, value) else {
1306        return;
1307    };
1308    let source = {
1309        let node = node.borrow();
1310        if node.value.tag_name() != Some("iframe") {
1311            return;
1312        }
1313        let Some(source) = node.value.get_attr("src").map(str::trim) else {
1314            return;
1315        };
1316        if source.is_empty() {
1317            return;
1318        }
1319        source.to_string()
1320    };
1321    let resolved = with_host(vm, |host| resolved_iframe_url(&host.document_url, &source))
1322        .flatten()
1323        .unwrap_or(source);
1324    let _ = with_host_mut(vm, |host| {
1325        // When src changes, discard any in-flight fetch and stale content
1326        // document so the new URL is fetched immediately.
1327        host.pending_iframe_fetches.remove(&node_id);
1328        host.failed_iframe_fetches.remove(&node_id);
1329        host.iframe_documents.remove(&node_id);
1330        host.pending_iframe_fetches.insert(node_id);
1331        host.iframe_fetch_requests.push(JsIframeFetchRequest {
1332            dom_id: node_id,
1333            url: resolved,
1334        });
1335    });
1336}

1502/// Collects the form controls that belong to `form` (its `input`, `select`,
1503/// `textarea` and `button` descendants).
1504fn form_controls_of(form: &NodeRef<HtmlNodeType>) -> Vec<NodeRef<HtmlNodeType>> {
1505    let mut controls = Vec::new();
1506    let mut stack = form.borrow().children().to_vec();
1507    while let Some(child) = stack.pop() {
1508        let tag = child.borrow().value.tag_name().map(str::to_ascii_lowercase);
1509        match tag.as_deref() {
1510            Some("input") | Some("select") | Some("textarea") | Some("button") => {
1511                controls.push(Rc::clone(&child));
1512            }
1513            // Do not descend into nested forms (invalid HTML anyway).
1514            Some("form") => {}
1515            _ => stack.extend(child.borrow().children().iter().cloned()),
1516        }
1517    }
1518    controls.reverse();
1519    controls
1520}

1522/// `form.elements.namedItem(name)` — the first control whose `name` or `id`
1523/// attribute matches, or `null`.
1524fn form_elements_named_item(vm: &mut VM, args: Vec<JSValue>) -> JSResult<JSValue> {
1525    let Some(receiver) = args.first().and_then(JSValue::as_object) else {
1526        return Ok(JSValue::null());
1527    };
1528    let Some(name) = args.get(1).and_then(JSValue::as_string) else {
1529        return Ok(JSValue::null());
1530    };
1531    // Resolve the collection's entries by index so we reuse the exposed
1532    // wrappers (and their dom ids) instead of re-exposing raw nodes.
1533    let length = receiver.borrow().get("length").to_number().max(0.0) as usize;
1534    for index in 0..length {
1535        let entry = receiver.borrow().get(&index.to_string());
1536        let matches = entry
1537            .as_object()
1538            .and_then(|object| {
1539                let dom_id = object.borrow().get("__orinium_dom_id").to_number();
1540                with_host(vm, |host| {
1541                    host.refs.get(&(dom_id as u64)).and_then(|w| w.upgrade())
1542                })
1543                .flatten()
1544            })
1545            .is_some_and(|node| {
1546                let borrowed = node.borrow();
1547                let name_attr = borrowed.value.get_attr("name").map(str::to_string);
1548                let id_attr = borrowed.value.get_attr("id").map(str::to_string);
1549                name_attr.as_deref() == Some(name) || id_attr.as_deref() == Some(name)
1550            });
1551        if matches {
1552            return Ok(entry);
1553        }
1554    }
1555    Ok(JSValue::null())
1556}

1558/// `form.requestSubmit()` / `form.submit()` — queues a top-level navigation to
1559/// the form's `action` URL with its controls serialized as query parameters
1560/// (GET forms). Reddit's challenge page uses this to submit its solution.
1561fn form_request_submit(vm: &mut VM, args: Vec<JSValue>) -> JSResult<JSValue> {
1562    let Some(form_value) = args.first().cloned() else {
1563        return Ok(JSValue::undefined());
1564    };
1565    let Some(form) = dom_node(vm, &form_value) else {
1566        return Ok(JSValue::undefined());
1567    };
1568    let (action, method, query) = {
1569        let borrowed = form.borrow();
1570        let action = borrowed.value.get_attr("action").unwrap_or("").to_string();
1571        let method = borrowed
1572            .value
1573            .get_attr("method")
1574            .unwrap_or("get")
1575            .to_ascii_lowercase();
1576        let pairs: Vec<(String, String)> = form_controls_of(&form)
1577            .iter()
1578            .filter_map(|control| {
1579                let borrowed = control.borrow();
1580                let name = borrowed.value.get_attr("name")?.to_string();
1581                let value = borrowed.value.get_attr("value").unwrap_or("").to_string();
1582                Some((name, value))
1583            })
1584            .collect();
1585        let query = if pairs.is_empty() {
1586            String::new()
1587        } else {
1588            let encoded: Vec<String> = pairs
1589                .iter()
1590                .map(|(name, value)| {
1591                    format!(
1592                        "{}={}",
1593                        crate::engine::js::web_apis::encoding::encode_query_component(name),
1594                        crate::engine::js::web_apis::encoding::encode_query_component(value)
1595                    )
1596                })
1597                .collect();
1598            format!("?{}", encoded.join("&"))
1599        };
1600        (action, method, query)
1601    };
1602    if action.is_empty() {
1603        return Ok(JSValue::undefined());
1604    }
1605    // Resolve the action against the document URL and queue the navigation.
1606    let base = with_host(vm, |host| host.document_url.clone()).unwrap_or_default();
1607    let target = url::Url::parse(&action)
1608        .or_else(|_| url::Url::parse(&base).and_then(|base| base.join(&action)))
1609        .map(|url| {
1610            if method == "get" && !query.is_empty() {
1611                url::Url::parse(&format!("{url}{query}")).unwrap_or(url)
1612            } else {
1613                url
1614            }
1615        });
1616    if let Ok(target) = target {
1617        with_host_mut(vm, |host| {
1618            host.navigation_requests.push(target.to_string());
1619        });
1620        log::info!(target: "FormNav", "form submit -> {target}");
1621    }
1622    Ok(JSValue::undefined())
1623}

1700pub(crate) fn element_dispatch_event(vm: &mut VM, args: Vec<JSValue>) -> JSResult<JSValue> {
1701    let Some(target) = args.first().and_then(JSValue::as_object) else {
1702        return Err(JSError::TypeError(
1703            "dispatchEvent called on incompatible receiver".to_string(),
1704        ));
1705    };
1706    let Some(event) = args.get(1).and_then(JSValue::as_object) else {
1707        return Err(JSError::TypeError(
1708            "dispatchEvent requires an Event".to_string(),
1709        ));
1710    };
1711    let event_type = event.borrow().get("type").to_string();
1712    if event_type.is_empty() {
1713        return Err(JSError::TypeError(
1714            "Event type must not be empty".to_string(),
1715        ));
1716    }
1717
1718    let target_obj = Rc::clone(&target);
1719
1720    // Build the ancestor path of exposed JS objects: [target, ..., root].
1721    let mut path: Vec<Rc<RefCell<JSObject>>> = vec![Rc::clone(&target_obj)];
1722    if let Some(mut node) = dom_node(vm, args.first().unwrap_or(&UNDEFINED)) {
1723        loop {
1724            let parent = node.borrow().parent();
1725            let Some(parent) = parent else { break };
1726            node = parent;
1727            if let Some(object) =
1728                expose_node(vm, Rc::clone(&node)).and_then(|value| value.as_object())
1729            {
1730                path.push(object);
1731            }
1732        }
1733    }
1734
1735    propagate_event(vm, &target_obj, &event, path)
1736}

1738/// Runs the capture/target/bubble propagation for an event whose `target` is
1739/// `target_obj` and whose ancestor path (target first) is `path`.
1740fn propagate_event(
1741    vm: &mut VM,
1742    target_obj: &Rc<RefCell<JSObject>>,
1743    event: &Rc<RefCell<JSObject>>,
1744    path: Vec<Rc<RefCell<JSObject>>>,
1745) -> JSResult<JSValue> {
1746    let event_type = event.borrow().get("type").to_string();
1747    let bubbles = event.borrow().get("bubbles").to_boolean();
1748
1749    // Reset propagation state for this dispatch.
1750    event.borrow_mut().set(
1751        "__orinium_immediate_propagation_stopped".to_string(),
1752        JSValue::from_bool(false),
1753    );
1754    event
1755        .borrow_mut()
1756        .set("cancelBubble".to_string(), JSValue::from_bool(false));
1757    event.borrow_mut().set(
1758        "target".to_string(),
1759        JSValue::from_object(Rc::clone(target_obj)),
1760    );
1761    event
1762        .borrow_mut()
1763        .set("__orinium_bubbles".to_string(), JSValue::from_bool(bubbles));
1764
1765    // Capture phase: from the root down to (but excluding) the target.
1766    for ancestor in path.iter().rev().skip(1) {
1767        if event_flag(event, "cancelBubble") {
1768            break;
1769        }
1770        dispatch_at_phase(vm, ancestor, event, &event_type, 1 /* capture */)?;
1771    }
1772
1773    // Target phase: the target's own handlers (both capturing and bubbling).
1774    if !event_flag(event, "cancelBubble") {
1775        dispatch_at_phase(vm, target_obj, event, &event_type, 2 /* target */)?;
1776    }
1777
1778    // Bubble phase: from the target's ancestors up to the root.
1779    if bubbles {
1780        for ancestor in path.iter().skip(1) {
1781            if event_flag(event, "cancelBubble") {
1782                break;
1783            }
1784            dispatch_at_phase(vm, ancestor, event, &event_type, 3 /* bubble */)?;
1785        }
1786    }
1787
1788    Ok(JSValue::from_bool(!event_flag(event, "defaultPrevented")))
1789}

1791/// Invokes the listeners/pending on-element handlers stored on `current` during
1792/// the given propagation phase.
1793///
1794/// `phase` is `1` (capture), `2` (target) or `3` (bubble). During capture only
1795/// capturing listeners run; during bubble only non-capturing listeners run; the
1796/// target runs both, plus any inline `on<type>` handler.
1797fn dispatch_at_phase(
1798    vm: &mut VM,
1799    current: &Rc<RefCell<JSObject>>,
1800    event: &Rc<RefCell<JSObject>>,
1801    event_type: &str,
1802    phase: u8,
1803) -> JSResult<()> {
1804    let current_js = JSValue::from_object(Rc::clone(current));
1805    event
1806        .borrow_mut()
1807        .set("currentTarget".to_string(), current_js.clone());
1808    event
1809        .borrow_mut()
1810        .set("eventPhase".to_string(), JSValue::from_number(phase as f64));
1811
1812    let dom_id = node_dom_id(&current_js).unwrap_or(0);
1813    if dom_id == 0 {
1814        return Ok(());
1815    }
1816    let listeners = with_host(vm, |host| {
1817        host.element_event_listeners
1818            .get(&dom_id)
1819            .and_then(|events| events.get(event_type))
1820            .cloned()
1821            .unwrap_or_default()
1822    })
1823    .unwrap_or_default();
1824
1825    if phase == 2 {
1826        let handler = current.borrow().get(&format!("on{event_type}"));
1827        if is_callable(&handler) {
1828            vm.call(
1829                handler,
1830                current_js.clone(),
1831                vec![JSValue::from_object(Rc::clone(event))],
1832            )?;
1833        }
1834    }
1835
1836    for (listener, capture) in listeners {
1837        if event_flag(event, "__orinium_immediate_propagation_stopped") {
1838            break;
1839        }
1840        if (phase == 1 && !capture) || (phase == 3 && capture) {
1841            continue;
1842        }
1843        vm.call(
1844            listener,
1845            current_js.clone(),
1846            vec![JSValue::from_object(Rc::clone(event))],
1847        )?;
1848    }
1849
1850    // stopImmediatePropagation halts all remaining propagation.
1851    if event_flag(event, "__orinium_immediate_propagation_stopped") {
1852        event
1853            .borrow_mut()
1854            .set("cancelBubble".to_string(), JSValue::from_bool(true));
1855    }

1872        return Ok(JSValue::undefined());

1889    event.set("defaultPrevented".to_string(), JSValue::from_bool(false));
1890    event.set("cancelBubble".to_string(), JSValue::from_bool(false));
1891    event.set("eventPhase".to_string(), JSValue::from_number(0.0));

1894        JSValue::from_native_function(super::events::event_prevent_default),

1902        JSValue::from_native_function(super::events::event_stop_immediate_propagation),
1903    );

1905
1906    // Build the ancestor path (target first) like element_dispatch_event.
1907    let target_obj = Rc::clone(&target);
1908    let mut path: Vec<Rc<RefCell<JSObject>>> = vec![Rc::clone(&target_obj)];

2467/// Collects the rows (tr) belonging to a table in DOM order, including the rows
2468/// nested inside its tbody/thead/tfoot sections but not rows of nested tables.
2469fn collect_table_rows(table: &NodeRef<HtmlNodeType>) -> Vec<NodeRef<HtmlNodeType>> {
2470    let mut rows = Vec::new();
2471    let children = table.borrow().children().to_vec();
2472    for child in children {
2473        match element_tag_name(&child).as_deref() {
2474            Some(tag @ ("tbody" | "thead" | "tfoot")) => {
2475                for tr in child.borrow().children() {
2476                    if element_tag_name(tr).as_deref() == Some("tr") {
2477                        rows.push(Rc::clone(tr));
2478                    }
2479                }
2480                let _ = tag;
2481            }
2482            Some("tr") => rows.push(Rc::clone(&child)),
2483            _ => {}
2484        }
2485    }
2486    rows
2487}

2610fn insert_row_impl(vm: &mut VM, args: Vec<JSValue>, section: bool) -> JSResult<JSValue> {
2611    let Some(node) = dom_node(vm, args.first().unwrap_or(&UNDEFINED)) else {
2612        return Ok(JSValue::null());
2613    };
2614    let index = args.get(1).and_then(JSValue::as_number).unwrap_or(-1.0);
2615    let tr = TreeNode::new(HtmlNodeType::Element {
2616        tag_name: "tr".to_string(),
2617        attributes: Vec::new(),
2618    });
2619
2620    if section {
2621        // Insert into the section's tr children.
2622        let section_children = collect_section_rows(&node);
2623        let ins = if index < 0.0 {
2624            section_children.len()
2625        } else {
2626            (index as usize).min(section_children.len())
2627        };
2628        let pos = ins_index_of(&node, node.borrow().children(), ins);
2629        TreeNode::insert_child_at(&node, pos, Rc::clone(&tr));
2630        mark_dom_dirty(vm);
2631        return Ok(expose_node(vm, tr).unwrap_or(JSValue::null()));
2632    }
2633
2634    // Insert into the table. If the position falls within a section, place the
2635    // row inside that section's tr children; otherwise append to the last
2636    // section (or the table directly if it has no sections).
2637    let rows = collect_table_rows(&node);
2638    let ins = if index < 0.0 {
2639        rows.len()
2640    } else {
2641        (index as usize).min(rows.len())
2642    };
2643    if ins < rows.len() {
2644        let Some(section_node) = rows[ins].borrow().parent() else {
2645            return Ok(JSValue::null());
2646        };
2647        let section_children = section_node.borrow().children().to_vec();
2648        let pos = ins_index_of(&section_node, &section_children, ins);
2649        TreeNode::insert_child_at(&section_node, pos, Rc::clone(&tr));
2650    } else if let Some(last_section) = node
2651        .borrow()
2652        .children()
2653        .iter()
2654        .rev()
2655        .find(|child| {
2656            matches!(
2657                element_tag_name(child).as_deref(),
2658                Some("tbody") | Some("thead") | Some("tfoot")
2659            )
2660        })
2661        .cloned()
2662    {
2663        let len = last_section.borrow().children().len();
2664        TreeNode::insert_child_at(&last_section, len, Rc::clone(&tr));
2665    } else {
2666        let len = node.borrow().children().len();
2667        TreeNode::insert_child_at(&node, len, Rc::clone(&tr));
2668    }
2669    mark_dom_dirty(vm);
2670    Ok(expose_node(vm, tr).unwrap_or(JSValue::null()))
2671}

3720/// The name shared by a radio group, which is scoped to the tree so that
3721/// same-named radios in different documents stay independent.
3722fn radio_group_name(node: &NodeRef<HtmlNodeType>) -> Option<String> {
3723    let name = node.borrow().value.get_attr("name")?.to_string();
3724    let root = tree_root_of(node);
3725    let mut stack = vec![root];
3726    let mut found = None;
3727    while let Some(current) = stack.pop() {
3728        let children = current.borrow().children().to_vec();
3729        for child in children {
3730            let is_radio = child.borrow().value.tag_name() == Some("input")
3731                && input_type(&child) == "radio"
3732                && child.borrow().value.get_attr("name") == Some(name.as_str());
3733            if is_radio && found.is_none() {
3734                found = Some(Rc::clone(&child));
3735            }
3736            stack.push(child);
3737        }
3738    }
3739    found.map(|peer| {
3740        peer.borrow()
3741            .value
3742            .get_attr("name")
3743            .unwrap_or("")
3744            .to_string()
3745    })
3746}

3748/// Applies `input.checked = value` semantics: the live state changes while the
3749/// `checked` attribute, which is `defaultChecked`, is left untouched. Checking a
3750/// radio also unchecks the rest of its group.
3751fn set_checked_state(node: &NodeRef<HtmlNodeType>, checked: bool) {
3752    if checked && input_type(node) == "radio" {
3753        let name = radio_group_name(node);
3754        if let Some(name) = name {
3755            let root = tree_root_of(node);
3756            let mut stack = vec![root];
3757            while let Some(current) = stack.pop() {
3758                let children = current.borrow().children().to_vec();
3759                for child in children {
3760                    if child.borrow().value.tag_name() == Some("input")
3761                        && input_type(&child) == "radio"
3762                        && child.borrow().value.get_attr("name") == Some(name.as_str())
3763                        && !Rc::ptr_eq(&child, node)
3764                    {
3765                        child.borrow().set_checkedness(false);
3766                    }
3767                    stack.push(child);
3768                }
3769            }
3770        }

4624        "arguments".to_string(),
