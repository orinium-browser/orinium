1//! Missing Web Platform API detector
2//!
3//! Investigation/diagnostic helper ("log mode"). Scripts probe a wide surface
4//! of Web Platform APIs that this engine has not implemented yet; when a page
5//! reaches an unimplemented API it fails with a generic `TypeError`. To make
6//! those gaps visible without breaking execution, this module emits a
7//! `log::info!` report of which global / document / Element.prototype APIs are
8//! *absent* at boot time.
9//!
10//! It is deliberately side-effect free: it only reads, never installs stubs or
11//! shadows already-present implementations. Call [`install_missing_api_report`]
12//! right after creating a [`crate::engine::js::JsRuntime`] in an investigation
13//! harness (`examples/yt.rs`, the youtube fetch test) and grep for
14//! `missing-api` in the logs.
15
16use crate::engine::js::JsRuntime;

60/// Methods probed on `document` (object style: `typeof document[name]`).
61pub(crate) const SUSPECT_DOCUMENT_METHODS: &[&str] = &[
62    "elementFromPoint",
63    "elementsFromPoint",
64    "execCommand",
65    "getSelection",
66    "pictureInPictureElement",
67];
68
69/// Methods probed on `Element.prototype` (object style:
70/// `typeof Element.prototype[name]`).
71pub(crate) const SUSPECT_ELEMENT_PROTOTYPE_METHODS: &[&str] = &[
72    "getAnimations",
73    "requestFullscreen",
74    "scrollIntoView",
75    "scrollTo",
76];
77
78/// Builds the JS prologue that reports which APIs are absent.
79fn prologue() -> String {
80    let mut code = String::new();
81    code.push_str("(function(){var missing = [];\n");
82    for name in SUSPECT_GLOBALS {
83        code.push_str(&format!(
84            "if (typeof {name} === 'undefined') missing.push('window.{name}');\n"
85        ));
86    }
87    for name in SUSPECT_DOCUMENT_METHODS {
88        code.push_str(&format!(
89            "if (typeof document.{name} === 'undefined') missing.push('document.{name}');\n"
90        ));
91    }
92    for name in SUSPECT_ELEMENT_PROTOTYPE_METHODS {
93        code.push_str(&format!(
94            "if (typeof Element !== 'undefined' && Element.prototype && typeof Element.prototype.{name} === 'undefined') missing.push('Element.prototype.{name}');\n"
95        ));
96    }
97    code.push_str("if (missing.length) console.warn('[missing-api] ' + missing.join(','));\n");
98    code.push_str("}).call(window);\n");
99    code
100}
101
102/// Runs the missing-API report on `runtime` and forwards each name to
103/// `log::info!(target: "MissingApi", ...)`.
104///
105/// Usage in a harness:
106/// ```no_run
107/// let (mut runtime, _dom) = ...;
108/// missing_api_detector::install_missing_api_report(&mut runtime);
109/// ```
110pub fn install_missing_api_report(runtime: &mut JsRuntime) {
111    runtime.run_script(&prologue());
112}
