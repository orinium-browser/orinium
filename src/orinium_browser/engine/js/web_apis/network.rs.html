32fn headers_constructor(_vm: &mut VM, args: Vec<JSValue>) -> JSResult<JSValue> {
33    let entries = args.get(1).map(extract_header_entries).unwrap_or_default();
34    Ok(JSValue::Object(make_headers(entries, false)))

37pub(crate) fn make_headers(
38    entries: Vec<(String, String)>,
39    immutable: bool,
40) -> Rc<RefCell<JSObject>> {
41    let data = Rc::new(RefCell::new(JSObject::new()));
42    for (name, value) in entries {
43        append_header_value(&mut data.borrow_mut(), &name, &value);
44    }
45
46    let mut headers = JSObject::new();
47    headers.set(HEADERS_DATA.to_string(), JSValue::Object(data));
48    headers.set(HEADERS_IMMUTABLE.to_string(), JSValue::Boolean(immutable));
49    headers.set("get".to_string(), JSValue::NativeFunction(headers_get));
50    headers.set("has".to_string(), JSValue::NativeFunction(headers_has));
51    headers.set("set".to_string(), JSValue::NativeFunction(headers_set));
52    headers.set(
53        "append".to_string(),
54        JSValue::NativeFunction(headers_append),
55    );
56    headers.set(
57        "delete".to_string(),
58        JSValue::NativeFunction(headers_delete),
59    );
60    Rc::new(RefCell::new(headers))
61}

63pub(crate) fn extract_header_entries(value: &JSValue) -> Vec<(String, String)> {
64    let JSValue::Object(object) = value else {
65        return Vec::new();
66    };
67    let object = object.borrow();
68    if let JSValue::Object(data) = object.get(HEADERS_DATA) {
69        let data = data.borrow();
70        return data
71            .keys()
72            .into_iter()
73            .map(|name| {
74                let value = data.get(&name).to_string();
75                (name, value)
76            })
77            .collect();
78    }
79    object
80        .keys()
81        .into_iter()
82        .map(|name| {
83            let value = object.get(&name).to_string();
84            (name, value)
85        })
86        .collect()
87}

89fn headers_data(args: &[JSValue]) -> JSResult<Rc<RefCell<JSObject>>> {
90    let Some(JSValue::Object(headers)) = args.first() else {
91        return Err(JSError::TypeError(
92            "Headers method called on incompatible receiver".to_string(),
93        ));
94    };
95    let data = headers.borrow().get(HEADERS_DATA);
96    match data {
97        JSValue::Object(data) => Ok(data),
98        _ => Err(JSError::TypeError(
99            "Headers method called on incompatible receiver".to_string(),
100        )),
101    }
102}

104fn ensure_headers_mutable(args: &[JSValue]) -> JSResult<()> {
105    let Some(JSValue::Object(headers)) = args.first() else {
106        return Err(JSError::TypeError(
107            "Headers method called on incompatible receiver".to_string(),
108        ));
109    };
110    if matches!(
111        headers.borrow().get(HEADERS_IMMUTABLE),
112        JSValue::Boolean(true)
113    ) {
114        return Err(JSError::TypeError(
115            "Response headers are immutable".to_string(),
116        ));
117    }
118    Ok(())

128fn normalize_header_name(name: &str) -> String {

133    value.trim().to_string()

136fn append_header_value(data: &mut JSObject, name: &str, value: &str) {
137    let name = normalize_header_name(name);
138    if name.is_empty() {
139        return;
140    }
141    let value = normalize_header_value(value);
142    let combined = match data.get(&name) {
143        JSValue::Undefined => value,
144        current => format!("{}, {}", current, value),
145    };
146    data.set(name, JSValue::String(combined));
147}

149fn headers_get(_vm: &mut VM, args: Vec<JSValue>) -> JSResult<JSValue> {
150    let data = headers_data(&args)?;
151    let name = normalize_header_name(&header_argument(&args, 1, "name")?);
152    Ok(match data.borrow().get(&name) {
153        JSValue::Undefined => JSValue::Null,
154        value => value,
155    })
156}

158fn headers_has(_vm: &mut VM, args: Vec<JSValue>) -> JSResult<JSValue> {
159    let data = headers_data(&args)?;
160    let name = normalize_header_name(&header_argument(&args, 1, "name")?);
161    let has = data.borrow().has_own_property(&name);
162    Ok(JSValue::Boolean(has))
163}

165fn headers_set(_vm: &mut VM, args: Vec<JSValue>) -> JSResult<JSValue> {
166    ensure_headers_mutable(&args)?;
167    let data = headers_data(&args)?;
168    let name = normalize_header_name(&header_argument(&args, 1, "name")?);
169    let value = normalize_header_value(&header_argument(&args, 2, "value")?);
170    if !name.is_empty() {
171        data.borrow_mut().set(name, JSValue::String(value));
172    }
173    Ok(JSValue::Undefined)
174}

176fn headers_append(_vm: &mut VM, args: Vec<JSValue>) -> JSResult<JSValue> {
177    ensure_headers_mutable(&args)?;
178    let data = headers_data(&args)?;
179    let name = header_argument(&args, 1, "name")?;
180    let value = header_argument(&args, 2, "value")?;
181    append_header_value(&mut data.borrow_mut(), &name, &value);
182    Ok(JSValue::Undefined)
183}

185fn headers_delete(_vm: &mut VM, args: Vec<JSValue>) -> JSResult<JSValue> {
186    ensure_headers_mutable(&args)?;
187    let data = headers_data(&args)?;
188    let name = normalize_header_name(&header_argument(&args, 1, "name")?);
189    data.borrow_mut().delete(&name);
190    Ok(JSValue::Undefined)

212fn request_constructor(_vm: &mut VM, args: Vec<JSValue>) -> JSResult<JSValue> {
213    let Some(input) = args.get(1) else {
214        return Err(JSError::TypeError("Request input is required".to_string()));
215    };
216    let parts = request_parts(input, args.get(2));
217    Ok(JSValue::Object(make_request(parts)))

220fn make_request(parts: RequestParts) -> Rc<RefCell<JSObject>> {
221    let mut request = JSObject::new();
222    request.define_property(
223        "url".to_string(),
224        Property::read_only(JSValue::String(parts.url)),
225    );
226    request.define_property(
227        "method".to_string(),
228        Property::read_only(JSValue::String(parts.method)),
229    );
230    request.define_property(
231        "headers".to_string(),
232        Property::read_only(JSValue::Object(make_headers(parts.headers, false))),
233    );
234    request.set(
235        REQUEST_BODY.to_string(),
236        JSValue::String(String::from_utf8_lossy(&parts.body).into_owned()),
237    );
238    request.set(REQUEST_MARKER.to_string(), JSValue::Boolean(true));
239    Rc::new(RefCell::new(request))
240}

242pub(crate) fn request_parts(input: &JSValue, init: Option<&JSValue>) -> RequestParts {
243    let mut parts = match input {
244        JSValue::Object(request)
245            if matches!(request.borrow().get(REQUEST_MARKER), JSValue::Boolean(true)) =>
246        {
247            let request = request.borrow();
248            RequestParts {
249                url: request.get("url").to_string(),
250                method: request.get("method").to_string(),
251                headers: extract_header_entries(&request.get("headers")),
252                body: match request.get(REQUEST_BODY) {
253                    JSValue::String(body) => body.into_bytes(),
254                    _ => Vec::new(),
255                },
256            }
257        }
258        value => RequestParts {
259            url: value.to_string(),
260            method: "GET".to_string(),
261            headers: Vec::new(),
262            body: Vec::new(),
263        },
264    };
265    apply_request_init(&mut parts, init);
266    parts
267}
268
269fn apply_request_init(parts: &mut RequestParts, init: Option<&JSValue>) {
270    let Some(JSValue::Object(init)) = init else {
271        return;
272    };
273    let init = init.borrow();
274    if init.has_own_property("method") {
275        match init.get("method") {
276            JSValue::Undefined | JSValue::Null => {}
277            value => parts.method = value.to_string().to_ascii_uppercase(),
278        }
279    }
280    if init.has_own_property("headers") {
281        parts.headers = extract_header_entries(&init.get("headers"));
282    }
283    if init.has_own_property("body") {
284        parts.body = match init.get("body") {
285            JSValue::Undefined | JSValue::Null => Vec::new(),
286            value => value.to_string().into_bytes(),
287        };
288    }
289}

314fn xml_http_request_constructor(_vm: &mut VM, _args: Vec<JSValue>) -> JSResult<JSValue> {
315    let mut xhr = JSObject::new();
316    xhr.set("readyState".to_string(), JSValue::Number(0.0));
317    xhr.set("status".to_string(), JSValue::Number(0.0));
318    xhr.set("statusText".to_string(), JSValue::String(String::new()));
319    xhr.set("responseText".to_string(), JSValue::String(String::new()));
320    xhr.set("response".to_string(), JSValue::String(String::new()));
321    xhr.set("responseType".to_string(), JSValue::String(String::new()));
322    xhr.set("withCredentials".to_string(), JSValue::Boolean(false));
323    xhr.set(
324        XHR_HEADERS.to_string(),
325        JSValue::Object(Rc::new(RefCell::new(JSObject::new()))),
326    );
327    xhr.set(
328        "open".to_string(),
329        JSValue::NativeFunction(xml_http_request_open),
330    );
331    xhr.set(
332        "send".to_string(),
333        JSValue::NativeFunction(xml_http_request_send),
334    );
335    xhr.set(
336        "setRequestHeader".to_string(),
337        JSValue::NativeFunction(xml_http_request_set_request_header),
338    );
339    xhr.set(
340        "getAllResponseHeaders".to_string(),
341        JSValue::NativeFunction(xml_http_request_get_all_response_headers),
342    );
343    // TODO: Cancel the in-flight network request and dispatch XMLHttpRequest abort events.
344    xhr.set("abort".to_string(), JSValue::NativeFunction(noop));
345    Ok(JSValue::Object(Rc::new(RefCell::new(xhr))))
346}

348fn xml_http_request_open(_vm: &mut VM, args: Vec<JSValue>) -> JSResult<JSValue> {
349    let Some(JSValue::Object(xhr)) = args.first() else {
350        return Err(JSError::TypeError(
351            "invalid XMLHttpRequest receiver".to_string(),
352        ));
353    };
354    let method = args
355        .get(1)
356        .cloned()
357        .unwrap_or(JSValue::String("GET".to_string()))
358        .to_string();
359    let url = args
360        .get(2)
361        .cloned()
362        .unwrap_or(JSValue::Undefined)
363        .to_string();
364    let mut xhr = xhr.borrow_mut();
365    xhr.set(
366        XHR_METHOD.to_string(),
367        JSValue::String(method.to_ascii_uppercase()),
368    );
369    xhr.set(XHR_URL.to_string(), JSValue::String(url));
370    xhr.set("readyState".to_string(), JSValue::Number(1.0));
371    Ok(JSValue::Undefined)
372}

396fn xml_http_request_send(vm: &mut VM, args: Vec<JSValue>) -> JSResult<JSValue> {
397    let Some(JSValue::Object(xhr)) = args.first() else {
398        return Err(JSError::TypeError(
399            "invalid XMLHttpRequest receiver".to_string(),
400        ));
401    };
402    let (url, method, headers) = {
403        let xhr_ref = xhr.borrow();
404        let headers = match xhr_ref.get(XHR_HEADERS) {
405            JSValue::Object(headers) => headers
406                .borrow()
407                .keys()
408                .into_iter()
409                .map(|name| {
410                    let value = headers.borrow().get(&name).to_string();
411                    (name, value)
412                })
413                .collect(),
414            _ => Vec::new(),
415        };
416        (
417            xhr_ref.get(XHR_URL).to_string(),
418            xhr_ref.get(XHR_METHOD).to_string(),
419            headers,
420        )
421    };
422    let body = match args.get(1) {
423        Some(JSValue::Undefined | JSValue::Null) | None => Vec::new(),
424        Some(value) => value.to_string().into_bytes(),
425    };
426    with_host_mut(vm, |host| {
427        host.next_fetch_id += 1;
428        let id = host.next_fetch_id;
429        host.xhr_requests.insert(id, Rc::clone(xhr));
430        host.fetch_requests.push(JsFetchRequest {
431            id,
432            url,
433            method,
434            headers,
435            body,
436        });
437    })
438    .ok_or_else(|| JSError::InternalError("XMLHttpRequest host is unavailable".to_string()))?;
439    Ok(JSValue::Undefined)
440}

453pub(crate) fn resolve_xml_http_request(
454    engine: &mut pixi_byte::JSEngine,
455    xhr: Rc<RefCell<JSObject>>,
456    response: JsFetchResponse,
457) {
458    let body = String::from_utf8_lossy(&response.body).into_owned();
459    let headers = response
460        .headers
461        .iter()
462        .map(|(name, value)| format!("{name}: {value}\r\n"))
463        .collect::<String>();
464    {
465        let mut xhr = xhr.borrow_mut();
466        xhr.set("readyState".to_string(), JSValue::Number(4.0));
467        xhr.set(
468            "status".to_string(),
469            JSValue::Number(response.status as f64),
470        );
471        xhr.set(
472            "statusText".to_string(),
473            JSValue::String(response.status_text),
474        );
475        xhr.set("responseURL".to_string(), JSValue::String(response.url));
476        xhr.set("responseText".to_string(), JSValue::String(body.clone()));
477        xhr.set("response".to_string(), JSValue::String(body));
478        xhr.set(
479            "__orinium_xhr_response_headers".to_string(),
480            JSValue::String(headers),
481        );
482    }
483    for name in ["onreadystatechange", "onload"] {
484        let handler = xhr.borrow().get(name);
485        if is_callable(&handler) {
486            let _ = engine.call(handler, JSValue::Object(Rc::clone(&xhr)), Vec::new());
487        }
488    }
489}

491fn fetch(vm: &mut VM, args: Vec<JSValue>) -> JSResult<JSValue> {
492    let input = args.get(1).cloned().unwrap_or(JSValue::Undefined);
493    let RequestParts {
494        url,
495        method,
496        headers,
497        body,
498    } = request_parts(&input, args.get(2));
499    let promise_constructor = vm.global_object.borrow().get("Promise");
500    let JSValue::Object(constructor) = &promise_constructor else {
501        return Err(JSError::InternalError(
502            "Promise constructor is unavailable".to_string(),
503        ));
504    };
505    let construct = constructor.borrow().get("__construct__");
506    let _ = with_host_mut(vm, |host| host.constructing_fetch_capability = None);
507    let promise = vm.call(
508        construct,
509        promise_constructor,
510        vec![JSValue::NativeFunction(capture_fetch_capability)],
511    )?;
512    let capability = with_host_mut(vm, |host| host.constructing_fetch_capability.take())
513        .flatten()
514        .ok_or_else(|| JSError::InternalError("Failed to create fetch Promise".to_string()))?;
515
516    let _ = with_host_mut(vm, |host| {
517        host.next_fetch_id += 1;
518        let id = host.next_fetch_id;
519        host.fetch_capabilities.insert(id, capability);
520        host.fetch_requests.push(JsFetchRequest {
521            id,
522            url,
523            method,
524            headers,
525            body,
526        });
527    });
528    Ok(promise)
529}

531fn capture_fetch_capability(vm: &mut VM, args: Vec<JSValue>) -> JSResult<JSValue> {
532    let resolve = args.get(1).cloned().unwrap_or(JSValue::Undefined);
533    let reject = args.get(2).cloned().unwrap_or(JSValue::Undefined);
534    let Some(()) = with_host_mut(vm, |host| {
535        host.constructing_fetch_capability = Some(JsFetchCapability { resolve, reject });
536    }) else {
537        return Err(JSError::InternalError(
538            "Fetch host state is unavailable".to_string(),
539        ));
540    };
541    Ok(JSValue::Undefined)
542}

544pub(crate) fn make_fetch_response(response: JsFetchResponse) -> Rc<RefCell<JSObject>> {
545    let body_bytes = response.body.clone();
546    let mut object = JSObject::new();
547    object.define_property(
548        "headers".to_string(),
549        Property::read_only(JSValue::Object(make_headers(response.headers, true))),
550    );
551    object.define_property(
552        "ok".to_string(),
553        Property::read_only(JSValue::Boolean((200..=299).contains(&response.status))),
554    );
555    object.define_property(
556        "status".to_string(),
557        Property::read_only(JSValue::Number(response.status as f64)),
558    );
559    object.define_property(
560        "statusText".to_string(),
561        Property::read_only(JSValue::String(response.status_text)),
562    );
563    object.define_property(
564        "redirected".to_string(),
565        Property::read_only(JSValue::Boolean(response.redirected)),
566    );
567    object.define_property(
568        "bodyUsed".to_string(),
569        Property {
570            value: JSValue::Undefined,
571            enumerable: true,
572            writable: false,
573            configurable: false,
574            getter: Some(JSValue::NativeFunction(fetch_response_body_used)),
575            setter: None,
576        },
577    );
578    object.define_property(
579        "url".to_string(),
580        Property::read_only(JSValue::String(response.url)),
581    );
582    object.define_property(
583        "text".to_string(),
584        Property::read_only(JSValue::NativeFunction(fetch_response_text)),
585    );
586    object.define_property(
587        "json".to_string(),
588        Property::read_only(JSValue::NativeFunction(fetch_response_json)),
589    );
590    object.define_property(
591        "arrayBuffer".to_string(),
592        Property::read_only(JSValue::NativeFunction(fetch_response_array_buffer)),
593    );
594    object.set(
595        "__orinium_response_body".to_string(),
596        JSValue::String(String::from_utf8_lossy(&response.body).into_owned()),
597    );
598    object.set(RESPONSE_BODY_USED.to_string(), JSValue::Boolean(false));
599    object.set(
600        RESPONSE_BODY_BYTES.to_string(),
601        JSArray::from_vec(
602            body_bytes
603                .into_iter()
604                .map(|byte| JSValue::Number(byte as f64))
605                .collect(),
606        )
607        .to_object(),
608    );
609    Rc::new(RefCell::new(object))
610}

612fn fetch_response_text(vm: &mut VM, args: Vec<JSValue>) -> JSResult<JSValue> {
613    let body = match consume_response_body(vm, &args, "text")? {
614        Ok(body) => body,
615        Err(rejection) => return Ok(rejection),
616    };
617    settle_promise(vm, "resolve", JSValue::String(body))
618}

620fn fetch_response_json(vm: &mut VM, args: Vec<JSValue>) -> JSResult<JSValue> {
621    let body = match consume_response_body(vm, &args, "json")? {
622        Ok(body) => body,
623        Err(rejection) => return Ok(rejection),
624    };
625    match serde_json::from_str::<serde_json::Value>(&body) {
626        Ok(value) => settle_promise(vm, "resolve", json_to_js_value(value)),
627        Err(error) => settle_promise(
628            vm,
629            "reject",
630            JSValue::String(format!("Failed to parse JSON: {error}")),
631        ),
632    }
633}

635fn fetch_response_array_buffer(vm: &mut VM, args: Vec<JSValue>) -> JSResult<JSValue> {
636    let response = match consume_response_object(vm, &args, "arrayBuffer") {
637        Ok(response) => response,
638        Err(JSError::Thrown(rejection)) => return Ok(rejection),
639        Err(error) => return Err(error),
640    };
641    let bytes = response.borrow().get(RESPONSE_BODY_BYTES);
642    settle_promise(vm, "resolve", make_array_buffer_from_value(&bytes))
643}

645fn fetch_response_body_used(_vm: &mut VM, args: Vec<JSValue>) -> JSResult<JSValue> {
646    let Some(JSValue::Object(response)) = args.first() else {
647        return Err(JSError::TypeError(
648            "Response.bodyUsed called on incompatible receiver".to_string(),
649        ));
650    };

652}

654fn consume_response_body(
655    vm: &mut VM,
656    args: &[JSValue],
657    method: &str,
658) -> JSResult<Result<String, JSValue>> {
659    let response = match consume_response_object(vm, args, method) {
660        Ok(response) => response,
661        Err(JSError::Thrown(value)) => return Ok(Err(value)),
662        Err(error) => return Err(error),
663    };
664    let response = response.borrow();
665    match response.get("__orinium_response_body") {
666        JSValue::String(body) => Ok(Ok(body)),
667        _ => Err(JSError::InternalError(
668            "Response body is unavailable".to_string(),
669        )),
670    }
671}
672
673fn consume_response_object(
674    vm: &mut VM,
675    args: &[JSValue],
676    method: &str,
677) -> JSResult<Rc<RefCell<JSObject>>> {
678    let Some(JSValue::Object(response)) = args.first() else {
679        return Err(JSError::TypeError(format!(
680            "Response.{method} called on incompatible receiver"
681        )));
682    };
683    if matches!(
684        response.borrow().get(RESPONSE_BODY_USED),
685        JSValue::Boolean(true)
686    ) {
687        let rejection = settle_promise(
688            vm,
689            "reject",
690            JSValue::String("Response body has already been consumed".to_string()),
691        )?;
692        return Err(JSError::Thrown(rejection));
693    }
694    response
695        .borrow_mut()
696        .set(RESPONSE_BODY_USED.to_string(), JSValue::Boolean(true));
697    Ok(Rc::clone(response))
698}
699
700fn settle_promise(vm: &mut VM, method: &str, value: JSValue) -> JSResult<JSValue> {
701    let promise = vm.global_object.borrow().get("Promise");
702    let JSValue::Object(constructor) = &promise else {
703        return Err(JSError::InternalError(
704            "Promise constructor is unavailable".to_string(),
705        ));
706    };
707    let settle = constructor.borrow().get(method);
708    vm.call(settle, promise, vec![value])
709}

711fn json_to_js_value(value: serde_json::Value) -> JSValue {
712    match value {
713        serde_json::Value::Null => JSValue::Null,
714        serde_json::Value::Bool(value) => JSValue::Boolean(value),
715        serde_json::Value::Number(value) => JSValue::Number(value.as_f64().unwrap_or(f64::NAN)),
716        serde_json::Value::String(value) => JSValue::String(value),
717        serde_json::Value::Array(values) => {
718            JSArray::from_vec(values.into_iter().map(json_to_js_value).collect()).to_object()
719        }
720        serde_json::Value::Object(properties) => {
721            let mut object = JSObject::new();
722            for (key, value) in properties {
723                object.set(key, json_to_js_value(value));
724            }
725            JSValue::Object(Rc::new(RefCell::new(object)))
726        }
727    }
