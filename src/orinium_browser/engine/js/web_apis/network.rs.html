38pub(crate) fn make_headers(
39    entries: Vec<(String, String)>,
40    immutable: bool,
41) -> Rc<RefCell<JSObject>> {
42    let data = Rc::new(RefCell::new(JSObject::new()));
43    for (name, value) in entries {
44        append_header_value(&mut data.borrow_mut(), &name, &value);
45    }
46
47    let mut headers = JSObject::new();
48    headers.set(HEADERS_DATA.to_string(), JSValue::from_object(data));
49    headers.set(HEADERS_IMMUTABLE.to_string(), JSValue::from_bool(immutable));
50    headers.set(
51        "get".to_string(),
52        JSValue::from_native_function(headers_get),
53    );
54    headers.set(
55        "has".to_string(),
56        JSValue::from_native_function(headers_has),
57    );
58    headers.set(
59        "set".to_string(),
60        JSValue::from_native_function(headers_set),
61    );
62    headers.set(
63        "append".to_string(),
64        JSValue::from_native_function(headers_append),
65    );
66    headers.set(
67        "delete".to_string(),
68        JSValue::from_native_function(headers_delete),
69    );
70    Rc::new(RefCell::new(headers))
71}

73pub(crate) fn extract_header_entries(value: &JSValue) -> Vec<(String, String)> {
74    let Some(object) = value.as_object() else {
75        return Vec::new();
76    };
77    let object = object.borrow();
78    if let Some(data) = object.get(HEADERS_DATA).as_object() {
79        let data = data.borrow();
80        return data
81            .keys()
82            .into_iter()
83            .map(|name| {
84                let value = data.get(&name).to_string();
85                (name, value)
86            })
87            .collect();
88    }
89    object
90        .keys()
91        .into_iter()
92        .map(|name| {
93            let value = object.get(&name).to_string();
94            (name, value)
95        })
96        .collect()
97}

99fn headers_data(args: &[JSValue]) -> JSResult<Rc<RefCell<JSObject>>> {

140fn append_header_value(data: &mut JSObject, name: &str, value: &str) {
141    let name = normalize_header_name(name);
142    if name.is_empty() {
143        return;
144    }
145    let value = normalize_header_value(value);
146    let combined = if data.get(&name).is_undefined() {
147        value
148    } else {
149        format!("{}, {}", data.get(&name), value)
150    };
151    data.set(name, JSValue::from_string(combined));
152}

155    let data = headers_data(&args)?;
156    let name = normalize_header_name(&header_argument(&args, 1, "name")?);
157    let value = data.borrow().get(&name);
158    Ok(if value.is_undefined() {
159        JSValue::null()
160    } else {
161        value

172fn headers_set(_vm: &mut VM, args: Vec<JSValue>) -> JSResult<JSValue> {
173    ensure_headers_mutable(&args)?;
174    let data = headers_data(&args)?;
175    let name = normalize_header_name(&header_argument(&args, 1, "name")?);
176    let value = normalize_header_value(&header_argument(&args, 2, "value")?);
177    if !name.is_empty() {
178        data.borrow_mut().set(name, JSValue::from_string(value));
179    }
180    Ok(JSValue::undefined())

220fn request_constructor(_vm: &mut VM, args: Vec<JSValue>) -> JSResult<JSValue> {
221    let Some(input) = args.get(1) else {
222        return Err(JSError::TypeError("Request input is required".to_string()));

228fn make_request(parts: RequestParts) -> Rc<RefCell<JSObject>> {
229    let mut request = JSObject::new();
230    request.define_property(
231        "url".to_string(),
232        Property::read_only(JSValue::from_string(parts.url)),
233    );
234    request.define_property(
235        "method".to_string(),
236        Property::read_only(JSValue::from_string(parts.method)),
237    );
238    request.define_property(
239        "headers".to_string(),
240        Property::read_only(JSValue::from_object(make_headers(parts.headers, false))),
241    );
242    if let Some(signal) = parts.signal {
243        request.define_property("signal".to_string(), Property::read_only(signal));
244    }
245    request.set(
246        REQUEST_BODY.to_string(),
247        JSValue::from_string(String::from_utf8_lossy(&parts.body).into_owned()),
248    );
249    request.set(REQUEST_MARKER.to_string(), JSValue::from_bool(true));
250    Rc::new(RefCell::new(request))
251}

253pub(crate) fn request_parts(input: &JSValue, init: Option<&JSValue>) -> RequestParts {
254    let mut parts = if let Some(request) = input.as_object()
255        && request.borrow().get(REQUEST_MARKER).as_boolean() == Some(true)
256    {
257        let request = request.borrow();
258        RequestParts {
259            url: request.get("url").to_string(),
260            method: request.get("method").to_string(),
261            headers: extract_header_entries(&request.get("headers")),
262            body: request
263                .get(REQUEST_BODY)
264                .as_string_owned()
265                .map(|body| body.into_bytes())
266                .unwrap_or_default(),
267            signal: request
268                .get("signal")
269                .as_object()
270                .map(|_| request.get("signal")),
271        }
272    } else {
273        RequestParts {
274            url: input.to_string(),
275            method: "GET".to_string(),
276            headers: Vec::new(),
277            body: Vec::new(),
278            signal: None,
279        }
280    };
281    apply_request_init(&mut parts, init);
282    parts
283}

285fn apply_request_init(parts: &mut RequestParts, init: Option<&JSValue>) {
286    let Some(init) = init.and_then(JSValue::as_object) else {
287        return;
288    };
289    let init = init.borrow();
290    if init.has_own_property("method") {
291        let method = init.get("method");
292        if !method.is_undefined() && !method.is_null() {
293            parts.method = method.to_string().to_ascii_uppercase();
294        }
295    }
296    if init.has_own_property("headers") {
297        parts.headers = extract_header_entries(&init.get("headers"));
298    }
299    if init.has_own_property("body") {
300        let body = init.get("body");
301        if body.is_undefined() || body.is_null() {
302            parts.body = Vec::new();
303        } else {
304            parts.body = body.to_string().into_bytes();
305        }
306    }
307    if init.has_own_property("signal") {
308        let signal = init.get("signal");
309        parts.signal = if signal.is_undefined() || signal.is_null() {
310            None
311        } else {
312            Some(signal)
313        };
314    }
315}

466fn xml_http_request_send(vm: &mut VM, args: Vec<JSValue>) -> JSResult<JSValue> {
467    let Some(xhr) = args.first().and_then(JSValue::as_object) else {
468        return Err(JSError::TypeError(
469            "invalid XMLHttpRequest receiver".to_string(),
470        ));
471    };
472    let (url, method, headers) = {
473        let xhr_ref = xhr.borrow();
474        let headers = xhr_ref
475            .get(XHR_HEADERS)
476            .as_object()
477            .map(|headers| {
478                headers
479                    .borrow()
480                    .keys()
481                    .into_iter()
482                    .map(|name| {
483                        let value = headers.borrow().get(&name).to_string();
484                        (name, value)
485                    })
486                    .collect()
487            })
488            .unwrap_or_default();
489        (
490            xhr_ref.get(XHR_URL).to_string(),
491            xhr_ref.get(XHR_METHOD).to_string(),
492            headers,
493        )
494    };
495    let body = match args.get(1) {
496        Some(value) if value.is_undefined() || value.is_null() => Vec::new(),
497        Some(value) => value.to_string().into_bytes(),
498        None => Vec::new(),
499    };
500    with_host_mut(vm, |host| {
501        host.next_fetch_id += 1;
502        let id = host.next_fetch_id;
503        host.xhr_requests.insert(id, Rc::clone(&xhr));
504        host.fetch_requests.push(JsFetchRequest {
505            id,
506            url,
507            method,
508            headers,
509            body,
510        });
511    })
512    .ok_or_else(|| JSError::InternalError("XMLHttpRequest host is unavailable".to_string()))?;
513    Ok(JSValue::undefined())
514}

528pub(crate) fn resolve_xml_http_request(
529    engine: &mut pixi_byte::JSEngine,
530    xhr: Rc<RefCell<JSObject>>,
531    response: JsFetchResponse,
532) {
533    let body = String::from_utf8_lossy(&response.body).into_owned();
534    {
535        let mut xhr = xhr.borrow_mut();
536        xhr.set("readyState".to_string(), JSValue::from_number(4.0));
537        xhr.set(
538            "status".to_string(),
539            JSValue::from_number(response.status as f64),
540        );
541        xhr.set(
542            "statusText".to_string(),
543            JSValue::from_string(response.status_text),
544        );
545        xhr.set(
546            "responseURL".to_string(),
547            JSValue::from_string(response.url),
548        );
549        xhr.set(
550            "responseText".to_string(),
551            JSValue::from_string(body.clone()),
552        );
553        xhr.set("response".to_string(), JSValue::from_string(body));
554        let sanitized = sanitize_response_headers(&response.headers)
555            .iter()
556            .map(|(name, value)| format!("{name}: {value}\r\n"))
557            .collect::<String>();
558        xhr.set(
559            "__orinium_xhr_response_headers".to_string(),
560            JSValue::from_string(sanitized),
561        );
562    }
563    for name in ["onreadystatechange", "onload"] {
564        let handler = xhr.borrow().get(name);
565        if is_callable(&handler) {
566            let _ = engine.call(handler, JSValue::from_object(Rc::clone(&xhr)), Vec::new());
567        }
568    }
569}

571fn fetch(vm: &mut VM, args: Vec<JSValue>) -> JSResult<JSValue> {
572    let input = args.get(1).cloned().unwrap_or(JSValue::undefined());
573    let RequestParts {
574        url,
575        method,
576        headers,
577        body,
578        signal,
579    } = request_parts(&input, args.get(2));
580    let promise_constructor = vm.global_object.borrow().get("Promise");
581    let Some(constructor) = promise_constructor.as_object() else {
582        return Err(JSError::InternalError(
583            "Promise constructor is unavailable".to_string(),
584        ));
585    };
586    let construct = constructor.borrow().get("__construct__");
587    let _ = with_host_mut(vm, |host| host.constructing_fetch_capability = None);
588    let promise = vm.call(
589        construct,
590        promise_constructor,
591        vec![JSValue::from_native_function(capture_fetch_capability)],
592    )?;
593    let capability = with_host_mut(vm, |host| host.constructing_fetch_capability.take())
594        .flatten()
595        .ok_or_else(|| JSError::InternalError("Failed to create fetch Promise".to_string()))?;
596
597    // A pre-aborted signal rejects the fetch without queueing a request.
598    if let Some(signal) = &signal
599        && signal_is_aborted(signal)
600    {
601        let reason = signal_abort_reason(signal);
602        if is_callable(&capability.reject) {
603            vm.call(capability.reject, JSValue::undefined(), vec![reason])?;
604        }
605        return Ok(promise);
606    }
607
608    let _ = with_host_mut(vm, |host| {
609        host.next_fetch_id += 1;
610        let id = host.next_fetch_id;
611        host.fetch_capabilities.insert(id, capability);
612        if let Some(signal) = signal.as_ref().and_then(JSValue::as_object) {
613            host.fetch_signals.insert(id, signal);
614        }
615        host.fetch_requests.push(JsFetchRequest {
616            id,
617            url,
618            method,
619            headers,
620            body,
621        });
622    });
623    Ok(promise)
624}

626fn capture_fetch_capability(vm: &mut VM, args: Vec<JSValue>) -> JSResult<JSValue> {
627    let resolve = args.get(1).cloned().unwrap_or(JSValue::undefined());
628    let reject = args.get(2).cloned().unwrap_or(JSValue::undefined());
629    let Some(()) = with_host_mut(vm, |host| {
630        host.constructing_fetch_capability = Some(JsFetchCapability { resolve, reject });
631    }) else {
632        return Err(JSError::InternalError(
633            "Fetch host state is unavailable".to_string(),
634        ));
635    };
636    Ok(JSValue::undefined())
637}

639fn sanitize_response_headers(headers: &[(String, String)]) -> Vec<(String, String)> {
640    headers
641        .iter()
642        .filter(|(name, _)| {
643            !name.eq_ignore_ascii_case("set-cookie") && !name.eq_ignore_ascii_case("set-cookie2")
644        })
645        .map(|(name, value)| (name.clone(), value.clone()))
646        .collect()
647}

649pub(crate) fn make_fetch_response(response: JsFetchResponse) -> Rc<RefCell<JSObject>> {
650    let body_bytes = response.body.clone();
651    // TODO: Filter cross-origin response headers according to Access-Control-Expose-Headers.
652    let sanitized_headers = sanitize_response_headers(&response.headers);
653    let mut object = JSObject::new();
654    object.set(RESPONSE_MARKER.to_string(), JSValue::from_bool(true));
655    object.define_property(
656        "headers".to_string(),
657        Property::read_only(JSValue::from_object(make_headers(sanitized_headers, true))),
658    );
659    object.define_property(
660        "ok".to_string(),
661        Property::read_only(JSValue::from_bool((200..=299).contains(&response.status))),
662    );
663    object.define_property(
664        "status".to_string(),
665        Property::read_only(JSValue::from_number(response.status as f64)),
666    );
667    object.define_property(
668        "statusText".to_string(),
669        Property::read_only(JSValue::from_string(response.status_text)),
670    );
671    object.define_property(
672        "redirected".to_string(),
673        Property::read_only(JSValue::from_bool(response.redirected)),
674    );
675    object.define_property(
676        "bodyUsed".to_string(),
677        Property {
678            value: JSValue::undefined(),
679            enumerable: true,
680            writable: false,
681            configurable: false,
682            getter: Some(JSValue::from_native_function(fetch_response_body_used)),
683            setter: None,
684        },
685    );
686    object.define_property(
687        "url".to_string(),
688        Property::read_only(JSValue::from_string(response.url)),
689    );
690    object.define_property(
691        "text".to_string(),
692        Property::read_only(JSValue::from_native_function(fetch_response_text)),
693    );
694    object.define_property(
695        "json".to_string(),
696        Property::read_only(JSValue::from_native_function(fetch_response_json)),
697    );
698    object.define_property(
699        "arrayBuffer".to_string(),
700        Property::read_only(JSValue::from_native_function(fetch_response_array_buffer)),
701    );
702    object.set(
703        "__orinium_response_body".to_string(),
704        JSValue::from_string(String::from_utf8_lossy(&response.body).into_owned()),
705    );
706    object.set(RESPONSE_BODY_USED.to_string(), JSValue::from_bool(false));
707    object.set(
708        RESPONSE_BODY_BYTES.to_string(),
709        JSArray::from_vec(
710            body_bytes
711                .into_iter()
712                .map(|byte| JSValue::from_number(byte as f64))
713                .collect(),
714        )
715        .to_object(),
716    );
717    Rc::new(RefCell::new(object))
718}

720// --- Response constructor ---
721
722const RESPONSE_MARKER: &str = "__orinium_response";
723
724/// `new Response(body?, init?)` — builds a Response from a string or Blob body
725/// with `status` / `statusText` / `headers` init fields. The instance methods
726/// (`text`, `json`, `arrayBuffer`) are the same natives the fetch-produced
727/// response objects use, attached to the instance because the VM links the
728/// returned object verbatim.
729fn response_constructor(vm: &mut VM, args: Vec<JSValue>) -> JSResult<JSValue> {
730    let body = args.get(1).cloned().unwrap_or(JSValue::undefined());
731    let init = args.get(2).and_then(JSValue::as_object);
732    let init_field = |name: &str| -> JSValue {
733        init.as_ref()
734            .map(|object| object.borrow().get(name))
735            .unwrap_or(JSValue::undefined())
736    };
737    // Per spec, a missing (or non-numeric) `status` defaults to 200; only an
738    // explicitly out-of-range number is an error.
739    let status_value = init_field("status").to_number();
740    let status = if status_value.is_nan() {
741        200
742    } else if status_value.is_finite() && (200.0..=599.0).contains(&status_value) {
743        status_value as u16
744    } else {
745        return Err(JSError::RangeError(
746            "Failed to construct 'Response': The status provided (…) is outside the range [200, 599]"
747                .to_string(),
748        ));
749    };
750    let status_text = init_field("statusText").to_string();
751    let headers = init
752        .map(|object| extract_header_entries(&JSValue::from_object(Rc::clone(&object))))
753        .unwrap_or_default();
754    let body_string = if let Some(blob) = body.as_object()
755        && blob.borrow().get("__orinium_blob").as_boolean() == Some(true)
756    {
757        blob.borrow().get("__orinium_blob_data").to_string()
758    } else if body.is_undefined() || body.is_null() {
759        String::new()
760    } else {
761        body.to_string()
762    };
763    let response = JsFetchResponse {
764        url: String::new(),
765        status,
766        status_text,
767        redirected: false,
768        body: body_string.into_bytes(),
769        headers,
770    };
771    let response_object = make_fetch_response(response);
772    let _ = vm;
773    Ok(JSValue::from_object(response_object))
774}

776/// Installs the `Response` global constructor.
777pub(crate) fn install_response(engine: &mut pixi_byte::JSEngine) {
778    let mut constructor = JSObject::new();
779    constructor.set(
780        "__construct__".to_string(),
781        JSValue::from_native_function(response_constructor),
782    );
783    constructor.set(
784        "error".to_string(),
785        JSValue::from_native_function(response_static_error),
786    );
787    constructor.set(
788        "redirect".to_string(),
789        JSValue::from_native_function(response_static_redirect),
790    );
791    engine.global_mut().borrow_mut().set(
792        "Response".to_string(),
793        JSValue::from_object(Rc::new(RefCell::new(constructor))),
794    );
795}

797/// `Response.error()` — a response whose `type` is "error".
798fn response_static_error(_vm: &mut VM, _args: Vec<JSValue>) -> JSResult<JSValue> {
799    let response = make_fetch_response(JsFetchResponse {

802        status_text: String::new(),

804        body: Vec::new(),
805        headers: Vec::new(),

814/// `Response.redirect(url, status?)` — a redirect response for `url`.
815fn response_static_redirect(_vm: &mut VM, args: Vec<JSValue>) -> JSResult<JSValue> {
816    let url = args
817        .get(1)
818        .map(JSValue::to_console_string)
819        .unwrap_or_default();
820    let status = match args.get(2).map(JSValue::to_number).unwrap_or(302.0) as u16 {
821        301 | 302 | 303 | 307 | 308 => status_argument(args.get(2)),
822        _ => {
823            return Err(JSError::RangeError(
824                "Failed to construct 'Response': Invalid redirect status code".to_string(),
825            ));
826        }
827    };
828    let response = make_fetch_response(JsFetchResponse {
829        url,
830        status,
831        status_text: String::new(),
832        redirected: true,
833        body: Vec::new(),
834        headers: Vec::new(),
835    });
836    response.borrow_mut().set(
837        "type".to_string(),
838        JSValue::from_string("default".to_string()),
839    );
840    Ok(JSValue::from_object(response))
841}

843fn status_argument(value: Option<&JSValue>) -> u16 {
844    value.map(JSValue::to_number).unwrap_or(302.0) as u16
845}

847fn fetch_response_text(vm: &mut VM, args: Vec<JSValue>) -> JSResult<JSValue> {
848    let body = match consume_response_body(vm, &args, "text")? {
849        Ok(body) => body,
850        Err(rejection) => return Ok(rejection),
851    };
852    settle_promise(vm, "resolve", JSValue::from_string(body))
853}
854
855fn fetch_response_json(vm: &mut VM, args: Vec<JSValue>) -> JSResult<JSValue> {
856    let body = match consume_response_body(vm, &args, "json")? {
857        Ok(body) => body,
858        Err(rejection) => return Ok(rejection),
859    };
860    match serde_json::from_str::<serde_json::Value>(&body) {
861        Ok(value) => settle_promise(vm, "resolve", json_to_js_value(value)),
862        Err(error) => settle_promise(
863            vm,
864            "reject",
865            JSValue::from_string(format!("Failed to parse JSON: {error}")),
866        ),
867    }
868}

870fn fetch_response_array_buffer(vm: &mut VM, args: Vec<JSValue>) -> JSResult<JSValue> {
871    let response = match consume_response_object(vm, &args, "arrayBuffer") {
872        Ok(response) => response,
873        Err(JSError::Thrown(rejection)) => return Ok(rejection),
874        Err(error) => return Err(error),
875    };
876    let bytes = response.borrow().get(RESPONSE_BODY_BYTES);
877    settle_promise(vm, "resolve", make_array_buffer_from_value(&bytes))
878}

880fn fetch_response_body_used(_vm: &mut VM, args: Vec<JSValue>) -> JSResult<JSValue> {
881    let Some(response) = args.first().and_then(JSValue::as_object) else {
882        return Err(JSError::TypeError(
883            "Response.bodyUsed called on incompatible receiver".to_string(),
884        ));
885    };
886    Ok(response.borrow().get(RESPONSE_BODY_USED))
887}
888
889fn consume_response_body(
890    vm: &mut VM,
891    args: &[JSValue],
892    method: &str,
893) -> JSResult<Result<String, JSValue>> {
894    let response = match consume_response_object(vm, args, method) {
895        Ok(response) => response,
896        Err(JSError::Thrown(value)) => return Ok(Err(value)),
897        Err(error) => return Err(error),
898    };
899    let response = response.borrow();
900    response
901        .get("__orinium_response_body")
902        .as_string_owned()
903        .map(Ok)
904        .ok_or_else(|| JSError::InternalError("Response body is unavailable".to_string()))
905}
906
907fn consume_response_object(
908    vm: &mut VM,
909    args: &[JSValue],
910    method: &str,
911) -> JSResult<Rc<RefCell<JSObject>>> {
912    let Some(response) = args.first().and_then(JSValue::as_object) else {
913        return Err(JSError::TypeError(format!(
914            "Response.{method} called on incompatible receiver"
915        )));
916    };
917    if response.borrow().get(RESPONSE_BODY_USED).as_boolean() == Some(true) {
918        let rejection = settle_promise(
919            vm,
920            "reject",
921            JSValue::from_string("Response body has already been consumed".to_string()),
922        )?;
923        return Err(JSError::Thrown(rejection));
924    }
925    response
926        .borrow_mut()
927        .set(RESPONSE_BODY_USED.to_string(), JSValue::from_bool(true));
928    Ok(Rc::clone(&response))
929}
930
931fn settle_promise(vm: &mut VM, method: &str, value: JSValue) -> JSResult<JSValue> {
932    let promise = vm.global_object.borrow().get("Promise");
933    let Some(constructor) = promise.as_object() else {
934        return Err(JSError::InternalError(
935            "Promise constructor is unavailable".to_string(),
936        ));
937    };
938    let settle = constructor.borrow().get(method);
939    vm.call(settle, promise, vec![value])
940}

942fn json_to_js_value(value: serde_json::Value) -> JSValue {
943    match value {
944        serde_json::Value::Null => JSValue::null(),
945        serde_json::Value::Bool(value) => JSValue::from_bool(value),
946        serde_json::Value::Number(value) => {
947            JSValue::from_number(value.as_f64().unwrap_or(f64::NAN))
948        }
949        serde_json::Value::String(value) => JSValue::from_string(value),
950        serde_json::Value::Array(values) => {
951            JSArray::from_vec(values.into_iter().map(json_to_js_value).collect()).to_object()
952        }
953        serde_json::Value::Object(properties) => {
954            let mut object = JSObject::new();
955            for (key, value) in properties {
956                object.set(key, json_to_js_value(value));
957            }
958            JSValue::from_object(Rc::new(RefCell::new(object)))
959        }
960    }
