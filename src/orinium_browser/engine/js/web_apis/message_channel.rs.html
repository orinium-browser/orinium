8const PEER: &str = "__message_port_peer";
9
10/// Installs the `MessageChannel` / `MessagePort` pair on the global scope.
11///
12/// This is a minimal single-threaded implementation: `postMessage` delivers
13/// the event to the peer port's `onmessage` handler asynchronously via the
14/// microtask queue, carrying a `{data, ports}` event object.
15pub(crate) fn install_message_channel(engine: &mut pixi_byte::JSEngine) {
16    let mut constructor = JSObject::new();
17    constructor.set(
18        "__construct__".to_string(),
19        JSValue::from_native_function(message_channel_constructor),
20    );
21    constructor.set(
22        "__call__".to_string(),
23        JSValue::from_native_function(message_channel_constructor),
24    );
25    engine.global_mut().borrow_mut().set(
26        "MessageChannel".to_string(),
27        JSValue::from_object(Rc::new(RefCell::new(constructor))),
28    );
29}

31fn make_port() -> JSValue {
32    let mut port = JSObject::new();
33    port.set(
34        "postMessage".to_string(),
35        JSValue::from_native_function(port_post_message),
36    );
37    port.set(
38        "start".to_string(),
39        JSValue::from_native_function(port_noop),
40    );
41    port.set(
42        "close".to_string(),
43        JSValue::from_native_function(port_noop),
44    );
45    JSValue::from_object(Rc::new(RefCell::new(port)))
46}
47
48fn message_channel_constructor(_vm: &mut VM, args: Vec<JSValue>) -> JSResult<JSValue> {
49    let port1 = make_port();
50    let port2 = make_port();
51    let p1 = port1.as_object().unwrap();
52    let p2 = port2.as_object().unwrap();
53    p1.borrow_mut()
54        .set(PEER.to_string(), JSValue::from_object(Rc::clone(&p2)));
55    p2.borrow_mut()
56        .set(PEER.to_string(), JSValue::from_object(Rc::clone(&p1)));
57    let mut channel = JSObject::new();
58    channel.set("port1".to_string(), port1);
59    channel.set("port2".to_string(), port2);
60    let _ = args; // receiver is unused
61    Ok(JSValue::from_object(Rc::new(RefCell::new(channel))))
62}
63
64fn port_noop(_vm: &mut VM, _args: Vec<JSValue>) -> JSResult<JSValue> {
65    Ok(JSValue::undefined())
66}
67
68fn port_post_message(vm: &mut VM, args: Vec<JSValue>) -> JSResult<JSValue> {
69    let Some(port) = args.first().and_then(JSValue::as_object) else {
70        return Err(pixi_byte::JSError::TypeError(
71            "MessagePort.postMessage: invalid receiver".to_string(),
72        ));
73    };
74    let data = args.get(1).cloned().unwrap_or(JSValue::undefined());
75    let peer = port.borrow().get(PEER);
76    let Some(peer_obj) = peer.as_object() else {
77        return Ok(JSValue::undefined());
78    };
79    let handler = peer_obj.borrow().get("onmessage");
80    if is_callable(&handler) {
81        let mut event = JSObject::new();
82        event.set("data".to_string(), data);
83        event.set(
84            "ports".to_string(),
85            vm.array_from_values(vec![JSValue::from_object(Rc::clone(&peer_obj))]),
86        );
87        event.set(
88            "target".to_string(),
89            JSValue::from_object(Rc::clone(&peer_obj)),
90        );
91        vm.enqueue_job(
92            handler,
93            JSValue::undefined(),
94            vec![JSValue::from_object(Rc::new(RefCell::new(event)))],
95        );
96    }
97    Ok(JSValue::undefined())
98}
