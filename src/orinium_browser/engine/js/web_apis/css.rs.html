11/// Installs the `CSS` global object and `CSSStyleDeclaration` constructor.
12pub(crate) fn install_css(engine: &mut pixi_byte::JSEngine) {
13    let global = engine.global_mut();
14
15    // CSSStyleDeclaration.prototype — shared by style and computed-style wrappers.
16    let csd_proto = Rc::new(RefCell::new(JSObject::new()));
17    csd_proto.borrow_mut().set(
18        "constructor".to_string(),
19        JSValue::from_object(Rc::clone(&csd_constructor_stub())),
20    );
21
22    // CSSStyleDeclaration() constructor — `new CSSStyleDeclaration()` creates
23    // an empty mutable declaration block.
24    let mut csd_constructor = JSObject::new();
25    csd_constructor.set(
26        "__call__".to_string(),
27        JSValue::from_native_function(css_style_declaration_construct),
28    );
29    csd_constructor.set(
30        "__construct__".to_string(),
31        JSValue::from_native_function(css_style_declaration_construct),
32    );
33    csd_constructor.define_property(
34        "prototype".to_string(),
35        Property::read_only(JSValue::from_object(Rc::clone(&csd_proto))),
36    );
37
38    // CSS global object.
39    let mut css = JSObject::new();
40    css.set(
41        "supports".to_string(),
42        JSValue::from_native_function(css_supports),
43    );
44    css.set(
45        "escape".to_string(),
46        JSValue::from_native_function(css_escape),
47    );
48
49    let mut global = global.borrow_mut();
50    global.set(
51        "CSSStyleDeclaration".to_string(),
52        JSValue::from_object(Rc::new(RefCell::new(csd_constructor))),
53    );
54    global.set(
55        "CSS".to_string(),
56        JSValue::from_object(Rc::new(RefCell::new(css))),
57    );
58}

60/// Stub prototype constructor for `CSSStyleDeclaration.prototype.constructor`.
61fn csd_constructor_stub() -> Rc<RefCell<JSObject>> {
62    let mut obj = JSObject::new();
63    obj.set(
64        "__call__".to_string(),
65        JSValue::from_native_function(css_style_declaration_construct),
66    );
67    obj.set(
68        "__construct__".to_string(),
69        JSValue::from_native_function(css_style_declaration_construct),
70    );
71    Rc::new(RefCell::new(obj))
72}

79/// `CSS.supports(property, value)` or `CSS.supports(conditionText)`.
80///
81/// Returns `true` when the implementation supports the given property-value
82/// pair.  A full CSS parser is not available, so validation is conservative:
83/// only well-known properties with simple values are recognised; everything
84/// else reports `false`.
85fn css_supports(_vm: &mut VM, args: Vec<JSValue>) -> JSResult<JSValue> {
86    let arg = args.get(1).cloned().unwrap_or(JSValue::undefined());
87    let text = match arg.as_string() {
88        Some(t) => t,
89        None => return Ok(JSValue::from_bool(false)),
90    };
91
92    // `CSS.supports("display: flex")` — single condition-text form.
93    if text.contains(':') {
94        return Ok(JSValue::from_bool(supports_condition_text(&text)));
95    }
96
97    // `CSS.supports(property, value)` — two-argument form.
98    let value = args.get(2).cloned().unwrap_or(JSValue::undefined());
99    let value = match value.as_string() {
100        Some(v) => v,
101        None => return Ok(JSValue::from_bool(false)),
102    };
103    Ok(JSValue::from_bool(supports_property_value(&text, &value)))
104}
105
106/// `CSS.supports(conditionText)` — parses "property: value" and checks.
107fn supports_condition_text(text: &str) -> bool {
108    let mut parts = text.splitn(2, ':');
109    let prop = parts.next().unwrap_or("").trim();
110    let value = parts.next().unwrap_or("").trim();
111    if prop.is_empty() || value.is_empty() {
112        return false;
113    }
114    supports_property_value(prop, value)
115}
116
117/// `CSS.supports(property, value)` — conservative property/value validation.
118fn supports_property_value(property: &str, value: &str) -> bool {
119    let prop = property.trim().to_ascii_lowercase();
120    let val = value.trim();
121
122    match prop.as_str() {
123        "display" => matches!(
124            val,
125            "none"
126                | "block"
127                | "inline"
128                | "inline-block"
129                | "flex"
130                | "inline-flex"
131                | "grid"
132                | "inline-grid"
133                | "contents"
134                | "run-in"
135        ),
136        "color" => is_css_color(val),
137        "background" => true, // shorthand, broadly supported
138        "background-color" => is_css_color(val),
139        "width" => is_css_dimension(val),
140        "height" => is_css_dimension(val),
141        "margin" | "padding" | "border" => true, // shorthands
142        "font" | "font-family" | "font-size" | "font-weight" | "font-style" => true,
143        "text-align" => matches!(
144            val,
145            "left" | "right" | "center" | "justify" | "start" | "end"
146        ),
147        "visibility" => matches!(val, "visible" | "hidden" | "collapse"),
148        "position" => matches!(val, "static" | "relative" | "absolute" | "fixed" | "sticky"),
149        "float" => matches!(
150            val,
151            "none" | "left" | "right" | "inline-start" | "inline-end"
152        ),
153        "clear" => matches!(
154            val,
155            "none" | "left" | "right" | "both" | "inline-start" | "inline-end"
156        ),
157        "overflow" => matches!(val, "visible" | "hidden" | "scroll" | "auto"),
158        "opacity" => is_css_number(val),
159        "z-index" => is_css_integer(val) || val == "auto",
160        _ => false,
161    }
162}

164/// `CSS.escape()` — escapes a string for use in a CSS selector.
165fn css_escape(_vm: &mut VM, args: Vec<JSValue>) -> JSResult<JSValue> {
166    let s = args.get(1).and_then(JSValue::as_string).unwrap_or_default();
167    let escaped = escape_identifier(s);
168    Ok(JSValue::from_string(escaped))
169}

171/// Escapes a string for use as a CSS identifier / selector fragment.
172fn escape_identifier(s: &str) -> String {
173    let mut out = String::with_capacity(s.len() + 8);
174    for ch in s.chars() {
175        match ch {
176            '\\' => out.push_str("\\\\"),
177            // Characters that must be escaped in CSS identifiers.
178            c if c.is_ascii_whitespace() => {
179                out.push('\\');
180                out.push_str(&format!("{:X} ", c as u32));
181            }
182            c if c == '.'
183                || c == '#'
184                || c == ':'
185                || c == '['
186                || c == ']'
187                || c == '('
188                || c == ')'
189                || c == ','
190                || c == '+'
191                || c == '~'
192                || c == '>'
193                || c == '*'
194                || c == '!'
195                || c == '@'
196                || c == '%'
197                || c == '^'
198                || c == '&'
199                || c == '`'
200                || c == '|'
201                || c == '='
202                || c == '"'
203                || c == '\''
204                || c == '/'
205                || c == '{'
206                || c == '}'
207                || c == '<'
208                || c == '>'
209                || c == '?'
210                || c == ';' =>
211            {
212                out.push('\\');
213                out.push(c);
214            }
215            // Control characters and non-ASCII are hex-escaped.
216            c if c.is_control() || (c as u32) > 0x7E => {
217                out.push('\\');
218                out.push_str(&format!("{:X} ", c as u32));
219            }
220            c => out.push(c),
221        }
222    }
223    // Remove trailing space added by hex escapes.
224    if out.ends_with(' ') {
225        out.pop();
226    }
227    out
228}

230/// Whether `value` looks like a CSS color (keyword, hex, rgb(), etc.).
231fn is_css_color(value: &str) -> bool {
232    let v = value.trim();
233    if v.is_empty() {
234        return false;
235    }
236    // Hex colors: #rgb, #rgba, #rrggbb, #rrggbbaa
237    if v.starts_with('#') {
238        let rest = &v[1..];
239        return !rest.is_empty()
240            && rest.chars().all(|c| c.is_ascii_hexdigit())
241            && matches!(rest.len(), 3 | 4 | 6 | 8);
242    }
243    // Functional colors: rgb(), rgba(), hsl(), hsla(), hwb(), lab(), lch(), color()
244    if v.ends_with(')')
245        && (v.starts_with("rgb(")
246            || v.starts_with("rgba(")
247            || v.starts_with("hsl(")
248            || v.starts_with("hsla(")
249            || v.starts_with("hwb(")

417/// Whether `value` looks like a CSS dimension (with unit or zero).
418fn is_css_dimension(value: &str) -> bool {
419    let v = value.trim();
420    if v == "0" {
421        return true;
422    }
423    // Must end with a unit (px, em, rem, %, etc.) or be a pure number.
424    let num_end = v
425        .chars()
426        .position(|c| !c.is_ascii_digit() && c != '.' && c != '-');
427    match num_end {
428        Some(i) => {
429            let unit = &v[i..];
430            !unit.is_empty()
431                && unit.chars().all(|c| c.is_ascii_alphabetic() || c == '%')
432                && v[..i].parse::<f64>().is_ok()
433        }
434        None => v.parse::<f64>().is_ok(),
435    }
436}
