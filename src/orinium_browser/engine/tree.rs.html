1//! Generic tree structure for DOM, render tree, or other hierarchical data.
2//!
3//! # Overview
4//! - `TreeNode<T>` stores a node value, parent, and children.
5//! - `Tree<T>` stores a root node and provides traversal, mapping, and searching utilities.
6
7use std::cell::{Cell, RefCell};
8use std::fmt::{self, Debug, Display, Formatter};
9use std::rc::{Rc, Weak};
10
11/// Alias for a reference-counted tree node
12pub type NodeRef<T> = Rc<RefCell<TreeNode<T>>>;
13
14/// A single tree node
15#[derive(Clone)]
16pub struct TreeNode<T> {
17    pub value: T,
18    parent: Option<Weak<RefCell<TreeNode<T>>>>,
19    children: Vec<NodeRef<T>>,
20}
21
22impl<T> TreeNode<T> {
23    /// Create a new node wrapped in Rc<RefCell<_>>
24    pub fn new(value: T) -> NodeRef<T> {
25        Rc::new(RefCell::new(Self {
26            value,
27            parent: None,
28            children: Vec::new(),
29        }))
30    }
31
32    /// Returns the parent node, if any
33    pub fn parent(&self) -> Option<NodeRef<T>> {
34        self.parent.as_ref().and_then(|w| w.upgrade())
35    }
36
37    /// Returns slice of child nodes
38    pub fn children(&self) -> &[NodeRef<T>] {
39        &self.children
40    }
41
42    /// Remove all children of this node
43    pub fn clear_children(&mut self) {
44        for child in &self.children {
45            child.borrow_mut().parent = None;
46        }
47        self.children.clear();
48    }
49
50    /// Add a child node
51    pub fn add_child(parent: &NodeRef<T>, child: NodeRef<T>) {
52        child.borrow_mut().parent = Some(Rc::downgrade(parent));
53        parent.borrow_mut().children.push(child);
54    }
55
56    /// Removes `child` from `parent`, returning the detached node when found.
57    pub fn remove_child(parent: &NodeRef<T>, child: &NodeRef<T>) -> Option<NodeRef<T>> {
58        let position = parent
59            .borrow()
60            .children
61            .iter()
62            .position(|candidate| Rc::ptr_eq(candidate, child))?;
63        let removed = parent.borrow_mut().children.remove(position);
64        removed.borrow_mut().parent = None;
65        Some(removed)
66    }
67
68    /// Detaches a node from its current parent.
69    pub fn detach(node: &NodeRef<T>) -> bool {
70        let Some(parent) = node.borrow().parent() else {
71            return false;
72        };
73        Self::remove_child(&parent, node).is_some()
74    }
75
76    /// Appends `child`, moving it from its current parent when necessary.
77    ///
78    /// Returns `false` when the operation would create a tree cycle.
79    pub fn append_child(parent: &NodeRef<T>, child: NodeRef<T>) -> bool {
80        if Self::is_inclusive_ancestor(&child, parent) {
81            return false;
82        }
83        Self::detach(&child);
84        Self::add_child(parent, child);
85        true
86    }
87
88    /// Inserts `child` immediately before `reference`, moving it from its
89    /// current parent when necessary.
90    pub fn insert_before(parent: &NodeRef<T>, child: NodeRef<T>, reference: &NodeRef<T>) -> bool {
91        if Rc::ptr_eq(&child, reference) {
92            return reference
93                .borrow()
94                .parent()
95                .is_some_and(|candidate| Rc::ptr_eq(&candidate, parent));
96        }
97        if Self::is_inclusive_ancestor(&child, parent) {
98            return false;
99        }
100        if !reference
101            .borrow()
102            .parent()
103            .is_some_and(|candidate| Rc::ptr_eq(&candidate, parent))
104        {
105            return false;
106        }
107
108        Self::detach(&child);
109        let Some(index) = parent
110            .borrow()
111            .children
112            .iter()
113            .position(|candidate| Rc::ptr_eq(candidate, reference))
114        else {
115            return false;
116        };
117        Self::insert_child_at(parent, index, child);
118        true
119    }
120
121    fn is_inclusive_ancestor(ancestor: &NodeRef<T>, node: &NodeRef<T>) -> bool {
122        let mut current = Some(Rc::clone(node));
123        while let Some(candidate) = current {
124            if Rc::ptr_eq(ancestor, &candidate) {
125                return true;
126            }
127            current = candidate.borrow().parent();
128        }
129        false
130    }
131
132    /// Insert a child at a given position
133    pub fn insert_child_at(parent: &NodeRef<T>, index: usize, child: NodeRef<T>) {
134        child.borrow_mut().parent = Some(Rc::downgrade(parent));
135        parent.borrow_mut().children.insert(index, child);
136    }
137
138    /// Create a child with value and add it to parent
139    pub fn add_child_value(parent: &NodeRef<T>, value: T) -> NodeRef<T> {
140        let child = Self::new(value);
141        Self::add_child(parent, Rc::clone(&child));
142        child
143    }
144
145    /// Replace child at given index
146    pub fn replace_child(
147        parent: &NodeRef<T>,
148        index: usize,
149        new_child: NodeRef<T>,
150    ) -> Option<NodeRef<T>> {
151        let mut p = parent.borrow_mut();
152        if index < p.children.len() {
153            let old_child = std::mem::replace(&mut p.children[index], new_child);
154            old_child.borrow_mut().parent = None;
155            Some(old_child)
156        } else {
157            None
158        }
159    }
160
161    /// Find direct children matching predicate
162    pub fn find_children_by<F>(&self, predicate: F) -> Vec<NodeRef<T>>
163    where
164        F: Fn(&T) -> bool,
165    {
166        self.children
167            .iter()
168            .filter(|c| predicate(&c.borrow().value))
169            .cloned()
170            .collect()
171    }
172
173    /// Clone node (optionally deep)
174    pub fn clone_node(&self, deep: bool) -> NodeRef<T>
175    where
176        T: Clone,
177    {
178        let new_node = Rc::new(RefCell::new(TreeNode {
179            value: self.value.clone(),
180            children: Vec::new(),
181            parent: None,
182        }));
183
184        if deep {
185            for child in &self.children {
186                let child_clone = child.borrow().clone_node(true);
187                child_clone.borrow_mut().parent = Some(Rc::downgrade(&new_node));
188                new_node.borrow_mut().children.push(child_clone);
189            }
190        }
191
192        new_node
193    }
194}

196/// Represents a tree with a single root node
197#[derive(Clone)]
198pub struct Tree<T> {
199    pub root: NodeRef<T>,
200    /// Monotonic counter bumped by [`Tree::mark_dirty`] on every mutation.
201    ///
202    /// Layout snapshots cache the DOM and are reused while the version is
203    /// unchanged, so any mutation (currently the text-input write-back, later
204    /// JS DOM manipulation) must call [`Tree::mark_dirty`] to invalidate them.
205    version: Cell<u64>,
206}
207
208impl<T: Clone> Tree<T> {
209    /// Create a new tree with root value
210    pub fn new(root_value: T) -> Self {
211        Self {
212            root: TreeNode::new(root_value),
213            version: Cell::new(0),
214        }
215    }
216
217    /// Wrap an already-built root in a fresh tree.
218    ///
219    /// The version counter starts at zero, so the resulting tree is treated as
220    /// a newly parsed document.
221    pub fn from_root(root: NodeRef<T>) -> Self {
222        Self {
223            root,
224            version: Cell::new(0),
225        }
226    }
227
228    /// Records a DOM mutation by bumping the tree's version counter.
229    pub fn mark_dirty(&self) {
230        self.version.set(self.version.get() + 1);
231    }
232
233    /// The current mutation version of the tree.
234    pub fn version(&self) -> u64 {
235        self.version.get()
236    }
237
238    /// Recursively traverse all nodes, applying a function
239    pub fn traverse<F>(&self, mut f: F)
240    where
241        F: FnMut(&NodeRef<T>),
242    {
243        fn visit<T, F>(node: &NodeRef<T>, f: &mut F)
244        where
245            F: FnMut(&NodeRef<T>),
246        {
247            f(node);
248            for child in &node.borrow().children {
249                visit(child, f);
250            }
251        }
252        visit(&self.root, &mut f);
253    }
254
255    /// Map each node value to another type, returning a new Tree
256    pub fn map<U, F>(&self, f: F) -> Tree<U>
257    where
258        F: Fn(&T) -> U,
259        U: Clone,
260    {
261        fn map_node<T, U, F>(node: &NodeRef<T>, f: &F) -> NodeRef<U>
262        where
263            F: Fn(&T) -> U,
264            U: Clone,
265        {
266            let n = node.borrow();
267            let new_node = TreeNode::new(f(&n.value));
268            for child in &n.children {
269                let mapped_child = map_node(child, f);
270                TreeNode::add_child(&new_node, mapped_child);
271            }
272            new_node
273        }
274
275        Tree {
276            root: map_node(&self.root, &f),
277            version: Cell::new(0),
278        }
279    }
280
281    /// Map using the NodeRef itself (for access to parent/children)
282    pub fn map_with_node<U, F>(&self, f: F) -> Tree<U>
283    where
284        F: Fn(&NodeRef<T>) -> U,
285        U: Clone,
286    {
287        fn map_node<T, U, F>(node: &NodeRef<T>, f: &F) -> NodeRef<U>
288        where
289            F: Fn(&NodeRef<T>) -> U,
290            U: Clone,
291        {
292            let new_node = TreeNode::new(f(node));
293            for child in &node.borrow().children {
294                let mapped_child = map_node(child, f);
295                TreeNode::add_child(&new_node, mapped_child);
296            }
297            new_node
298        }
299
300        Tree {
301            root: map_node(&self.root, &f),
302            version: Cell::new(0),
303        }
304    }
305
306    /// Find all nodes in the tree that satisfy a predicate
307    pub fn find_all<F>(&self, predicate: F) -> Vec<NodeRef<T>>
308    where
309        F: Fn(&T) -> bool,
310    {
311        let mut result = Vec::new();
312        self.traverse(|node| {
313            if predicate(&node.borrow().value) {
314                result.push(Rc::clone(node));
315            }
316        });
317        result
318    }
319}

321impl<T: Clone + Debug> Display for Tree<T> {
322    fn fmt(&self, f: &mut Formatter<'_>) -> fmt::Result {
323        fn fmt_node<T: Clone + Debug>(
324            node: &NodeRef<T>,
325            f: &mut Formatter<'_>,
326            prefix: &str,
327            is_last: bool,
328        ) -> fmt::Result {
329            let n = node.borrow();
330            let connector = if prefix.is_empty() {
331                ""
332            } else if is_last {
333                "└── "
334            } else {
335                "├── "
336            };
337            writeln!(f, "{}{}{:?}", prefix, connector, n.value)?;
338            let child_count = n.children.len();
339            for (i, child) in n.children.iter().enumerate() {
340                let mut new_prefix = prefix.to_string();
341                new_prefix.push_str(if is_last { "    " } else { "│   " });
342                fmt_node(child, f, &new_prefix, i == child_count - 1)?;
343            }
344            Ok(())
345        }
346        fmt_node(&self.root, f, "", true)
347    }
348}
349
350impl<T: Clone + Debug> Debug for Tree<T> {
351    fn fmt(&self, f: &mut Formatter<'_>) -> fmt::Result {
352        write!(f, "{}", self)
353    }
354}

356#[cfg(test)]
357mod tests {
358    use super::*;
359
360    #[test]
361    fn append_child_reparents_and_prevents_cycles() {
362        let root = TreeNode::new("root");
363        let first = TreeNode::new("first");
364        let second = TreeNode::new("second");
365        TreeNode::add_child(&root, Rc::clone(&first));
366        TreeNode::add_child(&first, Rc::clone(&second));
367
368        assert!(TreeNode::append_child(&root, Rc::clone(&second)));
369        assert!(first.borrow().children().is_empty());
370        assert!(Rc::ptr_eq(&second.borrow().parent().unwrap(), &root));
371        assert!(!TreeNode::append_child(&second, Rc::clone(&root)));
372    }
373
374    #[test]
375    fn detach_and_clear_children_reset_parent_links() {
376        let root = TreeNode::new("root");
377        let first = TreeNode::new("first");
378        let second = TreeNode::new("second");
379        TreeNode::add_child(&root, Rc::clone(&first));
380        TreeNode::add_child(&root, Rc::clone(&second));
381
382        assert!(TreeNode::detach(&first));
383        assert!(first.borrow().parent().is_none());
384        root.borrow_mut().clear_children();
385        assert!(second.borrow().parent().is_none());
386    }
387
388    #[test]
389    fn insert_before_moves_nodes_and_preserves_order() {
390        let root = TreeNode::new("root");
391        let first = TreeNode::new("first");
392        let second = TreeNode::new("second");
393        let moving = TreeNode::new("moving");
394        TreeNode::add_child(&root, Rc::clone(&first));
395        TreeNode::add_child(&root, Rc::clone(&second));
396        TreeNode::add_child(&first, Rc::clone(&moving));
397
398        assert!(TreeNode::insert_before(&root, Rc::clone(&moving), &second));
399        let children = root.borrow().children().to_vec();
400        assert!(Rc::ptr_eq(&children[0], &first));
401        assert!(Rc::ptr_eq(&children[1], &moving));
402        assert!(Rc::ptr_eq(&children[2], &second));
403        assert!(first.borrow().children().is_empty());
404    }
