1//! Shared helpers for translating boxes after `ui_layout` has positioned them.
2//!
3//! Several post-layout passes move boxes the engine already laid out:
4//! [`table_layout`](super::table_layout) spreads column widths across rows, and
5//! the anchor-positioning fix-ups in
6//! [`builder::layout_fix`](super::builder) shift anchored elements. They must
7//! translate a box model and its inline line spans exactly the same way, so the
8//! algorithm lives in one place.
9//!
10//! Inline boxes carry an extra per-line position ([`ui_layout::LineSpan`]): the
11//! box model describes the outermost box, while each span records where that
12//! line actually sits. Moving an inline box without moving its spans would
13//! leave the rendered text behind, so every helper here shifts both.
14
15use ui_layout::{BoxModel, LayoutBox};
16
17/// Translates every layer (border, padding, content, children) of `model` by
18/// `dx`/`dy`.
19pub(crate) fn translate_box(model: &mut BoxModel, dx: f32, dy: f32) {
20    model.border_box.x += dx;
21    model.padding_box.x += dx;
22    model.content_box.x += dx;
23    model.children_box.x += dx;
24    model.border_box.y += dy;
25    model.padding_box.y += dy;
26    model.content_box.y += dy;
27    model.children_box.y += dy;
28}
29
30/// Translates a layout box — its box model together with any inline line
31/// spans — by `dx`/`dy`.
32pub(crate) fn shift_box(layout_box: &mut LayoutBox, dx: f32, dy: f32) {
33    match layout_box {
34        LayoutBox::None => {}
35        LayoutBox::BlockBox(model) => translate_box(model, dx, dy),
36        LayoutBox::InlineBox(inline) => {
37            translate_box(&mut inline.box_model, dx, dy);
38            for span in &mut inline.line_spans {
39                span.line_pos.0 += dx;
40                span.line_pos.1 += dy;
41            }
42        }
43    }
44}
45
46/// Moves `layout_box` so its border box's left edge lands on `target_x`.
47///
48/// The delta is derived from the box's current border edge, so this positions
49/// the box at an absolute coordinate rather than nudging it by a relative
50/// offset.
51pub(crate) fn move_box_x(layout_box: &mut LayoutBox, target_x: f32) {
52    let dx = match layout_box {
53        LayoutBox::None => return,
54        LayoutBox::BlockBox(model) => target_x - model.border_box.x,
55        LayoutBox::InlineBox(inline) => target_x - inline.box_model.border_box.x,
56    };
57    shift_box(layout_box, dx, 0.0);
58}
59
60/// Grows `layout_box` to `target_border_width` (never shrinks it).
61pub(crate) fn grow_box_width(layout_box: &mut LayoutBox, target_border_width: f32) {
62    let model = match layout_box {
63        LayoutBox::None => return,
64        LayoutBox::BlockBox(model) => model,
65        LayoutBox::InlineBox(inline) => &mut inline.box_model,
66    };
67    let delta = (target_border_width - model.border_box.width).max(0.0);
68    model.border_box.width += delta;
69    model.padding_box.width += delta;
70    model.content_box.width += delta;
71    model.children_box.width += delta;
72}

74#[cfg(test)]
75mod tests {
76    use ui_layout::{InlineBox, LineSpan, Rect};
77
78    use super::*;
79
80    fn model(x: f32, y: f32) -> BoxModel {
81        let rect = Rect {
82            x,
83            y,
84            width: 40.0,
85            height: 10.0,
86        };
87        rect.into()
88    }
89
90    #[test]
91    fn shift_block_box_moves_every_layer() {
92        let mut layout_box = LayoutBox::BlockBox(model(5.0, 7.0));
93        shift_box(&mut layout_box, 3.0, -2.0);
94        let b = layout_box.iter().next().unwrap();
95        assert_eq!(b.border_box.x, 8.0);
96        assert_eq!(b.padding_box.y, 5.0);
97        assert_eq!(b.content_box.x, 8.0);
98        assert_eq!(b.children_box.y, 5.0);
99    }
100
101    #[test]
102    fn shift_inline_box_moves_line_spans_too() {
103        let mut layout_box = LayoutBox::InlineBox(InlineBox {
104            box_model: model(5.0, 7.0),
105            line_spans: vec![LineSpan {
106                x_range: 0.0..10.0,
107                line_pos: (5.0, 7.0),
108                line_index: 0,
109            }],
110        });
111        shift_box(&mut layout_box, 3.0, 4.0);
112        let LayoutBox::InlineBox(inline) = &layout_box else {
113            panic!("expected InlineBox");
114        };
115        assert_eq!(inline.box_model.border_box.x, 8.0);
116        assert_eq!(inline.line_spans[0].line_pos, (8.0, 11.0));
117    }
118
119    #[test]
120    fn move_box_x_uses_absolute_target() {
121        let mut layout_box = LayoutBox::BlockBox(model(5.0, 7.0));
122        move_box_x(&mut layout_box, 100.0);
123        let b = layout_box.iter().next().unwrap();
124        assert_eq!(b.border_box.x, 100.0);
125        assert_eq!(b.border_box.y, 7.0);
126    }
127
128    #[test]
129    fn grow_box_width_never_shrinks() {
130        let mut layout_box = LayoutBox::BlockBox(model(0.0, 0.0));
131        grow_box_width(&mut layout_box, 100.0);
132        assert_eq!(layout_box.iter().next().unwrap().border_box.width, 100.0);
133        grow_box_width(&mut layout_box, 20.0);
134        assert_eq!(layout_box.iter().next().unwrap().border_box.width, 100.0);
135    }
136}
