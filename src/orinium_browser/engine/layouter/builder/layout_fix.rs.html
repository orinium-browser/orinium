12/// Returns `true` when `info` represents a whitespace-only text node that
13/// renders as a single collapsible space.
14pub fn is_collapsible_whitespace_info(info: &InfoNode) -> bool {
15    matches!(&info.kind, NodeKind::Text { text, .. } if text.trim().is_empty())

18/// Returns `true` when `child` is a block-level layout child — i.e. it has
19/// `outer: Block` display, or it is a shrink-to-fit `FlowRoot` (an anonymous
20/// flex/grid wrapper).
21pub fn is_block_layout_child(child: &LayoutChild) -> bool {
22    child.node().is_some_and(|node| {
23        node.style.display.outer() == Some(OuterDisplay::Block)
24            || (node.style.display.inner() == Some(InnerDisplay::FlowRoot)
25                && node.style.size.auto_behavior == AutoSizeBehavior::ShrinkToFit)
26    })
27}

33/// **Bug fix:** `grid-template-columns`, `width`
34///
35/// During the intrinsic grid pass `ui_layout` measures block flex containers
36/// with their containing width. In a template such as `1fr auto 1fr`, that
37/// makes the auto track take the entire grid and leaves both fraction tracks
38/// at zero. The first layout still records the flex contents' actual extent in
39/// `children_box`, so use that intrinsic width and let a second layout resolve
40/// the tracks correctly.
41pub fn constrain_auto_grid_track_items(node: &mut LayoutNode) -> bool {
42    let mut changed = false;
43    for child in &mut node.children {
44        if let LayoutChild::Node(child) = child {
45            changed |= constrain_auto_grid_track_items(child);
46        }
47    }
48
49    if node.style.display.inner() != Some(InnerDisplay::Grid)
50        || node.style.grid_template_columns.is_empty()
51    {
52        return changed;
53    }
54
55    let mut item_index = 0usize;
56    for child in &mut node.children {
57        let LayoutChild::Node(child) = child else {
58            continue;
59        };
60        if child.style.display == Display::None || child.style.position.kind.is_out_of_flow() {
61            continue;
62        }
63
64        let auto_track = node
65            .style
66            .grid_template_columns
67            .get(item_index)
68            .is_some_and(|track| matches!(track, GridTrack::Breadth(LengthOrAuto::Auto)));
69        item_index += 1;
70        let self_aligned = child.style.spacing.margin_left == LengthOrAuto::Auto
71            || child.style.spacing.margin_right == LengthOrAuto::Auto;
72        if (!auto_track && !self_aligned) || child.style.size.width != LengthOrAuto::Auto {
73            continue;
74        }
75
76        let Some(model) = child.layout_box.iter().next() else {
77            continue;
78        };
79        let intrinsic_width = model.children_box.width.max(0.0);
80        if intrinsic_width > 0.0 && intrinsic_width + 0.5 < model.content_box.width {
81            child.style.size.width = LengthOrAuto::Length(Length::Px(intrinsic_width));
82            changed = true;
83        }
84    }
85
86    changed
87}

89/// **Bug fix:** `display`, `margin-left`, `margin-right`
90///
91/// `ui_layout` advances past an inline flow-root using its content width.
92/// CSS inline-blocks advance by their margin-box width instead. Adjacent
93/// atomic inline boxes end up overlapping their padding or horizontal margins.
94pub fn correct_atomic_inline_spacing(node: &mut LayoutNode) {
95    correct_atomic_inline_spacing_impl(node, None);
96}

98/// Like [`correct_atomic_inline_spacing`], but accepts an [`InfoNode`] so that
99/// per-child `text-align` information is available during correction.
100pub fn correct_atomic_inline_spacing_with_info(node: &mut LayoutNode, info: &InfoNode) {
101    correct_atomic_inline_spacing_impl(node, Some(info));
102}

104/// Core implementation of atomic-inline spacing correction. When `info` is
105/// supplied, `text-align` from the container is respected for the first item on
106/// each line.
107fn correct_atomic_inline_spacing_impl(node: &mut LayoutNode, info: Option<&InfoNode>) {
108    let containing_rect = node.layout_box.iter().next().map(|model| model.content_box);
109    let containing_width = containing_rect.map(|rect| rect.width);
110    let text_align = info
111        .and_then(|info| match &info.kind {
112            NodeKind::Container { style, .. } => Some(style.text_align),
113            _ => None,
114        })
115        .unwrap_or_default();
116    let wraps_inline_content = matches!(
117        node.style.display.inner(),
118        Some(InnerDisplay::Flow | InnerDisplay::FlowRoot)
119    );
120    let mut previous: Option<(f32, f32)> = None;
121    let mut line_y: Option<f32> = None;
122    let mut line_start_x = 0.0;
123    let mut line_bottom = 0.0;
124    let mut line_margin_bottom = 0.0;
125    let mut preceding_block_bottom: Option<(f32, f32)> = None;
126
127    for (child_index, child) in node.children.iter_mut().enumerate() {
128        let LayoutChild::Node(child) = child else {
129            continue;
130        };
131
132        let child_info = info.and_then(|info| info.children.get(child_index));
133        correct_atomic_inline_spacing_impl(child, child_info);
134
135        let is_atomic_inline = child.style.display.outer() == Some(OuterDisplay::Inline)
136            && child.style.display.inner() != Some(InnerDisplay::Flow);
137
138        if is_atomic_inline && let Some(model) = child.layout_box.iter().next() {
139            let rect = model.border_box;
140            let margin_left = fixed_nonnegative_px(&child.style.spacing.margin_left);
141            let margin_right = fixed_nonnegative_px(&child.style.spacing.margin_right);
142            let margin_top = fixed_nonnegative_px(&child.style.spacing.margin_top);
143            let margin_bottom = fixed_nonnegative_px(&child.style.spacing.margin_bottom);
144
145            if line_y.is_none_or(|y| (y - rect.y).abs() >= 0.5) {
146                previous = None;
147                line_y = Some(rect.y);
148                line_start_x = rect.x;
149                line_bottom = rect.bottom();
150                line_margin_bottom = margin_bottom;
151            }
152
153            let mut desired_x = match previous {
154                Some((right, previous_margin_right)) => right + previous_margin_right + margin_left,
155                None => {
156                    let margin_width = margin_left + rect.width + margin_right;
157                    let aligned_x = containing_rect.map_or(rect.x, |containing| {
158                        let free_space = (containing.width - margin_width).max(0.0);
159                        containing.x
160                            + match text_align {
161                                TextAlign::Left => 0.0,
162                                TextAlign::Center => free_space / 2.0,
163                                TextAlign::Right => free_space,
164                            }
165                    });
166                    aligned_x + margin_left
167                }
168            };
169            // ui_layout positions atomic inline boxes at the line origin but
170            // does not include their vertical margins in that position. The
171            // margin box, rather than the border box, is what participates in
172            // inline formatting (notably a full-width inline-block <main>
173            // placed below a fixed header).
174            let mut desired_y = rect.y + margin_top;
175            if previous.is_none()
176                && let Some((block_bottom, block_margin_bottom)) = preceding_block_bottom
177            {
178                desired_y = desired_y.max(block_bottom + block_margin_bottom + margin_top);
179            }
180            let exceeds_line = previous.is_some()
181                && wraps_inline_content
182                && containing_width
183                    .is_some_and(|width| desired_x + rect.width + margin_right > width + 0.5);
184            if exceeds_line {
185                desired_x = line_start_x + margin_left;
186                desired_y = line_bottom + line_margin_bottom + margin_top;
187                line_y = Some(desired_y);
188                line_bottom = desired_y + rect.height;
189                line_margin_bottom = margin_bottom;
190            }
191
192            let shift_x = desired_x - rect.x;
193            if shift_x.abs() >= 0.01 {
194                shift_layout_box_x(&mut child.layout_box, shift_x);
195            }
196            let shift_y = desired_y - rect.y;
197            if shift_y.abs() >= 0.01 {
198                shift_layout_box_y(&mut child.layout_box, shift_y);
199            }
200
201            line_bottom = line_bottom.max(desired_y + rect.height);
202            line_margin_bottom = line_margin_bottom.max(margin_bottom);
203            previous = Some((desired_x + rect.width, margin_right));
204        } else if matches!(child.layout_box, ui_layout::LayoutBox::BlockBox(_)) {
205            previous = None;
206            line_y = None;
207            if !child.style.position.kind.is_out_of_flow()
208                && let Some(model) = child.layout_box.iter().next()
209            {
210                preceding_block_bottom = Some((
211                    model.border_box.bottom(),
212                    fixed_nonnegative_px(&child.style.spacing.margin_bottom),
213                ));
214            }
215        }
216    }
217
218    expand_auto_inline_width_to_children(node);
219    correct_single_row_grid_inline_alignment(node);
220}

222/// **Supplement:** `width`
223///
224/// `ui_layout` does not handle `width: auto` for inline containers wrapping
225/// block children. Grow the inline container's width to accommodate the
226/// widest child's margin box.
227fn expand_auto_inline_width_to_children(node: &mut LayoutNode) {
228    if node.style.display.outer() != Some(OuterDisplay::Inline)
229        || node.style.size.width != LengthOrAuto::Auto
230    {
231        return;
232    }
233    // Use the margin-box width of the widest child.
234    let required_width = node
235        .children
236        .iter()
237        .filter_map(LayoutChild::node)
238        .filter_map(|child| {
239            child.layout_box.iter().next().map(|model| {
240                model.border_box.right() + fixed_nonnegative_px(&child.style.spacing.margin_right)
241            })
242        })
243        .fold(0.0, f32::max);
244    if required_width <= 0.0 {
245        return;
246    }
247    if let Some(model) = node.layout_box.iter().next() {
248        let extra = required_width - model.content_box.width;
249        if extra > 0.0 {
250            match &mut node.layout_box {
251                ui_layout::LayoutBox::BlockBox(model) => {
252                    model.content_box.width += extra;
253                    model.padding_box.width += extra;
254                    model.border_box.width += extra;
255                    model.children_box.width = model.children_box.width.max(required_width);
256                }
257                ui_layout::LayoutBox::InlineBox(inline) => {
258                    inline.box_model.content_box.width += extra;
259                    inline.box_model.padding_box.width += extra;
260                    inline.box_model.border_box.width += extra;
261                    inline.box_model.children_box.width =
262                        inline.box_model.children_box.width.max(required_width);
263                    if let Some(last) = inline.line_spans.last_mut() {
264                        last.x_range.end += extra;
265                    }
266                }
267                ui_layout::LayoutBox::None => {}
268            }
269        }
270    }
271}

277/// **Supplement:** custom flex item `layout` method
278///
279/// `ui_layout` measures and positions direct custom flex items but does not
280/// call their `layout` method. Render-time text-flow lookup therefore finds no
281/// spans for text directly inside an `inline-flex` element. Walk the tree and
282/// invoke `layout` for any custom flex item whose result is missing.
283pub fn refresh_missing_text_layout_results(
284    layout: &mut LayoutNode,
285    info: &InfoNode,
286    viewport: (f32, f32),
287) {
288    let containing = layout
289        .layout_box
290        .iter()
291        .next()
292        .map(|model| (model.content_box.width, model.content_box.height))
293        .unwrap_or(viewport);
294
295    for (layout_child, info_child) in layout.children.iter_mut().zip(&info.children) {
296        match (layout_child, &info_child.kind) {
297            (LayoutChild::Node(child), _) => {
298                refresh_missing_text_layout_results(child, info_child, viewport);
299            }
300            (LayoutChild::Custom(custom), NodeKind::Text { text_id, .. })
301                if TextFlowLayouter::get_result(*text_id).is_none() =>
302            {
303                let Some(box_model) = custom.result().map(|result| result.box_model.clone()) else {
304                    continue;
305                };
306                let line_height = custom
307                    .style()
308                    .line_height
309                    .resolve_with(Some(containing.0), viewport.0, viewport.1)
310                    .unwrap_or(box_model.border_box.height);
311                let _ = custom.layouter_mut().layout(&ui_layout::LayoutContext {
312                    containing_block_width: Some(containing.0),
313                    containing_block_height: Some(containing.1),
314                    start_pos: (box_model.border_box.x, box_model.border_box.y),
315                    available_inline_size: box_model.border_box.width.max(1.0),
316                    float_space: None,
317                    line_height,
318                    viewport_width: viewport.0,
319                    viewport_height: viewport.1,
320                });
321            }
322            _ => {}
323        }
324    }
325}

327/// **Supplement:** `margin-left`, `margin-right`, `column-gap`
328///
329/// `ui_layout` does not implement `margin-inline: auto` for grid items. When
330/// all grid items land in a single row, resolve auto left/right margins within
331/// each track so that items are horizontally centered or right-aligned.
332fn correct_single_row_grid_inline_alignment(node: &mut LayoutNode) {
333    if node.style.display.inner() != Some(InnerDisplay::Grid) {
334        return;
335    }
336    let Some(content_box) = node.layout_box.iter().next().map(|model| model.content_box) else {
337        return;
338    };
339    let column_gap = fixed_nonnegative_px(&node.style.column_gap);
340    let item_indices: Vec<usize> = node
341        .children
342        .iter()
343        .enumerate()
344        .filter_map(|(index, child)| {
345            child.node().and_then(|child| {
346                (child.style.display != Display::None
347                    && !child.style.position.kind.is_out_of_flow())
348                .then_some(index)
349            })
350        })
351        .collect();
352    if item_indices.len() > node.style.grid_template_columns.len() {
353        return;
354    }
355
356    for (position, index) in item_indices.iter().copied().enumerate() {
357        let next_x = item_indices
358            .get(position + 1)
359            .and_then(|next| node.children[*next].node())
360            .and_then(|next| next.layout_box.iter().next())
361            .map(|model| model.border_box.x - column_gap)
362            .unwrap_or(content_box.width);
363        let child = node.children[index].node_mut().expect("grid item");
364        let left_auto = child.style.spacing.margin_left == LengthOrAuto::Auto;
365        let right_auto = child.style.spacing.margin_right == LengthOrAuto::Auto;
366        if !left_auto && !right_auto {
367            continue;
368        }
369        let Some(model) = child.layout_box.iter().next() else {
370            continue;
371        };
372        let track_start = model.border_box.x;
373        let free_space = (next_x - track_start - model.border_box.width).max(0.0);
374        let offset = match (left_auto, right_auto) {
375            (true, true) => free_space / 2.0,
376            (true, false) => free_space,
377            _ => 0.0,
378        };
379        shift_layout_box_x(&mut child.layout_box, offset);
380    }
381}

387/// Shift every layer (content, padding, border, children) of a layout box
388/// horizontally by `shift_x` pixels.
389fn shift_layout_box_x(layout_box: &mut ui_layout::LayoutBox, shift_x: f32) {
390    let shift_model = |model: &mut ui_layout::BoxModel| {
391        model.border_box.x += shift_x;
392        model.padding_box.x += shift_x;
393        model.content_box.x += shift_x;
394        model.children_box.x += shift_x;
395    };
396    match layout_box {
397        ui_layout::LayoutBox::None => {}
398        ui_layout::LayoutBox::BlockBox(model) => shift_model(model),
399        ui_layout::LayoutBox::InlineBox(inline) => {
400            shift_model(&mut inline.box_model);
401            for span in &mut inline.line_spans {
402                span.line_pos.0 += shift_x;
403            }
404        }
405    }
406}

408/// Shift every layer (content, padding, border, children) of a layout box
409/// vertically by `shift_y` pixels.
410fn shift_layout_box_y(layout_box: &mut ui_layout::LayoutBox, shift_y: f32) {
411    let shift_model = |model: &mut ui_layout::BoxModel| {
412        model.border_box.y += shift_y;
413        model.padding_box.y += shift_y;
414        model.content_box.y += shift_y;
415        model.children_box.y += shift_y;
416    };
417    match layout_box {
418        ui_layout::LayoutBox::None => {}
419        ui_layout::LayoutBox::BlockBox(model) => shift_model(model),
420        ui_layout::LayoutBox::InlineBox(inline) => {
421            shift_model(&mut inline.box_model);
422            for span in &mut inline.line_spans {
423                span.line_pos.1 += shift_y;
424            }
425        }
426    }
427}

429/// Extract a non-negative pixel value from a `LengthOrAuto`, returning `0.0`
430/// for `Auto` or non-pixel lengths.
431fn fixed_nonnegative_px(value: &LengthOrAuto) -> f32 {
432    match value {
433        LengthOrAuto::Length(Length::Px(value)) => value.max(0.0),
434        _ => 0.0,
435    }
