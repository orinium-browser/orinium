93pub(crate) fn element_info(html_node: &HtmlNodeType) -> Option<ElementInfo> {
94    let HtmlNodeType::Element {
95        tag_name,
96        attributes,
97        ..
98    } = html_node
99    else {
100        return None;
101    };
102    Some(ElementInfo {
103        tag_name: tag_name.as_str().into(),
104        id: attributes
105            .iter()
106            .find(|attribute| attribute.name == "id")
107            .map(|attribute| Arc::from(attribute.value.as_str())),
108        classes: attributes
109            .iter()
110            .find(|attribute| attribute.name == "class")
111            .map(|attribute| {
112                split_html_space(&attribute.value)
113                    .map(Arc::<str>::from)
114                    .collect()
115            })
116            .unwrap_or_default(),
117        attributes: attributes
118            .iter()
119            .map(|attribute| {
120                (
121                    Arc::<str>::from(attribute.name.as_str()),
122                    Arc::<str>::from(attribute.value.as_str()),
123                )
124            })
125            .collect(),
126        element_index: 1,
127        element_count: 1,
128        type_index: 1,
129        type_count: 1,
130        previous_siblings: ElementChain::default(),
131        // A lone node carries no tree context, so both are filled in by the
132        // callers that hold a snapshot. `has_parent` is optimistic here because
133        // the structural pseudo-classes only use it to reject a tree root,
134        // which a snapshot-less caller never describes.
135        has_parent: true,
136        is_empty: false,
137        lang: None,
138        checked: false,
139        hovered: false,
140        active: false,
141    })
142}

162/// The Selectors 4 `:empty` test: no element children, and no text node whose
163/// data is non-empty. Comments and processing instructions are ignored, as are
164/// empty text nodes.
165pub(crate) fn is_empty_element(snapshot: &DomSnapshot, id: NodeId) -> bool {
166    snapshot
167        .children(id)
168        .iter()
169        .all(|&child| match &snapshot.node(child).kind {
170            HtmlNodeType::Text(text) => text.is_empty(),
171            HtmlNodeType::Element { .. } => false,
172            _ => true,
173        })
174}

176pub(crate) fn element_sibling_infos(
177    snapshot: &DomSnapshot,
178    children: &[NodeId],
179    parent_lang: Option<&str>,
180    interaction: &InteractionState,
181) -> Vec<Option<ElementInfo>> {
182    let mut type_counts = HashMap::<Arc<str>, usize>::new();
183    for &child in children {
184        if let Some(tag_name) = snapshot.node(child).kind.tag_name() {
185            *type_counts.entry(Arc::from(tag_name)).or_default() += 1;
186        }
187    }
188    let element_count = type_counts.values().sum();
189
190    let mut seen_types = HashMap::<Arc<str>, usize>::new();
191    let mut previous_siblings = ElementChain::default();
192    let mut element_index = 0;
193    children
194        .iter()
195        .map(|&child| {
196            let mut info = match element_info(&snapshot.node(child).kind) {
197                Some(info) => info,
198                None => return None,
199            };
200            element_index += 1;
201            let seen = seen_types.entry(Arc::clone(&info.tag_name)).or_default();
202            *seen += 1;
203            info.element_index = element_index;
204            info.element_count = element_count;
205            info.type_index = *seen;
206            info.type_count = type_counts[&info.tag_name];
207            info.previous_siblings = previous_siblings.clone();
208            info.is_empty = is_empty_element(snapshot, child);
209            info.checked = snapshot.node(child).checked;
210            // `:lang()` sees the element's own tag, or the one inherited from
211            // the nearest ancestor that declares one.
212            info.lang = own_lang(snapshot, child).or_else(|| parent_lang.map(Arc::<str>::from));
213            info.hovered = interaction.hovered_path.iter().any(|&id| id == child);
214            info.active = interaction.active_path.iter().any(|&id| id == child);
215
216            let mut sibling = info.clone();
217            sibling.previous_siblings = ElementChain::default();
218            previous_siblings = previous_siblings.prepend(Some(sibling));
219            Some(info)
220        })
221        .collect()
222}

241/// Convert a resolved `Length` to an absolute pixel value for `LineHeight::Px`.
242pub(super) fn length_to_px(len: &Length, font_size: f32) -> f32 {
243    match len {
244        Length::Px(v) => *v,
245        Length::Percent(v) => v * font_size / 100.0,
246        Length::Add(a, b) => length_to_px(a, font_size) + length_to_px(b, font_size),
247        Length::Sub(a, b) => length_to_px(a, font_size) - length_to_px(b, font_size),
248        Length::Mul(a, f) => length_to_px(a, font_size) * f,
249        Length::Div(a, f) => length_to_px(a, font_size) / f,
250        _ => font_size * DEFAULT_LINE_FACTOR,
251    }
252}

304/// Builds layout and render trees with decoded images keyed by their `src` value.
305///
306/// `viewport` supplies the viewport size used to evaluate `@media` queries while
307/// resolving the style cascade (e.g. `@media (min-width: 768px)`).
308#[allow(clippy::too_many_arguments)]
309pub fn build_layout_and_info_with_images(
310    dom: &NodeRef<HtmlNodeType>,
311    resolved_styles: &ResolvedStyles,
312    measurer: Arc<dyn text::TextMeasurer>,
313    audio_sinks: Arc<dyn AudioSinkFactory>,

324    let (layout, info, _) = build_layout_and_info_from_snapshot(

328        measurer,
329        audio_sinks,

336        None,
337        None,
338        &InteractionState::default(),
339        &HashMap::new(),
340        &HashMap::new(),
341    );
342    (layout, info)
343}

345/// Builds layout and render trees from a [`DomSnapshot`].
346///
347/// `write_back_sender` (when set) is cloned per text input so value changes
348/// are reported as `(node id, value)` on the channel instead of mutating the
349/// DOM directly, which allows this function to run off the UI thread.
350///
351/// `system_color_scheme` seeds the root element's used color scheme and is
352/// used to resolve `color-scheme: light dark` and `light-dark()` values.
353///
354/// `match_cache` (when set) memoizes selector match results across builds that
355/// share the same DOM snapshot and rule set. The caller is responsible for
356/// clearing it when either input changes.
357#[allow(clippy::too_many_arguments)]
358pub fn build_layout_and_info_from_snapshot(
359    snapshot: &DomSnapshot,
360    root: NodeId,
361    rule_set: &RuleSet,
362    measurer: Arc<dyn text::TextMeasurer>,
363    audio_sinks: Arc<dyn AudioSinkFactory>,
364    parent: InheritedCss,
365    mut chain: ElementChain,
366    system_color_scheme: ColorScheme,
367    scripting_mode: ScriptingMode,
368    images: &HashMap<String, Image>,
369    audio: &HashMap<String, Arc<[u8]>>,
370    write_back_sender: Option<DomWriteBack>,
371    mut match_cache: Option<&mut SelectorMatchCache>,
372    interaction: &InteractionState,
373    existing_custom_nodes: &HashMap<u32, Arc<dyn CustomNode>>,
374    animated_overrides: &HashMap<NodeId, HashMap<String, CssValue>>,
375) -> (LayoutNode, InfoNode, crate::engine::animation::AnimationInputs) {
376    perf_scope!(total);
377
378    #[cfg(any(feature = "profile", debug_assertions))]
379    let mut css_match_time = std::time::Duration::ZERO;
380    #[cfg(any(feature = "profile", debug_assertions))]
381    let mut apply_decl_time = std::time::Duration::ZERO;
382    #[cfg(any(feature = "profile", debug_assertions))]
383    let mut custom_node_time = std::time::Duration::ZERO;
384    #[cfg(any(feature = "profile", debug_assertions))]
385    let mut text_layout_time = std::time::Duration::ZERO;
386    #[cfg(any(feature = "profile", debug_assertions))]
387    let mut exit_phase_time = std::time::Duration::ZERO;
388    #[cfg(any(feature = "profile", debug_assertions))]
389    let mut sibling_info_time = std::time::Duration::ZERO;
390    #[cfg(any(feature = "profile", debug_assertions))]
391    let mut whitespace_keep_time = std::time::Duration::ZERO;
392    #[cfg(any(feature = "profile", debug_assertions))]
393    let mut enter_prep_time = std::time::Duration::ZERO;
394    #[cfg(any(feature = "profile", debug_assertions))]
395    let mut child_slot_build_time = std::time::Duration::ZERO;
396    #[cfg(any(feature = "profile", debug_assertions))]
397    let mut leaf_assemble_time = std::time::Duration::ZERO;
398    #[cfg(any(feature = "profile", debug_assertions))]
399    let mut push_children_time = std::time::Duration::ZERO;
400    #[cfg(any(feature = "profile", debug_assertions))]
401    let mut exit_setup_time = std::time::Duration::ZERO;
402    #[cfg(any(feature = "profile", debug_assertions))]
403    let mut exit_build_time = std::time::Duration::ZERO;
404    #[cfg(any(feature = "profile", debug_assertions))]
405    let mut node_count = 0u64;
406
407    #[cfg(any(feature = "profile", debug_assertions))]
408    let mut cand_stats = CandidateMetrics::default();
409
410    let registry = ComponentRegistry::new();
411    /*
412     * Build the initial element chain for the root node.
413     */
414    if let Some(mut info) = element_info(&snapshot.node(root).kind) {
415        // The snapshot root has no parent, so it must not satisfy
416        // `:first-child` and friends.
417        info.has_parent = false;
418        info.is_empty = is_empty_element(snapshot, root);
419        info.hovered = interaction.hovered_path.iter().any(|&id| id == root);
420        info.active = interaction.active_path.iter().any(|&id| id == root);
421        chain = chain.prepend(Some(info));
422    }
423
424    // ── Explicit post-order stack (index-based to avoid borrow conflicts) ──
425
426    let mut stack: Vec<StackFrame> = Vec::new();
427    stack.push(StackFrame {
428        dom: root,
429        lang: own_lang(snapshot, root),
430        chain,
431        child: Arc::new(parent),
432        kind: None,
433        style: None,
434        parent_style: Style::default(),
435        parent_container_style: ContainerStyle::default(),
436        child_slots: Vec::new(),
437        element_children: Vec::new(),
438    });
439
440    let mut results: HashMap<NodeId, (LayoutNode, InfoNode)> = HashMap::new();
441    let mut animation_inputs: AnimationInputs = HashMap::new();
442
443    // We use an index instead of .last_mut() so that push/pop don't conflict
444    // with the mutable reference to the current frame.
445    while let Some(top_idx) = {
446        if stack.is_empty() {
447            None
448        } else {
449            Some(stack.len() - 1)
450        }
451    } {
452        // Phase check must happen BEFORE borrowing stack[top_idx] mutably.
453        let is_entered = stack[top_idx].kind.is_some();
454
455        if !is_entered {
456            // ── ENTER phase ──────────────────────────────────────────────
457            // Read frame state we need before taking any mutable action.
458            let chain_for_css = stack[top_idx].chain.clone();
459            let child_css = Arc::clone(&stack[top_idx].child);
460
461            let html_node = &snapshot.node(stack[top_idx].dom).kind;
462            let node_id = stack[top_idx].dom;
463            let mut text_style = child_css.text_style.clone();
464            let mut text_flow_style = child_css.text_flow_style;
465            let mut container_style = ContainerStyle::default();
466            let mut style = Style::default();
467            let mut overflow = Overflow::default();
468            let parent_style = stack[top_idx].parent_style.clone();
469            let parent_container_style = stack[top_idx].parent_container_style.clone();
470            let parent_text_style = text_style.clone();
471            let parent_text_flow_style = text_flow_style;
472
473            // Inherit container visibility
474            container_style.visibility = parent_container_style.visibility;
475
476            // Collect CSS candidates.
477            perf_scope!(css_match);
478            let (candidates, custom_property_candidates, pseudo_candidates) =
479                if let HtmlNodeType::Element { .. } = html_node {
480                    let (candidates, custom_property_candidates, pseudo_candidates) =
481                        collect_candidates(
482                            rule_set,
483                            &chain_for_css,
484                            node_id,
485                            match_cache.as_deref_mut(),
486                            #[cfg(any(feature = "profile", debug_assertions))]
487                            &mut cand_stats,
488                        );
489                    (
490                        Some(candidates),
491                        Some(custom_property_candidates),
492                        pseudo_candidates,
493                    )
494                } else {
495                    (None, None, HashMap::new())
496                };
497            #[cfg(any(feature = "profile", debug_assertions))]
498            {
499                css_match_time += css_match.elapsed();
500            }
501
502            perf_scope!(enter_prep);
503            // Inherit custom properties by sharing the parent's map unless this
504            // element introduces new `--var` values (copy-on-write applies
505            // cascade-discovered custom properties lazily).
506            let mut custom_properties = Arc::clone(&child_css.custom_props);
507            if let Some(own) = custom_property_candidates
508                && !own.is_empty()
509            {
510                Arc::make_mut(&mut custom_properties).extend(own);
511            }
512
513            // Resolve the used color scheme for this element. `light-dark()`
514            // and system colors resolve against it, and it is inherited by
515            // descendants that do not set `color-scheme` themselves.
516            let used_color_scheme = {
517                let declaration = candidates
518                    .as_ref()
519                    .and_then(|c| c.get("color-scheme"))
520                    .and_then(|d| {
521                        DeclarationResolver::resolve_var(
522                            &d.value,
523                            &custom_properties,
524                            &mut HashSet::new(),
525                        )
526                    });
527                resolve_used_color_scheme(
528                    declaration.as_ref(),
529                    child_css.color_scheme,
530                    system_color_scheme,
531                )
532            };
533            #[cfg(any(feature = "profile", debug_assertions))]
534            {
535                enter_prep_time += enter_prep.elapsed();
536            }
537
538            // Apply CSS declarations.
539            perf_scope!(apply_decl);
540            let mut anim_effective: std::collections::HashMap<String, CssValue> = std::collections::HashMap::new();
541            let mut anim_transition = anim_parse::TransitionInputs::default();
542            let mut anim_animation = anim_parse::AnimationInputs::default();
543            if let Some(candidates) = &candidates {
544                // The candidates map dedupes per property name (cascade winner),
545                // but a shorthand and its longhand (e.g. `padding` and
546                // `padding-top`) are distinct keys. Iterating a HashMap applies
547                // them in random order, so re-sort by source order to make the
548                // cascade deterministic.
549                let mut candidates: Vec<_> = candidates.values().collect();
550                candidates.sort_by_key(|declaration| declaration.order);
551
552                for declaration in candidates {
553                    if declaration.name.starts_with("--") {
554                        continue;
555                    }
556                    let Some(value) = DeclarationResolver::resolve_var(
557                        &declaration.value,
558                        &custom_properties,
559                        &mut HashSet::new(),
560                    ) else {
561                        continue;
562                    };
563                    apply_declaration(
564                        &declaration.name,
565                        &value,
566                        &mut style,
567                        &mut container_style,
568                        &mut text_style,
569                        &mut text_flow_style,
570                        &parent_style,
571                        &parent_container_style,
572                        &parent_text_style,
573                        &parent_text_flow_style,
574                        &mut overflow,
575                        used_color_scheme,
576                    );
577                    collect_animation_decl(
578                        &declaration.name,
579                        &value,
580                        &mut anim_effective,
581                        &mut anim_transition,
582                        &mut anim_animation,
583                    );
584                }
585            }
586
587            // Apply the element's inline `style` attribute. Inline styles are
588            // author-origin declarations with the highest specificity, so they
589            // override stylesheet rules — unless the stylesheet rule was
590            // `!important`, which still wins over a non-`!important` inline
591            // declaration.
592            if let Some(style_attr) = html_node.get_attr("style") {
593                let inline_declarations = resolve_inline_style(style_attr);
594                for (name, value, important) in &inline_declarations {
595                    if name.starts_with("--") {
596                        let stylesheet_important = candidates
597                            .as_ref()
598                            .and_then(|c| c.get(name))
599                            .is_some_and(|declaration| declaration.important);
600                        if *important || !stylesheet_important {
601                            set_inline_custom_property(
602                                Arc::make_mut(&mut custom_properties),
603                                name.clone(),
604                                value.clone(),
605                                *important,
606                            );
607                        }
608                    }
609                }
610                for (name, value, important) in inline_declarations {
611                    if name.starts_with("--") {
612                        continue;
613                    }
614                    let stylesheet_important = candidates
615                        .as_ref()
616                        .and_then(|c| c.get(&name))
617                        .is_some_and(|declaration| declaration.important);

1034                            "border-top-color",
1035                            "border-right-color",
1036                            "border-bottom-color",
1037                            "border-left-color",
1038                        ] {

1044                            .entry("opacity".to_string())
1045                            .or_insert(CssValue::Number(container_opacity));
1046                        base_values
1047                            .entry("color".to_string())
1048                            .or_insert_with(|| {
1049                                CssValue::Color(format!(
1050                                    "{:02x}{:02x}{:02x}{:02x}",
1051                                    text_style.color.0,

1053                                    text_style.color.2,
1054                                    text_style.color.3

1058                            ptr,

1060                                transitions,
1061                                animations,

1220                        || matches!(child_node, HtmlNodeType::DocumentFragment)

1226                    }
1227                }

1245                    }
1246                }

1258                #[cfg(any(feature = "profile", debug_assertions))]

1263                    .into_iter()

1268                        }

1270                            ChildSlot::Inline(layout, info) => Some((layout, *info)),
1271                            ChildSlot::Element(_) => None,
1272                        }
1273                    })
1274                    .unzip();
1275                let layout = LayoutNode::with_children(style.clone(), layout_children);
1276                let info = InfoNode {
1277                    kind,
1278                    children: info_children,
1279                    dom_id: Some(stack[top_idx].dom),
1280                };
1281                let ptr = stack[top_idx].dom;
1282                results.insert(ptr, (layout, info));
1283                #[cfg(any(feature = "profile", debug_assertions))]
1284                {
1285                    leaf_assemble_time += leaf_assemble.elapsed();
1286                    node_count += 1;
1287                }
1288                stack.pop();
1289            } else {
1290                // ── Has element children → save state, push children ──
1291                perf_scope!(push_children);
1292                let parent_chain = stack[top_idx].chain.clone();
1293                let parent_container = match &kind {
1294                    NodeKind::Container { style, .. } => style.clone(),
1295                    _ => ContainerStyle::default(),
1296                };
1297                stack[top_idx].kind = Some(kind);
1298                stack[top_idx].parent_style = style.clone();
1299                stack[top_idx].parent_container_style = parent_container;
1300                stack[top_idx].style = Some(style);
1301                stack[top_idx].child = child;
1302                stack[top_idx].child_slots = child_slots;
1303                stack[top_idx].element_children = element_kids;
1304
1305                // Build child chains and push frames.
1306                // Clone element_kids before the immutable borrow below
1307                // so we don't hold &mut stack[] while pushing.
1308                let kids_for_push: Vec<_> = {
1309                    let f = &stack[top_idx];
1310                    f.element_children.clone()
1311                };
1312                let child_css = Arc::clone(&stack[top_idx].child);
1313                perf_scope!(sibling_info);
1314                let parent_lang = stack[top_idx]
1315                    .lang
1316                    .clone()
1317                    .or_else(|| own_lang(snapshot, stack[top_idx].dom));
1318                let mut kid_infos = element_sibling_infos(
1319                    snapshot,
1320                    &kids_for_push,
1321                    parent_lang.as_deref(),
1322                    interaction,
1323                );
1324                if !is_element_node(snapshot, stack[top_idx].dom) {
1325                    for info in kid_infos.iter_mut().flatten() {
1326                        info.has_parent = false;
1327                    }
1328                }
1329                #[cfg(any(feature = "profile", debug_assertions))]
1330                {
1331                    sibling_info_time += sibling_info.elapsed();
1332                }
1333                let parent_style_for_children = stack[top_idx].parent_style.clone();
1334                let parent_container_for_children = stack[top_idx].parent_container_style.clone();
1335                for (&kid, info) in kids_for_push.iter().zip(kid_infos).rev() {
1336                    let kid_lang = info.as_ref().and_then(|info| info.lang.clone());
1337                    stack.push(StackFrame {
1338                        dom: kid,
1339                        lang: kid_lang,
1340                        chain: parent_chain.prepend(info),
1341                        child: Arc::clone(&child_css),
1342                        kind: None,
1343                        style: None,
1344                        parent_style: parent_style_for_children.clone(),
1345                        parent_container_style: parent_container_for_children.clone(),
1346                        child_slots: Vec::new(),
1347                        element_children: Vec::new(),
1348                    });
1349                }
1350                #[cfg(any(feature = "profile", debug_assertions))]
1351                {
1352                    push_children_time += push_children.elapsed();
1353                }
1354            }
1355        } else {
1356            // ── EXIT phase ────────────────────────────────────────────────
1357            // Take ownership of frame data for building results.
1358            perf_scope!(exit_phase);
1359            perf_scope!(exit_setup);
1360
1361            let frame = stack.swap_remove(top_idx);
1362
1363            let style = frame.style.as_ref().unwrap().clone();
1364            let kind = frame.kind.as_ref().unwrap().clone();
1365
1366            // Collect element children results.
1367            let mut element_results: Vec<(LayoutChild, InfoNode)> = Vec::new();
1368
1369            for &kid in &frame.element_children {
1370                if let Some((child_layout, child_info)) = results.remove(&kid) {
1371                    element_results.push((child_layout.into(), child_info));
1372                }
1373            }
1374
1375            // Handle html→body background inheritance.
1376            let mut final_kind = kind;
1377            if snapshot.node(frame.dom).kind.tag_name() == Some("html") {
1378                let should_inherit = final_kind.is_container_with_transparent_bg();
1379                if should_inherit {
1380                    for (_, child_info) in &element_results {
1381                        if child_info
1382                            .dom_id
1383                            .and_then(|id| snapshot.node(id).kind.tag_name())
1384                            == Some("body")
1385                        {
1386                            let child_bg = child_info.kind.container_bg();
1387                            if let Some(bg) = child_bg
1388                                && let NodeKind::Container { ref mut style, .. } = final_kind
1389                            {
1390                                style.background = bg.clone();
1391                            }
1392                        }
1393                    }
1394                }
1395            }
1396
1397            #[cfg(any(feature = "profile", debug_assertions))]
1398            {
1399                exit_setup_time += exit_setup.elapsed();
1400            }
1401
1402            perf_scope!(exit_build);
1403            let mut element_results: Vec<_> = element_results.into_iter().map(Some).collect();
1404
1405            // Whitespace-only text nodes between two block-level siblings, or adjacent
1406            // to a `<br>`, would otherwise create stray inline boxes and spurious line
1407            // boxes in block, flex, and grid containers. Drop them now that every
1408            // sibling's display is resolved.
1409            perf_scope!(whitespace_keep);
1410            let keep = compute_whitespace_keep(&frame.child_slots, &element_results);
1411            #[cfg(any(feature = "profile", debug_assertions))]
1412            {
1413                whitespace_keep_time += whitespace_keep.elapsed();
1414            }
1415
1416            let mut all_layout: Vec<LayoutChild> = Vec::with_capacity(frame.child_slots.len());
1417            let mut all_info: Vec<InfoNode> = Vec::with_capacity(frame.child_slots.len());
1418
1419            for (i, slot) in frame.child_slots.into_iter().enumerate() {
1420                if !keep[i] {
1421                    continue;
1422                }
1423                let (lc, ic) = match slot {
1424                    ChildSlot::Inline(layout, info) => (layout, *info),
1425                    ChildSlot::Element(index) => element_results[index]
1426                        .take()
1427                        .expect("element child result must exist"),
1428                };
1429                all_layout.push(lc);
1430                all_info.push(ic);
1431            }
1432
1433            // Collapsible whitespace at a block boundary does not create an
1434            // anonymous line box. Keeping indentation-only DOM text here made
1435            // a block such as Scratch's `.page` start one default line-height
1436            // below the viewport.
1437            let keep: Vec<bool> = (0..all_layout.len())
1438                .map(|index| {
1439                    let collapsible = is_collapsible_whitespace_info(&all_info[index]);
1440                    let next_to_block = index == 0
1441                        || index + 1 == all_layout.len()
1442                        || index
1443                            .checked_sub(1)
1444                            .is_some_and(|previous| is_block_layout_child(&all_layout[previous]))
1445                        || all_layout.get(index + 1).is_some_and(is_block_layout_child);
1446                    !(collapsible && next_to_block)
1447                })
1448                .collect();
1449            all_layout = all_layout
1450                .into_iter()
1451                .zip(&keep)
1452                .filter_map(|(layout, keep)| keep.then_some(layout))
1453                .collect();
1454            all_info = all_info
1455                .into_iter()
1456                .zip(keep)
1457                .filter_map(|(info, keep)| keep.then_some(info))
1458                .collect();
1459
1460            // Grid and flex items are blockified by CSS Display. Keeping an
1461            // inline direct child makes its text-flow coordinates remain in
1462            // the parent's inline space, so item placement cannot move the
1463            // text with its box.
1464            if matches!(
1465                style.display.inner(),
1466                Some(InnerDisplay::Grid | InnerDisplay::Flex)
1467            ) {
1468                if style.display.inner() == Some(InnerDisplay::Grid) {
1469                    let columns = explicit_grid_track_count(&style.grid_template_columns);
1470                    let rows = explicit_grid_track_count(&style.grid_template_rows);
1471                    for child in &mut all_layout {
1472                        if let LayoutChild::Node(child) = child {
1473                            resolve_named_grid_area(child, &style.grid_template_areas);
1474                            resolve_grid_end_line(&mut child.style.grid_column, columns);
1475                            resolve_grid_end_line(&mut child.style.grid_row, rows);
1476                        }
1477                    }
1478                }
1479                for child in &mut all_layout {
1480                    if let LayoutChild::Node(child) = child
1481                        && child.style.display.outer() == Some(OuterDisplay::Inline)
1482                        && !child.style.position.kind.is_out_of_flow()
1483                        && let Display::OutsideInner { inner, .. } = child.style.display
1484                    {
1485                        child.style.display = Display::OutsideInner {
1486                            outer: OuterDisplay::Block,
1487                            inner,
1488                        };
1489                    }
1490                }
1491            }
1492
1493            let layout = LayoutNode::with_children(style, all_layout);
1494            let info = InfoNode {
1495                kind: final_kind,
1496                children: all_info,
1497                dom_id: Some(frame.dom),
1498            };
1499            let ptr = frame.dom;
1500            results.insert(ptr, (layout, info));
1501            #[cfg(any(feature = "profile", debug_assertions))]
1502            {
1503                exit_build_time += exit_build.elapsed();
1504                exit_phase_time += exit_phase.elapsed();
1505                node_count += 1;
1506            }
1507        }
1508    }
1509
1510    profile_log!(
1511        target: "LayoutRun",
1512        log::Level::Info,
1513        "[LayoutMetrics] total: {:?} (nodes: {})",
1514        total.elapsed(),
1515        node_count,
1516    );
1517    profile_log!(
1518        target: "LayoutRun",
1519        log::Level::Info,
1520        "[LayoutMetrics] css_match: {:?} | apply_decl: {:?}",
1521        css_match_time,
1522        apply_decl_time,
1523    );
1524    profile_log!(
1525        target: "LayoutRun",
1526        log::Level::Info,
1527        "[LayoutCandidates] elements: {} | examined: {} | matched: {}",
1528        cand_stats.elements_checked,
1529        cand_stats.candidates_examined,
1530        cand_stats.selectors_matched,
1531    );
1532    profile_log!(
1533        target: "LayoutRun",
1534        log::Level::Info,
1535        "[LayoutCandidates] query_time: {:?} | sel_match_time: {:?} | insert_time: {:?}",
1536        cand_stats.query_candidates_time,
1537        cand_stats.selector_match_time,
1538        cand_stats.cascade_insert_time,
1539    );
1540    profile_log!(
1541        target: "LayoutRun",
1542        log::Level::Info,
1543        "[LayoutMetrics] custom_node: {:?} | text_layout: {:?} | exit_phase: {:?}",
1544        custom_node_time,
1545        text_layout_time,
1546        exit_phase_time,
1547    );
1548    profile_log!(
1549        target: "LayoutRun",
1550        log::Level::Info,
1551        "[LayoutMetrics] sibling_info: {:?} | whitespace_keep: {:?}",
1552        sibling_info_time,
1553        whitespace_keep_time,
1554    );
1555    profile_log!(
1556        target: "LayoutRun",
1557        log::Level::Info,
1558        "[LayoutMetrics] enter_prep: {:?} | child_slot_build: {:?} | leaf_assemble: {:?}",
1559        enter_prep_time,
1560        child_slot_build_time,
1561        leaf_assemble_time,
1562    );
1563    profile_log!(
1564        target: "LayoutRun",
1565        log::Level::Info,
1566        "[LayoutMetrics] push_children: {:?} | exit_setup: {:?} | exit_build: {:?}",
1567        push_children_time,
1568        exit_setup_time,
1569        exit_build_time,
1570    );
1571    profile_log!(
1572        target: "LayoutRun",
1573        log::Level::Info,
1574        "[LayoutMetrics] rest: {:?}",
1575        total.elapsed().saturating_sub(
1576            css_match_time
1577                + apply_decl_time
1578                + custom_node_time
1579                + enter_prep_time
1580                + child_slot_build_time
1581                + leaf_assemble_time
1582                + push_children_time
1583                + exit_setup_time
1584                + exit_build_time
1585        ),
1586    );
1587
1588    let (layout, info_node) = results
1589        .remove(&root)
1590        .expect("root must have been processed");
1591    (layout, info_node, animation_inputs)
1592}

1596fn collect_animation_decl(
1597    name: &str,
1598    value: &CssValue,
1599    effective: &mut std::collections::HashMap<String, CssValue>,
1600    transition: &mut anim_parse::TransitionInputs,
1601    animation: &mut anim_parse::AnimationInputs,
1602) {
1603    let lname = name.to_ascii_lowercase();
1604    match lname.as_str() {
1605        "transition-property" => transition.property = Some(value.clone()),
1606        "transition-duration" => transition.duration = Some(value.clone()),
1607        "transition-delay" => transition.delay = Some(value.clone()),
1608        "transition-timing-function" => transition.timing_function = Some(value.clone()),
1609        "transition" => {
1610            // MVP shorthand: per-comma group parse property, times, easing.
1611            let groups = split_css_comma_groups(value);
1612            let mut props: Vec<CssValue> = Vec::new();
1613            let mut durs: Vec<CssValue> = Vec::new();
1614            let mut delays: Vec<CssValue> = Vec::new();
1615            let mut easings: Vec<CssValue> = Vec::new();
1616            for group in groups {
1617                let mut prop: Option<CssValue> = None;
1618                let mut times: Vec<CssValue> = Vec::new();
1619                let mut easing: Option<CssValue> = None;
1620                for item in flatten_css_value_items(&group) {
1621                    if anim_parse::parse_time_ms(&item).is_some() {
1622                        if times.len() < 2 {
1623                            times.push(item);
1624                        }
1625                        continue;
1626                    }
1627                    if easing.is_none() && anim_parse::parse_easing(&item).is_some() {
1628                        easing = Some(item);
1629                        continue;
1630                    }
1631                    if prop.is_none() {
1632                        if let CssValue::Keyword(_) | CssValue::String(_) = item {
1633                            prop = Some(item);
1634                        }
1635                    }
1636                }
1637                props.push(prop.unwrap_or_else(|| CssValue::Keyword("all".into())));
1638                durs.push(times.first().cloned().unwrap_or(CssValue::Length(0.0, Unit::S)));
1639                delays.push(times.get(1).cloned().unwrap_or(CssValue::Length(0.0, Unit::S)));
1640                easings.push(easing.unwrap_or_else(|| CssValue::Keyword("ease".into())));
1641            }
1642            if !props.is_empty() {
1643                transition.property = Some(join_css_comma_groups(props));
1644                transition.duration = Some(join_css_comma_groups(durs));
1645                transition.delay = Some(join_css_comma_groups(delays));
1646                transition.timing_function = Some(join_css_comma_groups(easings));
1647            }
1648        }
1649        "animation-name" => animation.name = Some(value.clone()),
1650        "animation-duration" => animation.duration = Some(value.clone()),
1651        "animation-delay" => animation.delay = Some(value.clone()),
1652        "animation-timing-function" => animation.timing_function = Some(value.clone()),
1653        "animation-iteration-count" => animation.iteration_count = Some(value.clone()),
1654        "animation-direction" => animation.direction = Some(value.clone()),
1655        "animation-fill-mode" => animation.fill_mode = Some(value.clone()),
1656        "animation-play-state" => animation.play_state = Some(value.clone()),
1657        "animation" => {
1658            // MVP shorthand: parse common tokens per comma group.
1659            let groups = split_css_comma_groups(value);
1660            let mut names: Vec<CssValue> = Vec::new();
1661            let mut durs: Vec<CssValue> = Vec::new();
1662            let mut delays: Vec<CssValue> = Vec::new();
1663            let mut easings: Vec<CssValue> = Vec::new();
1664            let mut iters: Vec<CssValue> = Vec::new();
1665            let mut dirs: Vec<CssValue> = Vec::new();
1666            let mut fills: Vec<CssValue> = Vec::new();
1667            let mut states: Vec<CssValue> = Vec::new();
1668            for group in groups {
1669                let mut name: Option<CssValue> = None;
1670                let mut times: Vec<CssValue> = Vec::new();
1671                let mut easing: Option<CssValue> = None;
1672                let mut iter: Option<CssValue> = None;
1673                let mut dir: Option<CssValue> = None;
1674                let mut fill: Option<CssValue> = None;
1675                let mut state: Option<CssValue> = None;
1676                for item in flatten_css_value_items(&group) {
1677                    if anim_parse::parse_time_ms(&item).is_some() {
1678                        if times.len() < 2 {
1679                            times.push(item);
1680                            continue;
1681                        }
1682                    }
1683                    if easing.is_none() && anim_parse::parse_easing(&item).is_some() {
1684                        easing = Some(item);
1685                        continue;
1686                    }
1687                    if iter.is_none() && anim_parse::parse_iteration_count(&item).is_some() {
1688                        iter = Some(item);
1689                        continue;
1690                    }
1691                    if dir.is_none() && anim_parse::parse_direction(&item).is_some() {
1692                        dir = Some(item);
1693                        continue;
1694                    }
1695                    if fill.is_none() && anim_parse::parse_fill_mode(&item).is_some() {
1696                        fill = Some(item);
1697                        continue;
1698                    }
1699                    if state.is_none() && anim_parse::parse_play_state(&item).is_some() {
1700                        state = Some(item);
1701                        continue;
1702                    }
1703                    if name.is_none() {
1704                        if let CssValue::Keyword(_) | CssValue::String(_) = item {
1705                            name = Some(item);
1706                        }
1707                    }
1708                }
1709                names.push(name.unwrap_or_else(|| CssValue::Keyword("none".into())));
1710                durs.push(times.first().cloned().unwrap_or(CssValue::Length(0.0, Unit::S)));
1711                delays.push(times.get(1).cloned().unwrap_or(CssValue::Length(0.0, Unit::S)));
1712                easings.push(easing.unwrap_or_else(|| CssValue::Keyword("ease".into())));
1713                iters.push(iter.unwrap_or(CssValue::Number(1.0)));
1714                dirs.push(dir.unwrap_or_else(|| CssValue::Keyword("normal".into())));
1715                fills.push(fill.unwrap_or_else(|| CssValue::Keyword("none".into())));
1716                states.push(state.unwrap_or_else(|| CssValue::Keyword("running".into())));
1717            }
1718            if !names.is_empty() {
1719                animation.name = Some(join_css_comma_groups(names));
1720                animation.duration = Some(join_css_comma_groups(durs));
1721                animation.delay = Some(join_css_comma_groups(delays));
1722                animation.timing_function = Some(join_css_comma_groups(easings));
1723                animation.iteration_count = Some(join_css_comma_groups(iters));
1724                animation.direction = Some(join_css_comma_groups(dirs));
1725                animation.fill_mode = Some(join_css_comma_groups(fills));
1726                animation.play_state = Some(join_css_comma_groups(states));
1727            }
1728        }
1729        _ => {}
1730    }
1731
1732    if matches!(
1733        lname.as_str(),
1734        "opacity"
1735            | "width"
1736            | "height"
1737            | "top"
1738            | "right"
1739            | "bottom"
1740            | "left"
1741            | "margin-top"
1742            | "margin-right"
1743            | "margin-bottom"
1744            | "margin-left"
1745            | "padding-top"
1746            | "padding-right"
1747            | "padding-bottom"
1748            | "padding-left"
1749            | "font-size"
1750            | "line-height"
1751            | "color"
1752            | "background-color"
1753            | "border-top-color"
1754            | "border-right-color"
1755            | "border-bottom-color"
1756            | "border-left-color"
1757    ) {
1758        effective.insert(lname, value.clone());
1759    }
1760}

1762fn split_css_comma_groups(value: &CssValue) -> Vec<Vec<CssValue>> {
1763    match value {
1764        CssValue::List(items) => {
1765            let mut groups: Vec<Vec<CssValue>> = vec![Vec::new()];
1766            for item in items {
1767                if matches!(item, CssValue::Keyword(k) if k == ",") {
1768                    groups.push(Vec::new());
1769                } else {
1770                    groups.last_mut().unwrap().push(item.clone());
1771                }
1772            }
1773            if groups.is_empty() {
1774                vec![vec![]]
1775            } else {
1776                groups
1777            }
1778        }
1779        other => vec![vec![other.clone()]],
1780    }
1781}
1782
1783fn flatten_css_value_items(items: &[CssValue]) -> Vec<CssValue> {
1784    let mut out = Vec::new();
1785    for item in items {
1786        match item {
1787            CssValue::List(inner) => out.extend(inner.clone()),
1788            _ => out.push(item.clone()),
1789        }
1790    }
1791    out
1792}
1793
1794fn join_css_comma_groups(groups: Vec<CssValue>) -> CssValue {
1795    if groups.len() <= 1 {
1796        return groups.into_iter().next().unwrap_or(CssValue::Keyword("none".into()));
1797    }
1798    let mut out: Vec<CssValue> = Vec::new();
1799    for (i, g) in groups.into_iter().enumerate() {
1800        if i > 0 {
1801            out.push(CssValue::Keyword(",".into()));
1802        }
1803        out.push(g);
1804    }
1805    CssValue::List(out)
1806}

1834/// Classification of the nearest layout-participating sibling of a slot.
1835#[derive(PartialEq)]
1836enum Neighbour {
1837    /// No participating sibling (container edge) or only `display:none` boxes.
1838    None,
1839    /// A block-level element box.
1840    Block,
1841    /// A `<br>` line break.
1842    LineBreak,
1843    /// Any other inline content (text, inline element, replaced, …).
1844    Inline,
1845}
1846
1847/// Inspects the nearest layout-participating sibling of `slot_index` in the
1848/// given `step` direction (`-1` = previous, `+1` = next).
1849///
1850/// Collapsible-whitespace-only text siblings are always skipped. `<br>`
1851/// siblings are skipped when `skip_line_break` is set, so the caller can tell
1852/// "adjacent to a `<br>`" apart from "the nearest block beyond a `<br>`".
1853fn neighbour(
1854    slots: &[ChildSlot],
1855    element_results: &[Option<(LayoutChild, InfoNode)>],
1856    slot_index: usize,
1857    step: isize,
1858    skip_line_break: bool,
1859) -> Neighbour {
1860    let mut idx = slot_index;
1861
1862    loop {
1863        let Some(next) = (if step < 0 {
1864            idx.checked_sub(1)
1865        } else {
1866            idx.checked_add(1)
1867        }) else {
1868            return Neighbour::None;
1869        };
1870        if next >= slots.len() {
1871            return Neighbour::None;
1872        }
1873        idx = next;
1874
1875        match &slots[idx] {
1876            ChildSlot::Element(element_index) => {
1877                let Some((child, _)) = element_results[*element_index].as_ref() else {
1878                    continue;
1879                };
1880
1881                // display:none elements do not participate in layout.
1882                if matches!(
1883                    child,
1884                    LayoutChild::Node(node)
1885                        if node.style.display == Display::None
1886                ) {
1887                    continue;
1888                }
1889
1890                return if matches!(
1891                    child,
1892                    LayoutChild::Node(node)
1893                        if node.style.display.outer() == Some(OuterDisplay::Block)
1894                ) {
1895                    Neighbour::Block
1896                } else {
1897                    Neighbour::Inline
1898                };
1899            }
1900            ChildSlot::Inline(_, info) => {
1901                if matches!(info.kind, NodeKind::LineBreak) {
1902                    if !skip_line_break {
1903                        return Neighbour::LineBreak;
1904                    }
1905                } else if !is_collapsible_whitespace_text(info) {
1906                    return Neighbour::Inline;
1907                }
1908            }
1909        }
1910    }
1911}

1913/// Per-child keep decision for whitespace-only text nodes.
1914///
1915/// A whitespace-only text node is dropped when it is adjacent on *either*
1916/// side to a block-level sibling or to a `<br>`; otherwise it would become a
1917/// stray inline box that creates a spurious line box in block containers.
1918fn compute_whitespace_keep(
1919    slots: &[ChildSlot],
1920    element_results: &[Option<(LayoutChild, InfoNode)>],
1921) -> Vec<bool> {
1922    (0..slots.len())
1923        .map(|i| match &slots[i] {
1924            ChildSlot::Inline(_, info) => {
1925                if !is_collapsible_whitespace_text(info) {
1926                    return true;
1927                }
1928                // Drop if either side is a block-level box or a `<br>` (a line
1929                // break starts a new line, so adjacent whitespace is spurious).
1930                let side_drops = |step: isize| {
1931                    neighbour(slots, element_results, i, step, true) == Neighbour::Block
1932                        || neighbour(slots, element_results, i, step, false) == Neighbour::LineBreak
1933                };
1934                !(side_drops(-1) || side_drops(1))
1935            }
1936            ChildSlot::Element(_) => true,
1937        })
1938        .collect()
1939}

1998/// Normalizes segment breaks and collapsible whitespace according to the
1999/// element's `white-space`.
2000///
2001/// Returns [`Cow::Borrowed`] when `text` is already in normalized form, which
2002/// is the common case for the many short text nodes whose content has no
2003/// segment breaks, repeated spaces or leading/trailing whitespace. Only the
2004/// text that actually needs rewriting is copied.
2005pub fn normalize_whitespace(text: &str, white_space: WhiteSpace) -> Cow<'_, str> {
2006    // `pre`-like modes only rewrite segment breaks (`\r\n`, `\r`, `\x0c` are
2007    // all replaced by `\n`); every other rune is preserved verbatim.
2008    let is_pre_like = matches!(
2009        white_space,
2010        WhiteSpace::Pre | WhiteSpace::PreWrap | WhiteSpace::BreakSpaces
2011    );
2012    let has_segment_break = text.as_bytes().iter().any(|&b| b == b'\r' || b == b'\x0c');
2013
2014    if is_pre_like {
2015        if !has_segment_break {
2016            return Cow::Borrowed(text);
2017        }
2018        // Only segment breaks change; avoid the throwaway second `replace`.
2019        let mut result = String::with_capacity(text.len());
2020        let mut chars = text.chars().peekable();
2021        while let Some(c) = chars.next() {
2022            match c {
2023                '\r' => {
2024                    if chars.peek() == Some(&'\n') {
2025                        chars.next();
2026                    }
2027                    result.push('\n');
2028                }
2029                '\x0c' => result.push('\n'),
2030                _ => result.push(c),
2031            }
2032        }
2033        return Cow::Owned(result);
2034    }
2035
2036    // `Normal`, `Nowrap` and `PreLine` all collapse runs of CSS whitespace:
2037    // every whitespace rune is mapped to a single `' '`. The text therefore
2038    // only needs rewriting when a segment break is present, when a whitespace
2039    // rune other than `' '` appears (tab, newline, …), or when two whitespace
2040    // runes are adjacent (the second would be dropped).
2041    let needs_rewrite = has_segment_break || {
2042        let mut prev_was_space = false;
2043        let mut rewrite = false;
2044        for c in text.chars() {
2045            let is_space = is_css_whitespace(c);
2046            if is_space && (prev_was_space || c != ' ') {
2047                rewrite = true;
2048                break;
2049            }
2050            prev_was_space = is_space;
2051        }
2052        rewrite
2053    };
2054
2055    if !needs_rewrite {
2056        return Cow::Borrowed(text);
2057    }
2058
2059    let text = text.replace("\r\n", "\n");
2060    let text = text.replace(['\r', '\x0c'], "\n");
2061
2062    let mut result = String::new();
2063    let mut prev_was_space = false;
2064
2065    for c in text.chars() {
2066        match white_space {
2067            WhiteSpace::Normal | WhiteSpace::Nowrap => {
2068                if is_css_whitespace(c) {
2069                    if !prev_was_space {
2070                        result.push(' ');
2071                    }
2072                    prev_was_space = true;
2073                } else {
2074                    result.push(c);
2075                    prev_was_space = false;
2076                }
2077            }
2078
2079            WhiteSpace::Pre | WhiteSpace::PreWrap | WhiteSpace::BreakSpaces => {
2080                result.push(c);
2081                prev_was_space = false;
2082            }
2083
2084            WhiteSpace::PreLine => {
2085                if c == '\n' {
2086                    result.push('\n');
2087                    prev_was_space = false;
2088                } else if is_css_whitespace(c) {
2089                    if !prev_was_space {
2090                        result.push(' ');
2091                    }
2092                    prev_was_space = true;
2093                } else {
2094                    result.push(c);
2095                    prev_was_space = false;
2096                }
2097            }
2098        }
2099    }
2100
2101    if white_space == WhiteSpace::PreLine {
2102        while result.ends_with('\n') {
2103            result.pop();
2104        }
2105    }
2106
2107    Cow::Owned(result)
2108}

2337/// How an element's `color-scheme` property constrains its used color scheme.
2338enum ColorSchemePref {
2339    /// `normal` or unset: fall back to the inherited (or system) scheme.
2340    Normal,
2341    Light,
2342    Dark,
2343    /// Both `light` and `dark` listed: follow the system preference.
2344    Both,
2345}
2346
2347/// Parses the winning `color-scheme` declaration into a preference.
2348fn color_scheme_pref(value: Option<&CssValue>) -> ColorSchemePref {
2349    let Some(value) = value else {
2350        return ColorSchemePref::Normal;
2351    };
2352    let mut light = false;
2353    let mut dark = false;
2354    let mut has = false;
2355    let mut push = |keyword: &str| {
2356        has = true;
2357        match keyword {
2358            "light" => light = true,
2359            "dark" => dark = true,
2360            // `only`, `normal`, unknown keywords are ignored here.
2361            _ => {}
2362        }
2363    };
2364    match value {
2365        CssValue::Keyword(k) => push(k),
2366        CssValue::List(items) => {
2367            for item in items {
2368                if let CssValue::Keyword(k) = item {
2369                    push(k);
2370                }
2371            }
2372        }
2373        _ => {}
2374    }
2375    if !has {
2376        ColorSchemePref::Normal
2377    } else if light && dark {
2378        ColorSchemePref::Both
2379    } else if light {
2380        ColorSchemePref::Light
2381    } else if dark {
2382        ColorSchemePref::Dark
2383    } else {
2384        ColorSchemePref::Normal
2385    }
2386}

2388/// Computes the used color scheme of an element from its `color-scheme`

2391    declaration: Option<&CssValue>,
2392    inherited: ColorScheme,
2393    system: ColorScheme,
2394) -> ColorScheme {
2395    match color_scheme_pref(declaration) {
2396        ColorSchemePref::Normal => inherited,
2397        ColorSchemePref::Light => ColorScheme::Light,
2398        ColorSchemePref::Dark => ColorScheme::Dark,
2399        ColorSchemePref::Both => system,
2400    }

2419    chain: &ElementChain,
2420    node_id: NodeId,
2421    mut match_cache: Option<&mut SelectorMatchCache>,
2422    #[cfg(any(feature = "profile", debug_assertions))] stats: &mut CandidateMetrics,
2423) -> (Properties, Properties, HashMap<PseudoElement, Properties>) {

2425    let mut custom_properties = HashMap::new();

2429        Some(el) => el,

2431    };
2432

2434    {
2435        stats.elements_checked += 1;

2439    // query time is measured separately from selector matching.

2442    let candidates_iter: Vec<_> = rule_set.query_candidates(element).collect();

2457        // (comparatively expensive) selector walk happens once per group.
2458        perf_scope!(sel_match);
2459        let matches_sel = match match_cache.as_deref_mut() {
2460            Some(cache) => {
2461                cache.get_or_insert(&group.selector, node_id, || group.selector.matches(chain))

2484                &mut properties
2485            };

2488            let should_replace = match target.get(&decl.name) {
2489                Some(current) => decl.outranks(current),

2491            };
2492

2494                target.insert(decl.name.clone(), decl.clone());
2495            }
2496
2497            #[cfg(any(feature = "profile", debug_assertions))]
2498            {
2499                stats.cascade_insert_time += cascade.elapsed();
2500            }
2501        }
2502    }
2503

2505}
