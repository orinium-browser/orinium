4fn mix_colors(colors: &[Color], weights: &[f32], space: &str) -> Color {
5    let n = colors.len().min(weights.len());
6    let mut acc = colors[0];
7    let mut acc_w = weights[0];
8    for i in 1..n {
9        if weights[i] <= 0.0 {
10            continue;
11        }
12        acc = match space {
13            "lch" => mix_two_lch(acc, colors[i], acc_w, weights[i]),
14            _ => mix_two_srgb(acc, colors[i], acc_w, weights[i]),
15        };
16        acc_w += weights[i];
17    }
18    acc
19}

21/// Interpolate between two colors in sRGB with premultiplied alpha.
22fn mix_two_srgb(a: Color, b: Color, wa: f32, wb: f32) -> Color {
23    let total = wa + wb;
24    if total <= 0.0 {
25        return Color(0, 0, 0, 0);
26    }
27    let f = wb / total;
28    let al = a.to_linear_f32_array();
29    let bl = b.to_linear_f32_array();
30    let mut c = [0.0f32; 4];
31    for i in 0..4 {
32        c[i] = al[i] * al[3] * (1.0 - f) + bl[i] * bl[3] * f;
33    }
34    let alpha = al[3] * (1.0 - f) + bl[3] * f;
35    if alpha > 0.0 {
36        for c_i in &mut c[..3] {
37            *c_i /= alpha;
38        }
39    }
40    c[3] = alpha;
41    Color::from_linear_f32_array(c)
42}

44/// Interpolate between two colors in LCH with premultiplied alpha.
45fn mix_two_lch(a: Color, b: Color, wa: f32, wb: f32) -> Color {
46    let total = wa + wb;
47    if total <= 0.0 {
48        return Color(0, 0, 0, 0);
49    }
50    let f = wb / total;
51    let a_alpha = a.3 as f32 / 255.0;
52    let b_alpha = b.3 as f32 / 255.0;
53    let (al, ac, ah) = rgb_to_lch(a);
54    let (bl, bc, bh) = rgb_to_lch(b);
55    let lm = al * a_alpha * (1.0 - f) + bl * b_alpha * f;
56    let cm = ac * a_alpha * (1.0 - f) + bc * b_alpha * f;
57    let hm = lerp_hue(ah, bh, f);
58    let alpha = a_alpha * (1.0 - f) + b_alpha * f;
59    let (l, c) = if alpha > 0.0 {
60        (lm / alpha, cm / alpha)
61    } else {
62        (lm, cm)
63    };
64    lch_to_color(l, c, hm, alpha)
65}

67/// Interpolate a hue angle along the shortest arc.
68fn lerp_hue(a: f32, b: f32, t: f32) -> f32 {
69    let mut d = b - a;
70    if d > 180.0 {
71        d -= 360.0;
72    } else if d < -180.0 {
73        d += 360.0;
74    }
75    (a + d * t).rem_euclid(360.0)
76}

78const D65_WHITE: (f32, f32, f32) = (0.95047, 1.0, 1.08883);

80fn lab_f(t: f32) -> f32 {
81    const EPS: f32 = 6.0 / 29.0;
82    if t > EPS * EPS * EPS {
83        t.cbrt()
84    } else {
85        t / (3.0 * EPS * EPS) + 4.0 / 29.0
86    }
87}
88
89fn lab_f_inv(t: f32) -> f32 {
90    const EPS: f32 = 6.0 / 29.0;
91    if t > EPS {
92        t * t * t
93    } else {
94        3.0 * EPS * EPS * (t - 4.0 / 29.0)
95    }
96}

98/// Convert sRGB (0..1 channels) to CIE XYZ (D65).
99fn srgb_to_xyz(r: f32, g: f32, b: f32) -> (f32, f32, f32) {
100    let r = Color::srgb_to_linear(r);
101    let g = Color::srgb_to_linear(g);
102    let b = Color::srgb_to_linear(b);
103    let x = 0.4124564 * r + 0.3575761 * g + 0.1804375 * b;
104    let y = 0.2126729 * r + 0.7151522 * g + 0.0721750 * b;
105    let z = 0.0193339 * r + 0.119_192 * g + 0.9503041 * b;
106    (x, y, z)
107}

109/// Convert CIE XYZ (D65) to sRGB (0..1 channels).
110fn xyz_to_srgb(x: f32, y: f32, z: f32) -> (f32, f32, f32) {
111    let r = 3.2404542 * x - 1.5371385 * y - 0.4985314 * z;
112    let g = -0.969_266 * x + 1.8760108 * y + 0.0415560 * z;
113    let b = 0.0556434 * x - 0.2040259 * y + 1.0572252 * z;
114    (
115        Color::linear_to_srgb(r),
116        Color::linear_to_srgb(g),
117        Color::linear_to_srgb(b),
118    )
119}

121fn xyz_to_lab(x: f32, y: f32, z: f32) -> (f32, f32, f32) {
122    let (wx, wy, wz) = D65_WHITE;
123    let fx = lab_f(x / wx);
124    let fy = lab_f(y / wy);
125    let fz = lab_f(z / wz);
126    let l = 116.0 * fy - 16.0;
127    let a = 500.0 * (fx - fy);
128    let b = 200.0 * (fy - fz);
129    (l, a, b)
130}

132fn lab_to_xyz(l: f32, a: f32, b: f32) -> (f32, f32, f32) {
133    let (wx, wy, wz) = D65_WHITE;
134    let fy = (l + 16.0) / 116.0;
135    let fx = fy + a / 500.0;
136    let fz = fy - b / 200.0;
137    (wx * lab_f_inv(fx), wy * lab_f_inv(fy), wz * lab_f_inv(fz))
138}

140/// Convert an sRGB color to LCH (L 0..100, C >= 0, H 0..360).
141fn rgb_to_lch(c: Color) -> (f32, f32, f32) {
142    let (x, y, z) = srgb_to_xyz(c.0 as f32 / 255.0, c.1 as f32 / 255.0, c.2 as f32 / 255.0);
143    let (l, a, b) = xyz_to_lab(x, y, z);
144    let chroma = (a * a + b * b).sqrt();
145    let hue = b.atan2(a).to_degrees().rem_euclid(360.0);
146    (l, chroma, hue)
147}

149/// Convert LCH (L 0..100, C >= 0, H 0..360) back to an sRGB color.
150fn lch_to_color(l: f32, chroma: f32, hue: f32, alpha: f32) -> Color {
151    let hr = hue.to_radians();
152    let a = chroma * hr.cos();
153    let b = chroma * hr.sin();
154    let (x, y, z) = lab_to_xyz(l, a, b);
155    let (r, g, b) = xyz_to_srgb(x, y, z);
156    Color(
157        (r.clamp(0.0, 1.0) * 255.0).round() as u8,
158        (g.clamp(0.0, 1.0) * 255.0).round() as u8,
159        (b.clamp(0.0, 1.0) * 255.0).round() as u8,
160        (alpha.clamp(0.0, 1.0) * 255.0).round() as u8,
161    )
162}

164/// Parse `color-mix(in <space>, <color> [<percentage>], ...)`.
165fn parse_color_mix(args: &[CssValue], name: &str, color_scheme: ColorScheme) -> Option<Color> {
166    // args: [Keyword("in"), Keyword("<space>"), <color>..., ...]
167    if !matches!(args.first(), Some(CssValue::Keyword(k)) if k.eq_ignore_ascii_case("in")) {
168        return None;
169    }
170    let space = match args.get(1) {
171        Some(CssValue::Keyword(k)) => k.to_ascii_lowercase(),
172        _ => return None,
173    };
174
175    let mut colors: Vec<Color> = Vec::new();
176    let mut weights: Vec<Option<f32>> = Vec::new();
177    let mut i = 2;
178    while i < args.len() {
179        let color = resolve_css_color(name, &args[i], color_scheme)?;
180        i += 1;
181        let mut weight = None;
182        if i < args.len() {
183            match &args[i] {
184                CssValue::Number(n) => {
185                    weight = Some(*n);
186                    i += 1;
187                }
188                CssValue::Length(p, Unit::Percent) => {
189                    weight = Some(*p);
190                    i += 1;
191                }
192                _ => {}
193            }
194        }
195        colors.push(color);
196        weights.push(weight);
197    }
198
199    if colors.len() < 2 {
200        return None;
201    }
202
203    // Resolve missing weights to the remaining percentage.
204    let specified_sum: f32 = weights.iter().flatten().sum();
205    let missing = weights.iter().filter(|w| w.is_none()).count();
206    let remainder = (100.0 - specified_sum).max(0.0);
207    let mut resolved: Vec<f32> = Vec::with_capacity(colors.len());
208    for weight in &weights {
209        match weight {
210            Some(v) => resolved.push(*v),
211            None if missing > 0 => resolved.push(remainder / missing as f32),
212            None => resolved.push(0.0),
213        }
214    }
215    let total: f32 = resolved.iter().sum();
216    let normalized: Vec<f32> = if total > 0.0 {
217        resolved.iter().map(|v| v / total).collect()
218    } else {
219        vec![1.0 / colors.len() as f32; colors.len()]
220    };
221
222    Some(mix_colors(&colors, &normalized, &space))
223}

225/// Resolve a single CSS component value to a plain `f32`.
226///
227/// Handles `calc()`, `min()`, `max()`, `clamp()`, plain numbers,
228/// and percentage lengths. Returns `None` for values that cannot
229/// be reduced to a number (e.g. keyword colors, px lengths).
230fn resolve_channel(value: &CssValue) -> Option<f32> {
231    match value {
232        CssValue::Number(n) => Some(*n),
233        CssValue::Length(p, Unit::Percent) => Some(*p),
234        CssValue::Function(fn_name, args) => resolve_channel_function(fn_name, args),
235        CssValue::List(items) => {
236            // A list acts like an inline calc expression: [a, +, b]
237            // or a calc result with a unit suffix: [calc(100 / 2), "%"]
238            if items.len() == 1 {
239                return resolve_channel(&items[0]);
240            }
241            // If the last element is a bare "%" keyword, the preceding
242            // elements form the numeric expression (the result is a percent).
243            if let Some(CssValue::Keyword(pct)) = items.last()
244                && pct == "%"
245            {
246                let expr: Vec<&CssValue> = items[..items.len() - 1].iter().collect();
247                return if expr.len() == 1 {
248                    resolve_channel(expr[0])
249                } else {
250                    resolve_channel_expr(&expr)
251                };
252            }
253            resolve_channel_slice(items)
254        }
255        _ => None,
256    }
257}

259/// Resolve the arguments of a CSS math function to a plain `f32`.
260fn resolve_channel_function(fn_name: &str, args: &[Vec<CssValue>]) -> Option<f32> {
261    match fn_name {
262        "calc" => {
263            // calc() has a single argument list: calc(a + b)
264            let flat: Vec<&CssValue> = args.iter().flatten().collect();
265            resolve_channel_expr(&flat)
266        }
267        "min" | "max" => {
268            // Each comma-separated argument is an independent expression.
269            let mut values = Vec::with_capacity(args.len());
270            for arg_group in args {
271                let flat: Vec<&CssValue> = arg_group.iter().collect();
272                values.push(resolve_channel_expr(&flat)?);
273            }
274            match fn_name {
275                "min" => values.into_iter().reduce(f32::min),
276                _ => values.into_iter().reduce(f32::max),
277            }
278        }
279        "clamp" => {
280            // clamp(min, val, max) — each argument may be a calc expression.
281            let refs: Vec<Vec<&CssValue>> = args.iter().map(|a| a.iter().collect()).collect();
282            let min = resolve_channel_expr(&refs[0])?;
283            let val = resolve_channel_expr(&refs[1])?;
284            let max = resolve_channel_expr(&refs[2])?;
285            Some(val.clamp(min, max))
286        }
287        _ => None,
288    }
289}

291/// Evaluate a flat expression slice (e.g. `[a, *, b, +, c]`) to a single `f32`.
292///
293/// Applies standard operator precedence: `*` and `/` bind tighter than `+` and `-`.
294fn resolve_channel_expr(components: &[&CssValue]) -> Option<f32> {
295    // Parse multiplication / division first (higher precedence).
296    fn parse_product<'a>(
297        iter: &mut std::iter::Peekable<impl Iterator<Item = &'a CssValue>>,
298    ) -> Option<f32> {
299        let mut result = resolve_channel(iter.next()?)?;
300        loop {
301            let op = match iter.peek() {
302                Some(CssValue::Keyword(k)) if k == "*" || k == "/" => iter.next()?,
303                _ => break,
304            };
305            let rhs = resolve_channel(iter.next()?)?;
306            result = match op {
307                CssValue::Keyword(o) if o == "*" => result * rhs,
308                CssValue::Keyword(o) if o == "/" => {
309                    if rhs == 0.0 {
310                        return None;
311                    }
312                    result / rhs
313                }
314                _ => return None,
315            };
316        }
317        Some(result)
318    }
319
320    let mut iter = components.iter().copied().peekable();
321    let mut result = parse_product(&mut iter)?;
322    loop {
323        let op = match iter.peek() {
324            Some(CssValue::Keyword(k)) if k == "+" || k == "-" => iter.next()?,
325            _ => break,
326        };
327        let rhs = parse_product(&mut iter)?;
328        result = match op {
329            CssValue::Keyword(o) if o == "+" => result + rhs,
330            CssValue::Keyword(o) if o == "-" => result - rhs,
331            _ => return None,
332        };
333    }
334    Some(result)
335}

337/// Resolve a slice of `&CssValue` to a plain `f32`.
338/// Used for flat inline expressions like `[Number(10), Keyword("+"), Number(3)]`.
339fn resolve_channel_slice(items: &[CssValue]) -> Option<f32> {
340    let refs: Vec<&CssValue> = items.iter().collect();
341    resolve_channel_expr(&refs)
342}

344/// Normalize a percentage-range value from `resolve_channel` to 0.0–1.0.
345///
346/// CSS spec: plain numbers > 1.0 in hsl/hwb are treated as percentages
347/// (e.g. `50` = 50% = 0.5). Values ≤ 1.0 are already in the 0.0–1.0 range.
348fn normalize_hsl_percent(v: f32) -> f32 {
349    if v > 1.0 {
350        (v / 100.0).clamp(0.0, 1.0)
351    } else {
352        v.clamp(0.0, 1.0)
353    }
354}

523    /// Convert HSL to RGB (0..255)
524    /// Convert hue degrees (0..360) to fully-saturated RGB (each 0.0..1.0).
525    fn hue_to_rgb(h: f32) -> (f32, f32, f32) {
526        let h = h.rem_euclid(360.0);
527        let sector = h / 60.0;
528        let x = 1.0 - (sector % 2.0 - 1.0).abs();
529        match sector as u32 {
530            0 => (1.0, x, 0.0),
531            1 => (x, 1.0, 0.0),
532            2 => (0.0, 1.0, x),
533            3 => (0.0, x, 1.0),
534            4 => (x, 0.0, 1.0),
535            _ => (1.0, 0.0, x),
536        }
537    }

539    /// HWB to RGBA conversion per CSS Color Level 4.
540    fn hwb_to_rgba(h: f32, w: f32, b: f32, a: f32) -> (u8, u8, u8, u8) {
541        let w = w.clamp(0.0, 1.0);
542        let b = b.clamp(0.0, 1.0);
543        let a = a.clamp(0.0, 1.0);
544
545        let (r, g, bl) = if w + b >= 1.0 {
546            let gray = w / (w + b);
547            (gray, gray, gray)
548        } else {
549            let (pr, pg, pb) = hue_to_rgb(h);
550            let factor = 1.0 - w - b;
551            (pr * factor + w, pg * factor + w, pb * factor + w)
552        };
553
554        let to_u8 = |v: f32| (v.clamp(0.0, 1.0) * 255.0).round() as u8;
555        (to_u8(r), to_u8(g), to_u8(bl), (a * 255.0).round() as u8)
556    }
557
558    fn hsla_to_rgba(h: f32, s: f32, l: f32, a: f32) -> (u8, u8, u8, u8) {
559        // 1. Compute Chroma
560        let c = (1.0 - (2.0 * l - 1.0).abs()) * s;
561        let h_prime = h / 60.0;
562        let x = c * (1.0 - ((h_prime % 2.0) - 1.0).abs());
563
564        // 2. Determine preliminary RGB values based on hue sector
565        let (r1, g1, b1) = match h_prime as u32 {
566            0 => (c, x, 0.0),
567            1 => (x, c, 0.0),
568            2 => (0.0, c, x),
569            3 => (0.0, x, c),
570            4 => (x, 0.0, c),
571            5 | 6 => (c, 0.0, x),
572            _ => (0.0, 0.0, 0.0),
573        };
574
575        // 3. Add m to match the lightness
576        let m = l - c / 2.0;
577        let r = ((r1 + m) * 255.0).round().clamp(0.0, 255.0) as u8;
578        let g = ((g1 + m) * 255.0).round().clamp(0.0, 255.0) as u8;
579        let b = ((b1 + m) * 255.0).round().clamp(0.0, 255.0) as u8;
580        let a = (a * 255.0).round().clamp(0.0, 255.0) as u8;
581
582        (r, g, b, a)
583    }

585    match css_color {
586        // Already parsed as an absolute color (rgb/rgba/hex, etc.)
587        CssValue::Color(_) => {
588            let (r, g, b, a) = css_color.to_rgba_tuple()?;
589            Some(Color(r, g, b, a))
590        }
591
592        // Named color keyword
593        CssValue::Keyword(value) => keyword_color_to_color(name, value),
594
595        // rgb() / rgba() unified
596        CssValue::Function(func, args) if func == "rgb" || func == "rgba" => {
597            let mut values = Vec::new();
598            let mut has_pct = false;
599            let mut alpha: Option<f32> = None;
600            let mut after_slash = false;
601
602            for arg in args.iter().flatten() {
603                match arg {
604                    CssValue::Keyword(k) if k == "/" => {
605                        after_slash = true;
606                    }
607                    CssValue::Keyword(k) if k == "%" => {
608                        // A bare % after a resolved value means that value is a
609                        // percentage. This handles `calc(100/2)%` from the parser.
610                        has_pct = true;
611                        if after_slash && !values.is_empty() {
612                            // Convert the last pushed number to a percentage alpha.
613                            let v = values.pop().unwrap();
614                            alpha = Some(v / 100.0);
615                        }
616                    }
617                    CssValue::Length(_p, Unit::Percent) => {
618                        has_pct = true;
619                        let v = resolve_channel(arg)?;
620                        if after_slash {
621                            alpha = Some(v / 100.0);
622                        } else {
623                            values.push(v);
624                        }
625                    }
626                    _ => {
627                        // Number, calc(), min(), max(), clamp(), var(), etc.
628                        let v = resolve_channel(arg)?;
629                        if after_slash {
630                            alpha = Some(v);
631                        } else {
632                            values.push(v);
633                        }
634                    }
635                }
636            }
637
638            // rgb(r, g, b) -> 3 values
639            // rgba(r, g, b, a) -> 4 values
640            // rgb(r g b / a) -> 3 values + after_slash
641            let (a, values) = if values.len() == 4 && alpha.is_none() {
642                (values[3], vec![values[0], values[1], values[2]])
643            } else if values.len() == 3 {
644                (alpha.unwrap_or(1.0), values)
645            } else {
646                return None;
647            };
648
649            // CSS stores rgb values as 0-255 integers or 0.0-1.0 floats
650            // or 0%-100% (already handled above).
651            let map_channel = |v: f32| -> f32 {
652                if has_pct {
653                    v / 100.0 * 255.0
654                } else if v > 1.0 {
655                    v.clamp(0.0, 255.0)
656                } else {
657                    v * 255.0
658                }
659            };
660
661            Some(Color(
662                map_channel(values[0]).round() as u8,
663                map_channel(values[1]).round() as u8,
664                map_channel(values[2]).round() as u8,
665                (a * 255.0).round() as u8,
666            ))
667        }
668
669        // hsl() / hsla() unified
670        CssValue::Function(func, args) if func == "hsl" || func == "hsla" => {
671            let mut hue_val: Option<f32> = None;
672            let mut sat_val: Option<f32> = None;
673            let mut light_val: Option<f32> = None;
674            let mut alpha: Option<f32> = None;
675            let mut after_slash = false;
676            let mut channel_index = 0u8;
677
678            for arg in args.iter().flatten() {
679                match arg {
680                    CssValue::Keyword(k) if k == "/" => {
681                        after_slash = true;
682                    }
683                    CssValue::Keyword(k) if k == "%" => {
684                        // Bare % means the *previous* channel is a percentage.
685                        // channel_index was already incremented past it.
686                        if after_slash {
687                            if let Some(ref mut a) = alpha {
688                                *a /= 100.0;
689                            }
690                        } else if channel_index > 0 {
691                            match channel_index - 1 {
692                                1 => {
693                                    if let Some(ref mut s) = sat_val {
694                                        *s /= 100.0;
695                                    }
696                                }
697                                2 => {
698                                    if let Some(ref mut l) = light_val {
699                                        *l /= 100.0;
700                                    }
701                                }
702                                _ => {}
703                            }
704                        }
705                    }
706                    CssValue::Length(percent, Unit::Percent) if after_slash => {
707                        alpha = Some(percent / 100.0);
708                    }
709                    CssValue::Length(value, Unit::Deg) if !after_slash => {
710                        hue_val = Some(value.rem_euclid(360.0));
711                        channel_index += 1;
712                    }
713                    CssValue::Length(value, Unit::Percent) if !after_slash => {
714                        let v = value / 100.0;
715                        match channel_index {
716                            0 => {
717                                hue_val = Some(v.rem_euclid(360.0));
718                            }
719                            1 => {
720                                sat_val = Some(v);
721                            }
722                            2 => {
723                                light_val = Some(v);
724                            }
725                            _ => {
726                                alpha = Some(v);
727                            }
728                        }
729                        channel_index += 1;
730                    }
731                    _ => {
732                        // Number, calc(), min(), max(), clamp(), etc.
733                        let v = resolve_channel(arg)?;
734                        if after_slash {
735                            alpha = Some(v);
736                        } else {
737                            match channel_index {
738                                0 => {
739                                    hue_val = Some(v.rem_euclid(360.0));
740                                }
741                                1 => {
742                                    sat_val = Some(normalize_hsl_percent(v));
743                                }
744                                2 => {
745                                    light_val = Some(normalize_hsl_percent(v));
746                                }
747                                3 => {
748                                    alpha = Some(v);
749                                }
750                                _ => {}
751                            }
752                            channel_index += 1;
753                        }
754                    }
755                }
756            }
757
758            let hue = hue_val?;
759            let saturation = sat_val?.clamp(0.0, 1.0);
760            let lightness = light_val?.clamp(0.0, 1.0);
761            let alpha = alpha.unwrap_or(1.0).clamp(0.0, 1.0);
762            let (r, g, b, a) = hsla_to_rgba(hue, saturation, lightness, alpha);
763
764            Some(Color(r, g, b, a))
765        }
766
767        // hwb() — hue, whiteness, blackness
768        CssValue::Function(func, args) if func == "hwb" => {
769            let mut hue_val: Option<f32> = None;
770            let mut white_val: Option<f32> = None;
771            let mut black_val: Option<f32> = None;
772            let mut alpha: Option<f32> = None;
773            let mut after_slash = false;
774            let mut channel_index = 0u8;
775
776            for arg in args.iter().flatten() {
777                match arg {
778                    CssValue::Keyword(k) if k == "/" => {
779                        after_slash = true;
780                    }
781                    CssValue::Keyword(k) if k == "%" => {
782                        if after_slash {
783                            if let Some(ref mut a) = alpha {
784                                *a /= 100.0;
785                            }
786                        } else if channel_index > 0 {
787                            match channel_index - 1 {
788                                1 => {
789                                    if let Some(ref mut w) = white_val {
790                                        *w /= 100.0;
791                                    }
792                                }
793                                2 => {
794                                    if let Some(ref mut b) = black_val {
795                                        *b /= 100.0;
796                                    }
797                                }
798                                _ => {}
799                            }
800                        }
801                    }
802                    CssValue::Length(percent, Unit::Percent) if after_slash => {
803                        alpha = Some(percent / 100.0);
804                    }
805                    CssValue::Length(value, Unit::Deg) if !after_slash => {
806                        hue_val = Some(value.rem_euclid(360.0));
807                        channel_index += 1;
808                    }
809                    CssValue::Length(value, Unit::Percent) if !after_slash => {
810                        let v = value / 100.0;
811                        match channel_index {
812                            0 => {
813                                hue_val = Some(v.rem_euclid(360.0));
814                            }
815                            1 => {
816                                white_val = Some(v);
817                            }
818                            2 => {
819                                black_val = Some(v);
820                            }
821                            3 => {
822                                alpha = Some(v);
823                            }
824                            _ => {}
825                        }
826                        channel_index += 1;
827                    }
828                    _ => {
829                        let v = resolve_channel(arg)?;
830                        if after_slash {
831                            alpha = Some(v);
832                        } else {
833                            match channel_index {
834                                0 => {
835                                    hue_val = Some(v.rem_euclid(360.0));
836                                }
837                                1 => {
838                                    white_val = Some(normalize_hsl_percent(v));
839                                }
840                                2 => {
841                                    black_val = Some(normalize_hsl_percent(v));
842                                }
843                                3 => {
844                                    alpha = Some(v);
845                                }
846                                _ => {}
847                            }
848                            channel_index += 1;
849                        }
850                    }
851                }
852            }
853
854            let hue = hue_val?;
855            let w = white_val?.clamp(0.0, 1.0);
856            let b = black_val?.clamp(0.0, 1.0);
857            let alpha = alpha.unwrap_or(1.0).clamp(0.0, 1.0);
858
859            let (r, g, bl, a) = hwb_to_rgba(hue, w, b, alpha);
860            Some(Color(r, g, bl, a))
861        }
862
863        // light-dark(<light-color>, <dark-color>)
864        CssValue::Function(func, args) if func == "light-dark" && args.len() == 2 => {
865            let chosen = match color_scheme {
866                ColorScheme::Light => args[0].first()?,
867                ColorScheme::Dark => args[1].first()?,
868            };
869            resolve_css_color(name, chosen, color_scheme)
870        }
871
872        // color-mix(in <space>, <color> [<percentage>], <color> [<percentage>])
873        CssValue::Function(func, args) if func == "color-mix" => {
874            let flat_args: Vec<CssValue> = args.iter().flatten().cloned().collect();
875            parse_color_mix(&flat_args, name, color_scheme)
876        }
877
878        // Any other value reaching here is a pipeline error
879        _ => {
880            log::error!(
881                target: "Layouter",
882                "Unexpected CSS color value for `{}` at layout stage: {:?}",
883                name,
884                css_color
885            );
886            None
887        }
888    }
