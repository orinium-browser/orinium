38/// Convert a resolved `Length` to an absolute pixel value for `LineHeight::Px`.
39fn length_to_px(len: &Length, font_size: f32) -> f32 {
40    match len {
41        Length::Px(v) => *v,
42        Length::Percent(v) => v * font_size / 100.0,
43        Length::Add(a, b) => length_to_px(a, font_size) + length_to_px(b, font_size),
44        Length::Sub(a, b) => length_to_px(a, font_size) - length_to_px(b, font_size),
45        Length::Mul(a, f) => length_to_px(a, font_size) * f,
46        Length::Div(a, f) => length_to_px(a, font_size) / f,
47        _ => font_size * DEFAULT_LINE_FACTOR,
48    }
49}

51/// Builds a layout tree (`LayoutNode`) and a render info tree (`InfoNode`) from the DOM.
52///
53/// # Overview
54/// - Recursively traverses the HTML DOM
55/// - Applies resolved CSS declarations
56/// - Computes layout-related styles
57/// - Collects render-time information (color, font size, text)
58///
59/// # Style resolution order (low → high priority)
60///
61/// 1. **Inherited values from parent**
62///    - `text_style`
63///
64/// 2. **Resolved CSS declarations**
65///    - Overrides inherited values when specified
66///
67/// 3. **HTML defaults / semantics**
68///    - `display` (block, inline, etc.)
69///    - Text measurement for text nodes
70///
71/// # Inherited properties
72///
73/// Only the following properties are inherited explicitly:
74///
75/// - `text_style`
76///
77/// All other style fields are initialized per node and are **not inherited**.
78///
79/// # Parameters
80///
81/// - `parent_text_style`
82///
83/// These values must be passed from the computed result of the parent when
84/// calling this function recursively.
85///
86/// # Returns
87///
88/// A tuple of:
89/// - `LayoutNode`: used by the layout engine
90/// - `InfoNode`: used for rendering (text, color, font size)
91pub fn build_layout_and_info(
92    dom: &Rc<RefCell<TreeNode<HtmlNodeType>>>,
93    resolved_styles: &ResolvedStyles,
94    measurer: &dyn text::TextMeasurer<TextStyle>,
95    parent: InheritedCss,
96    mut chain: ElementChain,
97) -> (LayoutNode, InfoNode) {
98    let html_node = dom.borrow().value.clone();
99
100    let mut text_style = parent.text_style;
101    let mut container_style = ContainerStyle::default();
102    let mut style = Style::default();
103
104    /* -----------------------------
105       Build element chain
106    ----------------------------- */
107    if let HtmlNodeType::Element {
108        tag_name,
109        attributes,
110        ..
111    } = &html_node
112    {
113        let id = attributes
114            .iter()
115            .find(|a| a.name == "id")
116            .map(|a| a.value.clone());
117
118        let class_list: Vec<String> = attributes
119            .iter()
120            .find(|attr| attr.name == "class")
121            .map(|attr| {
122                attr.value
123                    .split_whitespace()
124                    .map(|s| s.to_string())
125                    .collect()
126            })
127            .unwrap_or_default();
128
129        chain.insert(
130            0,
131            ElementInfo {
132                tag_name: tag_name.clone(),
133                id,
134                classes: class_list,
135            },
136        );
137    }
138
139    /* -----------------------------
140       Collect CSS candidates
141    ----------------------------- */
142    let candidates: Option<HashMap<String, (CssValue, (u32, u32, u32), usize)>> =
143        if let HtmlNodeType::Element { .. } = &html_node {
144            Some(collect_candidates(resolved_styles, &chain))
145        } else {
146            None
147        };
148
149    /* -----------------------------
150       Phase 1: Apply element-specific CSS declarations
151    ----------------------------- */
152    if let Some(candidates) = &candidates {
153        for (name, (value, _, _)) in candidates {
154            if name.starts_with("--") {
155                continue;
156            }
157            apply_declaration(
158                name,
159                value,
160                &mut style,
161                &mut container_style,
162                &mut text_style,
163            );
164        }
165    }
166
167    /* -----------------------------
168       Phase 2: Resolve line-height using final font_size.
169       text_style.line_height was either inherited from parent
170       (via parent.text_style) or set by an explicit declaration
171       in Phase 1 (via apply_declaration).
172    ----------------------------- */
173    style.line_height = match text_style.line_height {
174        LineHeight::Number(factor) => Length::Px(text_style.font_size * factor),
175        LineHeight::Normal => Length::Px(text_style.font_size * DEFAULT_LINE_FACTOR),
176        LineHeight::Px(px) => Length::Px(px),
177    };
178
179    let child = InheritedCss { text_style };
180
181    let (mut kind, inline_fragments_opt) = if let HtmlNodeType::Text(t) = &html_node {
182        let t = normalize_whitespace(t);
183
184        let _t = std::time::Instant::now();
185        let measured = measurer
186            .measure(&text::TextMeasureRequest {
187                text: t.clone(),
188                style: text_style,
189            })
190            .expect("text measure failed");
191        let preview = if t.len() > 40 {
192            let cut = t.floor_char_boundary(40);
193            format!("{}...", &t[..cut])
194        } else {
195            t.clone()
196        };
197        log::info!(
198            target: "Layouter",
199            "measure inline: text={:?} len={} took={:?}",
200            preview,
201            t.len(),
202            _t.elapsed(),
203        );
204
205        let kind = NodeKind::Text {
206            texts: measured.iter().map(|f| f.text.clone()).collect(),
207            style: text_style,
208        };
209
210        let inline_fragments: Vec<ItemFragment> = measured
211            .into_iter()
212            .map(|f| {
213                ItemFragment::Fragment(Fragment {
214                    width: f.width,
215                    height: f.height,
216                })
217            })
218            .collect();
219
220        (kind, Some(inline_fragments))
221    } else if let Some(name) = html_node.tag_name()
222        && name == "a"
223        && let Some(href) = html_node.get_attr("href")
224    {
225        (
226            NodeKind::Container {
227                scroll_x: false,
228                scroll_y: false,
229                scroll_offset_x: 0.0,
230                scroll_offset_y: 0.0,
231                style: container_style,
232                role: ContainerRole::Link {
233                    href: href.to_string(),
234                },
235            },
236            None,
237        )
238    } else {
239        (
240            NodeKind::Container {
241                scroll_x: false,
242                scroll_y: false,
243                scroll_offset_x: 0.0,
244                scroll_offset_y: 0.0,
245                style: container_style,
246                role: ContainerRole::Normal,
247            },
248            None,
249        )
250    };
251
252    // Process Children if there are no inline fragments (i.e. text nodes).
253    let (layout, info) = if let Some(inline_fragments) = inline_fragments_opt {
254        /* -----------------------------
255           Text Node with inline fragments
256        ----------------------------- */
257
258        let style = Style {
259            display: Display {
260                outer: OuterDisplay::Inline,
261                inner: InnerDisplay::Flow,
262            },
263            ..style
264        };
265
266        let layout = LayoutNode::with_children(style, inline_fragments);
267
268        let info = InfoNode {
269            kind,
270            children: vec![],
271        };
272
273        (layout, info)
274    } else {
275        /* -----------------------------
276           Children
277        ----------------------------- */
278
279        // NOTE:
280        // Table 要素は未実装。
281        // 暫定的に Flex に置き換える。
282        // TODO: 将来的には TableLayout 実装に置き換える。
283        let mut layout_children: Vec<LayoutChild> = Vec::new();
284        let mut info_children = Vec::new();
285
286        if style.display.outer != OuterDisplay::None {
287            // Table 要素は暫定的に Flex に置き換える。
288            match &html_node {
289                HtmlNodeType::Element { tag_name, .. }
290                    if tag_name == "table"
291                        || tag_name == "tbody"
292                        || tag_name == "thead"
293                        || tag_name == "tfoot" =>
294                {
295                    style.display = Display {
296                        outer: OuterDisplay::Block,
297                        inner: InnerDisplay::Flex,
298                    };
299                    style.flex_direction = FlexDirection::Column;
300                }
301                HtmlNodeType::Element { tag_name, .. } if tag_name == "tr" => {
302                    style.display = Display {
303                        outer: OuterDisplay::Block,
304                        inner: InnerDisplay::Flex,
305                    };
306                    style.flex_direction = FlexDirection::Row;
307                }
308                _ => {}
309            }
310
311            for child_dom in dom.borrow().children() {
312                let child_node = child_dom.borrow().value.clone();
313
314                if let HtmlNodeType::Text(t) = &child_node {
315                    let t = normalize_whitespace(t);
316                    let _t = std::time::Instant::now();
317                    let request = &text::TextMeasureRequest {
318                        text: t.clone(),
319                        style: text_style,
320                    };
321                    let measured = measurer.measure(request).unwrap_or_else(|_|
322                            // FallbackTextMeasurer won't return any errors.
323                            FallbackTextMeasurer.measure(request).unwrap());
324                    let preview = if t.len() > 40 {
325                        let cut = t.floor_char_boundary(40);
326                        format!("{}...", &t[..cut])
327                    } else {
328                        t.clone()
329                    };
330                    log::info!(
331                        target: "Layouter",
332                        "measure child: text={:?} len={} took={:?}",
333                        preview,
334                        t.len(),
335                        _t.elapsed(),
336                    );
337
338                    let text_kind = NodeKind::Text {
339                        texts: measured.iter().map(|f| f.text.clone()).collect(),
340                        style: text_style,
341                    };
342
343                    for fragment in &measured {
344                        layout_children.push(generate_fragment_node(fragment).into());
345                    }
346
347                    info_children.push(InfoNode {
348                        kind: text_kind,
349                        children: vec![],
350                    });
351                } else {
352                    if child_dom.borrow().value.tag_name() == Some("br") {
353                        layout_children.push(ItemFragment::LineBreak.into());
354                        info_children.push(InfoNode {
355                            kind: NodeKind::LineBreak,
356                            children: vec![],
357                        });
358                        continue;
359                    }
360
361                    let (child_layout, child_info) = build_layout_and_info(
362                        child_dom,
363                        resolved_styles,
364                        measurer,
365                        child,
366                        chain.clone(),
367                    );
368
369                    if dom.borrow().value.tag_name() == Some("html")
370                        && child_dom.borrow().value.tag_name() == Some("body")
371                        && let NodeKind::Container { style, .. } = &mut kind
372                        && style.background == Background::Color(Color(0, 0, 0, 0))
373                    {
374                        let background = {
375                            let NodeKind::Container { style, .. } = &child_info.kind else {
376                                continue;
377                            };
378                            style.background.clone()
379                        };
380                        // html 要素の body 子要素に背景色が指定されていない場合、
381                        // body の背景色を html の背景色で上書きする
382                        style.background = background;
383                    }
384
385                    layout_children.push(child_layout.into());
386                    info_children.push(child_info);
387                }
388            }
389        }
390
391        let layout = LayoutNode::with_children(style, layout_children);
392
393        let info = InfoNode {
394            kind,
395            children: info_children,
396        };
397
398        (layout, info)
399    };
400
401    (layout, info)
402}

404fn normalize_whitespace(text: &str) -> String {
405    let mut result = String::new();
406    let mut prev_was_space = false;
407
408    for c in text.chars() {
409        if c.is_whitespace() {
410            if !prev_was_space {
411                result.push(' ');
412                prev_was_space = true;
413            }
414        } else {
415            result.push(c);
416            prev_was_space = false;
417        }
418    }
419
420    result
421}

423fn generate_fragment_node(fragment: &MeasuredFragment) -> ItemFragment {
424    if fragment.text == "\n" {
425        ItemFragment::LineBreak
426    } else {
427        ItemFragment::Fragment(Fragment {
428            width: fragment.width,
429            height: fragment.height,
430        })
431    }

434fn collect_candidates(
435    resolved_styles: &ResolvedStyles,
436    chain: &ElementChain,
437) -> HashMap<String, (CssValue, (u32, u32, u32), usize)> {
438    let mut candidates: HashMap<String, (CssValue, (u32, u32, u32), usize)> = HashMap::new();
439
440    for decl in resolved_styles {
441        if decl.selector.matches(chain) {
442            let entry = candidates.get(&decl.name);
443
444            let should_replace = match entry {
445                None => true,
446                Some((_, spec, order)) => {
447                    decl.specificity > *spec || (decl.specificity == *spec && decl.order > *order)
448                }
449            };
450
451            if should_replace {
452                candidates.insert(
453                    decl.name.clone(),
454                    (decl.value.clone(), decl.specificity, decl.order),
455                );
456            }
457        }
458    }
459
460    candidates
461}

463fn apply_declaration(
464    name: &str,
465    value: &CssValue,
466    style: &mut Style,
467    container_style: &mut ContainerStyle,
468    text_style: &mut TextStyle,
469) -> Option<()> {
470    fn expand_box<T: Clone, F>(
471        name: &str,
472        value: &CssValue,
473        text_style: &TextStyle,
474        resolve: &impl Fn(&str, &CssValue, &TextStyle) -> Option<T>,
475        mut set: F,
476    ) -> Option<()>
477    where
478        F: FnMut(T, T, T, T),
479    {
480        let resolve = |v: &CssValue| -> Option<T> { resolve(name, v, text_style) };
481
482        match value {
483            CssValue::List(values) => {
484                let vals: Vec<T> = values.iter().map(resolve).collect::<Option<_>>()?;
485
486                match vals.as_slice() {
487                    [a] => set(a.clone(), a.clone(), a.clone(), a.clone()),
488                    [v, h] => set(v.clone(), h.clone(), v.clone(), h.clone()),
489                    [t, h, b] => set(t.clone(), h.clone(), b.clone(), h.clone()),
490                    [t, r, b, l] => set(t.clone(), r.clone(), b.clone(), l.clone()),
491                    _ => return None,
492                }
493            }
494
495            _ => {
496                let v = resolve(value)?;
497                set(v.clone(), v.clone(), v.clone(), v);
498            }
499        }
500
501        Some(())
502    }
503
504    fn parse_border_shorthand(
505        name: &str,
506        value: &CssValue,
507        text_style: &TextStyle,
508    ) -> Option<(Option<Length>, Option<BorderStyle>, Option<Color>)> {
509        let mut width: Option<Length> = None;
510        let mut style_v: Option<BorderStyle> = None;
511        let mut color_v: Option<Color> = None;
512
513        let items: Vec<&CssValue> = match value {
514            CssValue::List(values) => values.iter().collect(),
515            _ => vec![value],
516        };
517
518        for v in items {
519            let token = v;
520
521            // try as length (numeric lengths)
522            if width.is_none()
523                && let Some(l) = resolve_css_len(name, token, text_style)
524            {
525                width = Some(l);
526                continue;
527            }
528
529            // try as width keyword (thin/medium/thick). Check keywords before style keywords.
530            if width.is_none()
531                && let CssValue::Keyword(s) = token
532            {
533                match s.as_str().to_ascii_lowercase().as_str() {
534                    "thin" => {
535                        width = Some(Length::Px(1.0));
536                        continue;
537                    }
538                    "medium" => {
539                        width = Some(Length::Px(3.0));
540                        continue;
541                    }
542                    "midium" => {
543                        width = Some(Length::Px(3.0));
544                        continue;
545                    } // common misspelling
546                    "thick" => {
547                        width = Some(Length::Px(5.0));
548                        continue;
549                    }
550                    _ => {}
551                }
552            }
553
554            // try as style keyword
555            if style_v.is_none()
556                && let CssValue::Keyword(s) = token
557            {
558                let s_lower = s.as_str();
559                let parsed = match s_lower {
560                    "none" => Some(BorderStyle::None),
561                    "solid" => Some(BorderStyle::Solid),
562                    "dashed" => Some(BorderStyle::Dashed),
563                    "dotted" => Some(BorderStyle::Dotted),
564                    _ => None,
565                };
566
567                if let Some(p) = parsed {
568                    style_v = Some(p);
569                    continue;
570                }
571            }
572
573            // try as color
574            if color_v.is_none()
575                && let Some(c) = resolve_css_color(name, token)
576            {
577                color_v = Some(c);
578                continue;
579            }
580
581            // unknown token: ignore
582        }
583
584        Some((width, style_v, color_v))
585    }
586
587    match (name, value) {
588        /* ======================
589         * Display
590         * ====================== */
591        ("display", CssValue::Keyword(v)) => {
592            if let Some(parsed_display) = Display::from_css_name(v.as_str()) {
593                style.display = parsed_display;
594                if parsed_display.inner == InnerDisplay::Flex {
595                    style.flex_direction = FlexDirection::Row;
596                }
597            }
598        }
599
600        /* ======================
601         * Color / Text
602         * ====================== */
603        ("background-color", _) => {
604            container_style.background = match value {
605                CssValue::Keyword(kw) if kw.eq_ignore_ascii_case("inherit") => {
606                    Background::Color(text_style.color)
607                }
608                CssValue::Keyword(kw) if kw.eq_ignore_ascii_case("currentColor") => {
609                    Background::Color(text_style.color)
610                }
611                _ => Background::Color(resolve_css_color(name, value)?),
612            };
613        }
614
615        ("background", _) => {
616            container_style.background = parse_background_shorthand(name, value, text_style)?;
617        }
618
619        ("color", _) => {
620            text_style.color = match value {
621                CssValue::Keyword(kw) if kw.eq_ignore_ascii_case("inherit") => {
622                    // inherit: use parent's color
623                    text_style.color
624                }
625                CssValue::Keyword(kw) if kw.eq_ignore_ascii_case("currentColor") => {
626                    text_style.color
627                }
628                _ => resolve_css_color(name, value)?,
629            }
630        }
631
632        ("font-size", CssValue::Length(_, _)) => {
633            // TODO: Add other size
634            let len = resolve_css_len(name, value, text_style)?;
635            let px = match &len {
636                Length::Px(v) => *v,
637                Length::Percent(v) => *v * text_style.font_size / 100.0,
638                _ => {
639                    log::error!(target: "Layouter", "Unknown size type for `{}`: {:?}", name, len);
640                    return None;
641                }
642            };
643            text_style.font_size = px;
644        }
645
646        ("line-height", CssValue::Number(factor)) => {
647            text_style.line_height = LineHeight::Number(*factor);
648            style.line_height = Length::Px(text_style.font_size * factor);
649        }
650        ("line-height", CssValue::Keyword(v)) if v == "normal" => {
651            text_style.line_height = LineHeight::Normal;
652            style.line_height = Length::Px(text_style.font_size * DEFAULT_LINE_FACTOR);
653        }
654        ("line-height", _) => {
655            let len = resolve_css_len(name, value, text_style)?;
656            text_style.line_height = LineHeight::Px(length_to_px(&len, text_style.font_size));
657            style.line_height = len;
658        }
659
660        ("font-weight", CssValue::Keyword(v)) => {
661            text_style.font_weight = match v.as_str() {
662                "normal" => FontWeight::NORMAL,
663                "bold" => FontWeight::BOLD,
664                _ => text_style.font_weight,
665            };
666        }
667        ("font-weight", CssValue::Number(v)) => {
668            text_style.font_weight = FontWeight(*v as u16);
669        }
670
671        ("font-style", CssValue::Keyword(v)) => {
672            text_style.font_style = match v.as_str() {
673                "normal" => FontStyle::Normal,
674                "italic" => FontStyle::Italic,
675                "oblique" => FontStyle::Oblique,
676                _ => text_style.font_style,
677            };
678        }
679
680        ("text-decoration", CssValue::Keyword(v)) => {
681            text_style.text_decoration = match v.as_str() {
682                "none" => TextDecoration::None,
683                "underline" => TextDecoration::Underline,
684                "line-through" => TextDecoration::LineThrough,
685                "overline" => TextDecoration::Overline,
686                _ => TextDecoration::None,
687            };
688        }
689
690        ("text-align", CssValue::Keyword(v)) if v == "left" => {
691            text_style.text_align = TextAlign::Left;
692        }
693        ("text-align", CssValue::Keyword(v)) if v == "center" => {
694            text_style.text_align = TextAlign::Center;
695        }
696        ("text-align", CssValue::Keyword(v)) if v == "right" => {
697            text_style.text_align = TextAlign::Right;
698        }
699
700        /* ======================
701         * Box Model
702         * ====================== */
703        ("box-sizing", CssValue::Keyword(v)) => {
704            style.box_sizing = match v.as_str() {
705                "content-box" => BoxSizing::ContentBox,
706                "border-box" => BoxSizing::BorderBox,
707                _ => BoxSizing::ContentBox,
708            };
709        }
710
711        ("border-style", CssValue::Keyword(v)) => {
712            let s = match v.as_str() {
713                "none" => BorderStyle::None,
714                "solid" => BorderStyle::Solid,
715                "dashed" => BorderStyle::Dashed,
716                "dotted" => BorderStyle::Dotted,
717                _ => BorderStyle::None,
718            };
719
720            container_style.border_style.top = s;
721            container_style.border_style.right = s;
722            container_style.border_style.bottom = s;
723            container_style.border_style.left = s;
724        }
725
726        ("margin", v) => {
727            expand_box(
728                name,
729                v,
730                text_style,
731                &|_, cv, ts| match cv {
732                    CssValue::Keyword(s) if s == "auto" => Some(ui_layout::LengthOrAuto::Auto),
733                    _ => resolve_css_len(name, cv, ts).map(ui_layout::LengthOrAuto::Length),
734                },
735                |t, r, b, l| {
736                    style.spacing.margin_top = t;
737                    style.spacing.margin_right = r;
738                    style.spacing.margin_bottom = b;
739                    style.spacing.margin_left = l;
740                },
741            )?;
742        }
743        ("margin-top", CssValue::Keyword(s)) if s == "auto" => {
744            style.spacing.margin_top = LengthOrAuto::Auto;
745        }
746        ("margin-top", _) => {
747            style.spacing.margin_top =
748                LengthOrAuto::Length(resolve_css_len(name, value, text_style)?);
749        }
750        ("margin-right", CssValue::Keyword(s)) if s == "auto" => {
751            style.spacing.margin_right = LengthOrAuto::Auto;
752        }
753        ("margin-right", _) => {
754            style.spacing.margin_right =
755                LengthOrAuto::Length(resolve_css_len(name, value, text_style)?);
756        }
757        ("margin-bottom", CssValue::Keyword(s)) if s == "auto" => {
758            style.spacing.margin_bottom = LengthOrAuto::Auto;
759        }
760        ("margin-bottom", _) => {
761            style.spacing.margin_bottom =
762                LengthOrAuto::Length(resolve_css_len(name, value, text_style)?);
763        }
764        ("margin-left", CssValue::Keyword(s)) if s == "auto" => {
765            style.spacing.margin_left = LengthOrAuto::Auto;
766        }
767        ("margin-left", _) => {
768            style.spacing.margin_left =
769                LengthOrAuto::Length(resolve_css_len(name, value, text_style)?);
770        }
771
772        ("border", v) => {
773            let (maybe_width, maybe_style, maybe_color) = if let CssValue::Keyword(k) = v
774                && (k.eq_ignore_ascii_case("inset") || k.eq_ignore_ascii_case("initial"))
775            {
776                (Some(Length::Px(0.0)), None, None)
777            } else {
778                parse_border_shorthand(name, v, text_style)?
779            };
780
781            if let Some(w) = maybe_width {
782                style.spacing.border_top = w.clone();
783                style.spacing.border_right = w.clone();
784                style.spacing.border_bottom = w.clone();
785                style.spacing.border_left = w;
786            }
787
788            if let Some(s) = maybe_style {
789                container_style.border_style.top = s;
790                container_style.border_style.right = s;
791                container_style.border_style.bottom = s;
792                container_style.border_style.left = s;
793            }
794
795            if let Some(c) = maybe_color {
796                container_style.border_color.top = c;
797                container_style.border_color.right = c;
798                container_style.border_color.bottom = c;
799                container_style.border_color.left = c;
800            }
801        }
802        ("border-top", _) => {
803            let (maybe_width, maybe_style, maybe_color) =
804                parse_border_shorthand(name, value, text_style)?;
805            if let Some(w) = maybe_width {
806                style.spacing.border_top = w;
807            }
808            if let Some(s) = maybe_style {
809                container_style.border_style.top = s;
810            }
811            if let Some(c) = maybe_color {
812                container_style.border_color.top = c;
813            }
814        }
815        ("border-right", _) => {
816            let (maybe_width, maybe_style, maybe_color) =
817                parse_border_shorthand(name, value, text_style)?;
818            if let Some(w) = maybe_width {
819                style.spacing.border_right = w;
820            }
821            if let Some(s) = maybe_style {
822                container_style.border_style.right = s;
823            }
824            if let Some(c) = maybe_color {
825                container_style.border_color.right = c;
826            }
827        }
828        ("border-bottom", _) => {
829            let (maybe_width, maybe_style, maybe_color) =
830                parse_border_shorthand(name, value, text_style)?;
831            if let Some(w) = maybe_width {
832                style.spacing.border_bottom = w;
833            }
834            if let Some(s) = maybe_style {
835                container_style.border_style.bottom = s;
836            }
837            if let Some(c) = maybe_color {
838                container_style.border_color.bottom = c;
839            }
840        }
841        ("border-left", _) => {
842            let (maybe_width, maybe_style, maybe_color) =
843                parse_border_shorthand(name, value, text_style)?;
844            if let Some(w) = maybe_width {
845                style.spacing.border_left = w;
846            }
847            if let Some(s) = maybe_style {
848                container_style.border_style.left = s;
849            }
850            if let Some(c) = maybe_color {
851                container_style.border_color.left = c;
852            }
853        }
854
855        ("padding", v) => {
856            expand_box(
857                name,
858                v,
859                text_style,
860                &|_, v, ts| resolve_css_len(name, v, ts),
861                |t, r, b, l| {
862                    style.spacing.padding_top = t;
863                    style.spacing.padding_right = r;
864                    style.spacing.padding_bottom = b;
865                    style.spacing.padding_left = l;
866                },
867            )?;
868        }
869        ("padding-top", _) => {
870            style.spacing.padding_top = resolve_css_len(name, value, text_style)?;
871        }
872        ("padding-right", _) => {
873            style.spacing.padding_right = resolve_css_len(name, value, text_style)?;
874        }
875        ("padding-bottom", _) => {
876            style.spacing.padding_bottom = resolve_css_len(name, value, text_style)?;
877        }
878        ("padding-left", _) => {
879            style.spacing.padding_left = resolve_css_len(name, value, text_style)?;
880        }
881
882        /* ======================
883         * Size
884         * ====================== */
885        ("width", CssValue::Keyword(s)) if s == "auto" => {
886            style.size.width = LengthOrAuto::Auto;
887        }
888        ("width", _) => {
889            style.size.width = LengthOrAuto::Length(resolve_css_len(name, value, text_style)?);
890        }
891        ("height", CssValue::Keyword(s)) if s == "auto" => {
892            style.size.height = LengthOrAuto::Auto;
893        }
894        ("height", _) => {
895            style.size.height = LengthOrAuto::Length(resolve_css_len(name, value, text_style)?);
896        }
897        ("min-width", CssValue::Keyword(s)) if s == "auto" => {
898            style.size.min_width = LengthOrAuto::Auto;
899        }
900        ("min-width", _) => {
901            style.size.min_width = LengthOrAuto::Length(resolve_css_len(name, value, text_style)?);
902        }
903        ("min-height", CssValue::Keyword(s)) if s == "auto" => {
904            style.size.min_height = LengthOrAuto::Auto;
905        }
906        ("min-height", _) => {
907            style.size.min_height = LengthOrAuto::Length(resolve_css_len(name, value, text_style)?);
908        }
909        ("max-width", CssValue::Keyword(s)) if s == "auto" => {
910            style.size.max_width = LengthOrAuto::Auto;
911        }
912        ("max-width", _) => {
913            style.size.max_width = LengthOrAuto::Length(resolve_css_len(name, value, text_style)?);
914        }
915        ("max-height", CssValue::Keyword(s)) if s == "auto" => {
916            style.size.max_height = LengthOrAuto::Auto;
917        }
918        ("max-height", _) => {
919            style.size.max_height = LengthOrAuto::Length(resolve_css_len(name, value, text_style)?);
920        }
921
922        /* ======================
923         * Flex
924         * ====================== */
925        ("flex-direction", CssValue::Keyword(v)) => {
926            if style.display.inner == InnerDisplay::Flex {
927                style.flex_direction = match v.as_str() {
928                    "row" => FlexDirection::Row,
929                    "column" => FlexDirection::Column,
930                    _ => return None,
931                };
932            }
933        }
934
935        ("justify-content", CssValue::Keyword(v)) => {
936            style.justify_content = match v.as_str() {
937                "flex-start" => JustifyContent::Start,
938                "center" => JustifyContent::Center,
939                "flex-end" => JustifyContent::End,
940                "space-between" => JustifyContent::SpaceBetween,
941                "space-around" => JustifyContent::SpaceAround,
942                _ => return None,
943            };
944        }
945
946        ("align-items", CssValue::Keyword(v)) => {
947            style.align_items = match v.as_str() {
948                "stretch" => AlignItems::Stretch,
949                "flex-start" => AlignItems::Start,
950                "center" => AlignItems::Center,
951                "flex-end" => AlignItems::End,
952                _ => return None,
953            };
954        }
955
956        ("gap", CssValue::Keyword(s)) if s == "auto" => {
957            style.row_gap = LengthOrAuto::Auto;
958            style.column_gap = LengthOrAuto::Auto;
959        }
960        ("gap", _) => {
961            let gap = LengthOrAuto::Length(resolve_css_len(name, value, text_style)?);
962            style.row_gap = gap.clone();
963            style.column_gap = gap;
964        }
965
966        ("align-self", CssValue::Keyword(v)) => {
967            style.item_style.align_self = match v.as_str() {
968                "stretch" => Some(AlignItems::Stretch),
969                "flex-start" => Some(AlignItems::Start),
970                "center" => Some(AlignItems::Center),
971                "flex-end" => Some(AlignItems::End),
972                _ => return None,
973            };
974        }
975
976        ("flex-grow", CssValue::Number(v)) => {
977            style.item_style.flex_grow = *v;
978        }
979
980        ("flex-basis", CssValue::Keyword(s)) if s == "auto" => {
981            style.item_style.flex_basis = LengthOrAuto::Auto;
982        }
983        ("flex-basis", _) => {
984            style.item_style.flex_basis =
985                LengthOrAuto::Length(resolve_css_len(name, value, text_style)?);
986        }
987
988        _ => {}
989    }
990    Some(())
991}

993// =========================
994//   Background Shorthand
995// =========================
996
997fn parse_background_shorthand(
998    name: &str,
999    value: &CssValue,
1000    text_style: &TextStyle,
1001) -> Option<Background> {
1002    let items: Vec<&CssValue> = match value {
1003        CssValue::List(values) => values.iter().collect(),
1004        _ => vec![value],
1005    };
1006
1007    let mut maybe_color: Option<Color> = None;
1008    let mut maybe_gradient: Option<Gradient> = None;
1009
1010    for v in items {
1011        // inherit
1012        if let CssValue::Keyword(kw) = v {
1013            if kw.eq_ignore_ascii_case("inherit") {
1014                maybe_color = Some(text_style.color);
1015                continue;
1016            }
1017            if kw.eq_ignore_ascii_case("currentColor") {
1018                maybe_color = Some(text_style.color);
1019                continue;
1020            }
1021        }
1022
1023        if let CssValue::Number(0.0) = v {
1024            maybe_color = Some(Color(0, 0, 0, 0));
1025            continue;
1026        }
1027
1028        // gradient
1029        if let CssValue::Function(fn_name, args) = v {
1030            if fn_name == "linear-gradient" || fn_name == "radial-gradient" {
1031                maybe_gradient = Some(parse_gradient(fn_name, args, text_style)?);
1032                continue;
1033            }
1034        }
1035
1036        // color
1037        if let Some(c) = resolve_css_color(name, v) {
1038            maybe_color = Some(c);
1039            continue;
1040        }
1041    }
1042
1043    if let Some(g) = maybe_gradient {
1044        return Some(Background::Gradient(g));
1045    }
1046    if let Some(c) = maybe_color {
1047        return Some(Background::Color(c));
1048    }
1049
1050    None
1051}

1053// =========================
1054//   Gradient Parsing
1055// =========================
1056
1057fn parse_gradient(fn_name: &str, args: &[CssValue], text_style: &TextStyle) -> Option<Gradient> {
1058    match fn_name {
1059        "linear-gradient" => parse_linear_gradient(args, text_style),
1060        "radial-gradient" => parse_radial_gradient(args, text_style),
1061        _ => None,
1062    }
1063}
1064
1065fn parse_linear_gradient(args: &[CssValue], _text_style: &TextStyle) -> Option<Gradient> {
1066    if args.is_empty() {
1067        return None;
1068    }
1069
1070    let (skip, angle) = parse_linear_direction(args);
1071    let angle = angle.unwrap_or(180.0);
1072    let stops = parse_color_stops(&args[skip..])?;
1073
1074    Some(Gradient {
1075        kind: GradientKind::Linear { angle },
1076        stops,
1077    })
1078}
1079
1080/// Returns (number_of_consumed_args, optional_angle_in_degrees).
1081fn parse_linear_direction(args: &[CssValue]) -> (usize, Option<f32>) {
1082    if args.is_empty() {
1083        return (0, None);
1084    }
1085
1086    // <angle>
1087    if let CssValue::Length(v, Unit::Deg) = &args[0] {
1088        return (1, Some(*v));
1089    }
1090
1091    // "to" <side-or-corner>
1092    if let CssValue::Keyword(k) = &args[0] {
1093        if k.as_str() == "to" && args.len() > 1 {
1094            let mut idx = 1;
1095            let mut sides: Vec<&str> = Vec::new();
1096            while idx < args.len() {
1097                if let CssValue::Keyword(k) = &args[idx] {
1098                    match k.as_str() {
1099                        "top" | "bottom" | "left" | "right" => {
1100                            sides.push(k.as_str());
1101                            idx += 1;
1102                        }
1103                        _ => break,
1104                    }
1105                } else {
1106                    break;
1107                }
1108            }
1109            if !sides.is_empty() {
1110                let angle = match sides.as_slice() {
1111                    ["top"] => Some(0.0),
1112                    ["top", "left"] => Some(315.0),
1113                    ["top", "right"] => Some(45.0),
1114                    ["bottom"] => Some(180.0),
1115                    ["bottom", "left"] => Some(225.0),
1116                    ["bottom", "right"] => Some(135.0),
1117                    ["left"] => Some(270.0),
1118                    ["right"] => Some(90.0),
1119                    _ => None,
1120                };
1121                return (idx, angle);
1122            }
1123        }
1124    }
1125
1126    (0, None)
1127}
1128
1129fn parse_radial_gradient(args: &[CssValue], _text_style: &TextStyle) -> Option<Gradient> {
1130    let mut shape = RadialShape::Ellipse;
1131    let mut size = RadialSizeKind::default();
1132    let mut position = (0.5f32, 0.5f32);
1133
1134    let mut idx = 0;
1135
1136    // Consume known radial keywords before color stops
1137    while idx < args.len() {
1138        if let CssValue::Keyword(k) = &args[idx] {
1139            match k.as_str() {
1140                "circle" => {
1141                    shape = RadialShape::Circle;
1142                    idx += 1;
1143                    continue;
1144                }
1145                "ellipse" => {
1146                    shape = RadialShape::Ellipse;
1147                    idx += 1;
1148                    continue;
1149                }
1150                "closest-side" => {
1151                    size = RadialSizeKind::ClosestSide;
1152                    idx += 1;
1153                    continue;
1154                }
1155                "farthest-side" => {
1156                    size = RadialSizeKind::FarthestSide;
1157                    idx += 1;
1158                    continue;
1159                }
1160                "closest-corner" => {
1161                    size = RadialSizeKind::ClosestCorner;
1162                    idx += 1;
1163                    continue;
1164                }
1165                "farthest-corner" => {
1166                    size = RadialSizeKind::FarthestCorner;
1167                    idx += 1;
1168                    continue;
1169                }
1170                _ => break,
1171            }
1172        } else {
1173            break;
1174        }
1175    }
1176
1177    // Optional "at <position>" — simplified to "at center" / "at top left" etc.
1178    if idx < args.len() && args[idx] == CssValue::Keyword("at".into()) {
1179        idx += 1; // skip "at"
1180        if idx < args.len() {
1181            if let CssValue::Keyword(k) = &args[idx] {
1182                // Parse position keywords
1183                match k.as_str() {
1184                    "center" => position = (0.5, 0.5),
1185                    "top" => position = (0.5, 0.0),
1186                    "bottom" => position = (0.5, 1.0),
1187                    "left" => position = (0.0, 0.5),
1188                    "right" => position = (1.0, 0.5),
1189                    _ => {} // ignore unknown
1190                }
1191                idx += 1;
1192                // Optional second keyword (e.g. "top left")
1193                if idx < args.len() {
1194                    if let CssValue::Keyword(k2) = &args[idx] {
1195                        match (k.as_str(), k2.as_str()) {
1196                            ("top", "left") | ("left", "top") => position = (0.0, 0.0),
1197                            ("top", "right") | ("right", "top") => position = (1.0, 0.0),
1198                            ("bottom", "left") | ("left", "bottom") => position = (0.0, 1.0),
1199                            ("bottom", "right") | ("right", "bottom") => position = (1.0, 1.0),
1200                            _ => {}
1201                        }
1202                        idx += 1;
1203                    }
1204                }
1205            }
1206        }
1207    }
1208
1209    let stops = parse_color_stops(&args[idx..])?;
1210    if stops.is_empty() {
1211        return None;
1212    }
1213    Some(Gradient {
1214        kind: GradientKind::Radial {
1215            shape,
1216            size,
1217            position,
1218        },
1219        stops,
1220    })
1221}
1222
1223fn parse_color_stops(args: &[CssValue]) -> Option<Vec<ColorStop>> {
1224    let mut stops = Vec::new();
1225    let mut i = 0;
1226
1227    while i < args.len() {
1228        let color = resolve_css_color("gradient", &args[i])?;
1229        i += 1;
1230
1231        let position = if i < args.len() {
1232            match &args[i] {
1233                CssValue::Length(v, Unit::Percent) => {
1234                    i += 1;
1235                    Some((*v / 100.0).clamp(0.0, 1.0))
1236                }
1237                CssValue::Length(_v, Unit::Px) => {
1238                    i += 1;
1239                    None
1240                }
1241                _ => None,
1242            }
1243        } else {
1244            None
1245        };
1246
1247        stops.push(ColorStop { color, position });
1248    }
1249
1250    Some(stops)
1251}

1253/// Resolve CssValue to Length.
1254fn resolve_css_len(name: &str, css_len: &CssValue, text_style: &TextStyle) -> Option<Length> {
1255    match &css_len {
1256        CssValue::Length(v, Unit::Em) => Some(Length::Px(text_style.font_size * v)),
1257        CssValue::Length(v, Unit::Rem) => Some(Length::Px(16.0 * v)), // html sont-size 仮値
1258        CssValue::Length(v, u) => match u {
1259            Unit::Percent => Some(Length::Percent(*v)),
1260            Unit::Px => Some(Length::Px(*v)),
1261            Unit::Vw => Some(Length::Vw(*v)),
1262            Unit::Vh => Some(Length::Vh(*v)),
1263            Unit::Em | Unit::Rem => unreachable!(),
1264            Unit::Deg => {
1265                log::error!(target: "Layouter", "Unexpected deg unit for `{}` (expected length)", name);
1266                return None;
1267            }
1268        },
1269        CssValue::Number(0.0) => Some(Length::Px(0.0)),
1270        CssValue::Keyword(_) => None,
1271        CssValue::Function(fn_name, args) if fn_name == "calc" && !args.is_empty() => {
1272            let mut iter = args.iter();
1273            let mut result = resolve_css_len(name, iter.next().unwrap(), text_style)?;
1274
1275            while let (Some(op), Some(val)) = (iter.next(), iter.next()) {
1276                match op {
1277                    CssValue::Keyword(o) if o == "+" => {
1278                        let val_resolved = resolve_css_len(name, val, text_style)?;
1279                        result = Length::Add(Box::new(result), Box::new(val_resolved));
1280                    }
1281                    CssValue::Keyword(o) if o == "-" => {
1282                        let val_resolved = resolve_css_len(name, val, text_style)?;
1283                        result = Length::Sub(Box::new(result), Box::new(val_resolved));
1284                    }
1285                    CssValue::Keyword(o) if o == "*" => {
1286                        if let CssValue::Number(factor) = val {
1287                            result = Length::Mul(Box::new(result), *factor);
1288                        } else {
1289                            log::error!(target: "Layouter", "Invalid operand for multiplication in calc() for `{}`: {:?}", name, val);
1290                            return None;
1291                        }
1292                    }
1293                    CssValue::Keyword(o) if o == "/" => {
1294                        if let CssValue::Number(factor) = val {
1295                            if *factor == 0.0 {
1296                                log::error!(target: "Layouter", "Division by zero in calc() for `{}`", name);
1297                                return None;
1298                            }
1299                            result = Length::Div(Box::new(result), *factor);
1300                        } else {
1301                            log::error!(target: "Layouter", "Invalid operand for division in calc() for `{}`: {:?}", name, val);
1302                            return None;
1303                        }
1304                    }
1305                    _ => {
1306                        log::error!(target: "Layouter", "Unknown operator in calc() for `{}`: {:?}", name, op);
1307                        return None;
1308                    }
1309                }
1310            }
1311
1312            Some(result)
1313        }
1314        CssValue::Color(_) => None,
1315        _ => {
1316            log::error!(target: "Layouter", "Unknown CSS Length type for `{}`: {:?}", name, css_len);
1317            None
1318        }
1319    }
1320}

1322/// Resolve a computed CssValue into a final RGBA Color.
1323///
1324/// Assumptions:
1325/// - This function is called *after* cascade and inheritance resolution.
1326/// - Keywords like `currentColor`, `inherit`, `initial`, `unset`
1327///   must NOT reach this stage.
1328/// - The returned Color is always absolute RGBA.
1329fn resolve_css_color(name: &str, css_color: &CssValue) -> Option<Color> {
1330    fn keyword_color_to_color(name: &str, keyword: &str) -> Option<Color> {
1331        // NOTE:
1332        // Keyword matching is case-insensitive according to CSS specs.
1333        // Keep this list limited to commonly used CSS Color Level 3 keywords.
1334        match keyword.to_ascii_lowercase().as_str() {
1335            // ===== Basic =====
1336            "black" => Some(Color(0, 0, 0, 255)),
1337            "silver" => Some(Color(192, 192, 192, 255)),
1338            "gray" | "grey" => Some(Color(128, 128, 128, 255)),
1339            "white" => Some(Color(255, 255, 255, 255)),
1340
1341            // ===== Red =====
1342            "maroon" => Some(Color(128, 0, 0, 255)),
1343            "red" => Some(Color(255, 0, 0, 255)),
1344            "firebrick" => Some(Color(178, 34, 34, 255)),
1345            "crimson" => Some(Color(220, 20, 60, 255)),
1346            "indianred" => Some(Color(205, 92, 92, 255)),
1347            "lightcoral" => Some(Color(240, 128, 128, 255)),
1348            "salmon" => Some(Color(250, 128, 114, 255)),
1349            "darksalmon" => Some(Color(233, 150, 122, 255)),
1350            "lightsalmon" => Some(Color(255, 160, 122, 255)),
1351

1469    /// Convert HSL to RGB (0..255)
1470    fn hsla_to_rgba(h: f32, s: f32, l: f32, a: f32) -> (u8, u8, u8, u8) {
1471        // 1. Compute Chroma
1472        let c = (1.0 - (2.0 * l - 1.0).abs()) * s;
1473        let h_prime = h / 60.0;
1474        let x = c * (1.0 - ((h_prime % 2.0) - 1.0).abs());
1475
1476        // 2. Determine preliminary RGB values based on hue sector
1477        let (r1, g1, b1) = match h_prime as u32 {
1478            0 => (c, x, 0.0),
1479            1 => (x, c, 0.0),
1480            2 => (0.0, c, x),
1481            3 => (0.0, x, c),
1482            4 => (x, 0.0, c),
1483            5 | 6 => (c, 0.0, x),
1484            _ => (0.0, 0.0, 0.0),
1485        };
1486
1487        // 3. Add m to match the lightness
1488        let m = l - c / 2.0;
1489        let r = ((r1 + m) * 255.0).round().clamp(0.0, 255.0) as u8;
1490        let g = ((g1 + m) * 255.0).round().clamp(0.0, 255.0) as u8;
1491        let b = ((b1 + m) * 255.0).round().clamp(0.0, 255.0) as u8;
1492        let a = (a * 255.0).round().clamp(0.0, 255.0) as u8;
1493
1494        (r, g, b, a)
1495    }

1497    match css_color {
1498        // Already parsed as an absolute color (rgb/rgba/hex, etc.)
1499        CssValue::Color(_) => {
1500            let (r, g, b, a) = css_color.to_rgba_tuple()?;
1501            Some(Color(r, g, b, a))
1502        }
1503
1504        // Named color keyword
1505        CssValue::Keyword(value) => keyword_color_to_color(name, value),
1506
1507        // rgb() / rgba() unified
1508        CssValue::Function(func, args) if func == "rgb" || func == "rgba" => {
1509            // Extract numeric components, ignoring commas and handling '/'
1510            let mut numbers = Vec::new();
1511            let mut alpha: Option<f32> = None;
1512            let mut after_slash = false;
1513
1514            for arg in args {
1515                match arg {
1516                    CssValue::Keyword(k) if k == "/" => {
1517                        after_slash = true;
1518                    }
1519                    CssValue::Number(n) => {
1520                        if after_slash {
1521                            alpha = Some(*n);
1522                        } else {
1523                            numbers.push(*n);
1524                        }
1525                    }
1526                    _ => return None,
1527                }
1528            }
1529
1530            if numbers.len() != 3 {
1531                return None;
1532            }
1533
1534            let a = alpha.unwrap_or(1.0);
1535
1536            Some(Color(
1537                (numbers[0] * 255.0).round() as u8,
1538                (numbers[1] * 255.0).round() as u8,
1539                (numbers[2] * 255.0).round() as u8,
1540                (a * 255.0).round() as u8,
1541            ))
1542        }
1543
1544        // hsl() / hsla() unified
1545        CssValue::Function(func, args) if func == "hsl" || func == "hsla" => {
1546            // Collect h, s, l and optional alpha
1547            let mut numbers = Vec::new();
1548            let mut alpha: Option<f32> = None;
1549            let mut after_slash = false;
1550
1551            for arg in args {
1552                match arg {
1553                    CssValue::Keyword(k) if k == "/" => {
1554                        after_slash = true;
1555                    }
1556                    CssValue::Number(n) => {
1557                        if after_slash {
1558                            alpha = Some(*n);
1559                        } else {
1560                            numbers.push(*n);
1561                        }
1562                    }
1563                    _ => return None,
1564                }
1565            }
1566
1567            if numbers.len() != 3 {
1568                return None;
1569            }
1570
1571            let a = alpha.unwrap_or(1.0);
1572            let (r, g, b, a) = hsla_to_rgba(numbers[0], numbers[1], numbers[2], a);
1573
1574            Some(Color(r, g, b, a))
1575        }
1576
1577        // Any other value reaching here is a pipeline error
1578        _ => {
1579            log::error!(
1580                target: "Layouter",
1581                "Unexpected CSS color value for `{}` at layout stage: {:?}",
1582                name,
