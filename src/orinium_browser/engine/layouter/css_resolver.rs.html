10struct Declaration {
11    name: String,
12    value: CssValue,
13    important: bool,
14}

16/// A single CSS declaration after selector resolution and value processing.
17///
18/// `ResolvedDeclaration` represents one property-value pair that has been
19/// fully associated with a selector and enriched with all information
20/// required for CSS cascade resolution.
21///
22/// This structure is produced after:
23/// - Parsing selectors
24/// - Resolving `var()` using custom properties
25/// - Computing selector specificity
26///
27/// During the cascade phase, multiple `ResolvedDeclaration`s with the same
28/// property name may compete. The winner is determined by comparing:
29///
30/// 1. `specificity` (higher specificity wins)
31/// 2. `order` (later declarations win)
32#[derive(Debug, Clone)]
33pub struct ResolvedDeclaration {
34    pub selector: ComplexSelector,
35    pub name: String,
36    pub value: CssValue,
37    pub specificity: (u32, u32, u32),
38    pub order: usize,
39    pub important: bool,
40}
41
42pub type ResolvedStyles = Vec<ResolvedDeclaration>;
43
44// ============================================================
45//  CssResolver — tree walk + rule resolution
46// ============================================================
47
48pub struct CssResolver;
49
50impl CssResolver {
51    pub fn resolve(stylesheet: &CssNode) -> ResolvedStyles {
52        let mut styles = Vec::new();
53        let mut order = 0;
54        Self::walk(stylesheet, &mut styles, &mut order);
55        styles
56    }
57
58    fn walk(node: &CssNode, styles: &mut ResolvedStyles, order: &mut usize) {
59        Self::resolve_rule(node, styles, order);
60
61        if Self::should_recurse(node) {
62            for child in node.children() {
63                Self::walk(child, styles, order);
64            }
65        }
66    }
67
68    fn resolve_rule(node: &CssNode, styles: &mut ResolvedStyles, order: &mut usize) {
69        let CssNodeType::Rule { selectors } = node.node() else {
70            return;
71        };
72
73        let declarations = DeclarationResolver::collect(node);
74
75        for selector in selectors {
76            Self::push_resolved(selector, &declarations, styles, order);
77        }
78    }
79
80    fn push_resolved(
81        selector: &ComplexSelector,
82        declarations: &[Declaration],
83        styles: &mut ResolvedStyles,
84        order: &mut usize,
85    ) {
86        let specificity = selector.specificity();
87
88        for decl in declarations {
89            styles.push(ResolvedDeclaration {
90                selector: selector.clone(),
91                name: decl.name.clone(),
92                value: decl.value.clone(),
93                specificity,
94                order: *order,
95                important: decl.important,
96            });
97            *order += 1;
98        }
99    }
100
101    fn should_recurse(node: &CssNode) -> bool {
102        if let CssNodeType::AtRule { name, params } = node.node() {
103            Self::evaluate_at_rule(name, params)
104        } else {
105            true
106        }
107    }
108
109    fn evaluate_at_rule(name: &str, params: &AtQuery) -> bool {
110        if name.eq_ignore_ascii_case("supports") {
111            SupportsEvaluator::evaluate(params)
112        } else {
113            true
114        }
115    }
116}

118// ============================================================
119//  SupportsEvaluator — `@supports` condition evaluation
120// ============================================================
121
122struct SupportsEvaluator;
123
124impl SupportsEvaluator {
125    /// Dispatch on the `AtQuery` AST variant.
126    fn evaluate(query: &AtQuery) -> bool {
127        match query {
128            // `@supports (display: grid)` — a parenthesised group
129            AtQuery::Group(items) => Self::evaluate_group(items),
130            // `@supports (display: grid)` — the inner condition after unwrapping
131            AtQuery::Condition { name, value } => Self::is_supported(name, value),
132            // Stray keyword outside a group (malformed input)
133            AtQuery::Keyword(_) => false,
134        }
135    }
136
137    /// Evaluate a group of `@supports` items.
138    ///
139    /// The parser produces flat groups like:
140    /// - `(display: grid)` → `[Group([Condition])]`
141    /// - `(A) and (B)` → `[Group, Keyword("and"), Group]`
142    /// - `not (A)` → `[Keyword("not"), Group]`
143    /// - `(A) or (B)` → `[Group, Keyword("or"), Group]`
144    fn evaluate_group(items: &[AtQuery]) -> bool {
145        if items.is_empty() {
146            return false;
147        }
148
149        // `not (display: grid)` — negate
150        if matches!(items.first(), Some(AtQuery::Keyword(k)) if k.eq_ignore_ascii_case("not")) {
151            return items.len() > 1 && !Self::evaluate(&AtQuery::Group(items[1..].to_vec()));
152        }
153
154        // `(display: flex) and (gap: 10px)` — all operands must be supported
155        if let Some(operands) = Self::split_by_keyword(items, "and") {
156            operands.iter().all(|g| Self::evaluate(g))
157        // `(display: flex) or (display: grid)` — at least one must be supported
158        } else if let Some(operands) = Self::split_by_keyword(items, "or") {
159            operands.iter().any(|g| Self::evaluate(g))
160        // Single group — unwrap one level
161        } else if items.len() == 1 {
162            Self::evaluate(&items[0])
163        } else {
164            items.iter().all(|g| Self::evaluate(g))
165        }
166    }
167
168    fn is_supported(name: &str, value: &CssValue) -> bool {
169        super::builder::apply_declaration(
170            name,
171            value,
172            &mut ui_layout::Style::default(),
173            &mut super::types::ContainerStyle::default(),
174            &mut super::types::TextStyle::default(),
175        )
176        .is_some()
177    }
178
179    fn split_by_keyword<'a>(items: &'a [AtQuery], keyword: &str) -> Option<Vec<&'a AtQuery>> {
180        let has = items
181            .iter()
182            .any(|item| matches!(item, AtQuery::Keyword(k) if k.eq_ignore_ascii_case(keyword)));
183        if !has {
184            return None;
185        }
186        Some(
187            items
188                .iter()
189                .filter(
190                    |item| !matches!(item, AtQuery::Keyword(k) if k.eq_ignore_ascii_case(keyword)),
191                )
192                .collect(),
193        )
194    }
195}

197// ============================================================
198//  DeclarationResolver — `!important` extraction, `var()` resolution
199// ============================================================
200
201struct DeclarationResolver;
202
203impl DeclarationResolver {
204    fn collect(rule_node: &CssNode) -> Vec<Declaration> {
205        let mut custom_props: CustomProperties = HashMap::new();
206        let mut result = Vec::new();
207
208        for child in rule_node.children() {
209            if let CssNodeType::Declaration { name, value } = &child.node() {
210                if name.starts_with("--") {
211                    custom_props.insert(name.into(), value.clone());
212                }
213            }
214        }
215
216        for child in rule_node.children() {
217            if let CssNodeType::Declaration { name, value } = &child.node() {
218                let (raw_value, important) = Self::extract_important(value);
219
220                if name.starts_with("--") {
221                    // `--accent: blue` — custom property, keep as-is
222                    result.push(Declaration {
223                        name: name.clone(),
224                        value: raw_value,
225                        important,
226                    });
227                } else if let Some(resolved) =
228                    Self::resolve_var(&raw_value, &custom_props, &mut HashSet::new())
229                {
230                    // `color: var(--accent)` — resolve var() then emit
231                    result.push(Declaration {
232                        name: name.clone(),
233                        value: resolved,
234                        important,
235                    });
236                }
237            }
238        }
239
240        result
241    }
242
243    /// Extract `!important` from a CSS value.
244    ///
245    /// `border: 1px solid black !important` is parsed as a `List` where the
246    /// last two items are `Keyword("!")` and `Keyword("important")`.
247    fn extract_important(value: &CssValue) -> (CssValue, bool) {
248        match value {
249            CssValue::List(list) if list.len() >= 2 => {
250                let len = list.len();
251                let is_important = matches!(
252                    (&list[len - 2], &list[len - 1]),
253                    (
254                        CssValue::Keyword(bang),
255                        CssValue::Keyword(ident)
256                    )
257                    if bang == "!" && ident.eq_ignore_ascii_case("important")
258                );
259
260                if is_important {
261                    let value = if len - 2 == 1 {
262                        list.iter().next().unwrap().clone()
263                    } else {
264                        CssValue::List(list[..len - 2].to_vec())
265                    };
266                    (value, true)
267                } else {
268                    (value.clone(), false)
269                }
270            }
271            _ => (value.clone(), false),
272        }
273    }
274
275    fn resolve_var(
276        value: &CssValue,
277        custom_props: &CustomProperties,
278        visited: &mut HashSet<CssIdent>,
279    ) -> Option<CssValue> {
280        match value {
281            // `var(--accent)` / `var(--missing, red)` — resolve the custom property
282            CssValue::Function(name, args) if name == "var" => {
283                let var_name = match args.first() {
284                    Some(CssValue::Keyword(name)) => name,
285                    _ => return None,
286                };
287
288                if !visited.insert(var_name.clone()) {
289                    return None;
290                }
291
292                let result = if let Some(v) = custom_props.get(var_name) {
293                    Self::resolve_var(v, custom_props, visited)
294                } else if let Some(fallback) = args.get(1) {
295                    Self::resolve_var(fallback, custom_props, visited)
296                } else {
297                    None
298                };
299
300                visited.remove(var_name);
301                result
302            }
303
304            // `rgb(var(--r), var(--g), var(--b))` — resolve args independently
305            CssValue::Function(name, args) => {
306                let resolved_args = args
307                    .iter()
308                    .map(|v| Self::resolve_var(v, custom_props, &mut visited.clone()))
309                    .collect::<Option<Vec<_>>>()?;
310                Some(CssValue::Function(name.clone(), resolved_args))
311            }
312
313            // `1px solid var(--color)` — resolve each item independently
314            CssValue::List(list) => {
315                let resolved = list
316                    .iter()
317                    .map(|v| Self::resolve_var(v, custom_props, &mut visited.clone()))
318                    .collect::<Option<Vec<_>>>()?;
319                Some(CssValue::List(resolved))
320            }
321
322            // `10px` / `"hello"` / `#fff` — already concrete, pass through
323            _ => Some(value.clone()),
324        }
325    }
326}
