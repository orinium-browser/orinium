6use std::collections::{HashMap, HashSet};

8type CustomProperties = HashMap<CssIdent, CssValue>;

10struct Declaration {
11    name: String,
12    value: CssValue,
13    important: bool,
14}

16/// Origin of a declaration in the CSS cascade.
17#[derive(Debug, Clone, Copy, PartialEq, Eq, PartialOrd, Ord)]
18pub enum StyleOrigin {
19    /// Browser-provided default styling.
20    UserAgent,
21    /// Styling supplied by the loaded document.
22    Author,
23}
24
25/// A single CSS declaration after selector resolution and value processing.
26///
27/// `ResolvedDeclaration` represents one property-value pair that has been
28/// fully associated with a selector and enriched with all information
29/// required for CSS cascade resolution.
30///
31/// This structure is produced after:
32/// - Parsing selectors
33/// - Resolving `var()` using custom properties
34/// - Computing selector specificity
35///
36/// During the cascade phase, multiple `ResolvedDeclaration`s with the same
37/// property name may compete. The winner is determined by comparing:
38///
39/// 1. `important` declarations
40/// 2. cascade `origin`
41/// 3. `specificity` (higher specificity wins)
42/// 4. `order` (later declarations win)
43#[derive(Debug, Clone)]
44pub struct ResolvedDeclaration {
45    pub selector: ComplexSelector,
46    pub name: String,
47    pub value: CssValue,
48    pub specificity: (u32, u32, u32),
49    pub order: usize,
50    pub important: bool,
51    pub origin: StyleOrigin,
52}
53
54pub type ResolvedStyles = Vec<ResolvedDeclaration>;
55
56impl ResolvedDeclaration {
57    /// Returns whether this declaration wins over another matching declaration.
58    pub fn outranks(&self, other: &Self) -> bool {
59        (self.important, self.origin, self.specificity, self.order)
60            > (
61                other.important,
62                other.origin,
63                other.specificity,
64                other.order,
65            )
66    }
67}
68
69/// Appends resolved declarations while preserving source order across stylesheets.
70pub fn append_resolved_styles(target: &mut ResolvedStyles, mut incoming: ResolvedStyles) {
71    let next_order = target
72        .iter()
73        .map(|declaration| declaration.order)
74        .max()
75        .map_or(0, |order| order + 1);
76    for declaration in &mut incoming {
77        declaration.order += next_order;
78    }
79    target.extend(incoming);
80}

82/// Resolves a `style` attribute's declarations into (name, value, important)
83/// triples, applying `var()` and `!important` handling like a rule block.
84///
85/// Inline styles participate in the cascade as author-origin declarations with
86/// the highest specificity, so callers should apply them after stylesheet
87/// declarations for the same element.
88pub fn resolve_inline_style(style_attr: &str) -> Vec<(String, CssValue, bool)> {
89    let mut parser = crate::engine::css::parser::Parser::new(style_attr);
90    let Ok(nodes) = parser.parse_declarations() else {
91        return Vec::new();
92    };
93
94    DeclarationResolver::collect(&nodes)
95        .into_iter()
96        .map(|declaration| (declaration.name, declaration.value, declaration.important))
97        .collect()
98}
99
100pub fn resolve_inline_value(value: &str) -> Option<CssValue> {
101    let mut tokenizer = crate::engine::css::tokenizer::Tokenizer::new(value);
102    let mut tokens = Vec::new();
103
104    loop {
105        let token = tokenizer.next_token();
106        if token == crate::engine::css::tokenizer::Token::EOF {
107            break;
108        }
109
110        tokens.push(token);
111    }
112
113    let Ok(value) = crate::engine::css::parser::Parser::parse_tokens_to_css_value(tokens) else {
114        return None;
115    };
116
117    Some(value)
118}

120// ============================================================
121//  CssResolver — tree walk + rule resolution
122// ============================================================
123
124pub struct CssResolver;
125
126impl CssResolver {
127    /// Resolves an author stylesheet into declarations used by layout.
128    pub fn resolve(stylesheet: &CssNode) -> ResolvedStyles {
129        Self::resolve_with_origin(stylesheet, StyleOrigin::Author)
130    }
131
132    /// Resolves a stylesheet using the supplied cascade origin.
133    pub fn resolve_with_origin(stylesheet: &CssNode, origin: StyleOrigin) -> ResolvedStyles {
134        let mut styles = Vec::new();
135        let mut order = 0;
136        Self::walk(stylesheet, &mut styles, &mut order, origin);
137        styles
138    }
139
140    /// Parses declarations from an HTML `style` attribute.
141    ///
142    /// Inline declarations use author origin but outrank selector-based author
143    /// rules of the same importance. Author `!important` declarations still
144    /// outrank normal inline declarations, as required by the cascade.
145    pub fn resolve_inline_style(style: &str) -> ResolvedStyles {
146        let source = format!("* {{ {style} }}");
147        let Ok(stylesheet) = crate::engine::css::parser::Parser::new(&source).parse() else {
148            return Vec::new();
149        };
150        let mut declarations = Self::resolve_with_origin(&stylesheet, StyleOrigin::Author);
151        for declaration in &mut declarations {
152            declaration.specificity = (u32::MAX, u32::MAX, u32::MAX);
153            declaration.order = usize::MAX;
154        }
155        declarations
156    }
157
158    fn walk(node: &CssNode, styles: &mut ResolvedStyles, order: &mut usize, origin: StyleOrigin) {
159        Self::resolve_rule(node, styles, order, origin);
160
161        if Self::should_recurse(node) {
162            for child in node.children() {
163                Self::walk(child, styles, order, origin);
164            }
165        }
166    }
167
168    fn resolve_rule(
169        node: &CssNode,
170        styles: &mut ResolvedStyles,
171        order: &mut usize,
172        origin: StyleOrigin,
173    ) {
174        let CssNodeType::Rule { selectors } = node.node() else {
175            return;
176        };
177
178        let declarations = DeclarationResolver::collect(node.children());
179
180        for selector in selectors {
181            Self::push_resolved(selector, &declarations, styles, order, origin);
182        }
183    }
184
185    fn push_resolved(
186        selector: &ComplexSelector,
187        declarations: &[Declaration],
188        styles: &mut ResolvedStyles,
189        order: &mut usize,
190        origin: StyleOrigin,
191    ) {
192        let specificity = selector.specificity();
193
194        for decl in declarations {
195            styles.push(ResolvedDeclaration {
196                selector: selector.clone(),
197                name: decl.name.clone(),
198                value: decl.value.clone(),
199                specificity,
200                order: *order,
201                important: decl.important,
202                origin,
203            });
204            *order += 1;
205        }
206    }
207
208    fn should_recurse(node: &CssNode) -> bool {
209        if let CssNodeType::AtRule { name, params } = node.node() {
210            Self::evaluate_at_rule(name, params)
211        } else {
212            true
213        }
214    }
215
216    fn evaluate_at_rule(name: &str, params: &AtQuery) -> bool {
217        if name.eq_ignore_ascii_case("supports") {
218            SupportsEvaluator::evaluate(params)
219        } else {
220            true
221        }
222    }
223}

286// ============================================================
287//  SupportsEvaluator — `@supports` condition evaluation
288// ============================================================
289
290struct SupportsEvaluator;
291
292impl SupportsEvaluator {
293    /// Dispatch on the `AtQuery` AST variant.
294    fn evaluate(query: &AtQuery) -> bool {
295        match query {
296            // `@supports (display: grid)` — a parenthesised group
297            AtQuery::Group(items) => Self::evaluate_group(items),
298            // `@supports (display: grid)` — the inner condition after unwrapping
299            AtQuery::Condition { name, value } => Self::is_supported(name, value),
300            // Stray keyword outside a group (malformed input)
301            AtQuery::Keyword(_) => false,
302        }
303    }
304
305    /// Evaluate a group of `@supports` items.
306    ///
307    /// The parser produces flat groups like:
308    /// - `(display: grid)` → `[Group([Condition])]`
309    /// - `(A) and (B)` → `[Group, Keyword("and"), Group]`
310    /// - `not (A)` → `[Keyword("not"), Group]`
311    /// - `(A) or (B)` → `[Group, Keyword("or"), Group]`
312    fn evaluate_group(items: &[AtQuery]) -> bool {
313        if items.is_empty() {
314            return false;
315        }
316
317        // `not (display: grid)` — negate
318        if matches!(items.first(), Some(AtQuery::Keyword(k)) if k.eq_ignore_ascii_case("not")) {
319            return items.len() > 1 && !Self::evaluate(&AtQuery::Group(items[1..].to_vec()));
320        }
321
322        // `(display: flex) and (gap: 10px)` — all operands must be supported
323        if let Some(operands) = Self::split_by_keyword(items, "and") {
324            operands.iter().all(|g| Self::evaluate(g))
325        // `(display: flex) or (display: grid)` — at least one must be supported
326        } else if let Some(operands) = Self::split_by_keyword(items, "or") {
327            operands.iter().any(|g| Self::evaluate(g))
328        // Single group — unwrap one level
329        } else if items.len() == 1 {
330            Self::evaluate(&items[0])
331        } else {
332            items.iter().all(Self::evaluate)
333        }
334    }
335
336    fn is_supported(name: &str, value: &CssValue) -> bool {
337        super::builder::apply_declaration(
338            name,
339            value,
340            &mut ui_layout::Style::default(),
341            &mut super::types::ContainerStyle::default(),
342            &mut super::types::TextStyle::default(),
343            &mut super::types::Overflow::default(),
344            super::types::ColorScheme::Light,
345        )
346        .is_some()
347    }
348
349    fn split_by_keyword<'a>(items: &'a [AtQuery], keyword: &str) -> Option<Vec<&'a AtQuery>> {
350        let has = items
351            .iter()
352            .any(|item| matches!(item, AtQuery::Keyword(k) if k.eq_ignore_ascii_case(keyword)));
353        if !has {
354            return None;
355        }
356        Some(
357            items
358                .iter()
359                .filter(
360                    |item| !matches!(item, AtQuery::Keyword(k) if k.eq_ignore_ascii_case(keyword)),
361                )
362                .collect(),
363        )
364    }
365}

367// ============================================================
368//  DeclarationResolver — `!important` extraction, `var()` resolution
369// ============================================================
370
371struct DeclarationResolver;
372
373impl DeclarationResolver {
374    fn collect(children: &[CssNode]) -> Vec<Declaration> {
375        let mut custom_props: CustomProperties = HashMap::new();
376        let mut result = Vec::new();
377
378        for child in children {
379            if let CssNodeType::Declaration { name, value } = &child.node()
380                && name.starts_with("--")
381            {
382                custom_props.insert(name.into(), value.clone());
383            }
384        }
385
386        for child in children {
387            if let CssNodeType::Declaration { name, value } = &child.node() {
388                let (raw_value, important) = Self::extract_important(value);
389
390                if name.starts_with("--") {
391                    // `--accent: blue` — custom property, keep as-is
392                    result.push(Declaration {
393                        name: name.clone(),
394                        value: raw_value,
395                        important,
396                    });
397                } else if let Some(resolved) =
398                    Self::resolve_var(&raw_value, &custom_props, &mut HashSet::new())
399                {
400                    // `color: var(--accent)` — resolve var() then emit
401                    result.push(Declaration {
402                        name: name.clone(),
403                        value: resolved,
404                        important,
405                    });
406                }
407            }
408        }
409
410        result
411    }
412
413    /// Extract `!important` from a CSS value.
414    ///
415    /// `border: 1px solid black !important` is parsed as a `List` where the
416    /// last two items are `Keyword("!")` and `Keyword("important")`.
417    fn extract_important(value: &CssValue) -> (CssValue, bool) {
418        match value {
419            CssValue::List(list) if list.len() >= 2 => {
420                let len = list.len();
421                let is_important = matches!(
422                    (&list[len - 2], &list[len - 1]),
423                    (
424                        CssValue::Keyword(bang),
425                        CssValue::Keyword(ident)
426                    )
427                    if bang == "!" && ident.eq_ignore_ascii_case("important")
428                );
429
430                if is_important {
431                    let value = if len - 2 == 1 {
432                        list.iter().next().unwrap().clone()
433                    } else {
434                        CssValue::List(list[..len - 2].to_vec())
435                    };
436                    (value, true)
437                } else {
438                    (value.clone(), false)
439                }
440            }
441            _ => (value.clone(), false),
442        }
443    }
444
445    fn resolve_var(
446        value: &CssValue,
447        custom_props: &CustomProperties,
448        visited: &mut HashSet<CssIdent>,
449    ) -> Option<CssValue> {
450        match value {
451            // `var(--accent)` / `var(--missing, red)` — resolve the custom property
452            CssValue::Function(name, args) if name == "var" => {
453                let var_name = match args.first() {
454                    Some(CssValue::Keyword(name)) => name,
455                    _ => return None,
456                };
457
458                if !visited.insert(var_name.clone()) {
459                    return None;
460                }
461
462                let result = if let Some(v) = custom_props.get(var_name) {
463                    Self::resolve_var(v, custom_props, visited)
464                } else if let Some(fallback) = args.get(1) {
465                    Self::resolve_var(fallback, custom_props, visited)
466                } else {
467                    None
468                };
469
470                visited.remove(var_name);
471                result
472            }
473
474            // `rgb(var(--r), var(--g), var(--b))` — resolve args independently
475            CssValue::Function(name, args) => {
476                let resolved_args = args
477                    .iter()
478                    .map(|v| Self::resolve_var(v, custom_props, &mut visited.clone()))
479                    .collect::<Option<Vec<_>>>()?;
480                Some(CssValue::Function(name.clone(), resolved_args))
481            }
482
483            // `1px solid var(--color)` — resolve each item independently
484            CssValue::List(list) => {
485                let resolved = list
486                    .iter()
487                    .map(|v| Self::resolve_var(v, custom_props, &mut visited.clone()))
488                    .collect::<Option<Vec<_>>>()?;
489                Some(CssValue::List(resolved))
490            }
491
492            // `10px` / `"hello"` / `#fff` — already concrete, pass through
493            _ => Some(value.clone()),
494        }
495    }
