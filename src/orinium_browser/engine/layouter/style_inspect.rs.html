13/// One property declaration as shown in the Styles panel.
14#[derive(Debug, Clone)]
15pub struct InspectedDeclaration {
16    pub name: String,
17    pub value: CssValue,
18    pub important: bool,
19    /// Whether this declaration is the cascade winner for its property.
20    pub applied: bool,
21}
22
23/// A rule (or the inline `style` attribute) whose declarations were cascaded.
24#[derive(Debug, Clone)]
25pub struct MatchedRule {
26    /// Rendered selector text, or `"element.style"` for inline styles.
27    pub selector: String,
28    pub origin: StyleOrigin,
29    /// Whether this rule is the synthetic inline-style entry.
30    pub inline: bool,
31    pub declarations: Vec<InspectedDeclaration>,
32}
33
34/// Total ordering mirroring the builder's cascade semantics: `!important`
35/// first, then origin (`Author` ranks above `UserAgent`), then inline styles
36/// over non-important stylesheet rules, then specificity and source order.
37///
38/// Inline entries use source orders starting at [`INLINE_ORDER_BASE`] so that
39/// repeated inline declarations resolve last-wins while still ranking above
40/// every normal stylesheet declaration.
41#[derive(Clone, PartialEq, Eq, PartialOrd, Ord)]
42struct CascadeKey {
43    important: bool,
44    origin: StyleOrigin,
45    inline: bool,
46    specificity: (u32, u32, u32),
47    order: usize,
48}
49
50const INLINE_ORDER_BASE: usize = usize::MAX / 2;
51
52impl CascadeKey {
53    fn from_declaration(declaration: &ResolvedDeclaration) -> Self {
54        Self {
55            important: declaration.important,
56            origin: declaration.origin,
57            inline: false,
58            specificity: declaration.specificity,
59            order: declaration.order,
60        }
61    }
62}
63
64struct Pending {
65    name: String,
66    key: CascadeKey,
67    rule: usize,
68    declaration: usize,
69}
70
71/// Collects every rule matching `target` and flags winning declarations.
72///
73/// `inline_style_attr` is the target's raw `style` attribute value, taken
74/// from the same snapshot the layout was built from; pass `None` when the
75/// element has no attribute. `target` must be an element node.
76pub fn collect_matched_rules(
77    snapshot: &DomSnapshot,
78    target: NodeId,
79    rule_set: &RuleSet<'_>,
80    inline_style_attr: Option<&str>,
81) -> Vec<MatchedRule> {
82    let Some(chain) = chain_for_node(snapshot, target) else {
83        return Vec::new();
84    };
85    let Some(element) = chain.first() else {
86        return Vec::new();
87    };
88
89    // The synthetic inline entry leads the output like browser DevTools do.
90    let mut rules: Vec<MatchedRule> = Vec::new();
91    let mut pendings: Vec<Pending> = Vec::new();
92
93    if let Some(style_attr) = inline_style_attr {
94        let inline_declarations = resolve_inline_style(style_attr);
95        if !inline_declarations.is_empty() {
96            let mut declarations = Vec::with_capacity(inline_declarations.len());
97            for (offset, (name, value, important)) in inline_declarations.into_iter().enumerate() {
98                pendings.push(Pending {
99                    name: name.clone(),
100                    key: CascadeKey {
101                        important,
102                        origin: StyleOrigin::Author,
103                        inline: true,
104                        specificity: (u32::MAX, 0, 0),
105                        order: INLINE_ORDER_BASE + offset,
106                    },
107                    rule: 0,
108                    declaration: declarations.len(),
109                });
110                declarations.push(InspectedDeclaration {
111                    name,
112                    value,
113                    important,
114                    applied: false,
115                });
116            }
117            rules.push(MatchedRule {
118                selector: "element.style".to_string(),
119                origin: StyleOrigin::Author,
120                inline: true,
121                declarations,
122            });
123        }
124    }
125
126    let mut rule_index: HashMap<(String, StyleOrigin), usize> = HashMap::new();
127    for declaration in rule_set.query_candidates(element) {
128        if !declaration.selector.matches(&chain) {
129            continue;
130        }
131        let selector = declaration.selector.to_string();
132        let key = (selector, declaration.origin);
133        let rule = *rule_index.entry(key).or_insert_with(|| {
134            rules.push(MatchedRule {
135                selector: declaration.selector.to_string(),
136                origin: declaration.origin,
137                inline: false,
138                declarations: Vec::new(),
139            });
140            rules.len() - 1
141        });
142        pendings.push(Pending {
143            name: declaration.name.clone(),
144            key: CascadeKey::from_declaration(declaration),
145            rule,
146            declaration: rules[rule].declarations.len(),
147        });
148        rules[rule].declarations.push(InspectedDeclaration {
149            name: declaration.name.clone(),
150            value: declaration.value.clone(),
151            important: declaration.important,
152            applied: false,
153        });
154    }
155
156    mark_cascade_winners(&mut rules, &pendings);
157    rules
158}
159
160/// Flags the highest-ranking declaration per property name as applied.
161fn mark_cascade_winners(rules: &mut [MatchedRule], pendings: &[Pending]) {
162    let mut winners: HashMap<&str, (CascadeKey, usize)> = HashMap::new();
163    for (index, pending) in pendings.iter().enumerate() {
164        match winners.get(pending.name.as_str()) {
165            Some((existing_key, _)) if *existing_key >= pending.key => {}
166            _ => {
167                winners.insert(&pending.name, (pending.key.clone(), index));
168            }
169        }
170    }
171    for (_, (_, index)) in winners {
172        let pending = &pendings[index];
173        rules[pending.rule].declarations[pending.declaration].applied = true;
174    }
175}

177/// Rebuilds the [`ElementChain`] the builder would have walked with when
178/// styling `target`: ancestor elements outermost-first, each carrying the
179/// sibling indexes its structural pseudo-classes rely on.
180fn chain_for_node(snapshot: &DomSnapshot, target: NodeId) -> Option<ElementChain> {
181    let root = *snapshot.roots().first()?;
182    let mut path = Vec::new();
183    find_path(snapshot, root, target, &mut path)?;
184
185    let mut elements = Vec::with_capacity(path.len());
186    for (depth, &node_id) in path.iter().enumerate() {
187        let info = if depth == 0 {
188            element_info(&snapshot.node(node_id).kind)
189        } else {
190            let siblings = snapshot.children(path[depth - 1]);
191            let position = siblings.iter().position(|&sibling| sibling == node_id)?;
192            element_sibling_infos(snapshot, siblings)[position].clone()
193        };
194        elements.push(info);
195    }
196
197    // `from_vec` expects innermost-first; non-element ancestors (document,
198    // text) simply do not contribute chain links, matching the builder.
199    elements.reverse();
200    Some(ElementChain::from_vec(
201        elements.into_iter().flatten().collect(),
202    ))
203}
204
205fn find_path(
206    snapshot: &DomSnapshot,
207    current: NodeId,
208    target: NodeId,
209    path: &mut Vec<NodeId>,
210) -> Option<()> {
211    path.push(current);
212    if current == target {
213        return Some(());
214    }
215    for &child in snapshot.children(current) {
216        if find_path(snapshot, child, target, path).is_some() {
217            return Some(());
218        }
219    }
220    path.pop();
221    None
222}

232    /// Leaks a tiny fixture so `RuleSet` can borrow it for `'static`; keeps
233    /// test plumbing free of self-referential lifetimes.
234    fn leak(resolved: ResolvedStyles) -> &'static ResolvedStyles {
235        Box::leak(Box::new(resolved))
236    }
237
238    fn resolved_for(css: &str) -> ResolvedStyles {
239        CssResolver::resolve(&CssParser::new(css).parse().unwrap())
240    }
241
242    fn resolved_for_origin(css: &str, origin: StyleOrigin) -> ResolvedStyles {
243        CssResolver::resolve_with_origin(&CssParser::new(css).parse().unwrap(), origin)
244    }
245
246    fn media() -> MediaEnvironment {
247        MediaEnvironment::new((800.0, 600.0), ColorScheme::Light)
248    }
249
250    fn snapshot_for(html: &str) -> DomSnapshot {
251        let dom = HtmlParser::new(html).parse();
252        DomSnapshot::from_tree(&dom.root).0
253    }
254
255    fn node_with_tag(snapshot: &DomSnapshot, tag: &str) -> NodeId {
256        nodes_with_tag(snapshot, tag)
257            .into_iter()
258            .next()
259            .unwrap_or_else(|| panic!("no <{tag}> in fixture"))
260    }
261
262    /// Pre-order positions of every element with `tag`; text nodes sit
263    /// between element ids, so callers must not assume consecutive ids.
264    fn nodes_with_tag(snapshot: &DomSnapshot, tag: &str) -> Vec<NodeId> {
265        snapshot
266            .nodes()
267            .iter()
268            .enumerate()
269            .filter(|(_, node)| node.kind.tag_name() == Some(tag))
270            .map(|(index, _)| index as NodeId)
271            .collect()
272    }
273
274    fn declaration<'a>(rule: &'a MatchedRule, name: &str) -> &'a InspectedDeclaration {
275        rule.declarations
276            .iter()
277            .find(|declaration| declaration.name == name)
278            .unwrap_or_else(|| panic!("no {name} declared by {}", rule.selector))
279    }

281    #[test]
282    fn later_specific_rule_overrides_and_flags_the_loser() {
283        let snapshot = snapshot_for("<body><p class=\"box\">t</p></body>");
284        let styles = leak(resolved_for("p { color: red; } .box { color: blue; }"));
285        let rules = RuleSet::from_declarations(styles, &media());
286        let target = node_with_tag(&snapshot, "p");
287
288        let matched = collect_matched_rules(&snapshot, target, &rules, None);
289
290        let tag_rule = matched.iter().find(|rule| rule.selector == "p").unwrap();
291        let class_rule = matched.iter().find(|rule| rule.selector == ".box").unwrap();
292        assert!(!declaration(tag_rule, "color").applied);
293        assert!(declaration(class_rule, "color").applied);
294    }

296    #[test]
297    fn chain_reconstruction_carries_sibling_indexes() {
298        let snapshot = snapshot_for("<ul><li>a</li><li>b</li><span>s</span></ul>");
299
300        fn indexes(snapshot: &DomSnapshot, id: NodeId) -> (usize, usize) {
301            let chain = chain_for_node(snapshot, id).expect("chain");
302            let element = chain.first().unwrap();
303            (element.element_index, element.element_count)
304        }
305
306        let lis = nodes_with_tag(&snapshot, "li");
307        assert_eq!(lis.len(), 2);
308        assert_eq!(indexes(&snapshot, lis[0]), (1, 3));
309        assert_eq!(indexes(&snapshot, lis[1]), (2, 3));
310
311        // The chain reaches ancestors: first() is the node itself.
312        let chain = chain_for_node(&snapshot, lis[1]).unwrap();
313        assert_eq!(chain.first().unwrap().tag_name, "li");
314    }

346    #[test]
347    fn inline_styles_beat_author_rules_but_not_important_ones() {
348        let snapshot = snapshot_for("<body><p style=\"color: black\">t</p></body>");
349        let target = node_with_tag(&snapshot, "p");
350
351        let styles = leak(resolved_for("p { color: red; }"));
352        let rules = RuleSet::from_declarations(styles, &media());
353        let matched = collect_matched_rules(&snapshot, target, &rules, Some("color: black"));
354        let inline = matched.first().expect("inline entry leads");
355        assert!(inline.inline && inline.origin == StyleOrigin::Author);
356        assert_eq!(inline.selector, "element.style");
357        assert!(declaration(inline, "color").applied);
358
359        let styles = leak(resolved_for("p { color: red !important; }"));
360        let rules = RuleSet::from_declarations(styles, &media());
361        let matched = collect_matched_rules(&snapshot, target, &rules, Some("color: black"));
362        let inline = matched.first().unwrap();
363        let stylesheet = matched.iter().find(|rule| rule.selector == "p").unwrap();
364        assert!(!declaration(inline, "color").applied);
365        assert!(declaration(stylesheet, "color").applied);
366    }

369    fn user_agent_origin_ranks_below_author_origin() {
370        let snapshot = snapshot_for("<body><p>t</p></body>");
371        let mut combined = resolved_for_origin("p { margin-top: 0px; }", StyleOrigin::UserAgent);
372        combined.extend(resolved_for_origin(
373            "p { margin-top: 4px; }",
374            StyleOrigin::Author,
375        ));
376        let styles = leak(combined);
377        let rules = RuleSet::from_declarations(styles, &media());
378
379        let matched = collect_matched_rules(&snapshot, node_with_tag(&snapshot, "p"), &rules, None);
380        let ua = matched
381            .iter()
382            .find(|rule| rule.origin == StyleOrigin::UserAgent)
383            .expect("user-agent rule reported");
384        let author_rule = matched
385            .iter()
386            .find(|rule| rule.origin == StyleOrigin::Author)
387            .expect("author rule reported");
388        assert!(!declaration(ua, "margin-top").applied);
389        assert!(declaration(author_rule, "margin-top").applied);
