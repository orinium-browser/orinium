1//! SVG support shared by the inline `<svg>` painter and the standalone
2//! rasterizer.
3//!
4//! # Inline SVG
5//!
6//! The layout builder calls [`collect_svg`] while building the node for an
7//! `<svg>` element. The renderer then paints the collected shapes into the
8//! element's content box via [`emit_commands`], mapping the `viewBox`
9//! rectangle onto the box with [`viewbox_meet_transform`] (`xMidYMid meet`, the
10//! SVG default).
11//!
12//! # Standalone SVG documents
13//!
14//! Audio icons and `<img src="*.svg">` are rasterized to RGBA pixels by
15//! [`rasterize_from_bytes`], which replaces the removed `resvg` dependency.
16//!
17//! Coverage stays focused on solid-color fills: `<path d>` (including arc
18//! segments, expanded into cubic Béziers), `<rect>`, `<circle>`, `<ellipse>`,
19//! container `<g>`/`<a>`/`<symbol>` with composed `transform`/`opacity`
20//! attributes, `fill-rule`, and `opacity`/`fill-opacity` (approximated as a
21//! per-shape alpha multiplier). `<use>`/`<clipPath>`/gradients, strokes, and
22//! text are out of scope.

124/// Translates an SVG path `d` string into a [`Path`].
125///
126/// Supports the moveto/lineto/curveto/closepath commands in both absolute and
127/// relative forms: `M/m`, `L/l`, `H/h`, `V/v`, `C/c`, `S/s`, `Q/q`, `T/t`,
128/// `Z/z`. Arc segments (`A/a`) are approximated with cubic Bézier segments via
129/// the SVG endpoint→center parametrization (radii are scaled up when they are
130/// too small for the chord, and arcs larger than a quarter turn are split).
131/// Repeated coordinate sets after a command repeat that command (per the SVG
132/// grammar `M m x y ...` implicit lineto included). Malformed data is skipped
133/// defensively.
134pub fn path_from_d(d: &str) -> Path {
135    let mut path = Path::new();
136    let mut cur: Option<(f32, f32)> = None;
137    let mut start: Option<(f32, f32)> = None;
138    // Reflection anchor of the previous curve command (C/S/Q/T), used by the
139    // smooth variants. `None` when the previous command was not a curve.
140    let mut prev_control: Option<(f32, f32)> = None;
141    let mut prev_was_curve = false;
142
143    let mut nums = SvgNum::new(d);
144
145    while let Some(cmd) = nums.next_command() {
146        match cmd {
147            'M' | 'm' => {
148                let rel = cmd == 'm';
149                if let Some((x, y)) = nums.next_point(rel, cur) {
150                    path.move_to(x, y);
151                    cur = Some((x, y));
152                    start = Some((x, y));
153                    // Extra coordinate pairs are implicit relative lineto.
154                    while let Some((x, y)) = nums.next_point(rel, cur) {
155                        path.line_to(x, y);
156                        cur = Some((x, y));
157                    }
158                    prev_was_curve = false;
159                }
160            }
161            'L' | 'l' => {
162                let rel = cmd == 'l';
163                while let Some((x, y)) = nums.next_point(rel, cur) {
164                    path.line_to(x, y);
165                    cur = Some((x, y));
166                }
167                prev_was_curve = false;
168            }
169            'H' | 'h' => {
170                let rel = cmd == 'h';
171                while let Some(x) = nums.next_number() {
172                    let nx = if rel {
173                        cur.unwrap_or((0.0, 0.0)).0 + x
174                    } else {
175                        x
176                    };
177                    path.line_to(nx, cur.map_or(0.0, |p| p.1));
178                    cur = Some((nx, cur.map_or(0.0, |p| p.1)));
179                }
180                prev_was_curve = false;
181            }
182            'V' | 'v' => {
183                let rel = cmd == 'v';
184                while let Some(y) = nums.next_number() {
185                    let ny = if rel {
186                        cur.unwrap_or((0.0, 0.0)).1 + y
187                    } else {
188                        y
189                    };
190                    path.line_to(cur.map_or(0.0, |p| p.0), ny);
191                    cur = Some((cur.map_or(0.0, |p| p.0), ny));
192                }
193                prev_was_curve = false;
194            }
195            'C' | 'c' => {
196                let rel = cmd == 'c';
197                let rel_ref = cur.unwrap_or((0.0, 0.0));
198                while let (Some(c1), Some(c2), Some(p)) = (
199                    nums.next_point(rel, Some(rel_ref)),
200                    nums.next_point(rel, Some(rel_ref)),
201                    nums.next_point(rel, cur),
202                ) {
203                    path.cubic_to(c1, c2, p);
204                    cur = Some(p);
205                    prev_control = Some(c2);
206                    prev_was_curve = true;
207                }
208            }
209            'S' | 's' => {
210                let rel = cmd == 's';
211                let rel_ref = cur.unwrap_or((0.0, 0.0));
212                while let (Some(c2), Some(p)) = (
213                    nums.next_point(rel, Some(rel_ref)),
214                    nums.next_point(rel, cur),
215                ) {
216                    let c1 = if prev_was_curve {
217                        let c2_ref = prev_control.unwrap_or(rel_ref);
218                        (2.0 * rel_ref.0 - c2_ref.0, 2.0 * rel_ref.1 - c2_ref.1)
219                    } else {
220                        rel_ref
221                    };
222                    path.cubic_to(c1, c2, p);
223                    cur = Some(p);
224                    prev_control = Some(c2);
225                    prev_was_curve = true;
226                }
227            }
228            'Q' | 'q' => {
229                let rel = cmd == 'q';
230                let rel_ref = cur.unwrap_or((0.0, 0.0));
231                while let (Some(c), Some(p)) = (
232                    nums.next_point(rel, Some(rel_ref)),
233                    nums.next_point(rel, cur),
234                ) {
235                    path.quad_to(c, p);
236                    cur = Some(p);
237                    prev_control = Some(c);
238                    prev_was_curve = true;
239                }
240            }
241            'T' | 't' => {
242                let rel = cmd == 't';
243                let rel_ref = cur.unwrap_or((0.0, 0.0));
244                while let Some(p) = nums.next_point(rel, cur) {
245                    let c = if prev_was_curve {
246                        let c_ref = prev_control.unwrap_or(rel_ref);
247                        (2.0 * rel_ref.0 - c_ref.0, 2.0 * rel_ref.1 - c_ref.1)
248                    } else {
249                        rel_ref
250                    };
251                    path.quad_to(c, p);
252                    cur = Some(p);
253                    prev_control = Some(c);
254                    prev_was_curve = true;
255                }
256            }
257            'A' | 'a' => {
258                let rel = cmd == 'a';
259                while let (Some(rx), Some(ry), Some(rot), Some(large), Some(sweep), Some(p)) = (
260                    nums.next_number(),
261                    nums.next_number(),
262                    nums.next_number(),
263                    nums.next_number(),
264                    nums.next_number(),
265                    nums.next_point(rel, cur),
266                ) {
267                    match cur {
268                        Some(start) => append_arc(
269                            &mut path,
270                            start,
271                            rx.abs(),
272                            ry.abs(),
273                            rot,
274                            large != 0.0,
275                            sweep != 0.0,
276                            p,
277                        ),
278                        // An arc with no current point is a fresh subpath.
279                        None => path.move_to(p.0, p.1),
280                    }
281                    cur = Some(p);
282                }
283                prev_was_curve = false;
284            }
285            'Z' | 'z' => {
286                path.close();
287                cur = start;
288                prev_was_curve = false;
289            }
290            _ => break,
291        }
292    }
293
294    path
295}

297/// Approximates an SVG elliptical arc from `from` to `to` with cubic Bézier
298/// segments using the SVG 1.1 endpoint→center parametrization (F.6).
299///
300/// Radii are scaled up when they are too small for the chord, and the sweep is
301/// split so no single cubic spans more than a quarter turn, keeping the
302/// 4/3·tan(Δ/4) control-point approximation accurate.
303#[allow(clippy::too_many_arguments)]
304fn append_arc(
305    path: &mut Path,
306    from: (f32, f32),
307    rx: f32,
308    ry: f32,
309    phi_deg: f32,
310    large_arc: bool,
311    sweep: bool,
312    to: (f32, f32),
313) {
314    let (x1, y1) = from;
315    let (x2, y2) = to;
316    if (rx == 0.0 || ry == 0.0)
317        || ((x1 - x2).abs() < f32::EPSILON && (y1 - y2).abs() < f32::EPSILON)
318    {
319        path.line_to(x2, y2);
320        return;
321    }
322    let phi = phi_deg.to_radians();
323    let (cos_phi, sin_phi) = (phi.cos(), phi.sin());
324
325    // Endpoints in the rotated (axis-aligned ellipse) frame F.6.5.1.
326    let (dx2, dy2) = ((x1 - x2) * 0.5, (y1 - y2) * 0.5);
327    let x1p = cos_phi * dx2 + sin_phi * dy2;
328    let y1p = -sin_phi * dx2 + cos_phi * dy2;
329
330    // Scale the radii up when they can't span the chord F.6.6.
331    let (mut rx, mut ry) = (rx, ry);
332    let lambda = (x1p / rx).powi(2) + (y1p / ry).powi(2);
333    if lambda > 1.0 {
334        let scale = lambda.sqrt();
335        rx *= scale;
336        ry *= scale;
337    }
338
339    // Center in the rotated frame F.6.5.2.
340    let rx2 = rx * rx;
341    let ry2 = ry * ry;
342    let numerator = rx2 * ry2 - rx2 * y1p * y1p - ry2 * x1p * x1p;
343    let denominator = rx2 * y1p * y1p + ry2 * x1p * x1p;
344    let coef = if denominator <= 0.0 {
345        0.0
346    } else {
347        (numerator / denominator).max(0.0).sqrt()
348    };
349    let sign = if large_arc == sweep { -1.0 } else { 1.0 };
350    let (cxp, cyp) = (
351        sign * coef * (rx * y1p / ry),
352        sign * coef * (-ry * x1p / rx),
353    );
354
355    // World-space center F.6.5.3 and parameter angles F.6.5.5-6. The endpoint
356    // is at ellipse-frame angle θ where (x1p, y1p) = (cxp, cyp) + (rx cosθ, ry
357    // sinθ), so the angles are those of the radii-scaled vectors.
358    let cx = cos_phi * cxp - sin_phi * cyp + (x1 + x2) * 0.5;
359    let cy = sin_phi * cxp + cos_phi * cyp + (y1 + y2) * 0.5;
360    let theta1 = ((y1p - cyp) / ry).atan2((x1p - cxp) / rx);
361    let mut delta = ((-y1p - cyp) / ry).atan2((-x1p - cxp) / rx) - theta1;
362    if sweep && delta < 0.0 {
363        delta += std::f32::consts::TAU;
364    } else if !sweep && delta > 0.0 {
365        delta -= std::f32::consts::TAU;
366    }
367
368    // Split the sweep into ≤ quarter-turn cubics.
369    let segments = (delta.abs() / std::f32::consts::FRAC_PI_2).ceil().max(1.0) as usize;
370    for i in 0..segments {
371        let a = theta1 + delta * (i as f32 / segments as f32);
372        let b = theta1 + delta * ((i + 1) as f32 / segments as f32);
373        let k = (4.0 / 3.0) * ((b - a) * 0.25).tan();
374        let (sin_a, cos_a) = (a.sin(), a.cos());
375        let (sin_b, cos_b) = (b.sin(), b.cos());
376        let world = |(px, py): (f32, f32)| {
377            (
378                cos_phi * px - sin_phi * py + cx,
379                sin_phi * px + cos_phi * py + cy,
380            )
381        };
382        // The last segment ends exactly at `to` to avoid a gap from floating
383        // point drift in the trigonometric end-point computation.
384        let (p3x, p3y) = if i + 1 == segments {
385            to
386        } else {
387            world((rx * cos_b, ry * sin_b))
388        };
389        let (p1x, p1y) = world((rx * (cos_a - k * sin_a), ry * (sin_a + k * cos_a)));
390        let (p2x, p2y) = world((rx * (cos_b + k * sin_b), ry * (sin_b - k * cos_b)));
391        path.cubic_to((p1x, p1y), (p2x, p2y), (p3x, p3y));
392    }
393}

420/// Parses an SVG length, tolerating trailing unit suffixes (`px`, `%`, …) and
421/// scientific-notation exponents (`1.5e2`).
422pub(crate) fn parse_len(value: &str) -> Option<f32> {
423    let value = value.trim().trim_start_matches('+');
424    let bytes = value.as_bytes();
425    let mut i = 0;
426    if i < bytes.len() && bytes[i] == b'-' {
427        i += 1;
428    }
429    let digits_start = i;
430    while i < bytes.len() && bytes[i].is_ascii_digit() {
431        i += 1;
432    }
433    if i < bytes.len() && bytes[i] == b'.' {
434        i += 1;
435        while i < bytes.len() && bytes[i].is_ascii_digit() {
436            i += 1;
437        }
438    }
439    if i == digits_start {
440        return None;
441    }
442    if i < bytes.len() && (bytes[i] == b'e' || bytes[i] == b'E') {
443        let mut j = i + 1;
444        if j < bytes.len() && (bytes[j] == b'+' || bytes[j] == b'-') {
445            j += 1;
446        }
447        if j < bytes.len() && bytes[j].is_ascii_digit() {
448            while j < bytes.len() && bytes[j].is_ascii_digit() {
449                j += 1;
450            }
451            i = j;
452        }
453    }
454    value[..i].parse::<f32>().ok()
455}

457/// Parses an SVG `transform` attribute into a single affine transform.
458///
459/// Supported functions: `matrix`, `translate`, `scale`, `rotate`, `skewX`,
460/// `skewY`. Multiple functions compose left-to-right per the SVG spec.
461pub(crate) fn parse_transform(value: &str) -> Result<AffineTransform, String> {
462    let mut out = AffineTransform::identity();
463    let mut rest = value;
464    loop {
465        let trimmed = rest.trim_start();
466        if trimmed.is_empty() {
467            return Ok(out);
468        }
469        let open = trimmed
470            .find('(')
471            .ok_or_else(|| format!("transform: missing '(' in {value:?}"))?;
472        let function = trimmed[..open].trim();
473        let inner = &trimmed[open + 1..];
474        let close = inner
475            .find(')')
476            .ok_or_else(|| format!("transform: missing ')' after {function:?} in {value:?}"))?;
477        let args = &inner[..close];
478        let next = parse_function(function, args).ok_or_else(|| {
479            format!("transform: unsupported or malformed {function:?} in {value:?}")
480        })?;
481        out = out.then(&next);
482        rest = &inner[close + 1..];
483    }
484}

486fn parse_function(name: &str, args: &str) -> Option<AffineTransform> {
487    let nums = args
488        .split(|c: char| c == ',' || c.is_whitespace())
489        .filter(|token| !token.trim().is_empty())
490        .filter_map(|token| token.trim().parse::<f32>().ok())
491        .collect::<Vec<f32>>();
492    match name {
493        "matrix" if nums.len() >= 6 => Some(AffineTransform {
494            m11: nums[0],
495            m12: nums[2],
496            m21: nums[1],
497            m22: nums[3],
498            dx: nums[4],
499            dy: nums[5],
500        }),
501        "translate" if !nums.is_empty() => Some(AffineTransform::translate(
502            nums[0],
503            nums.get(1).copied().unwrap_or(0.0),
504        )),
505        "scale" if !nums.is_empty() => Some(AffineTransform::scale(
506            nums[0],
507            nums.get(1).copied().unwrap_or(nums[0]),
508        )),
509        "rotate" if !nums.is_empty() => {
510            let angle = nums[0].to_radians();
511            if nums.len() >= 3 {
512                let (cx, cy) = (nums[1], nums[2]);
513                Some(
514                    AffineTransform::identity()
515                        .then(&AffineTransform::translate(-cx, -cy))
516                        .then(&AffineTransform::rotate(angle))
517                        .then(&AffineTransform::translate(cx, cy)),
518                )
519            } else {
520                Some(AffineTransform::rotate(angle))
521            }
522        }
523        "skewX" if nums.len() == 1 => Some(AffineTransform {
524            m12: nums[0].to_radians().tan(),
525            ..AffineTransform::identity()
526        }),
527        "skewY" if nums.len() == 1 => Some(AffineTransform {
528            m21: nums[0].to_radians().tan(),
529            ..AffineTransform::identity()
530        }),
531        _ => None,
532    }
533}

535/// Resolves an SVG presentation value such as `fill="#006AFF"` to a color,
536/// without round-tripping through a full CSS declaration block.
537///
538/// `none`/`transparent`/`initial`/`unset` resolve to a transparent color;
539/// `currentColor` resolves to the element's computed text color (`inherit`
540/// falls back to the SVG default, black, because the tree walker already
541/// propagates inherited fills); every other value goes through the standard
542/// CSS color pipeline. Unknown values return `None` so callers can apply the
543/// SVG default (black).
544pub(crate) fn parse_svg_fill(
545    value: &str,
546    color_scheme: ColorScheme,
547    current_color: Color,
548) -> Option<Color> {
549    let css = resolve_inline_value(value.trim())?;
550    match &css {
551        CssValue::Keyword(kw) if is_transparent_keyword(kw) => Some(Color(0, 0, 0, 0)),
552        CssValue::Keyword(kw) if kw.eq_ignore_ascii_case("currentColor") => Some(current_color),
553        CssValue::Keyword(kw) if kw.eq_ignore_ascii_case("inherit") => None,
554        // `url(...)` names a gradient paint server, never a solid color. Falling
555        // through to the CSS pipeline would log a spurious error and return
556        // `None` anyway, so short-circuit to keep the log clean.
557        CssValue::Function(name, _) if name.eq_ignore_ascii_case("url") => None,
558        _ => resolve_css_color("fill", &css, color_scheme),
559    }
560}

562/// Extracts the fragment `id` from a `url(#id)` paint-server reference like
563/// `fill="url(#blinearGradient971)"`. Returns `None` for any other value.
564pub(crate) fn parse_svg_url(value: &str) -> Option<String> {
565    let css = resolve_inline_value(value.trim())?;
566    let CssValue::Function(name, args) = &css else {
567        return None;
568    };
569    if !name.eq_ignore_ascii_case("url") {
570        return None;
571    }
572    // The fragment is either an unquoted identifier or a quoted string.
573    let fragment = args.first()?.iter().find_map(|arg| match arg {
574        CssValue::Keyword(kw) => Some(kw.to_string()),
575        CssValue::String(s) => Some(s.clone()),
576        _ => None,
577    })?;

580}

652/// Resolves the `stroke` presentation value into a concrete stroke, threading
653/// the current stroke down the tree: a `stroke: none` (or transparent) stops
654/// stroking, `currentColor` resolves to `current_color`, and the remaining
655/// parameters keep whatever the ancestors declared when this element does not
656/// override them. A non-positive `stroke-width` suppresses the stroke. Returns
657/// `None` when nothing paints.
658pub(crate) fn resolve_stroke(
659    el: &impl SvgElement,
660    color_scheme: ColorScheme,
661    current_color: Color,
662    gradients: &HashMap<String, Gradient>,
663    inherited: Option<&SvgStroke>,
664) -> Option<SvgStroke> {
665    let mut stroke = SvgStroke::inherited_from(inherited);
666    // Only produce a stroke when the element (or an ancestor) actually
667    // declared one; the SVG default is `stroke: none`.
668    let mut declared = inherited.is_some();
669    if let Some(value) = presentation_value(el, "stroke") {
670        if is_transparent_keyword(value.trim()) {
671            return None;
672        }
673        declared = true;
674        if let Some(id) = parse_svg_url(value) {
675            // A gradient stroke paints even when the reference resolves; an
676            // unresolved reference also paints nothing.
677            stroke.gradient = Some(gradients.get(&id).cloned()?);
678        } else {
679            stroke.gradient = None;
680            match parse_svg_fill(value, color_scheme, current_color) {
681                Some(color) => stroke.color = color,
682                None => stroke.color = Color(0, 0, 0, 255),
683            }
684        }
685    }
686    if let Some(width) = numeric_presentation_value(el, "stroke-width") {
687        stroke.width = width;
688    }
689    if let Some(value) = presentation_value(el, "stroke-linecap") {
690        stroke.cap = match value.trim() {
691            v if v.eq_ignore_ascii_case("round") => StrokeCap::Round,
692            v if v.eq_ignore_ascii_case("square") => StrokeCap::Square,
693            _ => StrokeCap::Butt,
694        };
695    }
696    if let Some(value) = presentation_value(el, "stroke-linejoin") {
697        stroke.join = match value.trim() {
698            v if v.eq_ignore_ascii_case("round") => StrokeJoin::Round,
699            v if v.eq_ignore_ascii_case("bevel") => StrokeJoin::Bevel,
700            _ => StrokeJoin::Miter,
701        };
702    }
703    if let Some(limit) = numeric_presentation_value(el, "stroke-miterlimit") {
704        stroke.miter_limit = limit.max(1.0);
705    }
706    if let Some(value) = presentation_value(el, "stroke-dasharray") {
707        stroke.dash = parse_dasharray(value);
708    }
709    if let Some(offset) = numeric_presentation_value(el, "stroke-dashoffset") {
710        stroke.dash_offset = offset;
711    }
712    if !declared || stroke.width <= 0.0 {
713        return None;
714    }
715    let paints = stroke.gradient.is_some() || stroke.color.3 != 0;
716    if !paints {
717        return None;
718    }
719    Some(stroke)
720}

722/// Parses a `stroke-dasharray` value: `none` and empty lists yield `None`;
723/// otherwise the whitespace/comma separated non-negative lengths. A pattern
724/// that is entirely zero-length paints nothing.
725pub(crate) fn parse_dasharray(value: &str) -> Option<Vec<f32>> {
726    let value = value.trim();
727    if value.is_empty() || value.eq_ignore_ascii_case("none") {
728        return None;
729    }
730    let lengths: Vec<f32> = value
731        .split(',')
732        .flat_map(|part| part.split_ascii_whitespace())
733        .filter_map(|n| n.parse::<f32>().ok())
734        .map(|n| n.max(0.0))
735        .collect();
736    if lengths.is_empty() || lengths.iter().all(|n| *n <= 0.0) {
737        None
738    } else {
739        Some(lengths)
740    }
741}

743/// The inherited paint properties SVG carries down the tree, so a container's
744/// `fill`/`stroke` (and companions) are picked up by its descendants.
745#[derive(Debug, Clone)]
746pub(crate) struct PaintState {
747    pub fill: Option<Color>,
748    pub fill_gradient: Option<Gradient>,
749    pub fill_opacity: f32,
750    pub rule: FillRule,
751    pub stroke: Option<SvgStroke>,
752    pub stroke_opacity: f32,
753}
754
755impl PaintState {
756    pub(crate) fn new() -> Self {
757        PaintState {
758            fill: None,
759            fill_gradient: None,
760            fill_opacity: 1.0,
761            rule: FillRule::NonZero,
762            stroke: None,
763            stroke_opacity: 1.0,
764        }
765    }
766
767    /// Overlays `el`'s declarations on the incoming state. `gradients` resolves
768    /// `url(#id)` paint-server references to their gradient definitions.
769    pub(crate) fn apply(
770        &mut self,
771        el: &impl SvgElement,
772        color_scheme: ColorScheme,
773        current_color: Color,
774        gradients: &HashMap<String, Gradient>,
775    ) {
776        if let Some(value) = presentation_value(el, "fill") {
777            match parse_svg_url(value) {
778                Some(id) => {
779                    // A gradient fill overrides any inherited solid fill; an
780                    // unresolved reference paints nothing (`none`).
781                    if let Some(gradient) = gradients.get(&id) {
782                        self.fill_gradient = Some(gradient.clone());
783                        self.fill = None;
784                    } else {
785                        self.fill_gradient = None;
786                        self.fill = Some(Color(0, 0, 0, 0));
787                    }
788                }
789                None => {
790                    self.fill_gradient = None;
791                    self.fill = parse_svg_fill(value, color_scheme, current_color)
792                        .or(Some(Color(0, 0, 0, 255)));
793                }
794            }
795        }
796        self.fill_opacity *= numeric_presentation_value(el, "fill-opacity")
797            .unwrap_or(1.0)
798            .clamp(0.0, 1.0);
799        if let Some(value) = presentation_value(el, "fill-rule") {
800            self.rule = if value.eq_ignore_ascii_case("evenodd") {
801                FillRule::EvenOdd
802            } else {
803                FillRule::NonZero
804            };
805        }
806        self.stroke_opacity *= numeric_presentation_value(el, "stroke-opacity")
807            .unwrap_or(1.0)
808            .clamp(0.0, 1.0);
809        self.stroke = resolve_stroke(
810            el,
811            color_scheme,
812            current_color,
813            gradients,
814            self.stroke.as_ref(),
815        );
816    }
817}

874/// Hashes everything about the SVG subtree that affects its collected content:
875/// the `viewBox` and every descendant's tag name and attributes.
876fn svg_subtree_hash(snapshot: &DomSnapshot, svg_id: NodeId) -> u64 {
877    let mut h: u64 = 0xcbf29ce484222325;
878    fn hash_node(h: &mut u64, snapshot: &DomSnapshot, id: NodeId) {
879        let mut mix_byte = |b: u8| {
880            *h ^= b as u64;
881            *h = (*h).wrapping_mul(0x100000001b3);
882        };
883        let node = &snapshot.node(id).kind;
884        if let crate::engine::html::HtmlNodeType::Element {
885            tag_name,
886            attributes,
887            ..
888        } = node
889        {
890            for b in tag_name.bytes() {
891                mix_byte(b);
892            }
893            let len = attributes.len() as u32;
894            for b in len.to_le_bytes() {
895                mix_byte(b);
896            }
897            for attr in attributes {
898                for b in attr.name.bytes() {
899                    mix_byte(b);
900                }
901                for b in attr.value.bytes() {
902                    mix_byte(b);
903                }
904            }
905        }
906        for &child in snapshot.children(id) {
907            hash_node(h, snapshot, child);
908        }
909    }
910    hash_node(&mut h, snapshot, svg_id);
911    h
912}

914/// Collects the inline SVG content of an `<svg>` element: its `viewBox` and
915/// the solid-color shapes of its content, recursively.
916///
917/// Container elements (`g`, `a`, `symbol`) are walked with their `transform`
918/// and `opacity` attributes composed down the tree. `path`/`rect`/`circle`/
919/// `ellipse` become [`SvgShape`]s whose paths are pre-transformed into
920/// `viewBox` space, so [`emit_commands`] only maps the `viewBox` onto the
921/// content box. `text_color` is the `<svg>` element's computed CSS `color`,
922/// used to resolve `fill="currentColor"`. Returns `None` when the element has
923/// no usable `viewBox` or no paintable subpaths.
924pub fn collect_svg(
925    snapshot: &DomSnapshot,
926    svg_id: NodeId,
927    color_scheme: ColorScheme,
928    text_color: Color,
929) -> Option<SvgContent> {
930    let mut signature = svg_subtree_hash(snapshot, svg_id);
931    signature = signature
932        .wrapping_add(color_scheme as u64)
933        .wrapping_add(text_color.0 as u64)
934        .wrapping_add((text_color.1 as u64) << 8)
935        .wrapping_add((text_color.2 as u64) << 16)
936        .wrapping_add((text_color.3 as u64) << 24);
937    {
938        let mut cache = svg_content_cache().lock().unwrap();
939        if let Some((cached_sig, content)) = cache.get(&svg_id)
940            && *cached_sig == signature
941        {
942            return Some(content.clone());
943        }
944    }
945    #[cfg(test)]
946    COLLECT_SVG_MISSES.with(|c| c.fetch_add(1, std::sync::atomic::Ordering::Relaxed));
947
948    let content = {
949        let view_box = parse_view_box(snapshot.node(svg_id).kind.get_attr("viewBox")?);
950        if view_box.2 <= 0.0 || view_box.3 <= 0.0 {
951            return None;
952        }
953        let mut gradients = HashMap::new();
954        for &child in snapshot.children(svg_id) {
955            collect_gradient_defs(snapshot, child, color_scheme, text_color, &mut gradients);
956        }
957        let mut id_map = HashMap::new();
958        for &child in snapshot.children(svg_id) {
959            collect_id_map(snapshot, child, &mut id_map);
960        }
961        let mut clip_paths = HashMap::new();
962        for &child in snapshot.children(svg_id) {
963            collect_clip_path_defs(snapshot, child, &mut clip_paths);
964        }
965        let mut shapes = Vec::new();
966        let mut use_stack: Vec<NodeId> = Vec::new();
967        for &child in snapshot.children(svg_id) {
968            collect_content(
969                snapshot,
970                child,
971                &AffineTransform::identity(),
972                1.0,
973                color_scheme,
974                text_color,
975                PaintState::new(),
976                &gradients,
977                &id_map,
978                &clip_paths,
979                &mut use_stack,
980                &mut shapes,
981            );
982        }
983        if shapes.is_empty() {
984            return None;
985        }
986        SvgContent { view_box, shapes }
987    };
988    svg_content_cache()
989        .lock()
990        .unwrap()
991        .put(svg_id, (signature, content.clone()));
992    Some(content)
993}

995/// Collects `<clipPath>` definitions under `root` into `out`, keyed by their
996/// `id` attribute. All shape descendants of a `<clipPath>` are combined into a
997/// single path (transformed by the clipPath element's `transform` attribute).
998fn collect_clip_path_defs(snapshot: &DomSnapshot, id: NodeId, out: &mut HashMap<String, ClipPath>) {
999    let el = &snapshot.node(id).kind;
1000    if el.tag_name() == Some("clipPath")
1001        && let Some(clip_id) = el.get_attr("id").filter(|id| !id.is_empty())
1002    {
1003        let mut path = Path::new();
1004        let transform = match el.get_attr("transform") {
1005            Some(value) => parse_transform(value).unwrap_or(AffineTransform::identity()),
1006            None => AffineTransform::identity(),
1007        };
1008        let rule = resolve_fill_rule(el);
1009        collect_clip_path_shapes(snapshot, id, &transform, &mut path);
1010        if !path.commands().is_empty() {

1012                .or_insert(ClipPath { path, rule });
1013        }
1014    }
1015    for &child in snapshot.children(id) {
1016        collect_clip_path_defs(snapshot, child, out);
1017    }

1020/// Recursively collects all shape descendants under `id` into `path`, applying
1021/// `transform` to each shape's geometry.
1022fn collect_clip_path_shapes(
1023    snapshot: &DomSnapshot,
1024    id: NodeId,
1025    transform: &AffineTransform,
1026    path: &mut Path,
1027) {
1028    let el = &snapshot.node(id).kind;
1029    if let Some(shape_path) = element_shape(el) {
1030        let transformed = transform_path(&shape_path, transform);
1031        // Append the transformed subpaths into the combined path.
1032        for subpath in transformed.subpaths() {
1033            if let Some(&first) = subpath.first() {
1034                path.move_to(first.0, first.1);
1035                for &(x, y) in &subpath[1..] {
1036                    path.line_to(x, y);
1037                }
1038                path.close();
1039            }
1040        }
1041    }
1042    for &child in snapshot.children(id) {
1043        collect_clip_path_shapes(snapshot, child, transform, path);
1044    }
1045}

1100/// Resolves a gradient's geometry from its attributes. Both gradient kinds share
1101/// the `gradientUnits`/`spreadMethod` handling.
1102fn gradient_units_and_spread(el: &impl SvgElement) -> (bool, GradientSpread) {
1103    let user_space = el
1104        .svg_attr("gradientUnits")
1105        .map(|v| v.trim().eq_ignore_ascii_case("userSpaceOnUse"))
1106        .unwrap_or(false);
1107    let spread = match el
1108        .svg_attr("spreadMethod")
1109        .map(|v| v.trim())
1110        .unwrap_or("pad")
1111    {
1112        v if v.eq_ignore_ascii_case("repeat") => GradientSpread::Repeat,
1113        v if v.eq_ignore_ascii_case("reflect") => GradientSpread::Reflect,
1114        _ => GradientSpread::Pad,
1115    };
1116    (user_space, spread)
1117}

1119/// Parses an `<linearGradient>` element's geometry around already-resolved
1120/// color stops. The defs live in `viewBox`/bbox space, so no transform is
1121/// applied here.
1122fn parse_linear_gradient(
1123    el: &impl SvgElement,
1124    stops: &[ColorStop],
1125    _color_scheme: ColorScheme,
1126    _text_color: Color,
1127) -> Option<Gradient> {
1128    let (user_space, spread) = gradient_units_and_spread(el);
1129    if stops.is_empty() {
1130        return None;
1131    }
1132    Some(Gradient {
1133        kind: GradientKind::SvgLinear {
1134            x1: el.svg_attr("x1").and_then(parse_len).unwrap_or(0.0),
1135            y1: el.svg_attr("y1").and_then(parse_len).unwrap_or(0.0),
1136            x2: el.svg_attr("x2").and_then(parse_len).unwrap_or(1.0),
1137            y2: el.svg_attr("y2").and_then(parse_len).unwrap_or(0.0),
1138            user_space,
1139        },
1140        stops: stops.to_vec(),
1141        spread,
1142    })
1143}

1145/// Parses a `<radialGradient>` element's geometry. A missing focal point
1146/// degrades to the center (`fx/fy` inherit `cx/cy` per the SVG spec).
1147fn parse_radial_gradient(
1148    el: &impl SvgElement,
1149    stops: &[ColorStop],
1150    _color_scheme: ColorScheme,
1151    _text_color: Color,
1152) -> Option<Gradient> {
1153    let (user_space, spread) = gradient_units_and_spread(el);
1154    let cx = el.svg_attr("cx").and_then(parse_len).unwrap_or(0.5);
1155    let cy = el.svg_attr("cy").and_then(parse_len).unwrap_or(0.5);
1156    if stops.is_empty() {
1157        return None;
1158    }
1159    Some(Gradient {
1160        kind: GradientKind::SvgRadial {
1161            cx,

1174/// Resolves the direct `<stop>` children of the gradient element `id`,
1175/// resolving each stop's color, opacity, and offset. Unparsable stops are
1176/// skipped.
1177fn collect_stops(
1178    snapshot: &DomSnapshot,
1179    id: NodeId,
1180    color_scheme: ColorScheme,
1181    text_color: Color,
1182) -> Vec<ColorStop> {
1183    let mut out = Vec::new();
1184    for &child in snapshot.children(id) {
1185        let el = &snapshot.node(child).kind;
1186        if el.tag_name() != Some("stop") {
1187            continue;
1188        }
1189        let offset = el.get_attr("offset").and_then(parse_stop_offset);
1190        let color = presentation_value(el, "stop-color")
1191            .and_then(|value| parse_svg_fill(value, color_scheme, text_color))
1192            .unwrap_or(Color(0, 0, 0, 255));
1193        let opacity = numeric_presentation_value(el, "stop-opacity")
1194            .unwrap_or(1.0)
1195            .clamp(0.0, 1.0);
1196        out.push(ColorStop {
1197            color: with_alpha(color, opacity),
1198            position: offset,
1199        });
1200    }
1201    out
1202}

1204/// Parses an SVG gradient stop `offset` (`0.5`, `50%`, …) into a 0..1 fraction.

1213/// Recursively gathers the paintable shapes under `id`, composing `transform`
1214/// and `opacity` into the inherited values as it walks containers. `paint`
1215/// carries the inherited `fill`/`stroke` state that containers set for their
1216/// descendants.
1217#[allow(clippy::too_many_arguments)]
1218fn collect_content(
1219    snapshot: &DomSnapshot,
1220    id: NodeId,
1221    inherited: &AffineTransform,
1222    opacity: f32,
1223    color_scheme: ColorScheme,
1224    text_color: Color,
1225    mut paint: PaintState,
1226    gradients: &HashMap<String, Gradient>,
1227    id_map: &HashMap<String, NodeId>,
1228    clip_paths: &HashMap<String, ClipPath>,
1229    use_stack: &mut Vec<NodeId>,
1230    shapes: &mut Vec<SvgShape>,
1231) {
1232    let el = &snapshot.node(id).kind;
1233    let eff = match el.get_attr("transform") {
1234        Some(value) => match parse_transform(value) {
1235            Ok(transform) => inherited.then(&transform),
1236            Err(_) => *inherited,
1237        },
1238        None => *inherited,
1239    };
1240    let element_opacity = numeric_presentation_value(el, "opacity").unwrap_or(1.0);
1241    let opacity = opacity * element_opacity.clamp(0.0, 1.0);
1242    paint.apply(el, color_scheme, text_color, gradients);
1243
1244    match el.tag_name() {
1245        Some("g" | "a" | "symbol") => {
1246            for &child in snapshot.children(id) {
1247                collect_content(
1248                    snapshot,
1249                    child,
1250                    &eff,
1251                    opacity,
1252                    color_scheme,
1253                    text_color,
1254                    paint.clone(),
1255                    gradients,
1256                    id_map,
1257                    clip_paths,
1258                    use_stack,
1259                    shapes,
1260                );
1261            }
1262        }
1263        Some("use") => {
1264            // `<use href="#id">` clones the referenced element's content at the
1265            // use element's position, with x/y translating the content.
1266            let href = el.svg_attr("href").or_else(|| el.svg_attr("xlink:href"));
1267            let target_id = href
1268                .and_then(parse_use_ref)
1269                .and_then(|frag| id_map.get(&frag).copied());
1270            if let Some(target) = target_id {
1271                // Circular reference detection: bail out if the target is
1272                // already being expanded by an ancestor `<use>`.
1273                if !use_stack.contains(&target) {
1274                    let x = el.svg_attr("x").and_then(parse_len).unwrap_or(0.0);
1275                    let y = el.svg_attr("y").and_then(parse_len).unwrap_or(0.0);
1276                    let use_transform = eff.then(&AffineTransform::translate(x, y));
1277                    use_stack.push(target);
1278                    collect_content(
1279                        snapshot,
1280                        target,
1281                        &use_transform,
1282                        opacity,
1283                        color_scheme,
1284                        text_color,
1285                        paint.clone(),
1286                        gradients,
1287                        id_map,
1288                        clip_paths,
1289                        use_stack,
1290                        shapes,
1291                    );
1292                    use_stack.pop();
1293                }
1294            }
1295        }
1296        Some("path" | "rect" | "circle" | "ellipse" | "polygon" | "polyline" | "line") => {
1297            let Some(path) = element_shape(el) else {
1298                return;
1299            };
1300            let path = transform_path(&path, &eff);
1301            if path.commands().is_empty() {
1302                return;
1303            }
1304            let fill_alpha = opacity * paint.fill_opacity;
1305            let fill = match &paint.fill_gradient {
1306                Some(gradient) => Brush::Gradient(with_stop_alpha(gradient.clone(), fill_alpha)),
1307                None => Brush::Solid(with_alpha(
1308                    paint.fill.unwrap_or(Color(0, 0, 0, 255)),
1309                    fill_alpha,
1310                )),
1311            };
1312            let stroke = paint.stroke.clone().map(|mut stroke| {
1313                if let Some(gradient) = &mut stroke.gradient {
1314                    *gradient = with_stop_alpha(gradient.clone(), opacity * paint.stroke_opacity);
1315                } else {
1316                    stroke.color = with_alpha(stroke.color, opacity * paint.stroke_opacity);
1317                }
1318                stroke
1319            });
1320            let clip_path = el
1321                .svg_attr("clip-path")
1322                .and_then(parse_use_ref)
1323                .and_then(|frag| clip_paths.get(&frag).cloned());
1324            shapes.push(SvgShape {
1325                path,
1326                fill,
1327                rule: paint.rule,
1328                stroke,
1329                clip_path,
1330            });
1331        }
1332        _ => {}
1333    }
1334}

1347/// Builds the shape-space path an element describes, or `None` when the element
1348/// is malformed or has no area.
1349fn element_shape(el: &impl SvgElement) -> Option<Path> {
1350    match el.svg_tag_name()? {
1351        "path" => {
1352            let d = el.svg_attr("d")?;
1353            let path = path_from_d(d);
1354            (!path.commands().is_empty()).then_some(path)
1355        }
1356        "rect" => {
1357            let x = el.svg_attr("x").and_then(parse_len).unwrap_or(0.0);
1358            let y = el.svg_attr("y").and_then(parse_len).unwrap_or(0.0);
1359            let w = el.svg_attr("width").and_then(parse_len).unwrap_or(0.0);
1360            let h = el.svg_attr("height").and_then(parse_len).unwrap_or(0.0);
1361            let rx = el
1362                .svg_attr("rx")
1363                .and_then(parse_len)
1364                .unwrap_or(0.0)
1365                .max(0.0);
1366            let ry = el
1367                .svg_attr("ry")
1368                .and_then(parse_len)
1369                .unwrap_or(0.0)
1370                .max(0.0);
1371            if w <= 0.0 || h <= 0.0 {
1372                return None;
1373            }
1374            if rx > 0.0 || ry > 0.0 {
1375                // A single rx/ry applies to every corner, per the SVG grammar.
1376                let radius = (rx, ry);
1377                Some(rounded_rect_path(
1378                    x, y, w, h, radius, radius, radius, radius,
1379                ))
1380            } else {
1381                Some(rect_path(x, y, w, h))
1382            }
1383        }
1384        "circle" | "ellipse" => {
1385            let cx = el.svg_attr("cx").and_then(parse_len).unwrap_or(0.0);
1386            let cy = el.svg_attr("cy").and_then(parse_len).unwrap_or(0.0);
1387            let (rx, ry) = if el.svg_tag_name() == Some("circle") {
1388                let r = el.svg_attr("r").and_then(parse_len).unwrap_or(0.0);
1389                (r, r)
1390            } else {
1391                (
1392                    el.svg_attr("rx").and_then(parse_len).unwrap_or(0.0),
1393                    el.svg_attr("ry").and_then(parse_len).unwrap_or(0.0),
1394                )
1395            };
1396            (rx > 0.0 && ry > 0.0).then_some(ellipse_path(cx, cy, rx, ry))
1397        }
1398        "polygon" => {
1399            let points = parse_points(el.svg_attr("points").unwrap_or_default());
1400            (!points.is_empty()).then(|| polygon_path(&points))
1401        }
1402        "polyline" => {
1403            let points = parse_points(el.svg_attr("points").unwrap_or_default());
1404            (!points.is_empty()).then(|| polyline_path(&points))
1405        }
1406        "line" => {
1407            let (x1, y1) = (
1408                el.svg_attr("x1").and_then(parse_len).unwrap_or(0.0),
1409                el.svg_attr("y1").and_then(parse_len).unwrap_or(0.0),
1410            );
1411            let (x2, y2) = (
1412                el.svg_attr("x2").and_then(parse_len).unwrap_or(0.0),
1413                el.svg_attr("y2").and_then(parse_len).unwrap_or(0.0),
1414            );
1415            Some(line_path(x1, y1, x2, y2))
1416        }
1417        _ => None,
1418    }
1419}

1421/// Builds the transform mapping `viewBox` coordinates into a viewport using
1422/// `xMidYMid meet` (the SVG default: fit the whole box inside the viewport,
1423/// centered). Without a usable `viewBox`, coordinates map 1:1.
1424pub fn viewbox_meet_transform(view_box: ViewBox, vw: f32, vh: f32) -> AffineTransform {
1425    let (vx, vy, vbw, vbh) = view_box;
1426    if vbw <= 0.0 || vbh <= 0.0 {
1427        return AffineTransform::identity();
1428    }
1429    let scale = (vw / vbw).min(vh / vbh);
1430    let tx = (vw - vbw * scale) * 0.5;
1431    let ty = (vh - vbh * scale) * 0.5;
1432    AffineTransform {
1433        m11: scale,
1434        m22: scale,
1435        dx: tx - vx * scale,
1436        dy: ty - vy * scale,
1437        ..AffineTransform::identity()
1438    }
1439}

1441/// Paints the shapes of an inline SVG into its content box.
1442///
1443/// Pushes a transform mapping the `viewBox` rectangle onto the content box
1444/// with [`viewbox_meet_transform`] (letterboxing when the aspect ratios
1445/// differ), emits one `Fill` per shape in document order, then pops the
1446/// transform. The caller must have already pushed the box model:
1447/// `push_box_model` translates block boxes to their content-box origin, and
1448/// inline boxes push no transform (so `is_inline` paints in absolute
1449/// coordinates). Paints nothing when the box or `viewBox` is degenerate.
1450pub fn emit_commands(
1451    cmd_buf: &mut Vec<DrawCommand>,
1452    content_box: (f32, f32, f32, f32),
1453    is_inline: bool,
1454    view_box: ViewBox,
1455    shapes: &[SvgShape],
1456    opacity: f32,
1457) {
1458    let (cx, cy, cw, ch) = content_box;
1459    if cw <= 0.0 || ch <= 0.0 {
1460        return;
1461    }
1462    let (ox, oy) = if is_inline { (cx, cy) } else { (0.0, 0.0) };
1463    let mut transform = viewbox_meet_transform(view_box, cw, ch);
1464    transform.dx += ox;
1465    transform.dy += oy;
1466    cmd_buf.push(DrawCommand::PushTransform { transform });
1467    for shape in shapes {
1468        if let Some(clip) = &shape.clip_path {
1469            cmd_buf.push(DrawCommand::PushClip {
1470                path: clip.path.clone(),
1471                rule: clip.rule,
1472            });
1473        }
1474        cmd_buf.push(DrawCommand::Fill {
1475            path: shape.path.clone(),
1476            paint: Paint {
1477                brush: shape.fill.clone(),
1478                opacity,
1479            },
1480            rule: shape.rule,
1481        });
1482        if let Some(stroke) = &shape.stroke {
1483            // The stroke outline is expanded in viewBox space (the same space
1484            // the shape's path lives in) and filled with the nonzero rule by
1485            // the same pipeline as the shape fill.
1486            let brush = match &stroke.gradient {
1487                Some(gradient) => Brush::Gradient(gradient.clone()),
1488                None => Brush::Solid(stroke.color),
1489            };
1490            cmd_buf.push(DrawCommand::Fill {
1491                path: stroke_path(&shape.path, &stroke.to_stroke()),
1492                paint: Paint { brush, opacity },
1493                rule: FillRule::NonZero,
1494            });
1495        }
1496        if shape.clip_path.is_some() {
1497            cmd_buf.push(DrawCommand::PopClip);
1498        }
1499    }
1500    cmd_buf.push(DrawCommand::PopTransform);
1501}

1503/// SVG path number lexer: reads sign-separated floats per the path `d`
1504/// grammar, tracking the command letter position.
1505struct SvgNum {
1506    chars: Vec<char>,
1507    pos: usize,
1508}
1509
1510impl SvgNum {
1511    fn new(d: &str) -> Self {
1512        SvgNum {
1513            chars: d.chars().collect(),
1514            pos: 0,
1515        }
1516    }
1517
1518    fn skip_ws(&mut self) {
1519        while self.pos < self.chars.len() {
1520            let c = self.chars[self.pos];
1521            if c.is_ascii_whitespace() || c == ',' {
1522                self.pos += 1;
1523            } else {
1524                break;
1525            }
1526        }
1527    }
1528
1529    /// Next command letter (consuming it), or `None` at end of data.
1530    fn next_command(&mut self) -> Option<char> {
1531        self.skip_ws();
1532        let c = *self.chars.get(self.pos)?;
1533        if c.is_ascii_alphabetic() {
1534            self.pos += 1;
1535            Some(c)
1536        } else {
1537            None
1538        }
1539    }
1540
1541    fn next_number(&mut self) -> Option<f32> {
1542        self.skip_ws();
1543        if self.pos >= self.chars.len() {
1544            return None;
1545        }
1546        // A command letter terminates the current coordinate list.
1547        if self.chars[self.pos].is_ascii_alphabetic() {
1548            return None;
1549        }
1550        let start = self.pos;
1551        let mut i = self.pos;
1552        if i < self.chars.len() && (self.chars[i] == '+' || self.chars[i] == '-') {
1553            i += 1;
1554        }
1555        let mut digits = 0;
1556        while i < self.chars.len() && self.chars[i].is_ascii_digit() {
1557            i += 1;
1558            digits += 1;
1559        }
1560        if i < self.chars.len() && self.chars[i] == '.' {
1561            i += 1;
1562            while i < self.chars.len() && self.chars[i].is_ascii_digit() {
1563                i += 1;
1564                digits += 1;
1565            }
1566        }
1567        // A lone '.' is a valid zero in SVG data.
1568        if digits == 0 && start == self.pos {
1569            return None;
1570        }
1571        // Exponent.
1572        if i < self.chars.len() && (self.chars[i] == 'e' || self.chars[i] == 'E') {
1573            let mut j = i + 1;
1574            if j < self.chars.len() && (self.chars[j] == '+' || self.chars[j] == '-') {
1575                j += 1;
1576            }
1577            if j < self.chars.len() && self.chars[j].is_ascii_digit() {
1578                i = j;
1579                while i < self.chars.len() && self.chars[i].is_ascii_digit() {
1580                    i += 1;
1581                }
1582            }
1583        }
1584        // Reject trailing decimal point with no digits (e.g. "10.5M").
1585        let text: String = self.chars[start..i].iter().collect();
1586        match text.parse::<f32>() {
1587            Ok(v) => {
1588                self.pos = i;
1589                Some(v)
1590            }
1591            Err(_) => None,
1592        }
1593    }
1594
1595    /// A relative coordinate is relative to `ref` (the current point).
1596    fn next_value(&mut self, rel: bool, ref_val: Option<f32>) -> Option<f32> {
1597        let v = self.next_number()?;
1598        if rel {
1599            Some(ref_val.unwrap_or(0.0) + v)
1600        } else {
1601            Some(v)
1602        }
1603    }
1604
1605    fn next_point(&mut self, rel: bool, cur: Option<(f32, f32)>) -> Option<(f32, f32)> {
1606        let ref_pt = cur.unwrap_or((0.0, 0.0));
1607        let x = self.next_value(rel, Some(ref_pt.0))?;
1608        let y = self.next_value(rel, Some(ref_pt.1))?;
1609        Some((x, y))
1610    }
1611}

1789        // the ellipse whose center follows the spec's endpoint→center

1796        let num = 400.0 * 100.0 - 400.0 * y1p * y1p - 100.0 * x1p * x1p;

1802        for &(x, y) in ring {

1804            let (rx_local, ry_local) = (dx * cos_p + dy * sin_p, -dx * sin_p + dy * cos_p);

2460    /// Rasterizes `DrawCommand`s with a transform stack, mirroring how the GPU
2461    /// renderer lays out `PushTransform`/`Fill`/`PopTransform`.
2462    fn rasterize_commands(commands: &[DrawCommand], width: u32, height: u32) -> Vec<u8> {
2463        let mut rgba = vec![0u8; width as usize * height as usize * 4];
2464        let mut stack = vec![AffineTransform::identity()];
2465        let mut clip_stack: Vec<(Path, FillRule)> = Vec::new();
2466        for command in commands {
2467            match command {
2468                DrawCommand::PushTransform { transform } => {
2469                    let next = stack.last().unwrap().then(transform);
2470                    stack.push(next);
2471                }
2472                DrawCommand::PopTransform => {
2473                    stack.pop();
2474                }
2475                DrawCommand::PushClip { path, rule } => {
2476                    clip_stack.push((path.clone(), *rule));
2477                }
2478                DrawCommand::PopClip => {
2479                    clip_stack.pop();
2480                }
2481                DrawCommand::Fill { path, paint, rule } => {
2482                    let Brush::Solid(color) = &paint.brush else {
2483                        continue;
2484                    };
2485                    let fill = with_alpha(*color, paint.opacity);
2486                    let transformed = transform_path(path, stack.last().unwrap());
2487                    let clip_mask = if !clip_stack.is_empty() {
2488                        let mut combined = vec![255u8; (width as usize) * (height as usize) * 4];
2489                        for (clip_path, clip_rule) in &clip_stack {
2490                            let mut tmp = vec![0u8; (width as usize) * (height as usize) * 4];
2491                            let clip_transformed = transform_path(clip_path, stack.last().unwrap());
2492                            super::rasterize::fill_path_into(
2493                                &mut tmp,
2494                                width,
2495                                height,
2496                                &clip_transformed,
2497                                Color(255, 255, 255, 255),
2498                                *clip_rule,
2499                                None,
2500                            );
2501                            // Intersect: combined alpha *= tmp alpha.
2502                            for i in (3..combined.len()).step_by(4) {
2503                                combined[i] = ((combined[i] as u16 * tmp[i] as u16) / 255) as u8;
2504                            }
2505                        }
2506                        Some(combined)
2507                    } else {
2508                        None
2509                    };
2510                    super::rasterize::fill_path_into(
2511                        &mut rgba,
2512                        width,
2513                        height,
2514                        &transformed,
2515                        fill,
2516                        *rule,
2517                        clip_mask.as_deref(),
2518                    );
2519                }
2520                _ => {}
2521            }
2522        }
2523        rgba
2524    }
