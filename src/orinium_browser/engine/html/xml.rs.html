1//! Minimal XML parser that turns well-formed XML markup into a generic
2//! element tree.
3//!
4//! This is deliberately *not* a full XML spec implementation. It covers the
5//! structural subset that embedded documents (notably SVG) actually use:
6//! element/attribute/name constructs, self-closing tags, processing
7//! instructions (`<?...?>`), comments, and `<!DOCTYPE ...>` prologs. Element
8//! and attribute names are case-sensitive, and namespace prefixes are kept as
9//! literal name characters (`svg:path`), matching XML semantics rather than
10//! the HTML parser's case-insensitive handling.
11//!
12//! The parser is a thin tree-builder over [`super::tokenizer::Tokenizer`],
13//! reusing the shared HTML tokenizer for low-level lexing so that attribute
14//! value quoting, comment syntax, entity decoding, and whitespace handling
15//! all follow a single well-tested implementation.
16
17use super::tokenizer::{Token, Tokenizer};
18use crate::engine::tree::{NodeRef, Tree, TreeNode};
19use std::rc::Rc;
20
21/// A parsed XML element: its (case-sensitive, possibly namespace-qualified)
22/// tag name and its attribute list.
23#[derive(Debug, Clone, PartialEq)]
24pub struct XmlElement {
25    pub name: String,
26    pub attributes: Vec<(String, String)>,
27}
28
29impl XmlElement {
30    /// Returns the value of the attribute named `name`, or `None`.
31    pub fn attr(&self, name: &str) -> Option<&str> {
32        self.attributes
33            .iter()
34            .find(|(attr_name, _)| attr_name == name)
35            .map(|(_, value)| value.as_str())
36    }
37}
38
39/// Parses XML markup into an element tree.
40///
41/// The root of the returned tree is the document element (the first element of
42/// the document). Everything before it — the XML declaration `<?xml ...?>`,
43/// comments, and the `<!DOCTYPE ...>` prolog — is skipped. Returns an error
44/// when the markup is not well-formed.
45pub fn parse(markup: &str) -> Result<Tree<XmlElement>, String> {
46    let mut tokenizer = Tokenizer::new(markup);
47    let mut root: Option<NodeRef<XmlElement>> = None;
48    let mut stack: Vec<NodeRef<XmlElement>> = Vec::new();
49
50    while let Some(token) = tokenizer.next_token() {
51        match token {
52            Token::StartTag {
53                name,
54                attributes,
55                self_closing,
56            } => {
57                let node = TreeNode::new(XmlElement {
58                    name: name.into_owned(),
59                    attributes: attributes
60                        .into_iter()
61                        .map(|attr| (attr.name.into_owned(), attr.value.into_owned()))
62                        .collect(),
63                });
64                match stack.last() {
65                    Some(parent) => {
66                        if self_closing {
67                            TreeNode::add_child(parent, node);
68                        } else {
69                            TreeNode::add_child(parent, Rc::clone(&node));
70                            stack.push(Rc::clone(&node));
71                        }
72                    }
73                    None => {
74                        if self_closing {
75                            root = Some(node);
76                        } else {
77                            root = Some(Rc::clone(&node));
78                            stack.push(node);
79                        }
80                    }
81                }
82            }
83            Token::EndTag { name } => {
84                if let Some(top) = stack.last() {
85                    let name_str = name.to_string();
86                    if top.borrow().value.name == name.into_owned() {
87                        stack.pop();
88                    } else {
89                        return Err(format!(
90                            "XML: mismatched closing tag </{name_str}> for <{}>",
91                            top.borrow().value.name
92                        ));
93                    }
94                }
95            }
96            _ => {}
97        }
98    }
99
100    match root {
101        Some(root) if stack.is_empty() => Ok(Tree::from_root(root)),
102        Some(_) => Err(format!(
103            "XML: unclosed element <{}>",
104            stack.last().unwrap().borrow().value.name
105        )),
106        None => Err("XML: no document element".to_string()),
107    }
108}

110#[cfg(test)]
111mod tests {
112    use super::*;
113
114    #[test]
115    fn parses_svg_document_with_pi_and_comment() {
116        let tree = parse(
117            r##"<?xml version="1.0" encoding="UTF-8"?>
118               <!-- a comment -->
119               <svg width="12" height="8" xmlns="http://www.w3.org/2000/svg">
120                 <g id="layer">
121                   <rect width="12" height="8" fill="#ff0000"/>
122                   <path d="M 0 0 L 1 1"/>
123                 </g>
124               </svg>"##,
125        )
126        .expect("parses");
127        let root = tree.root.borrow();
128        assert_eq!(root.value.name, "svg");
129        assert_eq!(root.value.attr("width"), Some("12"));
130        let children = root.children().to_vec();
131        assert_eq!(children.len(), 1);
132        assert_eq!(children[0].borrow().value.name, "g");
133        let g_children = children[0].borrow().children().to_vec();
134        assert_eq!(g_children.len(), 2);
135        assert_eq!(g_children[0].borrow().value.name, "rect");
136        assert_eq!(g_children[0].borrow().value.attr("fill"), Some("#ff0000"));
137        assert_eq!(g_children[1].borrow().value.name, "path");
138    }
139
140    #[test]
141    fn keeps_namespace_qualified_names() {
142        let tree = parse(
143            r#"<svg xmlns:inkscape="http://www.inkscape.org/namespaces/inkscape">
144                 <inkscape:label foo:bar="1"/>
145               </svg>"#,
146        )
147        .expect("parses");
148        let root = tree.root.borrow();
149        assert_eq!(root.value.name, "svg");
150        let ns = &root.children()[0].borrow().value;
151        assert_eq!(ns.name, "inkscape:label");
152        assert_eq!(ns.attr("foo:bar"), Some("1"));
153    }
154
155    #[test]
156    fn element_names_are_case_sensitive() {
157        let tree = parse(r#"<svg><path d="M0 0"/><PATH d="x"/></svg>"#).expect("parses");
158        let root = tree.root.borrow();
159        let children = root.children().to_vec();
160        assert_eq!(children.len(), 2);
161        assert_eq!(children[0].borrow().value.name, "path");
162        assert_eq!(children[1].borrow().value.name, "PATH");
163    }
164
165    #[test]
166    fn accepts_well_formed_and_rejects_malformed() {
167        assert!(parse("<svg><rect/></svg>").is_ok());
168        assert!(parse("<svg/>").is_ok());
169        assert!(parse("<svg><rect/></svg>trailing").is_ok());
170        assert!(parse("<svg>").is_err());
171        assert!(parse("<svg><rect/></g>").is_err());
172        assert!(parse("plain text").is_err());
173    }
