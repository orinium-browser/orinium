11/// ヒットしたノード情報
12pub struct HitItem<'a> {
13    pub layout: &'a LayoutNode,
14    pub info: &'a InfoNode,
15}
16
17/// ヒットパス（子→親の順）
18pub type HitPath<'a> = Vec<HitItem<'a>>;
19
20/// Returns the innermost custom node on a hit path, if any.
21///
22/// The hit path is ordered child→parent, so the first `Custom` node found is
23/// the deepest node under the pointer.
24pub fn hit_custom_node<'a>(path: &'a HitPath<'a>) -> Option<&'a Rc<dyn CustomNode>> {
25    path.iter().find_map(|hit| match &hit.info.kind {
26        NodeKind::Custom { node, .. } => Some(node),
27        _ => None,
28    })
29}
30
31/// Dispatches a pointer event to the innermost custom node on the hit path.
32pub fn dispatch_pointer(path: &HitPath<'_>, event: PointerEvent) -> bool {
33    hit_custom_node(path).is_some_and(|node| node.on_pointer_event(event))
34}
35
36/// Updates the hover state of custom nodes after a pointer move.
37///
38/// Clears hover from the previously hovered node (if different) and sets it on
39/// the node under the pointer. Returns whether the hover target changed.
40pub fn update_hover(path: &HitPath<'_>, previous: Option<&Rc<dyn CustomNode>>) -> bool {
41    let current = hit_custom_node(path);
42    match (previous, current) {
43        (Some(prev), Some(curr)) if Rc::ptr_eq(prev, curr) => false,
44        (Some(prev), _) => {
45            prev.set_hovered(false);
46            if let Some(curr) = current {
47                curr.set_hovered(true);
48            }
49            true
50        }
51        (None, Some(curr)) => {
52            curr.set_hovered(true);
53            true
54        }
55        (None, None) => false,
56    }
57}

59/// x, y: グローバル座標
60pub fn hit_test<'a>(layout: &'a LayoutNode, info: &'a InfoNode, x: f32, y: f32) -> HitPath<'a> {
61    // layout_boxes が空なら何もヒットしない
62    if layout.layout_box.is_empty() {
63        return Vec::new();
64    }
65
66    for box_model in layout
67        .layout_box
68        .iter()
69        .collect::<Vec<_>>()
70        .into_iter()
71        .rev()
72    {
73        // 後ろの box が前面
74        let rect = box_model.padding_box;
75
76        // 1. rect 外なら次の box へ
77        if x < rect.x || y < rect.y || x > rect.x + rect.width || y > rect.y + rect.height {
78            continue;
79        }
80
81        // 2. ローカル座標に変換（スクロールオフセット考慮）
82        let mut local_x = x - box_model.content_box.x;
83        let mut local_y = y - box_model.content_box.y;
84
85        if let NodeKind::Container {
86            scroll_offset_x,
87            scroll_offset_y,
88            ..
89        }
90        | NodeKind::Custom {
91            scroll_offset_x,
92            scroll_offset_y,
93            ..
94        } = &info.kind
95        {
96            local_x += *scroll_offset_x;
97            local_y += *scroll_offset_y;
98        }
99
100        // 3. 子ノードを前面から探索
101        for (child_layout, child_info) in layout.children.iter().zip(&info.children).rev() {
102            if let Some(child_node) = child_layout.node() {
103                let mut path = hit_test(child_node, child_info, local_x, local_y);
104                if !path.is_empty() {
105                    // 子がヒット → 自分を末尾に追加
106                    path.push(HitItem { layout, info });
107                    return path;
108                }
109            } else if child_layout.object().is_some()
110                && let NodeKind::Custom {
111                    layout_id: Some(layout_id),
112                    ..
113                } = &child_info.kind
114                && let Some(result) = get_custom_inline_result(*layout_id)
115                && result.spans.iter().any(|span| {
116                    local_x >= span.x_range.start
117                        && local_x <= span.x_range.end
118                        && local_y >= span.line_pos.1
119                        && local_y <= span.line_pos.1 + result.height
120                })
121            {
122                return vec![
123                    HitItem {
124                        layout,
125                        info: child_info,
126                    },
127                    HitItem { layout, info },
128                ];
129            }
130        }
131
132        // 4. 子ノードに当たらなければこの box がヒット
133        return vec![HitItem { layout, info }];
134    }
135
136    // どの box にもヒットしなかった
137    Vec::new()
138}

140/// Focuses `target` and clears focus from every other text input.
141///
142/// Returns whether a text input received focus.
143pub fn focus_text_input(info: &InfoNode, target: Option<&Rc<dyn CustomNode>>) -> bool {
144    let mut focused = false;
145    if let NodeKind::Custom { node, .. } = &info.kind
146        && node.accepts_text_input()
147    {
148        let is_target = target.is_some_and(|target| Rc::ptr_eq(node, target));
149        node.set_focused(is_target);
150        focused |= is_target;
151    }
152    for child in &info.children {
153        focused |= focus_text_input(child, target);
154    }
155    focused
156}

158/// Sends an editing event to the focused text input, if one exists.
159pub fn dispatch_text_input(info: &InfoNode, event: TextInputEvent) -> bool {
160    if let NodeKind::Custom { node, .. } = &info.kind
161        && node.accepts_text_input()
162        && node.is_focused()
163    {
164        return node.handle_text_input(event);
165    }
166    for child in &info.children {
167        if dispatch_text_input(child, event.clone()) {
168            return true;
169        }
170    }
171    false
172}

174/// Returns whether the focused text input has an active IME composition.
175pub fn focused_text_input_is_composing(info: &InfoNode) -> bool {
176    if let NodeKind::Custom { node, .. } = &info.kind
177        && node.accepts_text_input()
178        && node.is_focused()
179    {
180        return node.is_composing();
181    }
182    info.children.iter().any(focused_text_input_is_composing)
183}

185/// Returns whether any custom node in the tree reports a pending repaint.
186///
187/// Consumes the repaint flags of the nodes it visits, so callers should only
188/// invoke this once per frame, right before deciding whether to redraw.
189pub fn any_custom_node_needs_repaint(info: &InfoNode) -> bool {
190    if let NodeKind::Custom { node, .. } = &info.kind
191        && node.needs_repaint()
192    {
193        return true;
194    }
195    info.children.iter().any(any_custom_node_needs_repaint)
196}

326        assert!(!any_custom_node_needs_repaint(&info_a));
