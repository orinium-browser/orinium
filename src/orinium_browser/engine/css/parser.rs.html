1//! CSS Parser
2//!
3//! Consumes tokens produced by the tokenizer and builds
4//! higher-level CSS syntax structures.
5//!
6//! ## Responsibilities
7//! - Parse token streams into structured CSS data
8//!   (selectors, declarations, component values)
9//! - Handle nesting such as blocks and functions
10//!
11//! ## Non-responsibilities
12//! - Tokenization of raw input
13//! - Semantic interpretation (length resolution, color computation, etc.)
14//!
15//! ## Design notes
16//! - No property-specific validation is performed here
17//! - Semantic meaning is assigned in later stages (style computation, layout)

26/// Node kinds used in the CSS syntax tree.
27///
28/// These nodes represent **syntactic structure only**.
29/// No semantic validation or value resolution is performed here.
30#[derive(Debug, Clone)]
31pub enum CssNodeType {
32    /// Root node of a CSS document
33    Stylesheet,
34
35    /// Qualified rule (e.g. `div { ... }`)
36    Rule {
37        /// Selectors associated with this rule
38        selectors: Vec<ComplexSelector>,
39    },
40
41    /// At-rule (e.g. `@media`, `@supports`)
42    AtRule {
43        /// At-rule name without `@`
44        name: String,
45
46        params: AtQuery,
47    },
48
49    /// Declaration inside a rule block (e.g. `color: red`)
50    Declaration {
51        /// Property name
52        name: String,
53
54        value: CssValue,
55    },
56}
57
58#[derive(Debug, Clone, PartialEq)]
59pub enum AtQuery {
60    Keyword(String), // screen, and, not
61    Condition {
62        name: String,    // max-width
63        value: CssValue, // 600px
64    },
65    Group(Vec<AtQuery>), // ( ... )
66}

68/// Node in the CSS syntax tree.
69///
70/// Each node represents a syntactic construct such as a rule,
71/// at-rule, or declaration, and may contain child nodes.
72#[derive(Debug)]
73pub struct CssNode {
74    /// Kind of this CSS node
75    node: CssNodeType,
76
77    /// Child nodes forming the tree structure
78    children: Vec<CssNode>,

193/// Parse the integer `An+B` grammar used by structural pseudo-classes.
194fn parse_an_plus_b(tokens: &[Token]) -> Option<(i32, i32)> {
195    let mut expression = String::new();
196    for token in tokens {
197        match token {
198            Token::Whitespace | Token::Comment(_) => {}
199            Token::Ident(value) => expression.push_str(&value.to_ascii_lowercase()),
200            Token::Number(value) if value.fract() == 0.0 => {
201                expression.push_str(&(*value as i32).to_string());
202            }
203            Token::Dimension(value, unit) if value.fract() == 0.0 => {
204                expression.push_str(&(*value as i32).to_string());
205                expression.push_str(&unit.to_ascii_lowercase());
206            }
207            Token::Delim(value @ ('+' | '-')) => expression.push(*value),
208            _ => return None,
209        }
210    }
211
212    match expression.as_str() {
213        "odd" => return Some((2, 1)),
214        "even" => return Some((2, 0)),
215        _ => {}
216    }
217
218    if let Some(n_index) = expression.find('n') {
219        if expression[n_index + 1..].contains('n') {
220            return None;
221        }
222        let coefficient = match &expression[..n_index] {
223            "" | "+" => 1,
224            "-" => -1,
225            value => value.parse().ok()?,
226        };
227        let offset = match &expression[n_index + 1..] {
228            "" => 0,
229            value => value.parse().ok()?,
230        };
231        Some((coefficient, offset))
232    } else {
233        Some((0, expression.parse().ok()?))
234    }
235}

237/// CSS parser consuming tokens and producing syntax structures.
238#[derive(Clone)]
239pub struct Parser<'a> {
240    /// Source of tokens produced by the tokenizer
241    tokenizer: Tokenizer<'a>,
242
243    /// Used to detect the start and end of rule blocks (`{}`).
244    brace_depth: usize,
245
246    /// Lookahead token (optional)
247    ///
248    /// Parser may need to peek the next token without consuming it.
249    lookahead: VecDeque<Token>,
250}

252/// Parser error kinds
253#[derive(Debug, Clone, PartialEq, Eq)]
254pub enum ParserErrorKind {
255    /// Expected a token but found something else
256    UnexpectedToken {
257        expected: &'static str,
258        found: String, // Token debug or value
259    },
260
261    /// Unexpected end of file
262    UnexpectedEOF,
263
264    /// Invalid or unsupported CSS syntax
265    InvalidSyntax,
266
267    /// Mismatched braces or parentheses

269}

273        write!(f, "{:?}", self)

312    /// Create a new CSS parser from a source string.
313    pub fn new(input: &'a str) -> Self {
314        Self {
315            tokenizer: Tokenizer::new(input),
316            brace_depth: 0,
317            lookahead: VecDeque::new(),
318        }

321    fn ensure_lookahead(&mut self, n: usize) {
322        while self.lookahead.len() <= n {
323            let tok = self.tokenizer.next_token();
324            self.lookahead.push_back(tok);
325        }
326    }
327
328    fn peek_next_token(&mut self, cursor_size: usize) -> &Token {
329        self.ensure_lookahead(cursor_size);
330        &self.lookahead[cursor_size]
331    }
332
333    /// Consume and return the next token.
334    fn peek_token(&mut self) -> &Token {
335        self.peek_next_token(0)
336    }

338    /// Parse a bare declaration list (e.g. the value of a `style` attribute).
339    ///
340    /// Unlike `parse()`, this does not expect selectors or a surrounding block.
341    /// It consumes declarations until EOF or `}` and returns them as
342    /// `Declaration` nodes, mirroring the body of a rule.
343    pub fn parse_declarations(&mut self) -> ParseResult<Vec<CssNode>> {
344        self.parse_declaration_list()
345    }

347    fn consume_token(&mut self) -> Token {
348        if let Some(tok) = self.lookahead.pop_front() {
349            tok
350        } else {
351            self.tokenizer.next_token()
352        }
353    }

355    /// Parse the entire CSS source into a syntax tree.
356    ///
357    /// This method consumes tokens until `Token::EOF` is reached and constructs
358    /// a `CssNode` representing the stylesheet root.
359    ///
360    /// Parsing behavior:
361    /// - Whitespace tokens are ignored
362    /// - Qualified rules and at-rules are parsed into child nodes
363    /// - No semantic validation is performed
364    pub fn parse(&mut self) -> ParseResult<CssNode> {
365        let mut stylesheet = CssNode {
366            node: CssNodeType::Stylesheet,
367            children: vec![],
368        };
369
370        loop {
371            let token = self.peek_token().clone();
372
373            match token {
374                Token::EOF => break,
375                Token::Whitespace | Token::Comment(_) => {
376                    self.consume_token();
377                }
378                Token::AtKeyword(_) => {
379                    let node = self
380                        .parse_at_rule()
381                        .map_err(|e| e.with_context("parse: failed to parse at-rule"))?;
382                    log::debug!(target: "CssParser", "AtRule parsed: {:?}", node);
383                    stylesheet.children.push(node);
384                }
385                _ => {
386                    let node = self
387                        .parse_rule()
388                        .map_err(|e| e.with_context("parse: failed to parse rule"))?;
389                    log::debug!(target: "CssParser", "Rule parsed: {:?}", node);
390                    stylesheet.children.push(node);
391                }
392            }
393        }
394
395        Ok(stylesheet)
396    }

398    /// Parses a stylesheet while recovering from unsupported top-level items.
399    ///
400    /// Browser stylesheets are frequently generated and may contain selectors
401    /// or declarations that this engine does not support yet. Dropping the
402    /// entire stylesheet for one such rule would also discard all compatible
403    /// rules, so this mode skips the failing item and resumes at the next
404    /// top-level boundary. The strict [`Self::parse`] entry point remains
405    /// available for validation and unit tests.
406    pub fn parse_lossy(&mut self) -> CssNode {
407        let mut stylesheet = CssNode {
408            node: CssNodeType::Stylesheet,
409            children: vec![],
410        };
411
412        loop {
413            let token = self.peek_token().clone();
414            let checkpoint = self.clone();
415            match token {
416                Token::EOF => break,
417                Token::Whitespace | Token::Comment(_) => {
418                    self.consume_token();
419                }
420                Token::AtKeyword(_) => match self.parse_at_rule() {
421                    Ok(node) => stylesheet.children.push(node),
422                    Err(error) => {
423                        log::warn!(
424                            target: "CssParser",
425                            "Skipping unsupported at-rule: {error}"
426                        );
427                        *self = checkpoint;
428                        self.recover_top_level_item();
429                    }
430                },
431                _ => match self.parse_rule() {
432                    Ok(node) => stylesheet.children.push(node),
433                    Err(error) => {
434                        log::warn!(
435                            target: "CssParser",
436                            "Skipping unsupported CSS rule: {error}"
437                        );
438                        *self = checkpoint;
439                        self.recover_top_level_item();
440                    }
441                },
442            }
443        }
444
445        stylesheet
446    }

448    fn recover_top_level_item(&mut self) {
449        let mut depth = 0_usize;
450        let mut consumed_any = false;
451        loop {
452            match self.peek_token().clone() {
453                Token::EOF => break,
454                // An at-rule starts a new top-level item. Preserve it when an
455                // invalid selector prefix (for example a concatenated BOM)
456                // was the item that failed, instead of swallowing the whole
457                // following @media block during recovery.
458                Token::AtKeyword(_) if depth == 0 && consumed_any => break,
459                Token::Delim('{') => {
460                    depth += 1;
461                    self.consume_token();
462                    consumed_any = true;
463                }
464                Token::Delim('}') => {
465                    self.consume_token();
466                    if depth <= 1 {
467                        break;
468                    }
469                    depth -= 1;
470                    consumed_any = true;
471                }
472                Token::Delim(';') if depth == 0 => {
473                    self.consume_token();
474                    break;
475                }
476                _ => {
477                    self.consume_token();
478                    consumed_any = true;
479                }
480            }
481        }
482    }

484    fn parse_at_rule(&mut self) -> ParseResult<CssNode> {
485        // 1. consume '@' token
486        let at_name = if let Token::AtKeyword(name) = self.consume_token() {
487            name
488        } else {
489            return Err(ParserError {
490                kind: ParserErrorKind::UnexpectedToken {
491                    expected: "@keyword",
492                    found: format!("{:?}", self.peek_token()),
493                },
494                context: vec![],
495            });
496        };
497
498        // 2. Collect prelude tokens (until '{' or ';'), handling nested parentheses
499        let mut prelude = vec![];
500        let mut paren_depth = 0;
501
502        loop {
503            match self.peek_token() {
504                Token::Delim('{') if paren_depth == 0 => break,
505                Token::Delim(';') if paren_depth == 0 => break,
506                Token::Delim('(') => {
507                    paren_depth += 1;
508                    prelude.push(self.consume_token());
509                }
510                Token::Delim(')') => {
511                    paren_depth -= 1;
512                    prelude.push(self.consume_token());
513                }
514                Token::EOF => break,
515                _ => prelude.push(self.consume_token()),
516            }
517        }
518
519        // 3. Convert prelude tokens to CssValue (handles functions and nested parentheses)
520        let params = Self::parse_at_query(prelude).map_err(|e| {
521            e.with_context("parse_at_rule: failed to parse params via parse_at_query")
522        })?;
523
524        // 4. Block vs semicolon
525        let children = if self.peek_token() == &Token::Delim('{') {
526            self.consume_token();
527            self.brace_depth += 1;
528
529            let mut children = vec![];
530            while self.peek_token() != &Token::Delim('}') {
531                match self.peek_token() {
532                    Token::EOF => {
533                        return Err(ParserError {
534                            kind: ParserErrorKind::UnexpectedEOF,
535                            context: vec![],
536                        });
537                    }
538                    Token::Whitespace => {
539                        self.consume_token();
540                    }
541                    Token::AtKeyword(_) => {
542                        let node = self.parse_at_rule().map_err(|e| {
543                            e.with_context("parse_at_rule: failed to parse nested at-rule")
544                        })?;
545                        children.push(node);
546                    }
547                    _ => {
548                        let mut cursor = 0;
549                        let mut is_declaration = false;
550
551                        loop {
552                            match self.peek_next_token(cursor) {
553                                Token::Delim('{') => {
554                                    break;
555                                }
556                                Token::Delim('}') => {
557                                    is_declaration = true;
558                                    break;
559                                }
560                                Token::EOF => {
561                                    return Err(ParserError {
562                                        kind: ParserErrorKind::UnexpectedEOF,
563                                        context: vec![],
564                                    });
565                                }
566                                _ => {}
567                            }
568                            cursor += 1;
569                        }
570
571                        let nodes = if is_declaration {
572                            self.parse_declaration_list().map_err(|e| {
573                                e.with_context(
574                                    "parse_at_rule: failed to parse declaration in block",
575                                )
576                            })?
577                        } else {
578                            vec![self.parse_rule().map_err(|e| {
579                                e.with_context("parse_at_rule: failed to parse rule in block")
580                            })?]
581                        };
582
583                        children.extend(nodes);
584                    }
585                }
586            }
587
588            self.consume_token(); // consume '}'
589            self.brace_depth -= 1;
590            children
591        } else {
592            if self.consume_token() != Token::Delim(';') {
593                return Err(ParserError {
594                    kind: ParserErrorKind::UnexpectedToken {
595                        expected: ";",
596                        found: format!("{:?}", self.peek_token()),
597                    },
598                    context: vec![],
599                });
600            }
601            vec![]
602        };
603
604        Ok(CssNode {
605            node: CssNodeType::AtRule {
606                name: at_name,
607                params,
608            },
609            children,
610        })
611    }

613    fn parse_at_query(tokens: Vec<Token>) -> ParseResult<AtQuery> {
614        let mut cursor = 0;
615        let items = Self::parse_at_query_list(&tokens, &mut cursor)?;
616        Ok(AtQuery::Group(items))
617    }
618
619    fn parse_at_query_list(tokens: &[Token], cursor: &mut usize) -> ParseResult<Vec<AtQuery>> {
620        let mut items = Vec::new();
621
622        while *cursor < tokens.len() {
623            match &tokens[*cursor] {
624                Token::Whitespace => {
625                    *cursor += 1;
626                }
627
628                Token::Delim('(') => {
629                    *cursor += 1;
630                    let group = Self::parse_at_query_list(tokens, cursor)?;
631                    items.push(AtQuery::Group(group));
632                }
633
634                Token::Delim(')') => {
635                    *cursor += 1;
636                    break;
637                }
638
639                Token::Ident(_) => {
640                    items.push(Self::parse_at_query_item(tokens, cursor)?);
641                }
642
643                Token::Delim(',') => {
644                    items.push(AtQuery::Keyword(",".into()));
645                    *cursor += 1;
646                }
647
648                _ => {
649                    *cursor += 1;
650                }
651            }
652        }
653
654        Ok(items)
655    }
656
657    fn parse_at_query_item(tokens: &[Token], cursor: &mut usize) -> ParseResult<AtQuery> {
658        let name = match &tokens[*cursor] {
659            Token::Ident(s) => s.clone(),
660            _ => unreachable!(),
661        };
662        *cursor += 1;
663
664        let mut colon = *cursor;
665        while matches!(
666            tokens.get(colon),
667            Some(Token::Whitespace | Token::Comment(_))
668        ) {
669            colon += 1;
670        }
671        if matches!(tokens.get(colon), Some(Token::Delim(':'))) {
672            *cursor = colon + 1;
673            let value = Self::parse_at_query_value(tokens, cursor)?;
674            Ok(AtQuery::Condition { name, value })
675        } else {
676            Ok(AtQuery::Keyword(name))
677        }
678    }
679
680    fn parse_at_query_value(tokens: &[Token], cursor: &mut usize) -> ParseResult<CssValue> {
681        let mut buf = Vec::new();
682        let mut paren_depth = 0;
683
684        while *cursor < tokens.len() {
685            match &tokens[*cursor] {
686                Token::Delim('(') => {
687                    paren_depth += 1;
688                    buf.push(tokens[*cursor].clone());
689                    *cursor += 1;
690                }
691                Token::Delim(')') if paren_depth == 0 => break,
692                Token::Delim(')') => {
693                    paren_depth -= 1;
694                    buf.push(tokens[*cursor].clone());
695                    *cursor += 1;
696                }
697                _ => {
698                    buf.push(tokens[*cursor].clone());
699                    *cursor += 1;
700                }
701            }
702        }
703
704        Self::parse_tokens_to_css_value(buf)
705    }

707    /// Parse a qualified rule (e.g., `div { color: red; }`).
708    ///
709    /// Parses the selector list first, then the block of declarations.
710    fn parse_rule(&mut self) -> ParseResult<CssNode> {
711        // 1. Parse selectors
712        let selectors = self.parse_selector_list();
713
714        // 2. Expect `{`
715        match self.consume_token() {
716            Token::Delim('{') => self.brace_depth += 1,
717            token => {
718                return Err(ParserError {
719                    kind: ParserErrorKind::UnexpectedToken {
720                        expected: "{",
721                        found: format!("{:?}", token),
722                    },
723                    context: vec![format!(
724                        "While parsing rule with selectors: {}",
725                        selectors
726                            .iter()
727                            .map(|s| format!("{:?}", s))
728                            .collect::<Vec<_>>()
729                            .join(", ")
730                    )],
731                });
732            }
733        }
734
735        // 3. Parse declarations inside the block
736        let mut children = vec![];
737        loop {
738            let token = self.peek_token().clone();
739            match token {
740                Token::Delim('}') => {
741                    self.consume_token();
742                    self.brace_depth -= 1;
743                    break;
744                }
745                Token::EOF => {
746                    return Err(ParserError {
747                        kind: ParserErrorKind::UnexpectedEOF,
748                        context: vec![],
749                    });
750                }
751                _ => {
752                    let mut decls = self.parse_declaration_list().map_err(|e| {
753                        e.with_context("parse_rule: failed to parse declaration list")
754                    })?;
755                    children.append(&mut decls);
756                }
757            }
758        }
759
760        Ok(CssNode {
761            node: CssNodeType::Rule { selectors },
762            children,
763        })
764    }

773    /// Parse a selector list up to a rule block or a functional pseudo-class
774    /// closing delimiter.
775    fn parse_selector_list_until(&mut self, terminator: Option<char>) -> Vec<ComplexSelector> {
776        let mut selectors = vec![];
777        let mut parts = vec![];
778
779        let mut current_selector: Option<Selector> = None;
780        let mut current_combinator: Option<Combinator> = None;
781
782        loop {
783            let token = self.peek_token().clone();
784            match token {
785                Token::Ident(name) => {
786                    let sel = current_selector.get_or_insert_with(|| Selector {
787                        tag: None,
788                        id: None,
789                        classes: vec![],
790                        attributes: vec![],
791                        pseudo_classes: vec![],
792                        pseudo_element: None,
793                    });
794
795                    if sel.tag.is_none() {
796                        sel.tag = Some(name);
797                    }
798
799                    self.consume_token();
800                }
801
802                Token::Hash(id) => {
803                    let sel = current_selector.get_or_insert_with(|| Selector {
804                        tag: None,
805                        id: None,
806                        classes: vec![],
807                        attributes: vec![],
808                        pseudo_classes: vec![],
809                        pseudo_element: None,
810                    });
811                    sel.id = Some(id);
812                    self.consume_token();
813                }
814
815                Token::Delim('.') => {
816                    self.consume_token();
817                    if let Token::Ident(class) = self.consume_token() {
818                        let sel = current_selector.get_or_insert_with(|| Selector {
819                            tag: None,
820                            id: None,
821                            classes: vec![],
822                            attributes: vec![],
823                            pseudo_classes: vec![],
824                            pseudo_element: None,
825                        });
826                        sel.classes.push(class);
827                    }
828                }
829
830                Token::Delim(':') => {
831                    self.consume_token();
832                    if self.peek_token() == &Token::Delim(':') {
833                        // pseudo-element
834                        self.consume_token();
835                        if let Token::Ident(name) = self.consume_token() {
836                            let sel = current_selector.get_or_insert_with(|| Selector {
837                                tag: None,
838                                id: None,
839                                classes: vec![],
840                                attributes: vec![],
841                                pseudo_classes: vec![],
842                                pseudo_element: None,
843                            });
844                            sel.pseudo_element = Some(name);
845                        }
846                    } else {
847                        let pseudo_class = match self.consume_token() {
848                            Token::Ident(name) => Some(PseudoClass::Simple(name)),
849                            Token::Function(name) => {
850                                if self.peek_token() == &Token::Delim('(') {
851                                    self.consume_token();
852                                }
853                                let lower_name = name.to_ascii_lowercase();
854                                let pseudo = match lower_name.as_str() {
855                                    "is" | "where" | "not" => PseudoClass::SelectorList {
856                                        name: lower_name,
857                                        selectors: self.parse_selector_list_until(Some(')')),
858                                    },
859                                    "nth-child" | "nth-last-child" | "nth-of-type"
860                                    | "nth-last-of-type" => {
861                                        let tokens = self.consume_until_closing_parenthesis();
862                                        let (a, b) =
863                                            parse_an_plus_b(&tokens).unwrap_or((0, i32::MIN));
864                                        PseudoClass::Nth {
865                                            name: lower_name,
866                                            a,
867                                            b,
868                                        }
869                                    }
870                                    _ => {
871                                        self.consume_until_closing_parenthesis();
872                                        PseudoClass::SelectorList {
873                                            name: lower_name,
874                                            selectors: Vec::new(),
875                                        }
876                                    }
877                                };
878                                if self.peek_token() == &Token::Delim(')') {
879                                    self.consume_token();
880                                }
881                                Some(pseudo)
882                            }
883                            _ => None,
884                        };
885                        if let Some(pseudo_class) = pseudo_class {
886                            let sel = current_selector.get_or_insert_with(|| Selector {
887                                tag: None,
888                                id: None,
889                                classes: vec![],
890                                attributes: vec![],
891                                pseudo_classes: vec![],
892                                pseudo_element: None,
893                            });
894                            sel.pseudo_classes.push(pseudo_class);
895                        }
896                    }
897                }
898
899                Token::Delim('[') => {
900                    self.consume_token();
901                    while matches!(self.peek_token(), Token::Whitespace | Token::Comment(_)) {
902                        self.consume_token();
903                    }
904
905                    let name = match self.consume_token() {
906                        Token::Ident(name) => name,
907                        _ => continue,
908                    };
909
910                    while matches!(self.peek_token(), Token::Whitespace | Token::Comment(_)) {
911                        self.consume_token();
912                    }
913
914                    let value = if self.peek_token() == &Token::Delim('=') {
915                        self.consume_token();
916                        while matches!(self.peek_token(), Token::Whitespace | Token::Comment(_)) {
917                            self.consume_token();
918                        }
919                        match self.consume_token() {
920                            Token::Ident(value) | Token::String(value) => Some(value),
921                            _ => continue,
922                        }
923                    } else {
924                        None
925                    };
926
927                    while matches!(self.peek_token(), Token::Whitespace | Token::Comment(_)) {
928                        self.consume_token();
929                    }
930                    if self.peek_token() == &Token::Delim(']') {
931                        self.consume_token();
932                        let sel = current_selector.get_or_insert_with(|| Selector {
933                            tag: None,
934                            id: None,
935                            classes: vec![],
936                            attributes: vec![],
937                            pseudo_classes: vec![],
938                            pseudo_element: None,
939                        });
940                        sel.attributes.push(AttributeSelector { name, value });
941                    }
942                }
943
944                Token::Whitespace | Token::Comment(_) => {
945                    // descendant combinator
946                    if let Some(sel) = current_selector.take() {
947                        parts.push(SelectorPart {
948                            selector: sel,
949                            combinator: current_combinator.take(),
950                        });
951                    }
952                    if current_combinator.is_none() {
953                        current_combinator = Some(Combinator::Descendant);
954                    }
955                    self.consume_token();
956                }
957
958                Token::Delim('>') => {
959                    if let Some(sel) = current_selector.take() {
960                        parts.push(SelectorPart {
961                            selector: sel,
962                            combinator: current_combinator.take(),
963                        });
964                    }
965                    current_combinator = Some(Combinator::Child);
966                    self.consume_token();
967                }
968
969                Token::Delim('+') | Token::Delim('~') => {
970                    if let Some(sel) = current_selector.take() {
971                        parts.push(SelectorPart {
972                            selector: sel,
973                            combinator: current_combinator.take(),
974                        });
975                    }
976                    current_combinator = Some(if token == Token::Delim('+') {
977                        Combinator::NextSibling
978                    } else {
979                        Combinator::SubsequentSibling
980                    });
981                    self.consume_token();
982                }
983
984                Token::Delim('*') => {
985                    current_selector.get_or_insert_with(|| Selector {
986                        tag: None,
987                        id: None,
988                        classes: vec![],
989                        attributes: vec![],
990                        pseudo_classes: vec![],
991                        pseudo_element: None,
992                    });
993                    self.consume_token();
994                }
995
996                Token::Delim(',') => {
997                    if let Some(sel) = current_selector.take() {
998                        parts.push(SelectorPart {
999                            selector: sel,
1000                            combinator: current_combinator.take(),
1001                        });
1002                    }
1003                    parts.reverse();
1004                    selectors.push(ComplexSelector {
1005                        parts: parts.clone(),
1006                    });
1007                    parts.clear();
1008                    current_combinator = None;
1009                    self.consume_token();
1010
1011                    while matches!(self.peek_token(), Token::Whitespace | Token::Comment(_)) {
1012                        self.consume_token();
1013                    }
1014                }
1015
1016                Token::Delim(')') if terminator == Some(')') => {
1017                    if let Some(sel) = current_selector.take() {
1018                        parts.push(SelectorPart {
1019                            selector: sel,
1020                            combinator: current_combinator.take(),
1021                        });
1022                    }
1023                    if !parts.is_empty() {
1024                        parts.reverse();
1025                        selectors.push(ComplexSelector { parts });
1026                    }
1027                    break;
1028                }
1029
1030                Token::Delim('{') | Token::EOF => {
1031                    if let Some(sel) = current_selector.take() {
1032                        parts.push(SelectorPart {
1033                            selector: sel,
1034                            combinator: current_combinator.take(),
1035                        });
1036                    }
1037                    if !parts.is_empty() {
1038                        parts.reverse();
1039                        selectors.push(ComplexSelector { parts });
1040                    }
1041                    break;
1042                }
1043
1044                // At-keywords cannot occur inside a qualified-rule prelude.
1045                // Leave the token untouched so parse_rule reports the invalid
1046                // prefix and lossy top-level recovery can resume at the at-rule.
1047                Token::AtKeyword(_) => break,
1048
1049                _ => {
1050                    self.consume_token();
1051                }
1052            }
1053        }
1054
1055        selectors
1056    }

1058    /// Consume tokens through the matching `)` of a functional pseudo-class.
1059    fn consume_until_closing_parenthesis(&mut self) -> Vec<Token> {
1060        let mut tokens = Vec::new();
1061        let mut depth = 0;
1062        loop {
1063            match self.peek_token() {
1064                Token::EOF => break,
1065                Token::Delim(')') if depth == 0 => break,
1066                Token::Delim('(') => {
1067                    depth += 1;
1068                    tokens.push(self.consume_token());
1069                }
1070                Token::Delim(')') => {
1071                    depth -= 1;
1072                    tokens.push(self.consume_token());
1073                }
1074                _ => tokens.push(self.consume_token()),
1075            }
1076        }
1077        tokens
1078    }

1080    /// Parse declaration until `Token::Delim('}')`.
1081    fn parse_declaration_list(&mut self) -> ParseResult<Vec<CssNode>> {
1082        let mut declarations = vec![];
1083        let mut parsing_name = true;
1084        let mut name = String::new();
1085        let mut value_tokens = vec![];
1086
1087        loop {
1088            let token = self.peek_token().clone();
1089            match token {
1090                Token::Delim(':') if parsing_name => {
1091                    parsing_name = false;
1092                    self.consume_token();
1093                }
1094                Token::Delim(';') if !parsing_name => {
1095                    self.consume_token(); // consume ;
1096                    declarations.push(CssNode {
1097                        node: CssNodeType::Declaration {
1098                            name: std::mem::take(&mut name),
1099                            value: Self::parse_tokens_to_css_value(std::mem::take(
1100                                &mut value_tokens,
1101                            ))
1102                            .map_err(|e| {
1103                                e.with_context(
1104                                    "parse_declaration: failed to parse declaration value list",
1105                                )
1106                            })?,
1107                        },
1108                        children: vec![],
1109                    });
1110                    parsing_name = true;
1111                }
1112                Token::Delim('}') | Token::EOF => {
1113                    if !parsing_name && !name.is_empty() {
1114                        declarations.push(CssNode {
1115                            node: CssNodeType::Declaration {
1116                                name: std::mem::take(&mut name),
1117                                value: Self::parse_tokens_to_css_value(std::mem::take(
1118                                    &mut value_tokens,
1119                                ))?,
1120                            },
1121                            children: vec![],
1122                        });
1123                    }
1124                    break;
1125                }
1126
1127                Token::Ident(s) if parsing_name => {
1128                    name.push_str(&s);
1129                    self.consume_token();
1130                }
1131                _ => {
1132                    if !parsing_name {
1133                        value_tokens.push(self.consume_token());
1134                    } else {
1135                        self.consume_token(); // skip unsupported token in name
1136                    }
1137                }
1138            }
1139        }
1140
1141        Ok(declarations)
1142    }

1144    pub fn parse_tokens_to_css_value(tokens: Vec<Token>) -> ParseResult<CssValue> {
1145        let mut values = vec![];
1146        let mut iter = tokens.into_iter().peekable();
1147
1148        while let Some(token) = iter.next() {
1149            log::debug!(target: "CssParser", "parse_tokens_to_css_value: token={:?}", token);
1150
1151            match token {
1152                Token::Ident(s) => values.push(CssValue::Keyword(s.into())),
1153
1154                Token::Delim(',') => {
1155                    // List separator
1156                    continue;
1157                }
1158
1159                Token::Delim('(') | Token::Delim(')') => {
1160                    // Function の構文用なので無視
1161                    continue;
1162                }
1163
1164                Token::Delim(c) => {
1165                    let mut s = [0_u8; 4];
1166                    let s = c.encode_utf8(&mut s);
1167
1168                    values.push(CssValue::Keyword(s.into()));
1169                }
1170
1171                Token::Number(n) => values.push(CssValue::Number(n)),
1172
1173                Token::String(s) => values.push(CssValue::String(s)),
1174
1175                Token::Dimension(value, unit) => {
1176                    let unit = match unit.as_str() {
1177                        "px" => Unit::Px,
1178                        "em" => Unit::Em,
1179                        "rem" => Unit::Rem,
1180                        "%" => Unit::Percent,
1181                        "vw" => Unit::Vw,
1182                        "vh" => Unit::Vh,
1183                        "deg" => Unit::Deg,
1184                        "fr" => Unit::Fr,
1185                        _ => Unit::Px,
1186                    };
1187                    values.push(CssValue::Length(value, unit));
1188                }
1189
1190                Token::Hash(s) => values.push(CssValue::Color(s)),
1191
1192                Token::Function(name) => {
1193                    // () の中をそのまま集める
1194                    let mut depth = 0;
1195                    let mut func_tokens = vec![];
1196
1197                    for tok in iter.by_ref() {
1198                        match &tok {
1199                            Token::Delim('(') => {
1200                                depth += 1;
1201                                func_tokens.push(tok);
1202                            }
1203                            Token::Delim(')') => {
1204                                func_tokens.push(tok);
1205                                depth -= 1;
1206                                if depth == 0 {
1207                                    break;
1208                                }
1209                            }
1210                            _ => func_tokens.push(tok),
1211                        }
1212                    }
1213
1214                    let arg_value = Self::parse_tokens_to_css_value(func_tokens)
1215                        .map_err(|e| e.with_context("parse function args"))?;
1216
1217                    let args = match arg_value {
1218                        CssValue::List(list) => list,
1219                        other => vec![other],
1220                    };
1221
1222                    values.push(CssValue::Function(name, args));
1223                }
1224
1225                _ => continue,
1226            }
1227        }
1228
1229        // 複数値なら List、単数ならそのまま
1230        Ok(match values.len() {
1231            0 => CssValue::Keyword(CssIdent::new_static("")),
1232            1 => values.remove(0),
1233            _ => CssValue::List(values),
1234        })
1235    }
1236}

1239impl fmt::Display for CssNode {
1240    fn fmt(&self, f: &mut fmt::Formatter<'_>) -> fmt::Result {
1241        fmt_tree_node(self, f, &[])
1242    }
1243}
1244
1245/// 再帰的にツリーを表示するヘルパー関数
1246fn fmt_tree_node(
1247    node: &CssNode,
1248    f: &mut fmt::Formatter<'_>,
1249    ancestors_last: &[bool],
1250) -> fmt::Result {
1251    let is_last = *ancestors_last.last().unwrap_or(&true);
1252    let connector = if ancestors_last.is_empty() {
1253        ""
1254    } else if is_last {
1255        "└── "
1256    } else {
1257        "├── "
1258    };
1259
1260    let mut prefix = String::new();
1261    for &ancestor_last in &ancestors_last[..ancestors_last.len().saturating_sub(1)] {
1262        prefix.push_str(if ancestor_last { "    " } else { "│   " });
1263    }
1264
1265    writeln!(f, "{}{}{:?}", prefix, connector, node.node())?;
1266
1267    let child_count = node.children().len();
1268    for (i, child) in node.children().iter().enumerate() {
1269        let child_is_last = i == child_count - 1;
1270        let mut new_ancestors = ancestors_last.to_vec();
1271        new_ancestors.push(child_is_last);
1272        fmt_tree_node(child, f, &new_ancestors)?;
1273    }
1274
1275    Ok(())
1276}
