7#[derive(Debug, Clone, Default)]
8pub struct ElementInfo {
9    pub tag_name: String,
10    pub id: Option<String>,
11    pub classes: Vec<String>,
12    pub attributes: Vec<(String, String)>,
13    /// One-based index among element siblings.
14    pub element_index: usize,
15    /// Number of element siblings including this element.
16    pub element_count: usize,
17    /// One-based index among siblings with the same tag name.
18    pub type_index: usize,
19    /// Number of siblings with the same tag name.
20    pub type_count: usize,
21    /// Element siblings preceding this element in document order.
22    pub previous_siblings: Arc<[ElementInfo]>,
23}

34fn matches_an_plus_b(index: usize, a: i32, b: i32) -> bool {
35    let index = index as i32;
36    if a == 0 {
37        return index == b;
38    }
39    let delta = index - b;
40    delta % a == 0 && delta / a >= 0
41}

43impl Selector {
44    /// Matches the non-structural portion of this selector against one element.
45    fn matches_base(&self, element: &ElementInfo) -> bool {
46        // tag
47        if let Some(tag) = &self.tag
48            && tag != &element.tag_name
49        {
50            return false;
51        }
52
53        // id
54        if let Some(expected_id) = &self.id {
55            match element.id.as_deref() {
56                Some(actual_id) if actual_id == expected_id => {}
57                _ => return false,
58            }
59        }
60
61        // class
62        for class in &self.classes {
63            if !element.classes.iter().any(|c| c == class) {
64                return false;
65            }
66        }
67
68        for expected in &self.attributes {
69            let actual = element
70                .attributes
71                .iter()
72                .find(|(name, _)| name.eq_ignore_ascii_case(&expected.name));
73            match (&expected.value, actual) {
74                (None, Some(_)) => {}
75                (Some(expected_value), Some((_, actual_value)))
76                    if actual_value == expected_value => {}
77                _ => return false,
78            }
79        }
80
81        true
82    }
83
84    fn matches_pseudo_classes(
85        &self,
86        chain: &[ElementInfo],
87        cursor: MatchCursor,
88        is_root: bool,
89    ) -> bool {
90        let element = ComplexSelector::element_at(chain, cursor);
91        self.pseudo_classes.iter().all(|pseudo| match pseudo {
92            PseudoClass::Simple(pseudo) => {
93                let has_attribute = |name: &str| {
94                    element
95                        .attributes
96                        .iter()
97                        .any(|(attribute, _)| attribute.eq_ignore_ascii_case(name))
98                };
99                let is_form_control = matches!(
100                    element.tag_name.as_str(),
101                    "button" | "fieldset" | "input" | "optgroup" | "option" | "select" | "textarea"
102                );
103                match pseudo.to_ascii_lowercase().as_str() {
104                    "root" => is_root,
105                    "link" | "any-link" => {
106                        matches!(element.tag_name.as_str(), "a" | "area") && has_attribute("href")
107                    }
108                    "disabled" => is_form_control && has_attribute("disabled"),
109                    "enabled" => is_form_control && !has_attribute("disabled"),
110                    "checked" => {
111                        (element.tag_name == "input" && has_attribute("checked"))
112                            || (element.tag_name == "option" && has_attribute("selected"))
113                    }
114                    "required" => is_form_control && has_attribute("required"),
115                    "optional" => is_form_control && !has_attribute("required"),
116                    "first-child" => element.element_index == 1,
117                    "last-child" => element.element_index == element.element_count,
118                    "only-child" => element.element_count == 1,
119                    "first-of-type" => element.type_index == 1,
120                    "last-of-type" => element.type_index == element.type_count,
121                    "only-of-type" => element.type_count == 1,
122                    // These require browsing history or live interaction state.
123                    "visited" | "active" | "focus" | "focus-visible" | "focus-within" | "hover" => {
124                        false
125                    }
126                    _ => false,
127                }
128            }
129            PseudoClass::SelectorList { name, selectors } => {
130                if selectors.is_empty() {
131                    return false;
132                }
133                let any_matches = selectors
134                    .iter()
135                    .any(|selector| selector.matches_from(chain, cursor, 0));
136                match name.as_str() {
137                    "is" | "where" => any_matches,
138                    "not" => !any_matches,
139                    _ => false,
140                }
141            }
142            PseudoClass::Nth { name, a, b } => {
143                let index = match name.as_str() {
144                    "nth-child" => element.element_index,
145                    "nth-last-child" => element
146                        .element_count
147                        .saturating_add(1)
148                        .saturating_sub(element.element_index),
149                    "nth-of-type" => element.type_index,
150                    "nth-last-of-type" => element
151                        .type_count
152                        .saturating_add(1)
153                        .saturating_sub(element.type_index),
154                    _ => return false,
155                };
156                matches_an_plus_b(index, *a, *b)
157            }
158        })
159    }
160
161    fn matches_at(&self, chain: &[ElementInfo], cursor: MatchCursor) -> bool {
162        let element = ComplexSelector::element_at(chain, cursor);
163        let is_root = cursor.sibling_index.is_none() && cursor.chain_index + 1 == chain.len();
164        if !self.matches_base(element) || !self.matches_pseudo_classes(chain, cursor, is_root) {
165            return false;
166        }
167        if let Some(_pseudo) = &self.pseudo_element {
168            // TODO
169            return false;
170        }
171
172        true
173    }
174}

176impl ComplexSelector {
177    pub fn matches(&self, chain: &[ElementInfo]) -> bool {
178        if chain.is_empty() || self.parts.is_empty() {
179            return false;
180        }
181        self.matches_from(
182            chain,
183            MatchCursor {
184                chain_index: 0,
185                sibling_index: None,
186            },
187            0,
188        )
189    }
190
191    fn element_at(chain: &[ElementInfo], cursor: MatchCursor) -> &ElementInfo {
192        match cursor.sibling_index {
193            Some(index) => &chain[cursor.chain_index].previous_siblings[index],
194            None => &chain[cursor.chain_index],
195        }
196    }
197
198    fn previous_sibling_cursor(chain: &[ElementInfo], cursor: MatchCursor) -> Option<MatchCursor> {
199        let position = cursor
200            .sibling_index
201            .unwrap_or(chain[cursor.chain_index].previous_siblings.len());
202        position.checked_sub(1).map(|sibling_index| MatchCursor {
203            chain_index: cursor.chain_index,
204            sibling_index: Some(sibling_index),
205        })
206    }
207
208    fn matches_from(
209        &self,
210        chain: &[ElementInfo],
211        cursor: MatchCursor,
212        selector_index: usize,
213    ) -> bool {
214        let part = &self.parts[selector_index];
215
216        if !part.selector.matches_at(chain, cursor) {
217            return false;
218        }
219
220        // セレクタが尽きた → 完全一致
221        if selector_index + 1 == self.parts.len() {
222            return true;
223        }
224
225        match part.combinator {
226            Some(Combinator::Descendant) => {
227                for next in (cursor.chain_index + 1)..chain.len() {
228                    if self.matches_from(
229                        chain,
230                        MatchCursor {
231                            chain_index: next,
232                            sibling_index: None,
233                        },
234                        selector_index + 1,
235                    ) {
236                        return true;
237                    }
238                }
239                false
240            }
241            Some(Combinator::Child) => {
242                cursor.chain_index + 1 < chain.len()
243                    && self.matches_from(
244                        chain,
245                        MatchCursor {
246                            chain_index: cursor.chain_index + 1,
247                            sibling_index: None,
248                        },
249                        selector_index + 1,
250                    )
251            }
252            Some(Combinator::NextSibling) => Self::previous_sibling_cursor(chain, cursor)
253                .is_some_and(|previous| self.matches_from(chain, previous, selector_index + 1)),
254            Some(Combinator::SubsequentSibling) => {
255                let mut previous = Self::previous_sibling_cursor(chain, cursor);
256                while let Some(candidate) = previous {
257                    if self.matches_from(chain, candidate, selector_index + 1) {
258                        return true;
259                    }
260                    previous = Self::previous_sibling_cursor(chain, candidate);
261                }
262                false
263            }
264            None => false,
265        }
266    }
267
268    pub fn specificity(&self) -> (u32, u32, u32) {
269        let mut a = 0; // id
270        let mut b = 0; // class / attr / pseudo-class
271        let mut c = 0; // tag / pseudo-element
272
273        for part in &self.parts {
274            let sel = &part.selector;
275
276            if sel.id.is_some() {
277                a += 1;
278            }
279            b += (sel.classes.len() + sel.attributes.len()) as u32;
280            for pseudo in &sel.pseudo_classes {
281                match pseudo {
282                    PseudoClass::Simple(_) | PseudoClass::Nth { .. } => b += 1,
283                    PseudoClass::SelectorList { name, selectors } if name == "where" => {}
284                    PseudoClass::SelectorList { selectors, .. } => {
285                        let nested = selectors
286                            .iter()
287                            .map(ComplexSelector::specificity)
288                            .max()
289                            .unwrap_or_default();
290                        a += nested.0;
291                        b += nested.1;
292                        c += nested.2;
293                    }
294                }
295            }
296            if sel.tag.is_some() {
297                c += 1;
298            }
299            c += u32::from(sel.pseudo_element.is_some());
300        }
301
302        (a, b, c)
303    }
