12/// A single element's identity as the selector matcher sees it.
13///
14/// The string-bearing fields are shared through `Arc` because the same element
15/// is described many times per layout build: once when its parent builds the
16/// sibling list and again inside the ancestor/sibling [`ElementChain`] links.
17/// Cheap `Clone` keeps that repeated description from re-cloning every
18/// attribute string on the hot path.
19#[derive(Debug, Clone, Default)]
20pub struct ElementInfo {
21    pub tag_name: Arc<str>,
22    pub id: Option<Arc<str>>,
23    pub classes: Arc<[Arc<str>]>,
24    pub attributes: Arc<[(Arc<str>, Arc<str>)]>,
25    /// One-based index among element siblings.
26    pub element_index: usize,
27    /// Number of element siblings including this element.
28    pub element_count: usize,
29    /// One-based index among siblings with the same tag name.
30    pub type_index: usize,
31    /// Number of siblings with the same tag name.
32    pub type_count: usize,
33    /// Element siblings preceding this element in document order, nearest
34    /// first. Links are shared through `Arc`, so building it per sibling is
35    /// O(1).
36    pub previous_siblings: ElementChain,
37    /// Whether this element has a parent element.
38    ///
39    /// The tree root is an element with no parent, and structural
40    /// pseudo-classes such as `:first-child` must not match it.
41    pub has_parent: bool,
42    /// Whether a checkbox or radio is currently checked. This is live
43    /// checkedness, not the `checked` content attribute, which is
44    /// `defaultChecked` and does not affect `:checked`.
45    pub checked: bool,
46    /// The language tag in effect for this element: its own `lang` attribute, or
47    /// failing that the nearest ancestor's. `:lang()` matches against this.
48    pub lang: Option<Arc<str>>,
49    /// Whether this element is `:empty` under the Selectors 4 definition:
50    /// no element children, and no text node with non-empty data. Comments and
51    /// processing instructions do not count.
52    pub is_empty: bool,
53    /// Whether this element or an ancestor is under the pointer.
54    pub hovered: bool,
55    /// Whether this element or an ancestor is the active (pressed) target.
56    pub active: bool,
57    /// Identity of the shadow tree this element lives in, if any. Stylesheets
58    /// are scoped by comparing it with the tree that declared them.
59    pub shadow_root_id: Option<u64>,
60    /// The shadow tree whose slot projects this element, when it is light-DOM
61    /// content assigned to a slot (`::slotted()` can reach it).
62    pub slotted_root: Option<u64>,
63}

65/// One link of an [`ElementChain`].
66#[derive(Debug)]
67struct ChainLink {
68    info: ElementInfo,
69    /// Ancestors (parent, grandparent, …), shared with the parent's chain.
70    next: Option<Arc<ChainLink>>,
71    /// Lazily built [`ChainSummary`] covering this link and every link after
72    /// it. Shared links compute it once per layout run.
73    summary: OnceLock<Box<ChainSummary>>,
74}
75
76/// Tag/id/class strings appearing anywhere along a chain, used to reject
77/// descendant and sibling walks without traversing them.
78#[derive(Debug, Default, Clone)]
79struct ChainSummary {
80    tags: HashSet<Box<str>>,
81    ids: HashSet<Box<str>>,
82    classes: HashSet<Box<str>>,
83}
84
85impl ChainSummary {
86    fn collect(start: &ChainLink) -> Box<Self> {
87        // Reuse an already-built summary further down the chain when present.
88        let mut summary = match start.next.as_deref().and_then(|next| next.summary.get()) {
89            Some(cached) => (**cached).clone(),
90            None => {
91                let mut summary = Self::default();
92                let mut link = start.next.as_deref();
93                while let Some(current) = link {
94                    summary.add(&current.info);
95                    link = current.next.as_deref();
96                }
97                summary
98            }
99        };
100        summary.add(&start.info);
101        Box::new(summary)
102    }
103
104    fn add(&mut self, info: &ElementInfo) {
105        self.tags.insert(info.tag_name.as_ref().into());
106        if let Some(id) = &info.id {
107            self.ids.insert(id.as_ref().into());
108        }
109        for class in info.classes.iter() {
110            self.classes.insert(class.as_ref().into());
111        }
112    }
113
114    /// Returns `false` only when no element in the chain can satisfy the
115    /// non-structural parts of `selector`: a required tag, id or class that
116    /// appears nowhere rules every candidate out at once.
117    fn could_match(&self, selector: &Selector) -> bool {
118        if let Some(tag) = &selector.tag
119            && !self.tags.contains(tag.as_str())
120        {
121            return false;
122        }
123        if let Some(id) = &selector.id
124            && !self.ids.contains(id.as_str())
125        {
126            return false;
127        }
128        selector
129            .classes
130            .iter()
131            .all(|class| self.classes.contains(class.as_str()))
132    }
133}

135impl ChainLink {
136    /// Summary of this link and every link after it, built on first use.
137    fn summary(&self) -> &ChainSummary {
138        self.summary.get_or_init(|| ChainSummary::collect(self))
139    }
140}

142/// 右（自分）→ 左（祖先）
143///
144/// Cloning is O(1) and prepending an element is O(1): descendant chains share
145/// their ancestor links through `Arc`.
146#[derive(Debug, Clone, Default)]
147pub struct ElementChain {
148    head: Option<Arc<ChainLink>>,
149}
150
151impl ElementChain {
152    /// Returns this chain extended with `info` as the innermost element.
153    ///
154    /// Passing `None` yields an equivalent chain without allocating.
155    pub fn prepend(&self, info: Option<ElementInfo>) -> Self {
156        match info {
157            Some(info) => Self {
158                head: Some(Arc::new(ChainLink {
159                    info,
160                    next: self.head.clone(),
161                    summary: OnceLock::new(),
162                })),
163            },
164            None => self.clone(),
165        }
166    }
167
168    /// The innermost (current) element.
169    pub fn first(&self) -> Option<&ElementInfo> {
170        self.head.as_deref().map(|link| &link.info)
171    }
172
173    /// Builds a chain from elements ordered innermost-first.
174    pub fn from_vec(elements: Vec<ElementInfo>) -> Self {
175        let mut chain = Self::default();
176        for info in elements.into_iter().rev() {
177            chain = chain.prepend(Some(info));
178        }
179        chain
180    }
181
182    /// Builds a chain by prepending `elements` in iteration order, so the
183    /// last element becomes the nearest link. Feeding siblings in document
184    /// order therefore yields a nearest-first chain.
185    pub fn from_document_order(elements: impl IntoIterator<Item = ElementInfo>) -> Self {
186        let mut chain = Self::default();
187        for info in elements {
188            chain = chain.prepend(Some(info));
189        }
190        chain
191    }
192}

195    /// Parsed arguments of `::slotted(...)` selectors, so the inner selector is
196    /// parsed once per spelling instead of once per matched element.
197    static SLOTTED_SELECTORS: RefCell<HashMap<String, Rc<Vec<ComplexSelector>>>> =
198        RefCell::new(HashMap::new());
199}

201/// Splits a functional pseudo-element such as `slotted(.a)` into its argument,
202/// comparing the function name case-insensitively without allocating.
203fn pseudo_argument<'a>(pseudo: &'a str, name: &str) -> Option<&'a str> {
204    let body = pseudo.strip_suffix(')')?;
205    let prefix = body.get(..name.len() + 1)?;
206    if !prefix[..name.len()].eq_ignore_ascii_case(name) || !prefix.ends_with('(') {
207        return None;
208    }
209    Some(&body[name.len() + 1..])
210}

212/// Builds an element chain (innermost-first) from a chain link.
213fn chain_from_link(link: &ChainLink) -> ElementChain {
214    let mut elements = Vec::new();
215    let mut current = Some(link);
216    while let Some(link) = current {
217        elements.push(link.info.clone());
218        current = link.next.as_deref();
219    }
220    ElementChain::from_vec(elements)
221}

223/// Returns the shadow host of `link` if that element exports `part_name` via
224/// its `part` attribute. The host is the nearest ancestor outside the element's
225/// shadow tree.
226fn exported_part_host<'a>(link: &'a ChainLink, part_name: &str) -> Option<&'a ChainLink> {
227    let exports = link.info.attributes.iter().any(|(name, value)| {
228        name.eq_ignore_ascii_case("part")
229            && value
230                .split_ascii_whitespace()
231                .any(|token| token == part_name)
232    });
233    if !exports {
234        return None;
235    }
236    let shadow_root_id = link.info.shadow_root_id;
237    let mut current = link.next.as_deref();
238    while let Some(ancestor) = current {
239        if ancestor.info.shadow_root_id == shadow_root_id {
240            current = ancestor.next.as_deref();
241            continue;
242        }
243        return Some(ancestor);
244    }
245    None
246}

248/// Matches the selector argument of `::slotted(sel)` against a projected
249/// element. An unparsable argument never matches, so a malformed rule cannot
250/// style unrelated content.
251fn slotted_inner_matches(inner: &str, cursor: MatchCursor<'_>) -> bool {
252    let selectors = SLOTTED_SELECTORS.with(|cache| {
253        let mut cache = cache.borrow_mut();
254        if let Some(selectors) = cache.get(inner) {
255            return Rc::clone(selectors);
256        }
257        let parsed = Rc::new(crate::engine::html::parser::parse_query_selectors(inner));
258        cache.insert(inner.to_string(), Rc::clone(&parsed));
259        parsed
260    });
261    if selectors.is_empty() {
262        return false;
263    }
264    let chain = chain_from_link(cursor.link);
265    selectors.iter().any(|selector| selector.matches(&chain))
266}

276/// Whether an element is an `input` whose type makes it a checkbox or a radio,
277/// the only inputs that can ever be `:checked`.
278fn is_checkable_input(element: &ElementInfo) -> bool {
279    if !element.tag_name.eq_ignore_ascii_case("input") {
280        return false;
281    }
282    match element
283        .attributes
284        .iter()
285        .find(|(name, _)| name.eq_ignore_ascii_case("type"))
286    {
287        Some((_, value)) => {
288            let value = value.trim();
289            value.eq_ignore_ascii_case("checkbox") || value.eq_ignore_ascii_case("radio")
290        }
291        // A missing `type` attribute means `type=text`.
292        None => false,
293    }
294}

296/// `:lang(<language>)` matches when the element's language tag is exactly the
297/// requested one, or a subtag of it: `:lang(en)` matches `en` and `en-GB`, but
298/// not `enx`. Comparison is ASCII case-insensitive, as HTML requires.
299fn matches_lang(element: &ElementInfo, language: &str) -> bool {
300    let Some(tag) = element.lang.as_deref() else {
301        return false;
302    };
303    if tag.eq_ignore_ascii_case(language) {
304        return true;
305    }
306    // Compare the byte prefix safely: `language.len()` may not fall on a char
307    // boundary of `tag`, in which case there is no match rather than a panic.
308    tag.get(..language.len())
309        .is_some_and(|prefix| prefix.eq_ignore_ascii_case(language))
310        && tag.as_bytes().get(language.len()) == Some(&b'-')
311}

313fn matches_an_plus_b(index: usize, a: i32, b: i32) -> bool {
314    // Widen to `i128` so neither `index - b` nor the modulo/division can
315    // overflow for large element indices or extreme `a`/`b` values.
316    let index = index as i128;
317    let a = i128::from(a);
318    let b = i128::from(b);
319    if a == 0 {
320        return index == b;
321    }
322    let delta = index - b;
323    delta % a == 0 && delta / a >= 0
324}

326impl Selector {
327    /// Matches the non-structural portion of this selector against one element.
328    fn matches_base(&self, element: &ElementInfo) -> bool {
329        // tag
330        if let Some(tag) = &self.tag
331            && tag.as_str() != element.tag_name.as_ref()
332        {
333            return false;
334        }
335
336        // id
337        if let Some(expected_id) = &self.id {
338            match element.id.as_deref() {
339                Some(actual_id) if actual_id == expected_id => {}
340                _ => return false,
341            }
342        }
343
344        // class
345        for class in &self.classes {
346            if !element.classes.iter().any(|c| c.as_ref() == class.as_str()) {
347                return false;
348            }
349        }
350
351        for expected in &self.attributes {
352            let actual = element
353                .attributes
354                .iter()
355                .find(|(name, _)| name.eq_ignore_ascii_case(expected.name.as_str()));
356
357            match (&expected.operator, &expected.value, actual) {
358                (AttributeSelectorOperator::Exists, _, Some(_)) => {}
359                (
360                    AttributeSelectorOperator::Equals,
361                    Some(expected_value),
362                    Some((_, actual_value)),
363                ) if actual_value.as_ref() == expected_value.as_str() => {}
364                (
365                    AttributeSelectorOperator::Includes,
366                    Some(expected_value),
367                    Some((_, actual_value)),
368                ) if actual_value
369                    .split_ascii_whitespace()
370                    .any(|value| value == expected_value.as_str()) => {}
371                (
372                    AttributeSelectorOperator::DashMatch,
373                    Some(expected_value),
374                    Some((_, actual_value)),
375                ) if actual_value.as_ref() == expected_value.as_str()
376                    || actual_value
377                        .strip_prefix(expected_value.as_str())
378                        .is_some_and(|rest| rest.starts_with('-')) => {}
379                (
380                    AttributeSelectorOperator::Prefix,
381                    Some(expected_value),
382                    Some((_, actual_value)),
383                ) if actual_value.starts_with(expected_value.as_str()) => {}
384                (
385                    AttributeSelectorOperator::Suffix,
386                    Some(expected_value),
387                    Some((_, actual_value)),
388                ) if actual_value.ends_with(expected_value.as_str()) => {}
389                (
390                    AttributeSelectorOperator::Substring,
391                    Some(expected_value),
392                    Some((_, actual_value)),
393                ) if actual_value.contains(expected_value.as_str()) => {}
394
395                _ => return false,
396            }
397        }
398
399        true
400    }

402    fn matches_pseudo_classes(&self, cursor: MatchCursor<'_>, is_root: bool) -> bool {
403        let element = ComplexSelector::element_at(cursor);
404        self.pseudo_classes.iter().all(|pseudo| match pseudo {
405            PseudoClass::Lang(language) => matches_lang(element, language),
406            PseudoClass::Simple(pseudo) => {
407                let has_attribute = |name: &str| {
408                    element
409                        .attributes
410                        .iter()
411                        .any(|(attribute, _)| attribute.eq_ignore_ascii_case(name))
412                };
413                let is_form_control = matches!(
414                    element.tag_name.as_ref(),
415                    "button" | "fieldset" | "input" | "optgroup" | "option" | "select" | "textarea"
416                );
417                if pseudo.eq_ignore_ascii_case("root") {
418                    is_root
419                } else if pseudo.eq_ignore_ascii_case("defined") {
420                    // Custom element tag names always contain a hyphen
421                    // (HTML spec), so an element with no hyphen is a
422                    // defined standard element. This keeps `:not(:defined)`
423                    // selectors (e.g. reddit's `visibility: hidden` on
424                    // unhydrated custom elements) from matching the whole
425                    // document tree.
426                    !element.tag_name.contains('-')
427                } else if pseudo.eq_ignore_ascii_case("link")
428                    || pseudo.eq_ignore_ascii_case("any-link")
429                {
430                    matches!(element.tag_name.as_ref(), "a" | "area") && has_attribute("href")
431                } else if pseudo.eq_ignore_ascii_case("disabled") {
432                    is_form_control && has_attribute("disabled")
433                } else if pseudo.eq_ignore_ascii_case("enabled") {
434                    is_form_control && !has_attribute("disabled")
435                } else if pseudo.eq_ignore_ascii_case("checked") {
436                    // Only checkboxes and radios are ever `:checked`; the
437                    // `checked` content attribute alone does not qualify,
438                    // because it only sets the default.
439                    is_checkable_input(element) && element.checked
440                        || (element.tag_name.as_ref() == "option" && has_attribute("selected"))
441                } else if pseudo.eq_ignore_ascii_case("required") {
442                    is_form_control && has_attribute("required")
443                } else if pseudo.eq_ignore_ascii_case("optional") {
444                    is_form_control && !has_attribute("required")
445                } else if pseudo.eq_ignore_ascii_case("first-child") {
446                    element.has_parent && element.element_index == 1
447                } else if pseudo.eq_ignore_ascii_case("last-child") {
448                    element.has_parent && element.element_index == element.element_count
449                } else if pseudo.eq_ignore_ascii_case("only-child") {
450                    element.has_parent && element.element_count == 1
451                } else if pseudo.eq_ignore_ascii_case("empty") {
452                    element.is_empty
453                } else if pseudo.eq_ignore_ascii_case("first-of-type") {
454                    element.has_parent && element.type_index == 1
455                } else if pseudo.eq_ignore_ascii_case("last-of-type") {
456                    element.has_parent && element.type_index == element.type_count
457                } else if pseudo.eq_ignore_ascii_case("only-of-type") {
458                    element.has_parent && element.type_count == 1
459                } else if pseudo.eq_ignore_ascii_case("hover") {
460                    element.hovered
461                } else if pseudo.eq_ignore_ascii_case("active") {
462                    element.active
463                } else {
464                    false
465                }
466            }
467            PseudoClass::SelectorList { name, selectors } => {
468                if selectors.is_empty() {
469                    return false;
470                }
471                let any_matches = selectors
472                    .iter()
473                    .any(|selector| selector.matches_from(cursor, 0));
474                match name.as_str() {
475                    "is" | "where" => any_matches,
476                    "not" => !any_matches,
477                    _ => false,
478                }
479            }
480            PseudoClass::Nth { name, a, b } => {
481                let index = match name.as_str() {
482                    "nth-child" => element.element_index,
483                    "nth-last-child" => element
484                        .element_count
485                        .saturating_add(1)
486                        .saturating_sub(element.element_index),
487                    "nth-of-type" => element.type_index,
488                    "nth-last-of-type" => element
489                        .type_count
490                        .saturating_add(1)
491                        .saturating_sub(element.type_index),
492                    _ => return false,
493                };
494                matches_an_plus_b(index, *a, *b)
495            }
496        })
497    }
498
499    fn matches_at(&self, cursor: MatchCursor<'_>, apply_pseudo: bool) -> bool {
500        let element = ComplexSelector::element_at(cursor);
501        let is_root = cursor.sibling_link.is_none() && cursor.link.next.is_none();
502        if !self.matches_base(element) || !self.matches_pseudo_classes(cursor, is_root) {
503            return false;
504        }
505        if !apply_pseudo {
506            return true;
507        }
508        let Some(pseudo) = &self.pseudo_element else {
509            return true;
510        };
511        // ::before / ::after always match their originating element.
512        if pseudo.eq_ignore_ascii_case("before") || pseudo.eq_ignore_ascii_case("after") {
513            return true;
514        }
515        // ::slotted(sel) only matches a light-DOM element that is actually
516        // assigned to a slot of a shadow host and itself matches `sel`.
517        // Elements that are not projected never match, so a rule such as
518        // `::slotted(.badge)` cannot style an unassigned child.
519        if let Some(inner) = pseudo_argument(pseudo, "slotted") {
520            if element.slotted_root.is_none() {
521                return false;
522            }
523            let inner = inner.trim();
524            if inner.is_empty() || inner == "*" {
525                return true;
526            }
527            return slotted_inner_matches(inner, cursor);
528        }
529        // `::part()` was resolved against the host chain before this compound
530        // was reached, and `::shadow` cannot bridge the stylesheet scope.
531        // Every other pseudo-element is not yet supported.
532        false
533    }
534}

536impl ComplexSelector {
537    pub fn matches(&self, chain: &ElementChain) -> bool {
538        self.matches_in_scope(chain, None)
539    }
540
541    /// Matches this selector for a declaration owned by `origin`: `None` for the
542    /// document's stylesheets, `Some(shadow tree id)` for a `<style>` inside
543    /// that shadow root. A stylesheet only styles its own tree; `::slotted()`
544    /// is the one bridge from a shadow root to the light content it projects,
545    /// and `::part()` reaches exported parts from outside.
546    pub fn matches_in_scope(&self, chain: &ElementChain, origin: Option<u64>) -> bool {
547        if self.parts.is_empty() {
548            return false;
549        }
550        let Some(link) = chain.head.as_deref() else {
551            return false;
552        };
553
554        // `::part(name)` is resolved against the part the shadow host exports,
555        // so the subject compound has to match the *host*, not the part element.
556        if let Some(part_name) = self.subject_part_name() {
557            return self.matches_shadow_part(link, part_name);
558        }
559
560        let in_scope = if self.subject_is_slotted() {
561            // `::slotted()` only exists inside a shadow tree's stylesheet, and
562            // then only reaches the light-DOM content that tree projects.
563            origin.is_some() && link.info.slotted_root == origin
564        } else {
565            link.info.shadow_root_id == origin
566        };
567        if !in_scope {
568            return false;
569        }
570
571        self.matches_from(
572            MatchCursor {
573                link,
574                sibling_link: None,
575            },
576            0,
577        )
578    }
579
580    /// The pseudo-element on the subject compound (`parts[0]`), if any.
581    fn subject_pseudo_element(&self) -> Option<&str> {
582        self.parts.first()?.selector.pseudo_element.as_deref()
583    }
584
585    /// Whether the subject compound is a `::slotted()` selector.
586    fn subject_is_slotted(&self) -> bool {
587        self.subject_pseudo_element()
588            .is_some_and(|pseudo| pseudo_argument(pseudo, "slotted").is_some())
589    }
590
591    /// If the subject compound carries `::part(name)`, returns the part name.
592    pub(crate) fn subject_part_name(&self) -> Option<&str> {
593        let name = pseudo_argument(self.subject_pseudo_element()?, "part")?.trim();
594        (!name.is_empty()).then_some(name)
595    }
596
597    /// Matches a `host::part(name)` selector: the tested element must export
598    /// `name`, and the rest of the selector must match the element's shadow
599    /// host (the first ancestor outside the element's shadow tree).
600    fn matches_shadow_part(&self, link: &ChainLink, part_name: &str) -> bool {
601        let Some(host) = exported_part_host(link, part_name) else {
602            return false;
603        };
604        // The host satisfies the compound that carried `::part()`, so the
605        // pseudo-element check is skipped for that compound only.
606        self.matches_from_inner(
607            MatchCursor {
608                link: host,
609                sibling_link: None,
610            },
611            0,
612            false,
613        )
614    }

616    fn element_at(cursor: MatchCursor<'_>) -> &ElementInfo {
617        match cursor.sibling_link {
618            Some(link) => &link.info,
619            None => &cursor.link.info,
620        }
621    }
622
623    fn previous_sibling_cursor(cursor: MatchCursor<'_>) -> Option<MatchCursor<'_>> {
624        let farther = match cursor.sibling_link {
625            Some(link) => link.next.as_deref(),
626            None => cursor.link.info.previous_siblings.head.as_deref(),
627        };
628        farther.map(|link| MatchCursor {
629            link: cursor.link,
630            sibling_link: Some(link),
631        })
632    }

634    fn matches_from(&self, cursor: MatchCursor<'_>, selector_index: usize) -> bool {
635        self.matches_from_inner(cursor, selector_index, true)
636    }
637
638    /// Walks the selector right-to-left. `apply_pseudo` is `false` only for the
639    /// compound whose `::part()` has already been resolved against the host.
640    fn matches_from_inner(
641        &self,
642        cursor: MatchCursor<'_>,
643        selector_index: usize,
644        apply_pseudo: bool,
645    ) -> bool {
646        let part = &self.parts[selector_index];
647
648        if !part.selector.matches_at(cursor, apply_pseudo) {
649            return false;
650        }
651
652        // セレクタが尽きた → 完全一致
653        if selector_index + 1 == self.parts.len() {
654            return true;
655        }
656
657        match part.combinator {
658            Some(Combinator::Descendant) => {
659                let next_selector = &self.parts[selector_index + 1].selector;
660                let Some(ancestors) = cursor.link.next.as_deref() else {
661                    return false;
662                };
663                // Fast-fail: no ancestor carries a required tag/id/class.
664                if !ancestors.summary().could_match(next_selector) {
665                    return false;
666                }
667                let mut ancestor = Some(ancestors);
668                while let Some(link) = ancestor {
669                    if self.matches_from(
670                        MatchCursor {
671                            link,
672                            sibling_link: None,
673                        },
674                        selector_index + 1,
675                    ) {
676                        return true;
677                    }
678                    ancestor = link.next.as_deref();
679                }
680                false
681            }
682            Some(Combinator::Child) => cursor.link.next.as_deref().is_some_and(|parent| {
683                self.matches_from(
684                    MatchCursor {
685                        link: parent,
686                        sibling_link: None,
687                    },
688                    selector_index + 1,
689                )
690            }),
691            Some(Combinator::NextSibling) => Self::previous_sibling_cursor(cursor)
692                .is_some_and(|previous| self.matches_from(previous, selector_index + 1)),
693            Some(Combinator::SubsequentSibling) => {
694                let next_selector = &self.parts[selector_index + 1].selector;
695                let remaining = match cursor.sibling_link {
696                    Some(link) => link.next.as_deref(),
697                    None => cursor.link.info.previous_siblings.head.as_deref(),
698                };
699                // Fast-fail: no preceding sibling carries a required tag/id/class.
700                if !remaining.is_some_and(|siblings| siblings.summary().could_match(next_selector))
701                {
702                    return false;
703                }
704                let mut previous = Self::previous_sibling_cursor(cursor);
705                while let Some(candidate) = previous {
706                    if self.matches_from(candidate, selector_index + 1) {
707                        return true;
708                    }
709                    previous = Self::previous_sibling_cursor(candidate);
710                }
711                false
712            }
713            None => false,
714        }
715    }
716
717    pub fn specificity(&self) -> (u32, u32, u32) {
718        let mut a = 0; // id
719        let mut b = 0; // class / attr / pseudo-class
720        let mut c = 0; // tag / pseudo-element
721
722        for part in &self.parts {
723            let sel = &part.selector;
724
725            if sel.id.is_some() {
726                a += 1;
727            }
728            b += (sel.classes.len() + sel.attributes.len()) as u32;
729            for pseudo in &sel.pseudo_classes {
730                match pseudo {
731                    PseudoClass::Simple(_) | PseudoClass::Nth { .. } | PseudoClass::Lang(_) => {
732                        b += 1
733                    }
734                    PseudoClass::SelectorList { name, selectors } if name == "where" => {}
735                    PseudoClass::SelectorList { selectors, .. } => {
736                        let nested = selectors
737                            .iter()
738                            .map(ComplexSelector::specificity)
739                            .max()
740                            .unwrap_or_default();
741                        a += nested.0;
742                        b += nested.1;
743                        c += nested.2;
744                    }
745                }
746            }
747            if sel.tag.is_some() {
748                c += 1;
749            }
750            c += u32::from(sel.pseudo_element.is_some());
751        }
752
753        (a, b, c)
754    }

1295    fn shadow_dom_css_encapsulation_allows_part_and_slotted() {
1296        // ::part(...) CAN cross the shadow boundary: it names a part the host
1297        // exports, and the rest of the selector must match that host.
1298        let mut el_in_shadow = element("div", &[], 1, 1, 1, 1);
1299        el_in_shadow.shadow_root_id = Some(42);
1300        el_in_shadow.attributes = vec![("part".into(), "foo".into())].into();
1301
1302        let selector = parse_selector("style-host::part(foo)");
1303        assert!(selector.matches(&chain([el_in_shadow.clone(), host("style-host")])));
1304        assert!(!selector.matches(&chain([el_in_shadow.clone(), host("other-host")])));
1305
1306        // Without a shadow host above it, the part is not exported to anyone.
1307        assert!(!parse_selector("::part(foo)").matches(&chain([el_in_shadow])));
