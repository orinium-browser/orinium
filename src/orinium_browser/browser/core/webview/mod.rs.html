95/// JavaScript execution policy for a page.
96///
97/// This drives both the HTML parser's scripting mode (so `<noscript>` fallbacks
98/// are either hidden or shown) and whether the WebView executes scripts.
99#[derive(Debug, Clone, Copy, PartialEq, Eq, Default)]
100pub enum JsPolicy {
101    /// Scripts run and `<noscript>` contents are kept as raw text.
102    #[default]
103    Enabled,
104    /// Scripts are never executed and `<noscript>` fallbacks are shown.
105    Disabled,
106}

110        match value {

112            JsPolicy::Disabled => ScriptingMode::Disabled,
113        }
114    }
115}

239/// A DOM snapshot paired with the tree mutation version it was built from.
240///
241/// The snapshot and its live references stay valid as long as the DOM has not
242/// mutated (`Tree::version()` unchanged). They are shared with layout tasks via
243/// `Arc` instead of being cloned per task.
244struct SnapshotCache {
245    dom_version: u64,
246    snapshot: Arc<DomSnapshot>,
247    dom_refs: Vec<Weak<RefCell<TreeNode<HtmlNodeType>>>>,
248}
249
250fn js_snapshot_from_tree(dom: &DomTree) -> (DomSnapshot, HashMap<usize, u64>) {
251    let mut dom_ids = HashMap::new();
252    let mut next_id = 1u64;
253    dom.traverse(|node| {
254        dom_ids.insert(Rc::as_ptr(node) as usize, next_id);
255        next_id += 1;
256    });
257    (DomSnapshot::from_mirror(&dom.root, &dom_ids), dom_ids)
258}

260impl std::fmt::Debug for SnapshotCache {
261    fn fmt(&self, f: &mut std::fmt::Formatter<'_>) -> std::fmt::Result {
262        f.debug_struct("SnapshotCache")
263            .field("dom_version", &self.dom_version)
264            .field("snapshot", &self.snapshot)
265            .finish()
266    }

384    /// Sets the JavaScript execution policy.
385    ///
386    /// The policy is applied to `<noscript>` parsing on the next document load
387    /// and takes effect immediately for execution: disabling it drops the JS
388    /// runtime, cancels pending script work, and stops running scripts.
389    pub fn set_js_policy(&mut self, policy: JsPolicy) {
390        if self.js_policy == policy {
391            return;
392        }
393        self.js_policy = policy;
394
395        if policy == JsPolicy::Disabled {
396            self.teardown_script_execution();
397        } else if self.js_processor.is_none()
398            && let Some(dom) = self.docment_info.as_ref().map(|info| Rc::clone(&info.dom))
399        {
400            // Re-enabling: install a processor so DOM APIs work again, without
401            // replaying scripts that were skipped while disabled.
402            let document_url = self
403                .docment_info
404                .as_ref()
405                .map(|info| info.document_url.to_string())
406                .unwrap_or_default();
407            let (snapshot, dom_ids) = js_snapshot_from_tree(&dom);
408            let processor = JsProcessor::new(snapshot);
409            processor.send(JsTask::SetDocumentUrl { url: document_url });
410            processor.send(JsTask::SetViewport {
411                width: self.viewport.0,
412                height: self.viewport.1,
413            });
414            processor.send(JsTask::SetLanguage {
415                language: locale::preferred_language(),
416            });
417            self.js_processor = Some(processor);
418            self.js_dom_ids = dom_ids;
419            self.pending_js_tasks = 3;
420        }
421    }

455    pub fn tick(&mut self) -> Vec<WebViewTask> {
456        let mut tasks = Vec::new();
457
458        match self.phase {
459            PagePhase::Init => {
460                let ua_styles = layouter::css_resolver::CssResolver::resolve_with_origin(
461                    &CssParser::new(USER_AGENT_CSS).parse().unwrap(),
462                    layouter::css_resolver::StyleOrigin::UserAgent,
463                );
464                layouter::css_resolver::append_resolved_styles(
465                    Arc::make_mut(&mut self.resolved_styles),
466                    ua_styles,
467                );
468
469                tasks.push(WebViewTask::AskTabHtml);
470
471                self.phase = PagePhase::BeforeHtmlParsing;
472            }
473
474            PagePhase::BeforeHtmlParsing => {}
475
476            PagePhase::HtmlParsed => {
477                // Phase 1: UA.css only layout
478                self.ensure_text_measurer();
479                self.update_layout();
480
481                for (source, url) in std::mem::take(&mut self.pending_images) {
482                    log::info!("Image fetch requested in WebView: url={}", url);
483                    tasks.push(WebViewTask::Fetch {
484                        url,
485                        kind: FetchKind::Image { source },
486                    });
487                }
488
489                for (source, url) in std::mem::take(&mut self.pending_audio) {
490                    log::info!("Audio fetch requested in WebView: url={}", url);
491                    tasks.push(WebViewTask::Fetch {
492                        url,
493                        kind: FetchKind::Audio { source },
494                    });
495                }
496
497                // CSS fetch を要求
498                if self.pending_css_urls.is_empty() {
499                    self.fragment_ready_version = Some(self.layout_requested_version);
500                    self.phase = PagePhase::CssApplied;
501                } else {
502                    for url in &self.pending_css_urls {
503                        log::info!("Fetch requested in WebView: url={}", url);
504                        tasks.push(WebViewTask::Fetch {
505                            url: url.clone(),
506                            kind: FetchKind::Css,
507                        });
508                    }
509
510                    self.phase = PagePhase::CssPending;
511                }
512            }
513
514            PagePhase::CssPending => {
515                // Poll for CSS processor results (Incremental strategy)
516                self.try_apply_css_results();
517            }
518
519            PagePhase::CssProcessing => {
520                // Poll for the single batch result (Batch strategy)
521                self.try_apply_batch_result();
522            }
523
524            PagePhase::CssApplied => {
525                self.advance_classic_scripts(&mut tasks);
526            }
527
528            PagePhase::ScriptApplied => {
529                // 安定状態
530            }
531        }
532
533        self.try_apply_js_results();
534        self.schedule_js_fetches(&mut tasks);
535        self.schedule_dynamic_scripts(&mut tasks);
536        for request in std::mem::take(&mut self.pending_devtools_requests) {
537            tasks.push(WebViewTask::DevToolsRequest {
538                id: request.id,
539                method: request.method,
540                params: request.params,
541            });
542        }
543        self.schedule_dynamic_styles(&mut tasks);
544        self.schedule_dynamic_images(&mut tasks);
545        self.schedule_pending_images(&mut tasks);
546        self.try_apply_decoded_images();
547        self.run_due_js_timers();
548        self.try_apply_layout_results();
549        self.drain_write_backs();
550        self.sync_dom_to_worker();
551
552        tasks
553    }

555    pub fn on_html_fetched(&mut self, html: String, document_url: Url) {
556        log::info!("Fetched HTML: {}", document_url);
557        self.pending_fragment_scroll = document_url.fragment().map(str::to_string);
558        self.fragment_ready_version = None;
559        let parsed = parse_html(&html, document_url, self.js_policy.into());
560
561        self.pending_css_urls = parsed.style_links;
562        self.pending_images = parsed.image_sources;
563        self.pending_audio = parsed.audio_sources;
564        self.classic_scripts = parsed.scripts;
565        self.next_script_index = 0;
566        self.pending_script_fetches.clear();
567        self.non_blocking_scripts_scheduled = false;
568        self.deferred_script_results.clear();
569        self.next_deferred_script_index = 0;
570        self.css_results_expected = self.pending_css_urls.len();
571
572        let mut initial_js_tasks = 0;
573        let mut initial_js_dom_ids = HashMap::new();
574        self.js_processor = if self.js_policy == JsPolicy::Enabled {
575            let (snapshot, dom_ids) = js_snapshot_from_tree(&parsed.dom);
576            initial_js_dom_ids = dom_ids;
577            let processor = JsProcessor::new(snapshot);
578            processor.send(JsTask::SetDocumentUrl {
579                url: parsed.document_url.to_string(),
580            });
581            processor.send(JsTask::SetViewport {
582                width: self.viewport.0,
583                height: self.viewport.1,
584            });
585            processor.send(JsTask::SetLanguage {
586                language: locale::preferred_language(),
587            });
588            initial_js_tasks = 3;
589            Some(processor)
590        } else {
591            None
592        };
593        self.pending_js_tasks = initial_js_tasks;
594        self.js_dom_ids = initial_js_dom_ids;
595        self.js_dom_dirty = false;
596        self.in_flight_timer_version = None;
597        self.pending_js_fetches.clear();
598        self.pending_devtools_requests.clear();
599        self.inspector_ids.borrow_mut().clear();
600        self.pending_dynamic_scripts.clear();
601        self.pending_dynamic_styles.clear();
602        self.pending_dynamic_images.clear();
603
604        let css_base_url = parsed.base_url.clone();
605        let docment_info = DocumentInfo {
606            document_url: parsed.document_url,
607            base_url: parsed.base_url,
608            dom: parsed.dom,
609            title: parsed.title,
610        };
611        self.docment_info = Some(docment_info);
612        self.snapshot_cache = None;
613
614        for inline_css in &parsed.inline_styles {
615            self.queue_css_images(inline_css, &css_base_url);
616            let sheet = CssParser::new(inline_css).parse_lossy();
617            layouter::css_resolver::append_resolved_styles(
618                Arc::make_mut(&mut self.resolved_styles),
619                layouter::css_resolver::CssResolver::resolve(&sheet),
620            );
621        }
622        self.phase = PagePhase::HtmlParsed;
623    }

634    pub fn on_css_fetched_from(&mut self, css: String, stylesheet_url: &Url) {
635        self.queue_css_images(&css, stylesheet_url);
636        self.linked_css.push(css.clone());
637        match self.css_strategy {
638            CssApplicationStrategy::Batch => {
639                self.loaded_css.push(css);
640
641                if self.loaded_css.len() == self.pending_css_urls.len() {
642                    let all_css = std::mem::take(&mut self.loaded_css);
643                    self.css_results_expected = 1;
644                    self.css_results_received = 0;
645                    self.css_processor.process(all_css);
646                    self.phase = PagePhase::CssProcessing;
647                }
648            }
649            CssApplicationStrategy::Incremental => {
650                self.css_processor.process(vec![css]);
651            }
652        }
653    }

667    /// Executes or queues a fetched external classic script by scheduling mode.
668    pub fn on_script_fetched(&mut self, index: usize, source: String) {
669        let Some(execution) = self.pending_script_fetches.get(&index).copied() else {
670            log::warn!("Ignoring unexpected classic script response at index {index}");
671            return;
672        };
673
674        if execution == ClassicScriptExecution::Default && self.next_script_index != index {
675            log::warn!("Ignoring out-of-order blocking script response at index {index}");
676            return;
677        }
678        self.pending_script_fetches.remove(&index);
679
680        match execution {
681            ClassicScriptExecution::Default => {
682                self.next_script_index += 1;
683                self.send_script(&source);
684            }
685            ClassicScriptExecution::Defer => {
686                self.deferred_script_results.insert(index, Some(source));
687            }
688            ClassicScriptExecution::Async => self.send_script(&source),
689        }
690    }

853    fn advance_classic_scripts(&mut self, tasks: &mut Vec<WebViewTask>) {
854        if self.js_policy == JsPolicy::Disabled {
855            self.phase = PagePhase::ScriptApplied;
856            return;
857        }
858
859        self.schedule_non_blocking_scripts(tasks);
860
861        while self.next_script_index < self.classic_scripts.len() {
862            match self.classic_scripts[self.next_script_index].clone() {
863                ClassicScript::Inline(source) => {
864                    self.next_script_index += 1;
865                    self.send_script(&source);
866                }
867                ClassicScript::External { url, execution } => {
868                    if execution != ClassicScriptExecution::Default {
869                        self.next_script_index += 1;
870                        continue;
871                    }
872
873                    let index = self.next_script_index;
874                    if let std::collections::hash_map::Entry::Vacant(entry) =
875                        self.pending_script_fetches.entry(index)
876                    {
877                        entry.insert(ClassicScriptExecution::Default);
878                        tasks.push(WebViewTask::Fetch {
879                            url,
880                            kind: FetchKind::Script { index },
881                        });
882                    }
883                    return;
884                }
885            }
886        }
887
888        self.advance_deferred_scripts();
889    }

892        if self.non_blocking_scripts_scheduled {

914        loop {
915            let Some(index) =
916                (self.next_deferred_script_index..self.classic_scripts.len()).find(|&index| {

918                        self.classic_scripts.get(index),
919                        Some(ClassicScript::External {

969        if self.js_dom_dirty {
970            // A write-back sync is owed. Pausing pokes keeps a timer snapshot
971            // from racing the pending sync; timers resume once it completes.

973        }

1118    /// Builds a snapshot and hands the heavy tree construction to the background.
1119    fn update_layout(&mut self) {
1120        if self.docment_info.is_none() {
1121            return;
1122        }
1123        self.ensure_text_measurer();
1124
1125        let doc_info = self.docment_info.as_ref().unwrap();
1126        let dom_version = doc_info.dom.version();
1127        let (snapshot, dom_refs) = match &self.snapshot_cache {
1128            // The DOM is unchanged since the last snapshot: reuse it instead of
1129            // re-cloning the whole tree (CSS/image relayouts dominate).
1130            Some(cache) if cache.dom_version == dom_version => {
1131                (Arc::clone(&cache.snapshot), cache.dom_refs.clone())
1132            }
1133            _ => {
1134                let (snapshot, dom_refs) = DomSnapshot::from_tree(&doc_info.dom.root);
1135                let snapshot = Arc::new(snapshot);
1136                self.snapshot_cache = Some(SnapshotCache {
1137                    dom_version,
1138                    snapshot: Arc::clone(&snapshot),
1139                    dom_refs: dom_refs.clone(),
1140                });
1141                (snapshot, dom_refs)
1142            }
1143        };
1144        let root = snapshot.roots()[0];
1145
1146        let media_environment =
1147            layouter::css_resolver::MediaEnvironment::new(self.viewport, self.system_color_scheme);
1148        let task = layouter::LayoutTask {
1149            snapshot,
1150            root,
1151            resolved_styles: Arc::clone(&self.resolved_styles),
1152            media_environment,
1153            measurer: self.text_measurer.clone().unwrap(),
1154            system_color_scheme: self.system_color_scheme,
1155            scripting_mode: self.js_policy.into(),
1156            images: self.images.clone(),
1157            audio: self.audio.clone(),
1158            parent: InheritedCss {
1159                text_flow_style: TextFlowStyle {
1160                    font_size: 16.0,
1161                    ..Default::default()
1162                },
1163                ..Default::default()
1164            },
1165            chain: ElementChain::default(),
1166            write_back_sender: Some(self.write_back_tx.clone()),
1167            version: 0,
1168        };
1169        self.layout_dom_refs = dom_refs;
1170        self.layout_requested_version = self.layout_processor.send(task);
1171        self.layout_pending = true;
1172    }

1189    /// Takes completed layout results from the thread and makes them drawable.
1190    fn try_apply_layout_results(&mut self) {
1191        while let Some(result) = self.layout_processor.try_receive() {
1192            let LayoutResult {
1193                layout,
1194                mut info,
1195                version,
1196            } = result;
1197            if version < self.layout_requested_version {
1198                continue;
1199            }
1200            // The builder initializes every node's scroll offset to 0, so a
1201            // rebuilt tree would otherwise drop the scroll position (e.g. the
1202            // viewport change on a window resize). Re-apply the offsets of the
1203            // previous tree before swapping the new one in.
1204            if let Some((_, old_info)) = self.layout_and_info.as_ref() {
1205                let mut scroll_offsets = HashMap::new();
1206                capture_scroll_offsets(old_info, &mut scroll_offsets);
1207                apply_scroll_offsets(&mut info, &scroll_offsets);
1208            }
1209
1210            self.layout_and_info = Some((layout, info));
1211            self.layout_applied_version = version;
1212            self.layout_pending = false;
1213            self.needs_redraw = true;
1214            // The fresh tree has no geometry yet (positioning normally happens
1215            // during draws). Position it right away so input events arriving
1216            // before the next redraw still hit-test against real boxes.
1217            self.position_layout_if_needed();
1218        }

1221    /// Takes completed JS results from the thread and commits them.
1222    ///
1223    /// A result that mutated the DOM carries a snapshot of the thread's mirror;
1224    /// committing it replaces the authoritative tree, re-registers the JS dom
1225    /// id map and triggers a relayout.
1226    fn try_apply_js_results(&mut self) {
1227        let results: Vec<JsTaskResult> = {
1228            let Some(processor) = self.js_processor.as_ref() else {
1229                return;
1230            };
1231            let mut results = Vec::new();
1232            while let Some(result) = processor.try_receive() {
1233                results.push(result);
1234            }
1235            results
1236        };
1237        for result in results {
1238            self.pending_js_tasks = self.pending_js_tasks.saturating_sub(1);
1239            self.pending_js_fetches.extend(result.fetch_requests);
1240            self.pending_devtools_requests
1241                .extend(result.devtools_requests);
1242            self.pending_dynamic_scripts
1243                .extend(result.dynamic_script_requests);
1244            self.pending_dynamic_styles
1245                .extend(result.dynamic_style_requests);
1246            self.pending_dynamic_images
1247                .extend(result.dynamic_image_requests);
1248
1249            if let Some(in_flight) = self.in_flight_timer_version
1250                && result.version >= in_flight
1251            {
1252                // The newest timer poke has been processed (run or superseded),
1253                // so its snapshot can no longer race the write-back sync.
1254                self.in_flight_timer_version = None;
1255            }
1256
1257            let Some(snapshot) = result.dom else {
1258                continue;
1259            };
1260            let Some(info) = self.docment_info.as_mut() else {
1261                continue;
1262            };
1263            // Commit the thread's mirror as the new authoritative tree. The
1264            // rebuilt tree starts with a fresh version, so the cached snapshot
1265            // and live layout references are stale and must be dropped.
1266            let (tree, dom_ids) = snapshot.into_tree();
1267            info.dom = Rc::new(tree);
1268            self.js_dom_ids = dom_ids;
1269            self.snapshot_cache = None;
1270            self.layout_dom_refs.clear();
1271
1272            self.rebuild_styles_and_layout();
1273            self.needs_redraw = true;
1274        }
1275    }

1277    /// Syncs UI-side DOM mutations (write-backs, `update_page`) to the JS thread.
1278    ///
1279    /// Only runs when no JS task is in flight: mid-task the thread's mirror
1280    /// legitimately diverges from the real tree, and committing an in-flight
1281    /// snapshot after this sync would clobber the thread's newer mutations.
1282    /// An in-flight `RunTimers` poke counts as in flight for the same reason:
1283    /// its callback may mutate the mirror and commit a stale snapshot.
1284    fn sync_dom_to_worker(&mut self) {
1285        if !self.js_dom_dirty
1286            || self.pending_js_tasks != 0
1287            || self.in_flight_timer_version.is_some()
1288        {
1289            return;
1290        }
1291        let Some(processor) = self.js_processor.as_ref() else {
1292            return;
1293        };
1294        let Some(doc_info) = self.docment_info.as_ref() else {
1295            return;
1296        };
1297        let snapshot = DomSnapshot::from_mirror(&doc_info.dom.root, &self.js_dom_ids);
1298        processor.send(JsTask::UpdateDom { snapshot });
1299        self.pending_js_tasks += 1;
1300        self.js_dom_dirty = false;
1301    }

1303    /// Applies value write-backs reported by text inputs to the live DOM.
1304    fn drain_write_backs(&mut self) {
1305        let mut applied = false;
1306        while let Ok((node_id, value)) = self.write_back_rx.try_recv() {
1307            if let Some(weak) = self.layout_dom_refs.get(node_id as usize)
1308                && let Some(node) = weak.upgrade()
1309            {
1310                node.borrow_mut().value.set_attr("value", value);
1311                applied = true;
1312            }
1313            self.needs_redraw = true;
1314        }
1315
1316        // The DOM mutated, so the cached snapshot is stale and must be rebuilt
1317        // on the next relayout. Future JS mutations must also bump the version.
1318        if applied && let Some(doc_info) = self.docment_info.as_mut() {
1319            doc_info.dom.mark_dirty();
1320        }
1321        // The JS thread's mirror must reflect the new input value; synced once
1322        // the in-flight JS tasks have all been applied.
1323        if applied {
1324            self.js_dom_dirty = true;
1325        }
1326    }

1397    /// Runs the box-positioning pass over the current layout tree unless it
1398    /// has already been positioned for the current version + viewport.
1399    ///
1400    /// Background layout results arrive unpositioned (geometry is computed on
1401    /// the main thread), so this must run before the tree is used for anything
1402    /// geometry-sensitive — drawing, but crucially also hit-testing. Without
1403    /// the eager call in [`WebView::try_apply_layout_results`], a click landing
1404    /// between a result being applied and the next draw would walk boxes with
1405    /// no geometry and find nothing.
1406    fn position_layout_if_needed(&mut self) {
1407        let viewport = self.viewport;
1408        if self.positioned_layout == Some((self.layout_applied_version, viewport)) {
1409            return;
1410        }
1411        let Some((layout, info)) = self.layout_and_info.as_mut() else {
1412            return;
1413        };
1414
1415        ui_layout::LayoutEngine::layout(layout, viewport.0, viewport.1);
1416        if layouter::constrain_auto_grid_track_items(layout) {
1417            ui_layout::LayoutEngine::layout(layout, viewport.0, viewport.1);
1418        }
1419        layouter::correct_atomic_inline_spacing_with_info(layout, info);
1420        layouter::align_table_columns(layout, info);
1421        layouter::refresh_missing_text_layout_results(layout, info, viewport);
1422
1423        self.positioned_layout = Some((self.layout_applied_version, viewport));
1424    }

1426    pub fn relayout(&mut self, viewport: (f32, f32)) {
1427        if self.viewport != viewport {
1428            self.viewport = viewport;
1429            if let Some(processor) = self.js_processor.as_ref() {
1430                processor.send(JsTask::SetViewport {
1431                    width: viewport.0,
1432                    height: viewport.1,
1433                });
1434                self.pending_js_tasks += 1;
1435            }
1436            self.update_layout();
1437        }
1438        self.position_layout_if_needed();
1439
1440        let fragment_target =
1441            if fragment_layout_is_ready(self.fragment_ready_version, self.layout_applied_version) {
1442                self.pending_fragment_scroll
1443                    .as_deref()
1444                    .and_then(|fragment| {
1445                        find_fragment_target_dom_id(&self.layout_dom_refs, fragment)
1446                    })
1447            } else {
1448                None
1449            };
1450
1451        let Some((layout, info)) = self.layout_and_info.as_mut() else {
1452            return;
1453        };
1454
1455        if fragment_target
1456            .is_some_and(|target| apply_fragment_scroll(layout, info, target, viewport.1))
1457        {
1458            self.pending_fragment_scroll = None;
1459            self.needs_redraw = true;
1460        }
1461
1462        let layout_metrics =
1463            collect_js_layout_metrics(layout, info, &self.layout_dom_refs, &self.js_dom_ids);
1464        if let Some(processor) = self.js_processor.as_ref() {
1465            processor.send(JsTask::SetLayoutMetrics {
1466                metrics: layout_metrics,
1467            });
1468            self.pending_js_tasks += 1;
1469        }
1470    }

1517/// Builds the geometry snapshot used by DOM measurement APIs from the same
1518/// layout boxes and scroll offsets consumed by painting and hit testing.
1519fn collect_js_layout_metrics(
1520    layout: &LayoutNode,
1521    info: &InfoNode,
1522    dom_refs: &[Weak<RefCell<TreeNode<HtmlNodeType>>>],
1523    js_dom_ids: &HashMap<usize, u64>,
1524) -> HashMap<u64, JsLayoutMetrics> {
1525    let mut metrics = HashMap::new();
1526    collect_js_layout_metrics_inner(
1527        layout,
1528        info,
1529        dom_refs,
1530        js_dom_ids,
1531        (0.0, 0.0),
1532        (0.0, 0.0),
1533        &mut metrics,
1534    );
1535    metrics
1536}
1537
1538fn collect_js_layout_metrics_inner(
1539    layout: &LayoutNode,
1540    info: &InfoNode,
1541    dom_refs: &[Weak<RefCell<TreeNode<HtmlNodeType>>>],
1542    js_dom_ids: &HashMap<usize, u64>,
1543    parent_content_origin: (f32, f32),
1544    inherited_scroll: (f32, f32),
1545    metrics: &mut HashMap<u64, JsLayoutMetrics>,
1546) {
1547    let is_fixed = layout.style.position.kind == ui_layout::Position::Fixed;
1548    let effective_scroll = if is_fixed {
1549        (0.0, 0.0)
1550    } else {
1551        inherited_scroll
1552    };
1553    let own_scroll = info.kind.scroll_offsets();
1554    let child_scroll = if is_fixed {
1555        own_scroll
1556    } else {
1557        (
1558            inherited_scroll.0 + own_scroll.0,
1559            inherited_scroll.1 + own_scroll.1,
1560        )
1561    };
1562
1563    let boxes: Vec<_> = layout.layout_box.iter().collect();
1564    if let Some(first) = boxes.first() {
1565        let mut page_left = parent_content_origin.0 + first.border_box.x;
1566        let mut page_top = parent_content_origin.1 + first.border_box.y;
1567        let mut page_right = page_left + first.border_box.width;
1568        let mut page_bottom = page_top + first.border_box.height;
1569        for model in boxes.iter().skip(1) {
1570            let left = parent_content_origin.0 + model.border_box.x;
1571            let top = parent_content_origin.1 + model.border_box.y;
1572            page_left = page_left.min(left);
1573            page_top = page_top.min(top);
1574            page_right = page_right.max(left + model.border_box.width);
1575            page_bottom = page_bottom.max(top + model.border_box.height);
1576        }
1577
1578        if let Some(node) = info
1579            .dom_id
1580            .and_then(|id| dom_refs.get(id as usize))
1581            .and_then(Weak::upgrade)
1582        {
1583            let node_key = Rc::as_ptr(&node) as usize;
1584            if let Some(dom_id) = js_dom_ids.get(&node_key).copied() {
1585                metrics.insert(
1586                    dom_id,
1587                    JsLayoutMetrics {
1588                        offset_left: first.border_box.x as f64,
1589                        offset_top: first.border_box.y as f64,
1590                        offset_width: (page_right - page_left) as f64,
1591                        offset_height: (page_bottom - page_top) as f64,
1592                        client_width: first.padding_box.width as f64,
1593                        client_height: first.padding_box.height as f64,
1594                        rect_left: (page_left - effective_scroll.0) as f64,
1595                        rect_top: (page_top - effective_scroll.1) as f64,
1596                        rect_width: (page_right - page_left) as f64,
1597                        rect_height: (page_bottom - page_top) as f64,
1598                    },
1599                );
1600            }
1601        }
1602
1603        // TODO: Apply CSS transforms and sticky-position paint offsets to DOMRect geometry.
1604        let child_origin = (
1605            parent_content_origin.0 + first.content_box.x,
1606            parent_content_origin.1 + first.content_box.y,
1607        );
1608        for (child_layout, child_info) in layout.children.iter().zip(&info.children) {
1609            if let Some(child_layout) = child_layout.node() {
1610                collect_js_layout_metrics_inner(
1611                    child_layout,
1612                    child_info,
1613                    dom_refs,
1614                    js_dom_ids,
1615                    child_origin,
1616                    child_scroll,
1617                    metrics,
1618                );
1619            }
1620        }
1621    }
1622}

1624/// Records the nonzero scroll offsets of every scrollable node in `info`,
1625/// keyed by the node's DOM snapshot id.
1626///
1627/// The layout builder initializes each node's `scroll_offset` to 0, so a
1628/// rebuild would otherwise drop the scroll position (e.g. after a window
1629/// resize). `dom_id` stays stable across rebuilds while the DOM is unchanged,
1630/// which makes it a reliable key for restoring state onto the new tree.
1631fn capture_scroll_offsets(info: &InfoNode, offsets: &mut HashMap<NodeId, (f32, f32)>) {
1632    let (x, y) = info.kind.scroll_offsets();
1633    if (x != 0.0 || y != 0.0)
1634        && let Some(dom_id) = info.dom_id
1635    {
1636        offsets.insert(dom_id, (x, y));
1637    }
1638    for child in &info.children {
1639        capture_scroll_offsets(child, offsets);
1640    }
1641}
1642
1643/// Restores scroll offsets captured by [`capture_scroll_offsets`] onto a newly
1644/// built tree.
1645///
1646/// Offsets are copied verbatim for every matching node regardless of its
1647/// `scroll_x`/`scroll_y` flags: the flags describe whether an axis *can*
1648/// scroll, not whether a scroll position was captured, so gating on them here
1649/// would drop positions (e.g. the viewport/page scroll carried by the root).
1650fn apply_scroll_offsets(info: &mut InfoNode, offsets: &HashMap<NodeId, (f32, f32)>) {
1651    if let Some((x, y)) = info.dom_id.and_then(|id| offsets.get(&id)) {
1652        match &mut info.kind {
1653            NodeKind::Container {
1654                scroll_offset_x,
1655                scroll_offset_y,
1656                ..
1657            }
1658            | NodeKind::Custom {
1659                scroll_offset_x,
1660                scroll_offset_y,
1661                ..
1662            } => {
1663                *scroll_offset_x = *x;
1664                *scroll_offset_y = *y;
1665            }
1666            _ => {}
1667        }
1668    }
1669    for child in &mut info.children {
1670        apply_scroll_offsets(child, offsets);
1671    }
1672}

1700fn fragment_target_y(
1701    layout: &LayoutNode,
1702    info: &InfoNode,
1703    target: NodeId,
1704    parent_content_y: f32,
1705) -> Option<f32> {
1706    let model = layout.layout_box.iter().next();
1707    if info.dom_id == Some(target) {
1708        return model.map(|model| parent_content_y + model.border_box.y);
1709    }
1710    let child_content_y =
1711        parent_content_y + model.as_ref().map_or(0.0, |model| model.content_box.y);
1712    layout
1713        .children
1714        .iter()
1715        .zip(&info.children)
1716        .find_map(|(layout_child, info_child)| {
1717            let LayoutChild::Node(layout_child) = layout_child else {
1718                return None;
1719            };
1720            fragment_target_y(layout_child, info_child, target, child_content_y)
1721        })
1722}

1724fn set_first_vertical_scroll_offset(
1725    layout: &LayoutNode,
1726    info: &mut InfoNode,
1727    target_y: f32,
1728    viewport_height: f32,
1729) -> bool {
1730    if let Some(model) = layout.layout_box.iter().next() {
1731        let offset = match &mut info.kind {
1732            NodeKind::Container {
1733                scroll_y: true,
1734                scroll_offset_y,
1735                ..
1736            }
1737            | NodeKind::Custom {
1738                scroll_y: true,
1739                scroll_offset_y,
1740                ..
1741            } => Some(scroll_offset_y),
1742            _ => None,
1743        };
1744        if let Some(offset) = offset {
1745            let max_scroll = (model.children_box.height
1746                - model.content_box.height.min(viewport_height))
1747            .max(0.0);
1748            *offset = target_y.clamp(0.0, max_scroll);
1749            return true;
1750        }
1751    }
1752
1753    layout
1754        .children
1755        .iter()
1756        .zip(&mut info.children)
1757        .any(|(layout_child, info_child)| {
1758            let LayoutChild::Node(layout_child) = layout_child else {
1759                return false;
1760            };
1761            set_first_vertical_scroll_offset(layout_child, info_child, target_y, viewport_height)
1762        })
1763}

1765fn collect_css_image_sources(css: &str) -> Vec<String> {
1766    fn collect_value(value: &CssValue, sources: &mut Vec<String>) {
1767        match value {
1768            CssValue::Function(name, arguments) if name.eq_ignore_ascii_case("url") => {
1769                if let Some(source) = arguments.iter().find_map(|argument| match argument {
1770                    CssValue::String(source) => Some(source.clone()),
1771                    CssValue::Keyword(source) => Some(source.to_string()),
1772                    _ => None,
1773                }) && !source.is_empty()
1774                    && !sources.contains(&source)
1775                {
1776                    sources.push(source);
1777                }
1778            }
1779            CssValue::Function(_, arguments) | CssValue::List(arguments) => {
1780                for argument in arguments {
1781                    collect_value(argument, sources);
1782                }
1783            }
1784            _ => {}
1785        }
1786    }
1787
1788    fn visit(node: &CssNode, sources: &mut Vec<String>) {
1789        if let CssNodeType::Declaration { name, value } = node.node()
1790            && matches!(
1791                name.to_ascii_lowercase().as_str(),
1792                "background" | "background-image"
1793            )
1794        {
1795            collect_value(value, sources);
1796        }
1797        for child in node.children() {
1798            visit(child, sources);
1799        }
1800    }
1801
1802    let stylesheet = CssParser::new(css).parse_lossy();
1803    let mut sources = Vec::new();
1804    visit(&stylesheet, &mut sources);
1805    sources
1806}

1808fn parse_html(html: &str, document_url: Url, scripting_mode: ScriptingMode) -> ParsedDocument {
1809    // --- DOM パース ---
1810    let mut parser = HtmlParser::new(html).with_scripting_mode(scripting_mode);
1811    let dom = Rc::new(parser.parse());
1812
1813    // --- base_url ---
1814    let base_url = dom
1815        .find_all(|n| n.tag_name() == Some("base"))
1816        .iter()
1817        .filter_map(|node_ref| {
1818            let html_node = &node_ref.borrow().value;
1819            let href = html_node.get_attr("href")?;
1820            document_url.join(href).ok()
1821        })
1822        .next()
1823        .unwrap_or_else(|| document_url.clone());
1824
1825    // --- title 抽出 ---
1826    let title = dom
1827        .collect_text_by_tag("title")
1828        .first()
1829        .cloned()
1830        .unwrap_or("".into());
1831
1832    // --- Style links ---
1833    // <link rel="stylesheet" href="...">
1834    let link_nodes = dom.find_all(|n| n.tag_name() == Some("link"));
1835    let mut style_links = Vec::new();
1836
1837    for node in link_nodes {
1838        let (rel, href) = {
1839            let node_ref = node.borrow();
1840            let html_node = &node_ref.value;
1841
1842            let rel = html_node.get_attr("rel").map(|s| s.to_string());
1843            let href = html_node.get_attr("href").map(|s| s.to_string());
1844            (rel, href)
1845        };
1846
1847        if let (Some(rel), Some(href)) = (rel, href)
1848            && rel == "stylesheet"
1849        {
1850            let css_url = match resolve_url(&base_url, &href) {
1851                Ok(url) => url,
1852                Err(_) => continue,
1853            };
1854            style_links.push(css_url);
1855        }
1856    }
1857
1858    // --- Inline styles ---
1859    let inline_styles = dom.collect_text_by_tag("style");
1860
1861    // --- Classic scripts ---
1862    let scripts = dom
1863        .collect_classic_script_descriptors()
1864        .into_iter()
1865        .filter_map(|script| match script.source {
1866            ClassicScriptSource::Inline(source) => Some(ClassicScript::Inline(source)),
1867            ClassicScriptSource::External(source) => {
1868                resolve_url(&base_url, &source)
1869                    .ok()
1870                    .map(|url| ClassicScript::External {
1871                        url,
1872                        execution: script.execution,
1873                    })
1874            }
1875        })
1876        .collect();
1877
1878    let image_sources = dom
1879        .get_elements_by_tag_name("img")
1880        .into_iter()
1881        .filter_map(|node| {
1882            let source = node.borrow().value.get_attr("src")?.to_string();
1883            let url = resolve_url(&base_url, &source).ok()?;
1884            Some((source, url))
1885        })
1886        .collect();
1887
1888    let audio_sources = dom
1889        .get_elements_by_tag_name("audio")
1890        .into_iter()
1891        .filter_map(|node| {
1892            let source = {
1893                let audio = node.borrow();
1894                audio.value.get_attr("src").map(str::to_string).or_else(|| {
1895                    audio.children().iter().find_map(|child| {
1896                        let child = child.borrow();
1897                        (child.value.tag_name() == Some("source"))
1898                            .then(|| child.value.get_attr("src").map(str::to_string))
1899                            .flatten()
1900                    })
1901                })
1902            }?;
1903            let url = resolve_url(&base_url, &source).ok()?;
1904            Some((source, url))
1905        })
1906        .collect();
1907
1908    ParsedDocument {
1909        document_url,
1910        base_url,
1911        dom,
1912        title,
1913        style_links,
1914        inline_styles,
1915        image_sources,
1916        audio_sources,
1917        scripts,
1918    }

1938    /// Drives `tick()` until `done` holds, or panics after a timeout.
1939    ///
1940    /// JS runs on a background thread, so effects (DOM commits, relayouts) are
1941    /// observed a few ticks after the task that produced them was sent.
1942    fn pump_until(webview: &mut WebView, mut done: impl FnMut(&mut WebView) -> bool, why: &str) {
1943        let deadline = Instant::now() + Duration::from_secs(5);
1944        while !done(webview) {
1945            assert!(Instant::now() < deadline, "timed out waiting for {why}");
1946            webview.tick();
1947            std::thread::sleep(Duration::from_millis(1));
1948        }
1949    }

1951    /// Drives `tick()` collecting tasks until `done` accepts one, then returns it.
1952    fn pump_for_task(
1953        webview: &mut WebView,
1954        mut done: impl FnMut(&WebViewTask) -> bool,
1955        why: &str,
1956    ) -> WebViewTask {
1957        let deadline = Instant::now() + Duration::from_secs(5);
1958        loop {
1959            let tasks = webview.tick();
1960            if let Some(task) = tasks.into_iter().find(&mut done) {
1961                return task;
1962            }
1963            assert!(Instant::now() < deadline, "timed out waiting for {why}");
1964            std::thread::sleep(Duration::from_millis(1));
1965        }
1966    }
