1//! ブラウザの描画機能。タブと chrome の描画リクエストを DrawCommand に変換し、
2//! プラットフォームの GPU レンダラへ送る。
3
4use crate::engine::renderer_model::{
5    self, AffineTransform, DrawCommand, FillRule, Rect, rect_path,
6};
7use crate::platform::renderer::gpu::GpuRenderer;
8
9use super::{BasicChrome, Chrome, RenderState};
10use crate::browser::core::tab::Tab;
11
12/// BrowserRenderer は実際の描画を担当する。
13///
14/// 責務:
15/// - アクティブタブのページとブラウザ chrome から DrawCommand を生成する
16/// - DrawCommand をプラットフォームの GPU レンダラへ渡し、描画を実行する
17/// - ウィンドウのサイズ・スケール・タイトルなどの描画状態を保持する
18///
19/// BrowserApp / BrowserUi は直接の描画実装を持たず、このレンダラへ処理を委譲する。
20#[derive(Debug)]
21pub struct BrowserRenderer {
22    /// ウィンドウの描画状態（DrawCommand、サイズ、スケール、タイトル）。
23    pub render_state: RenderState,
24    /// ブラウザ chrome（ツールバーなどの UI 描画）。既定は [`BasicChrome`]。
25    pub chrome: Box<dyn Chrome>,
26}
27
28impl Default for BrowserRenderer {
29    fn default() -> Self {
30        Self::new()
31    }
32}
33
34impl BrowserRenderer {
35    /// 既定の [`BasicChrome`] でレンダラを生成する。
36    pub fn new() -> Self {
37        Self::with_chrome(Box::new(BasicChrome::new()))
38    }
39
40    /// 任意の chrome 実装でレンダラを生成する。
41    pub fn with_chrome(chrome: Box<dyn Chrome>) -> Self {
42        Self {
43            render_state: RenderState::default(),
44            chrome,
45        }
46    }
47
48    /// ウィンドウの初期サイズ・スケール・タイトルでレンダラを生成する。
49    pub fn with_window(window_size: (u32, u32), scale_factor: f64, window_title: String) -> Self {
50        Self {
51            render_state: RenderState::new(window_size, scale_factor, window_title),
52            chrome: Box::new(BasicChrome::new()),
53        }
54    }
55
56    /// ウィンドウのサイズ・スケール・タイトルを設定する。
57    pub fn set_window(&mut self, window_size: (u32, u32), scale_factor: f64, window_title: String) {
58        self.render_state.window_size = window_size;
59        self.render_state.scale_factor = scale_factor;
60        self.render_state.window_title = window_title;
61    }
62
63    /// 指定されたアクティブタブのレイアウトを更新し、ページと chrome の
64    /// DrawCommand を再生成する。
65    pub fn rebuild(&mut self, tabs: &mut [Tab], active_tab: usize) {
66        let (width, height) = self.render_state.viewport();
67        let Rect {
68            x,
69            y,
70            width: content_width,
71            height: content_height,
72        } = self.chrome.content_rect(width, height);
73
74        // Reuse allocation
75        let mut draw_commands = std::mem::take(&mut self.render_state.draw_commands);
76        draw_commands.clear();
77
78        // Page area: below the chrome, clipped so page content never overlaps it.
79        draw_commands.push(DrawCommand::PushClip {
80            path: rect_path(x, y, content_width, content_height),
81            rule: FillRule::NonZero,
82        });
83        draw_commands.push(DrawCommand::PushTransform {
84            transform: AffineTransform::translate(x, y),
85        });
86
87        let title = if let Some(tab) = tabs.get_mut(active_tab) {
88            tab.relayout((content_width, content_height));
89
90            // Keep the chrome in sync with the active tab.
91            let url = tab.document_url().map(|url| url.to_string());
92            self.chrome.sync_url(url.as_deref());
93
94            if let Some((layout, info)) = tab.layout_and_info() {
95                self.chrome.debug_set_layout_node(layout);
96
97                renderer_model::generate_draw_commands(
98                    &mut draw_commands,
99                    layout,
100                    info,
101                    (content_width, content_height),
102                );
103                tab.clear_redraw_flag();
104                tab.title()
105            } else {
106                log::debug!("No layout/info available for tab {}", active_tab);
107                None
108            }
109        } else {
110            None
111        };
112
113        draw_commands.push(DrawCommand::PopTransform);
114        draw_commands.push(DrawCommand::PopClip);
115
116        // Chrome drawn on top of the page area.
117        self.chrome.draw(&mut draw_commands, width, height);
118
119        // Return reused buffer
120        self.render_state.draw_commands = draw_commands;
121
122        if let Some(title) = title {
123            self.render_state.window_title = title;
124        }
125    }
126
127    /// 現在の DrawCommand を GPU レンダラへ送る。
128    pub fn apply_draw_commands(&self, gpu: &mut GpuRenderer) {
129        gpu.parse_draw_commands(&self.render_state.draw_commands);
130    }
131
132    /// DrawCommand を再生成して GPU に送り、実際の描画を実行する。
133    pub fn redraw(&mut self, tabs: &mut [Tab], active_tab: usize, gpu: &mut GpuRenderer) {
134        self.rebuild(tabs, active_tab);
135        self.apply_draw_commands(gpu);
136        if let Err(e) = gpu.render() {
137            log::error!(target: "BrowserRenderer::redraw", "Render error occurred: {}", e);
138        }
139    }
140}
