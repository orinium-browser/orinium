34/// Render state for a browser window.
35#[derive(Debug, Clone)]
36pub struct RenderState {
37    /// List of draw commands generated from the layout engine.
38    pub draw_commands: Vec<DrawCommand>,
39    /// Current window size in pixels (width, height).
40    pub window_size: (u32, u32),
41    /// Current scale factor (for HiDPI displays).
42    pub scale_factor: f64,
43    /// Current window title.
44    pub window_title: String,
45}
46
47impl Default for RenderState {
48    fn default() -> Self {
49        Self {
50            draw_commands: Vec::new(),
51            window_size: (1280, 800),
52            scale_factor: 1.0,
53            window_title: String::new(),
54        }
55    }
56}

59    /// Creates a new `RenderState` with the specified size, scale factor, and title.
60    pub fn new(window_size: (u32, u32), scale_factor: f64, window_title: String) -> Self {
61        Self {
62            draw_commands: Vec::new(),
63            window_size,
64            scale_factor,
65            window_title,

69    /// Calculates the viewport dimensions in scaled logical pixels.
70    pub fn viewport(&self) -> (f32, f32) {
71        let sf = self.scale_factor as f32;
72        (
73            self.window_size.0 as f32 / sf,
74            self.window_size.1 as f32 / sf,
75        )
76    }

233    /// Ticks all tabs and collects fetch requests and redraw demands.
234    pub(crate) fn tick(&mut self) -> BrowserUiTick {
235        let mut fetches = Vec::new();
236        let mut needs_redraw = false;
237
238        let tab_count = self.tabs.len();
239        for tab_id in 0..tab_count {
240            let Some(tab) = self.tabs.get_mut(tab_id) else {
241                continue;
242            };
243            for task in tab.tick() {
244                match task {
245                    TabTask::Fetch { url, kind } => {
246                        log::info!("Fetch requested in BrowserUi: url={}", url);
247                        fetches.push(FetchRequest { tab_id, url, kind });
248                    }
249                    TabTask::NeedsRedraw => {
250                        needs_redraw = true;
251                    }
252                }
253            }
254        }
255
256        BrowserUiTick {
257            fetches,
258            needs_redraw,
259        }
260    }

262    /// Delivers a fetched resource to the target tab.
263    pub(crate) fn deliver_fetch(
264        &mut self,
265        tab_id: usize,
266        kind: FetchKind,
267        url: Url,
268        response: Result<BrowserResponse, BrowserNetworkError>,
269    ) {
270        let Some(tab) = self.tabs.get_mut(tab_id) else {
271            log::warn!("There is no Tab called id={}", tab_id);
272            return;
273        };
274
275        match response {
276            Ok(resp) => {
277                log::info!("Fetch Done in BrowserUi for tab_id={}", tab_id);
278
279                match kind {
280                    FetchKind::Html => {
281                        let html = String::from_utf8_lossy(&resp.body).to_string();
282                        tab.on_fetch_succeeded_html(html);
283                    }
284                    FetchKind::Css => {
285                        let css = String::from_utf8_lossy(&resp.body).to_string();
286                        tab.on_fetch_succeeded_css_from(css, &url);
287                    }
288                    FetchKind::Script { index } => {
289                        let source = String::from_utf8_lossy(&resp.body).to_string();
290                        tab.on_fetch_succeeded_script(index, source);
291                    }
292                    FetchKind::DynamicScript { node_id } => {
293                        let source = String::from_utf8_lossy(&resp.body).to_string();
294                        tab.on_fetch_succeeded_dynamic_script(node_id, source);
295                    }
296                    FetchKind::DynamicCss { node_id } => {
297                        let source = String::from_utf8_lossy(&resp.body).to_string();
298                        tab.on_fetch_succeeded_dynamic_style(node_id, source);
299                    }
300                    FetchKind::Image { source } => {
301                        tab.on_fetch_succeeded_image(source, &resp.body);
302                    }
303                    FetchKind::Audio { source } => {
304                        tab.on_fetch_succeeded_audio(source, &resp.body);
305                    }
306                    FetchKind::JavaScript { request_id, .. } => {
307                        let redirected = resp.url != url.as_str();
308                        tab.on_fetch_succeeded_js(request_id, resp, redirected);
309                    }
310                }
311            }
312            Err(err) => {
313                log::error!("NetworkError: {}", err);
314                match kind {
315                    FetchKind::Image { .. } | FetchKind::Audio { .. } => {
316                        log::warn!("Media fetch failed without aborting page load: {}", url);
317                    }
318                    FetchKind::Script { index } => {
319                        log::warn!("Classic script fetch failed without aborting page load: {url}");
320                        tab.on_fetch_failed_script(index);
321                    }
322                    FetchKind::DynamicScript { node_id } => {
323                        log::warn!("Dynamic script fetch failed without aborting page load: {url}");
324                        tab.on_fetch_failed_dynamic_script(node_id);
325                    }
326                    FetchKind::DynamicCss { node_id } => {
327                        log::warn!(
328                            "Dynamic stylesheet fetch failed without aborting page load: {url}"
329                        );
330                        tab.on_fetch_failed_dynamic_style(node_id);
331                    }
332                    FetchKind::JavaScript { request_id, .. } => {
333                        tab.on_fetch_failed_js(request_id, err.to_string());
334                    }
335                    FetchKind::Html | FetchKind::Css => tab.on_fetch_failed(err, url),
336                }
337            }
338        }
339    }

341    /// Handles a `winit` window event for this window and returns a `BrowserCommand`.
342    pub fn handle_window_event(
343        &mut self,
344        event: WindowEvent,
345        gpu: &mut GpuRenderer,
346    ) -> BrowserCommand {
347        match event {
348            WindowEvent::CloseRequested => BrowserCommand::Exit,
349
350            WindowEvent::RedrawRequested => {
351                self.redraw(gpu);
352                BrowserCommand::RenameWindowTitle
353            }
354
355            WindowEvent::Resized(size) => {
356                self.renderer.render_state.window_size = (size.width, size.height);
357                gpu.resize(size);
358                self.redraw(gpu);
359                BrowserCommand::RequestRedraw
360            }
361
362            WindowEvent::ScaleFactorChanged { scale_factor, .. } => {
363                gpu.set_scale_factor(scale_factor);
364                self.renderer.render_state.scale_factor = scale_factor;
365                self.redraw(gpu);
366                BrowserCommand::RequestRedraw
367            }
368
369            WindowEvent::MouseWheel { delta, .. } => {
370                self.handle_scroll(delta);
371                BrowserCommand::RequestRedraw
372            }
373
374            WindowEvent::CursorMoved { position, .. } => {
375                self.input.mouse_position = (position.x, position.y);
376                if self.handle_pointer_move(position.x, position.y) {
377                    BrowserCommand::RequestRedraw
378                } else {
379                    BrowserCommand::None
380                }
381            }
382
383            WindowEvent::MouseInput { button, state, .. } => self.handle_mouse_input(button, state),
384
385            WindowEvent::ModifiersChanged(modifiers) => {
386                self.input.modifiers = modifiers.state();
387                BrowserCommand::None
388            }
389
390            WindowEvent::KeyboardInput { event, .. } => self.handle_keyboard_input(event),
391
392            WindowEvent::Ime(event) => self.handle_ime_input(event),
393
394            _ => BrowserCommand::None,
395        }
396    }

398    /// Handles keyboard input events and returns a `BrowserCommand`.
399    fn handle_keyboard_input(&mut self, event: KeyEvent) -> BrowserCommand {
400        // TODO: あとで消す
401        const KEY_NEW_WINDOW: &str = "n";
402
403        if event.state != ElementState::Pressed {
404            return BrowserCommand::None;
405        }
406
407        let ctrl = self.input.modifiers.control_key();
408
409        if ctrl
410            && let winit::keyboard::Key::Character(ch) = &event.logical_key
411            && ch.as_str().eq_ignore_ascii_case(KEY_NEW_WINDOW)
412        {
413            return BrowserCommand::OpenNewWindow;
414        }
415
416        let tab_id = self.active_tab;
417
418        // While the chrome owns text input (e.g. the address bar is focused),
419        // keyboard input drives the chrome instead of the page.
420        if self.renderer.chrome.accepts_text_input() {
421            let action = self.renderer.chrome.key_event(&event, ctrl);
422            return match action {
423                ChromeAction::Navigate(url) => {
424                    if let Some(tab) = self.tabs.get_mut(tab_id) {
425                        tab.navigate(url);
426                    }
427                    BrowserCommand::RequestRedraw
428                }
429                ChromeAction::Repaint => BrowserCommand::RequestRedraw,
430                _ => BrowserCommand::None,
431            };
432        }
433
434        let Some((_, info)) = self.tabs.get(tab_id).and_then(Tab::layout_and_info) else {
435            return BrowserCommand::None;
436        };
437
438        let special = logical_key_to_special_event(&event.logical_key, ctrl);
439        let key = logical_key_to_text_key(&event.logical_key);
440        let handled = if let Some(special) = special {
441            crate::engine::input::dispatch_text_input(info, special)
442        } else if let Some(key) = key {
443            crate::engine::input::dispatch_text_input(info, InputTextEvent::Key(key))
444        } else if !ctrl && !crate::engine::input::focused_text_input_is_composing(info) {
445            event.text.as_ref().is_some_and(|text| {
446                crate::engine::input::dispatch_text_input(
447                    info,
448                    InputTextEvent::Insert(text.to_string()),
449                )
450            })
451        } else {
452            false
453        };
454
455        if handled {
456            BrowserCommand::RequestRedraw
457        } else {
458            BrowserCommand::None
459        }
460    }

462    /// Handles IME composition updates for the focused text input.
463    fn handle_ime_input(&mut self, event: Ime) -> BrowserCommand {
464        // While the chrome owns text input (e.g. the address bar is focused),
465        // IME events drive the chrome instead of the page.
466        if self.renderer.chrome.accepts_text_input() {
467            let action = self.renderer.chrome.ime_event(&event);
468            return match action {
469                ChromeAction::Repaint => BrowserCommand::RequestRedraw,
470                _ => BrowserCommand::None,
471            };
472        }
473
474        let event = match event {
475            Ime::Preedit(text, _) => InputTextEvent::Preedit(text),
476            Ime::Commit(text) => InputTextEvent::Commit(text),
477            Ime::Disabled => InputTextEvent::CancelComposition,
478            Ime::Enabled => return BrowserCommand::None,
479        };
480
481        let tab_id = self.active_tab;
482
483        let Some((_, info)) = self.tabs.get(tab_id).and_then(Tab::layout_and_info) else {
484            return BrowserCommand::None;
485        };
486
487        if crate::engine::input::dispatch_text_input(info, event) {
488            BrowserCommand::RequestRedraw
489        } else {
490            BrowserCommand::None
491        }
492    }

494    /// Handles mouse input events for the active tab.
495    ///
496    /// An open context menu intercepts every press and release before the
497    /// chrome and the page; a right-press over the web content opens it.
498    fn handle_mouse_input(&mut self, button: MouseButton, state: ElementState) -> BrowserCommand {
499        let (x, y, sf) = (
500            self.input.mouse_position.0,
501            self.input.mouse_position.1,
502            self.renderer.render_state.scale_factor,
503        );
504        let (px, py) = ((x / sf) as f32, (y / sf) as f32);
505
506        let width = self.renderer.render_state.viewport().0;
507        let height = self.renderer.render_state.viewport().1;
508
509        // An open context menu gets every press/release before the chrome
510        // and the page. Events it declines fall through to the normal flow.
511        if self.renderer.menu.is_open() {
512            let window_event = match state {
513                ElementState::Pressed => PointerEvent::Down { x: px, y: py },
514                ElementState::Released => PointerEvent::Up { x: px, y: py },
515            };
516            let result = self
517                .renderer
518                .menu
519                .pointer_event(width, height, window_event);
520            if result.consumed {
521                return self.dispatch_action(result.action, ActionSource::ContextMenu, x, y);
522            }
523        } else if button == MouseButton::Right {
524            // A right-press over the web content opens the context menu.
525            if state == ElementState::Pressed {
526                return self.open_context_menu(px, py);
527            }
528            return BrowserCommand::None;
529        }
530
531        if button != MouseButton::Left {
532            return BrowserCommand::None;
533        }
534
535        let tab_id = self.active_tab;
536
537        // Click inside the chrome.
538        let window_event = match state {
539            ElementState::Pressed => PointerEvent::Down { x: px, y: py },
540            ElementState::Released => PointerEvent::Up { x: px, y: py },
541        };
542        let result = self
543            .renderer
544            .chrome
545            .pointer_event(width, height, window_event);
546        if result.consumed {
547            return self.dispatch_action(result.action, ActionSource::Chrome, x, y);
548        }
549
550        // Content area: dispatch to the active tab in page coordinates.
551        let Rect { x: dx, y: dy, .. } = self.renderer.chrome.content_rect(width, height);
552
553        let (px, py) = (px - dx, py - dy);
554
555        if let Some(tab) = self.tabs.get_mut(tab_id) {
556            // Hit-test the content area, dispatch the pointer event to custom
557            // nodes, and remember which DOM element the press/release landed on.
558            let clicked_dom_id = tab.layout_and_info().and_then(|(layout, info)| {
559                let path = crate::engine::input::hit_test(layout, info, px, py);
560                // A press that does not belong to an open popup closes every
561                // open popup (top-layer overlays dismiss on outside clicks).
562                if matches!(state, ElementState::Pressed) {
563                    crate::engine::input::dismiss_open_popups(info, &path);
564                }
565                let event = match state {
566                    ElementState::Pressed => PointerEvent::Down { x: px, y: py },
567                    ElementState::Released => PointerEvent::Up { x: px, y: py },
568                };
569                crate::engine::input::dispatch_pointer(&path, event);
570                crate::engine::input::hit_dom_id(&path)
571            });
572
573            // A completed click (press and release on the same element) runs
574            // the element's JS `onclick` handler, if any.
575            let mut js_redraw = false;
576            match state {
577                ElementState::Pressed => {
578                    self.input.pressed_dom_id = clicked_dom_id;
579                }
580                ElementState::Released => {
581                    let pressed = self.input.pressed_dom_id.take();
582                    if let (Some(pressed), Some(released)) = (pressed, clicked_dom_id)
583                        && pressed == released
584                    {
585                        js_redraw = tab.on_js_click(released);
586                    }
587                }
588            }
589
590            // Clicking the page unfocuses the chrome's text input.
591            self.renderer.chrome.blur();
592
593            let input_focused = handle_mouse_click(tab, px, py);
594            if js_redraw {
595                BrowserCommand::RequestRedraw
596            } else {
597                BrowserCommand::SetImeAllowed {
598                    allowed: input_focused,
599                    position: (x, y),
600                }
601            }
602        } else {
603            BrowserCommand::None
604        }
605    }

607    /// Applies a [`ChromeAction`] produced by the chrome or the context menu
608    /// to the active tab.
609    ///
610    /// `source` decides whose repaint flag is consumed to close the event
611    /// handling; `(x, y)` are window coordinates for the IME request.
612    fn dispatch_action(
613        &mut self,
614        action: ChromeAction,
615        source: ActionSource,
616        x: f64,
617        y: f64,
618    ) -> BrowserCommand {
619        let tab_id = self.active_tab;
620        match action {
621            // Pressing the URL bar enables the OS IME so the caret and
622            // input methods work; the platform handler also requests a
623            // redraw.
624            ChromeAction::EnableIme => {
625                if let Some(tab) = self.tabs.get_mut(tab_id)
626                    && let Some((_, info)) = tab.layout_and_info()
627                {
628                    crate::engine::input::focus_text_input(info, None);
629                }
630                return BrowserCommand::SetImeAllowed {
631                    allowed: true,
632                    position: (x, y),
633                };
634            }
635            ChromeAction::Back => {
636                if let Some(tab) = self.tabs.get_mut(tab_id) {
637                    tab.go_back();
638                }
639            }
640            ChromeAction::Reload => {
641                if let Some(tab) = self.tabs.get_mut(tab_id) {
642                    tab.reload();
643                }
644            }
645            ChromeAction::Navigate(url) => {
646                if let Some(tab) = self.tabs.get_mut(tab_id) {
647                    tab.navigate(url);
648                }
649            }
650            ChromeAction::DumpLayoutNode => {
651                let node_opt = self
652                    .tabs
653                    .get(self.active_tab)
654                    .and_then(|t| t.layout_and_info().unzip().0);
655                if let Some(node) = node_opt {
656                    self.renderer.chrome.debug_set_layout_node(node);
657                }
658            }
659            ChromeAction::SetJsPolicy(policy) => {
660                if let Some(tab) = self.tabs.get_mut(tab_id) {
661                    tab.set_js_policy(policy);
662                    tab.reload();
663                }
664            }
665            ChromeAction::Repaint | ChromeAction::None => {}
666        }
667
668        let needs_repaint = match source {
669            ActionSource::Chrome => self.renderer.chrome.needs_repaint(),
670            ActionSource::ContextMenu => self.renderer.menu.needs_repaint(),
671        };
672        if needs_repaint {
673            BrowserCommand::RequestRedraw
674        } else {
675            BrowserCommand::SetImeAllowed {
676                allowed: false,
677                position: (x, y),
678            }
679        }
680    }

682    /// Opens the context menu for a right-press at window logical `(px, py)`.
683    ///
684    /// Builds a [`ClickContext`] (positions, link under the cursor, document
685    /// URL) and hands it to the menu. The menu only opens over the web
686    /// content area, never over the chrome.
687    fn open_context_menu(&mut self, px: f32, py: f32) -> BrowserCommand {
688        let width = self.renderer.render_state.viewport().0;
689        let height = self.renderer.render_state.viewport().1;
690
691        let Rect {
692            x: dx,
693            y: dy,
694            width: content_width,
695            height: content_height,
696        } = self.renderer.chrome.content_rect(width, height);
697        if px < dx || py < dy || px > dx + content_width || py > dy + content_height {
698            return BrowserCommand::None;
699        }
700
701        let page_pos = (px - dx, py - dy);
702
703        let Some(tab) = self.tabs.get(self.active_tab) else {
704            return BrowserCommand::None;
705        };
706        let document_url = tab.document_url().map(|url| url.to_string());
707        let link_url = tab.layout_and_info().and_then(|(layout, info)| {
708            let path = crate::engine::input::hit_test(layout, info, page_pos.0, page_pos.1);
709            link_href_at(&path)
710        });
711
712        let ctx = ClickContext {
713            window_pos: (px, py),
714            page_pos,
715            link_url,
716            document_url,
717        };
718
719        if self.renderer.menu.open(&ctx) {
720            BrowserCommand::RequestRedraw
721        } else {
722            BrowserCommand::None
723        }
724    }

726    /// Dispatches a pointer move and updates hover state for the active tab.
727    ///
728    /// Returns whether the move changed any visual state (and thus requires a
729    /// repaint).
730    fn handle_pointer_move(&mut self, x: f64, y: f64) -> bool {
731        let tab_id = self.active_tab;
732        let sf = self.renderer.render_state.scale_factor;
733        let (px, py) = ((x / sf) as f32, (y / sf) as f32);
734        let v_width = self.renderer.render_state.viewport().0;
735        let v_height = self.renderer.render_state.viewport().1;
736
737        // An open context menu intercepts every move before chrome/page.
738        if self.renderer.menu.is_open()
739            && self
740                .renderer
741                .menu
742                .pointer_event(v_width, v_height, PointerEvent::Move { x: px, y: py })
743                .consumed
744        {
745            return self.renderer.menu.needs_repaint();
746        }
747
748        // The chrome receives every move so it can track its own hover state.
749        let result = self.renderer.chrome.pointer_event(
750            v_width,
751            v_height,
752            PointerEvent::Move { x: px, y: py },
753        );
754        if result.consumed {
755            return self.renderer.chrome.needs_repaint();
756        }
757
758        // Content area: dispatch to the page in page coordinates.
759        let viewport = self.renderer.chrome.content_rect(v_width, v_height);
760        let Rect { x: px, y: py, .. } = viewport;
761
762        let Some(tab) = self.tabs.get_mut(tab_id) else {
763            return false;
764        };
765        let Some((layout, info)) = tab.layout_and_info() else {
766            return false;
767        };
768        let path = crate::engine::input::hit_test(layout, info, px, py);
769        let mut repaint =
770            crate::engine::input::dispatch_pointer(&path, PointerEvent::Move { x: px, y: py });
771        let previous = self.input.hovered.clone();
772        if crate::engine::input::update_hover(&path, previous.as_ref()) {
773            repaint = true;
774            self.input.hovered = crate::engine::input::hit_custom_node(&path).cloned();
775        }
776        repaint || self.renderer.chrome.needs_repaint()
777    }

779    /// Handles scrolling for the window's assigned tab, updating its layout container offsets.
780    fn handle_scroll(&mut self, delta: MouseScrollDelta) {
781        let (scroll_x, scroll_y) = match delta {
782            MouseScrollDelta::LineDelta(x, y) => (-x * 60.0, -y * 60.0),
783            MouseScrollDelta::PixelDelta(pos) => (-pos.x as f32, -pos.y as f32),
784        };
785
786        let (w_width, w_height) = self.renderer.render_state.viewport();
787        let Rect {
788            x: sx,
789            y: sy,
790            width,
791            height,
792        } = self.renderer.chrome.content_rect(w_width, w_height);
793        let sf = self.renderer.render_state.scale_factor;
794        let (mouse_x, mouse_y) = (
795            (self.input.mouse_position.0 / sf) as f32 - sx,
796            (self.input.mouse_position.1 / sf) as f32 - sy,
797        );
798
799        if mouse_x < sx || mouse_y < sy || mouse_x > sx + width || mouse_y > sy + height {
800            self.renderer
801                .chrome
802                .handle_scroll(w_width, w_height, mouse_x, mouse_y, scroll_x, scroll_y);
803        } else {
804            let tab_id = self.active_tab;
805            if let Some(tab) = self.tabs.get_mut(tab_id)
806                && let Some((layout, info)) = tab.layout_and_info_mut()
807            {
808                // Prefer the scrollable container under the cursor.
809                crate::engine::input::scroll_at(
810                    layout,
811                    info,
812                    (width, height),
813                    mouse_x,
814                    mouse_y,
815                    scroll_x,
816                    scroll_y,
817                );
818            }
819        }
820    }
821}
822/// Handles a mouse click in the given tab at the specified coordinates.
823fn handle_mouse_click(tab: &mut Tab, x: f32, y: f32) -> bool {
824    let hit_path = match tab.layout_and_info() {
825        Some((layout, info)) => crate::engine::input::hit_test(layout, info, x, y),
826        None => return false,
827    };
828
829    let input_target = hit_path.iter().find_map(|hit| {
830        if let layouter::types::NodeKind::Custom { node, .. } = &hit.info.kind
831            && node.accepts_text_input()
832        {
833            Some(Arc::clone(node))
834        } else {
835            None
836        }
837    });
838    let input_focused = tab.layout_and_info().is_some_and(|(_, info)| {
839        crate::engine::input::focus_text_input(info, input_target.as_ref())
840    });
841
842    let href_opt = link_href_at(&hit_path);
843
844    if let Some(href) = href_opt {
845        tab.move_to(&href)
846    }
847    input_focused
848}

850/// Returns the href of the innermost link on the hit path, if any.
851///
852/// The hit path is ordered child→parent, so the first match is the deepest
853/// link under the pointer.
854fn link_href_at(hit_path: &crate::engine::input::HitPath<'_>) -> Option<String> {
855    hit_path.iter().find_map(|hit| {
856        if let layouter::types::NodeKind::Container { role, .. } = &hit.info.kind
857            && let layouter::types::ContainerRole::Link { href } = role
858        {
859            Some(href.clone())
860        } else {
861            None
862        }
863    })
864}

870    /// The action was produced by the chrome.

872    /// The action was produced by the context menu.

874}

876/// Maps a logical key to a text-editing navigation key, if any.
877fn logical_key_to_text_key(key: &winit::keyboard::Key) -> Option<InputTextKey> {
878    match key {
879        winit::keyboard::Key::Named(winit::keyboard::NamedKey::Backspace) => {
880            Some(InputTextKey::Backspace)
881        }
882        winit::keyboard::Key::Named(winit::keyboard::NamedKey::Delete) => {
883            Some(InputTextKey::Delete)
884        }
885        winit::keyboard::Key::Named(winit::keyboard::NamedKey::ArrowLeft) => {
886            Some(InputTextKey::Left)
887        }
888        winit::keyboard::Key::Named(winit::keyboard::NamedKey::ArrowRight) => {
889            Some(InputTextKey::Right)
890        }
891        winit::keyboard::Key::Named(winit::keyboard::NamedKey::Home) => Some(InputTextKey::Home),
892        winit::keyboard::Key::Named(winit::keyboard::NamedKey::End) => Some(InputTextKey::End),
893        _ => None,
894    }
895}

897/// Maps a logical key to a text-editing special event (undo/redo/enter).
898fn logical_key_to_special_event(key: &winit::keyboard::Key, ctrl: bool) -> Option<InputTextEvent> {
899    if ctrl && let winit::keyboard::Key::Character(ch) = key {
900        match ch.as_str() {
901            "z" | "Z" => Some(InputTextEvent::Undo),
902            "y" | "Y" => Some(InputTextEvent::Redo),
903            _ => None,
904        }
905    } else if let winit::keyboard::Key::Named(winit::keyboard::NamedKey::Enter) = key {
906        Some(InputTextEvent::Enter)
907    } else {
908        None
909    }
