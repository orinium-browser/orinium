22/// Render state for a browser window.
23#[derive(Debug, Clone)]
24pub struct RenderState {
25    /// List of draw commands generated from the layout engine.
26    pub draw_commands: Vec<DrawCommand>,
27    /// Current window size in pixels (width, height).
28    pub window_size: (u32, u32),
29    /// Current scale factor (for HiDPI displays).
30    pub scale_factor: f64,
31    /// Current window title.
32    pub window_title: String,
33}
34
35impl Default for RenderState {
36    fn default() -> Self {
37        Self {
38            draw_commands: Vec::new(),
39            window_size: (800, 600),
40            scale_factor: 1.0,
41            window_title: String::new(),
42        }
43    }
44}

57    /// Calculates the viewport dimensions in scaled logical pixels.
58    pub fn viewport(&self) -> (f32, f32) {
59        let sf = self.scale_factor as f32;
60        (
61            self.window_size.0 as f32 / sf,
62            self.window_size.1 as f32 / sf,
63        )
64    }

204    /// Ticks all tabs and collects fetch requests and redraw demands.
205    pub(crate) fn tick(&mut self) -> BrowserUiTick {
206        let mut fetches = Vec::new();
207        let mut needs_redraw = false;
208
209        let tab_count = self.tabs.len();
210        for tab_id in 0..tab_count {
211            let Some(tab) = self.tabs.get_mut(tab_id) else {
212                continue;
213            };
214            for task in tab.tick() {
215                match task {
216                    TabTask::Fetch { url, kind } => {
217                        log::info!("Fetch requested in BrowserUi: url={}", url);
218                        fetches.push(FetchRequest { tab_id, url, kind });
219                    }
220                    TabTask::NeedsRedraw => {
221                        needs_redraw = true;
222                    }
223                }
224            }
225        }
226
227        BrowserUiTick {
228            fetches,
229            needs_redraw,
230        }
231    }

233    /// Delivers a fetched resource to the target tab.
234    pub(crate) fn deliver_fetch(
235        &mut self,
236        tab_id: usize,
237        kind: FetchKind,
238        url: Url,
239        response: Result<BrowserResponse, BrowserNetworkError>,
240    ) {
241        let Some(tab) = self.tabs.get_mut(tab_id) else {
242            log::warn!("There is no Tab called id={}", tab_id);
243            return;
244        };
245
246        match response {
247            Ok(resp) => {
248                log::info!("Fetch Done in BrowserUi for tab_id={}", tab_id);
249
250                match kind {
251                    FetchKind::Html => {
252                        let html = String::from_utf8_lossy(&resp.body).to_string();
253                        tab.on_fetch_succeeded_html(html);
254                    }
255                    FetchKind::Css => {
256                        let css = String::from_utf8_lossy(&resp.body).to_string();
257                        tab.on_fetch_succeeded_css(css);
258                    }
259                    FetchKind::Image { source } => {
260                        tab.on_fetch_succeeded_image(source, &resp.body);
261                    }
262                }
263            }
264            Err(err) => {
265                log::error!("NetworkError: {}", err);
266                if matches!(kind, FetchKind::Image { .. }) {
267                    log::warn!("Image fetch failed without aborting page load: {}", url);
268                } else {
269                    tab.on_fetch_failed(err, url);
270                }
271            }
272        }
273    }

275    /// Handles a `winit` window event for this window and returns a `BrowserCommand`.
276    pub fn handle_window_event(
277        &mut self,
278        event: WindowEvent,
279        gpu: &mut GpuRenderer,
280    ) -> BrowserCommand {
281        match event {
282            WindowEvent::CloseRequested => BrowserCommand::Exit,
283
284            WindowEvent::RedrawRequested => {
285                self.redraw(gpu);
286                BrowserCommand::RenameWindowTitle
287            }
288
289            WindowEvent::Resized(size) => {
290                self.renderer.render_state.window_size = (size.width, size.height);
291                gpu.resize(size);
292                self.redraw(gpu);
293                BrowserCommand::RequestRedraw
294            }
295
296            WindowEvent::ScaleFactorChanged { scale_factor, .. } => {
297                gpu.set_scale_factor(scale_factor);
298                self.renderer.render_state.scale_factor = scale_factor;
299                self.redraw(gpu);
300                BrowserCommand::RequestRedraw
301            }
302
303            WindowEvent::MouseWheel { delta, .. } => {
304                self.handle_scroll(delta);
305                BrowserCommand::RequestRedraw
306            }
307
308            WindowEvent::CursorMoved { position, .. } => {
309                self.input.mouse_position = (position.x, position.y);
310                if self.handle_pointer_move(position.x, position.y) {
311                    BrowserCommand::RequestRedraw
312                } else {
313                    BrowserCommand::None
314                }
315            }
316
317            WindowEvent::MouseInput { button, state, .. } => self.handle_mouse_input(button, state),
318
319            WindowEvent::ModifiersChanged(modifiers) => {
320                self.input.modifiers = modifiers.state();
321                BrowserCommand::None
322            }
323
324            WindowEvent::KeyboardInput { event, .. } => self.handle_keyboard_input(event),
325
326            WindowEvent::Ime(event) => self.handle_ime_input(event),
327
328            _ => BrowserCommand::None,
329        }
330    }

332    /// Handles keyboard input events and returns a `BrowserCommand`.
333    fn handle_keyboard_input(&mut self, event: KeyEvent) -> BrowserCommand {
334        // TODO: あとで消す
335        const KEY_NEW_WINDOW: &str = "n";
336
337        if event.state != ElementState::Pressed {
338            return BrowserCommand::None;
339        }
340
341        let ctrl = self.input.modifiers.control_key();
342
343        if ctrl
344            && let winit::keyboard::Key::Character(ch) = &event.logical_key
345            && ch.as_str().eq_ignore_ascii_case(KEY_NEW_WINDOW)
346        {
347            return BrowserCommand::OpenNewWindow;
348        }
349
350        let tab_id = self.active_tab;
351
352        // While the chrome owns text input (e.g. the address bar is focused),
353        // keyboard input drives the chrome instead of the page.
354        if self.renderer.chrome.accepts_text_input() {
355            let action = self.renderer.chrome.key_event(&event, ctrl);
356            return match action {
357                ChromeAction::Navigate(url) => {
358                    if let Some(tab) = self.tabs.get_mut(tab_id) {
359                        tab.navigate(url);
360                    }
361                    BrowserCommand::RequestRedraw
362                }
363                ChromeAction::Repaint => BrowserCommand::RequestRedraw,
364                _ => BrowserCommand::None,
365            };
366        }
367
368        let Some((_, info)) = self.tabs.get(tab_id).and_then(Tab::layout_and_info) else {
369            return BrowserCommand::None;
370        };
371
372        let special = logical_key_to_special_event(&event.logical_key, ctrl);
373        let key = logical_key_to_text_key(&event.logical_key);
374        let handled = if let Some(special) = special {
375            crate::engine::input::dispatch_text_input(info, special)
376        } else if let Some(key) = key {
377            crate::engine::input::dispatch_text_input(info, InputTextEvent::Key(key))
378        } else if !ctrl && !crate::engine::input::focused_text_input_is_composing(info) {
379            event.text.as_ref().is_some_and(|text| {
380                crate::engine::input::dispatch_text_input(
381                    info,
382                    InputTextEvent::Insert(text.to_string()),
383                )
384            })
385        } else {
386            false
387        };
388
389        if handled {
390            BrowserCommand::RequestRedraw
391        } else {
392            BrowserCommand::None
393        }
394    }

396    /// Handles IME composition updates for the focused text input.
397    fn handle_ime_input(&mut self, event: Ime) -> BrowserCommand {
398        // While the chrome owns text input (e.g. the address bar is focused),
399        // IME events drive the chrome instead of the page.
400        if self.renderer.chrome.accepts_text_input() {
401            let action = self.renderer.chrome.ime_event(&event);
402            return match action {
403                ChromeAction::Repaint => BrowserCommand::RequestRedraw,
404                _ => BrowserCommand::None,
405            };
406        }
407
408        let event = match event {
409            Ime::Preedit(text, _) => InputTextEvent::Preedit(text),
410            Ime::Commit(text) => InputTextEvent::Commit(text),
411            Ime::Disabled => InputTextEvent::CancelComposition,
412            Ime::Enabled => return BrowserCommand::None,
413        };
414
415        let tab_id = self.active_tab;
416
417        let Some((_, info)) = self.tabs.get(tab_id).and_then(Tab::layout_and_info) else {
418            return BrowserCommand::None;
419        };
420
421        if crate::engine::input::dispatch_text_input(info, event) {
422            BrowserCommand::RequestRedraw
423        } else {
424            BrowserCommand::None
425        }
426    }

428    /// Handles mouse input events, mainly left-clicks for the active tab.
429    fn handle_mouse_input(&mut self, button: MouseButton, state: ElementState) -> BrowserCommand {
430        if button != MouseButton::Left {
431            return BrowserCommand::None;
432        }
433
434        let (x, y, sf) = (
435            self.input.mouse_position.0,
436            self.input.mouse_position.1,
437            self.renderer.render_state.scale_factor,
438        );
439        let (px, py) = ((x / sf) as f32, (y / sf) as f32);
440
441        let tab_id = self.active_tab;
442        let width = self.renderer.render_state.viewport().0;
443
444        // Click inside the chrome.
445        let window_event = match state {
446            ElementState::Pressed => PointerEvent::Down { x: px, y: py },
447            ElementState::Released => PointerEvent::Up { x: px, y: py },
448        };
449        let result = self.renderer.chrome.pointer_event(width, window_event);
450        if result.consumed {
451            match result.action {
452                // Pressing the URL bar enables the OS IME so the caret and
453                // input methods work; the platform handler also requests a
454                // redraw.
455                ChromeAction::EnableIme => {
456                    if let Some(tab) = self.tabs.get_mut(tab_id)
457                        && let Some((_, info)) = tab.layout_and_info()
458                    {
459                        crate::engine::input::focus_text_input(info, None);
460                    }
461                    return BrowserCommand::SetImeAllowed {
462                        allowed: true,
463                        position: (x, y),
464                    };
465                }
466                ChromeAction::Back => {
467                    if let Some(tab) = self.tabs.get_mut(tab_id) {
468                        tab.go_back();
469                    }
470                }
471                ChromeAction::Reload => {
472                    if let Some(tab) = self.tabs.get_mut(tab_id) {
473                        tab.reload();
474                    }
475                }
476                ChromeAction::Navigate(url) => {
477                    if let Some(tab) = self.tabs.get_mut(tab_id) {
478                        tab.navigate(url);
479                    }
480                }
481                ChromeAction::Repaint | ChromeAction::None => {}
482            }
483
484            return if self.renderer.chrome.needs_repaint() {
485                BrowserCommand::RequestRedraw
486            } else {
487                BrowserCommand::SetImeAllowed {
488                    allowed: false,
489                    position: (x, y),
490                }
491            };
492        }
493
494        // Content area: dispatch to the active tab in page coordinates.
495        let chrome_height = self.renderer.chrome.chrome_height(width);
496        let (px, py) = (px, py - chrome_height);
497
498        if let Some(tab) = self.tabs.get_mut(tab_id) {
499            // Hit-test the content area, dispatch the pointer event to custom
500            // nodes, and remember which DOM element the press/release landed on.
501            let clicked_dom_id = tab.layout_and_info().and_then(|(layout, info)| {
502                let path = crate::engine::input::hit_test(layout, info, px, py);
503                let event = match state {
504                    ElementState::Pressed => PointerEvent::Down { x: px, y: py },
505                    ElementState::Released => PointerEvent::Up { x: px, y: py },
506                };
507                crate::engine::input::dispatch_pointer(&path, event);
508                crate::engine::input::hit_dom_id(&path)
509            });
510
511            // A completed click (press and release on the same element) runs
512            // the element's JS `onclick` handler, if any.
513            let mut js_redraw = false;
514            match state {
515                ElementState::Pressed => {
516                    self.input.pressed_dom_id = clicked_dom_id;
517                }
518                ElementState::Released => {
519                    let pressed = self.input.pressed_dom_id.take();
520                    if let (Some(pressed), Some(released)) = (pressed, clicked_dom_id)
521                        && pressed == released
522                    {
523                        js_redraw = tab.on_js_click(released);
524                    }
525                }
526            }
527
528            // Clicking the page unfocuses the chrome's text input.
529            self.renderer.chrome.blur();
530
531            let input_focused = handle_mouse_click(tab, px, py);
532            if js_redraw {
533                BrowserCommand::RequestRedraw
534            } else {
535                BrowserCommand::SetImeAllowed {
536                    allowed: input_focused,
537                    position: (x, y),
538                }
539            }
540        } else {
541            BrowserCommand::None
542        }
543    }

545    /// Dispatches a pointer move and updates hover state for the active tab.
546    ///
547    /// Returns whether the move changed any visual state (and thus requires a
548    /// repaint).
549    fn handle_pointer_move(&mut self, x: f64, y: f64) -> bool {
550        let tab_id = self.active_tab;
551        let sf = self.renderer.render_state.scale_factor;
552        let (px, py) = ((x / sf) as f32, (y / sf) as f32);
553        let width = self.renderer.render_state.viewport().0;
554
555        // The chrome receives every move so it can track its own hover state.
556        let result = self
557            .renderer
558            .chrome
559            .pointer_event(width, PointerEvent::Move { x: px, y: py });
560        if result.consumed {
561            return self.renderer.chrome.needs_repaint();
562        }
563
564        // Content area: dispatch to the page in page coordinates.
565        let chrome_height = self.renderer.chrome.chrome_height(width);
566        let (px, py) = (px, py - chrome_height);
567
568        let Some(tab) = self.tabs.get_mut(tab_id) else {
569            return false;
570        };
571        let Some((layout, info)) = tab.layout_and_info() else {
572            return false;
573        };
574        let path = crate::engine::input::hit_test(layout, info, px, py);
575        let mut repaint =
576            crate::engine::input::dispatch_pointer(&path, PointerEvent::Move { x: px, y: py });
577        let previous = self.input.hovered.clone();
578        if crate::engine::input::update_hover(&path, previous.as_ref()) {
579            repaint = true;
580            self.input.hovered = crate::engine::input::hit_custom_node(&path).cloned();
581        }
582        repaint || self.renderer.chrome.needs_repaint()
583    }

585    /// Handles scrolling for the window's assigned tab, updating its layout container offsets.
586    fn handle_scroll(&mut self, delta: MouseScrollDelta) {
587        let (scroll_x, scroll_y) = match delta {
588            MouseScrollDelta::LineDelta(x, y) => (-x * 60.0, -y * 60.0),
589            MouseScrollDelta::PixelDelta(pos) => (-pos.x as f32, -pos.y as f32),
590        };
591
592        let (width, window_height, sf) = (
593            self.renderer.render_state.viewport().0,
594            self.renderer.render_state.window_size.1 as f32,
595            self.renderer.render_state.scale_factor as f32,
596        );
597        let (mouse_x, mouse_y) = (
598            self.input.mouse_position.0 as f32,
599            self.input.mouse_position.1 as f32,
600        );
601
602        let tab_id = self.active_tab;
603        if let Some(tab) = self.tabs.get_mut(tab_id)
604            && let Some((layout, info)) = tab.layout_and_info_mut()
605        {
606            // Prefer the scrollable container under the cursor; the wheel
607            // event chains to the root when nothing nested consumes it.
608            if crate::engine::input::scroll_at(layout, info, mouse_x, mouse_y, scroll_x, scroll_y) {
609                return;
610            }
611
612            if let layouter::types::NodeKind::Container {
613                scroll_offset_y, ..
614            } = &mut info.kind
615            {
616                // The page is laid out below the chrome, so the visible height
617                // is the window height minus the chrome height.
618                let visible_height =
619                    (window_height / sf - self.renderer.chrome.chrome_height(width)).max(0.0);
620                *scroll_offset_y = (*scroll_offset_y + scroll_y).clamp(
621                    0.0,
622                    (layout
623                        .layout_box
624                        .iter()
625                        .map(|l| l.children_box.height)
626                        .sum::<f32>()
627                        - visible_height)
628                        .max(0.0),
629                );
630            }
631        }
632    }

642/// Handles a mouse click in the given tab at the specified coordinates.
643fn handle_mouse_click(tab: &mut Tab, x: f32, y: f32) -> bool {
644    let hit_path = match tab.layout_and_info() {
645        Some((layout, info)) => crate::engine::input::hit_test(layout, info, x, y),
646        None => return false,
647    };
648
649    let input_target = hit_path.iter().find_map(|hit| {
650        if let layouter::types::NodeKind::Custom { node, .. } = &hit.info.kind
651            && node.accepts_text_input()
652        {
653            Some(Arc::clone(node))
654        } else {
655            None
656        }
657    });
658    let input_focused = tab.layout_and_info().is_some_and(|(_, info)| {
659        crate::engine::input::focus_text_input(info, input_target.as_ref())
660    });
661
662    let href_opt = {
663        if let Some(hit) = hit_path.iter().find(|e| {
664            matches!(
665                e.info.kind,
666                layouter::types::NodeKind::Container { ref role, .. }
667                    if matches!(role, layouter::types::ContainerRole::Link { .. })
668            )
669        }) {
670            if let layouter::types::NodeKind::Container { role, .. } = &hit.info.kind
671                && let layouter::types::ContainerRole::Link { href } = role
672            {
673                Some(href.clone())
674            } else {
675                None
676            }
677        } else {
678            None
679        }
680    };
681
682    if let Some(href) = href_opt {
683        tab.move_to(&href)
684    }
685    input_focused
686}

688/// Maps a logical key to a text-editing navigation key, if any.
689fn logical_key_to_text_key(key: &winit::keyboard::Key) -> Option<InputTextKey> {
690    match key {
691        winit::keyboard::Key::Named(winit::keyboard::NamedKey::Backspace) => {
692            Some(InputTextKey::Backspace)
693        }
694        winit::keyboard::Key::Named(winit::keyboard::NamedKey::Delete) => {
695            Some(InputTextKey::Delete)
696        }
697        winit::keyboard::Key::Named(winit::keyboard::NamedKey::ArrowLeft) => {
698            Some(InputTextKey::Left)
699        }
700        winit::keyboard::Key::Named(winit::keyboard::NamedKey::ArrowRight) => {
701            Some(InputTextKey::Right)
702        }
703        winit::keyboard::Key::Named(winit::keyboard::NamedKey::Home) => Some(InputTextKey::Home),
704        winit::keyboard::Key::Named(winit::keyboard::NamedKey::End) => Some(InputTextKey::End),
705        _ => None,
706    }
707}

709/// Maps a logical key to a text-editing special event (undo/redo/enter).
710fn logical_key_to_special_event(key: &winit::keyboard::Key, ctrl: bool) -> Option<InputTextEvent> {
711    if ctrl && let winit::keyboard::Key::Character(ch) = key {
712        match ch.as_str() {
713            "z" | "Z" => Some(InputTextEvent::Undo),
714            "y" | "Y" => Some(InputTextEvent::Redo),
715            _ => None,
716        }
717    } else if let winit::keyboard::Key::Named(winit::keyboard::NamedKey::Enter) = key {
718        Some(InputTextEvent::Enter)
719    } else {
720        None
721    }
