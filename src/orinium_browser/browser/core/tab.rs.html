55/// Tab はブラウザで開かれた 1 つのページを表す構造体です。
56///
57/// 主な責務:
58/// - 現在表示しているページのタイトルの保持
59/// - ページ内容を扱う WebView の保持
60///
61/// WebView が「ページそのもの」の状態を管理するのに対し、
62/// Tab は UI 上のタブとしてのメタ情報（タイトルなど）を管理します。
63pub struct Tab {
64    title: Option<String>,
65    base_url: Option<Url>,
66    document_url: Option<Url>,
67    webview: Option<WebView>,
68
69    system_color_scheme: ColorScheme,
70
71    /// Policy controlling whether page scripts are executed and how
72    /// `<noscript>` contents are parsed.
73    js_policy: JsPolicy,
74
75    state: TabState,
76    /// Previously visited URLs, most recent last. Used by the back button.
77    history: Vec<Url>,
78
79    /// The custom node currently under the pointer, if any.
80    hovered: Option<Arc<dyn CustomNode>>,
81    /// The DOM node under the pointer when the left button was pressed.
82    ///
83    /// Used to detect a completed click (press and release on the same node),
84    /// which is forwarded to the page's JS `onclick` handler.
85    pressed_dom_id: Option<u32>,
86}

128    /// Tab 内の状態を 1 ステップ進める
129    ///
130    /// - WebView.tick() を呼び出す
131    /// - 発生した Task を BrowserApp に返す
132    pub fn tick(&mut self) -> Vec<TabTask> {
133        let mut tasks = Vec::new();
134        let Some(wv) = self.webview.as_mut() else {
135            return tasks;
136        };
137
138        for task in wv.tick() {
139            match task {
140                WebViewTask::Fetch { url, kind } => {
141                    log::info!("Fetch requested in Tab: url={}", url);
142                    tasks.push(TabTask::Fetch { url, kind });
143                }
144                WebViewTask::AskTabHtml => {
145                    tasks.push(TabTask::Fetch {
146                        url: self.document_url.as_ref().unwrap().clone(),
147                        kind: FetchKind::Html,
148                    });
149                }
150                WebViewTask::DevToolsRequest { id, method, params } => {
151                    tasks.push(TabTask::DevToolsRequest { id, method, params });
152                }
153            }
154        }
155
156        if wv.needs_redraw() {
157            tasks.push(TabTask::NeedsRedraw);
158        }
159
160        tasks
161    }

163    /// Delivers the result of a resource fetch to this tab.
164    ///
165    /// This is shared by regular browser tabs and the DevTools tab embedded in
166    /// the browser chrome so both follow the same loading and error handling
167    /// path.
168    pub(crate) fn deliver_fetch(
169        &mut self,
170        kind: FetchKind,
171        url: Url,
172        response: Result<BrowserResponse, BrowserNetworkError>,
173    ) {
174        match response {
175            Ok(resp) => match kind {
176                FetchKind::Html => {
177                    let html = String::from_utf8_lossy(&resp.body).to_string();
178                    self.on_fetch_succeeded_html(html);
179                }
180                FetchKind::Css => {
181                    let css = String::from_utf8_lossy(&resp.body).to_string();
182                    self.on_fetch_succeeded_css_from(css, &url);
183                }
184                FetchKind::Script { index } => {
185                    let source = String::from_utf8_lossy(&resp.body).to_string();
186                    self.on_fetch_succeeded_script(index, source);
187                }
188                FetchKind::DynamicScript { node_id } => {
189                    let source = String::from_utf8_lossy(&resp.body).to_string();
190                    self.on_fetch_succeeded_dynamic_script(node_id, source);
191                }
192                FetchKind::DynamicCss { node_id } => {
193                    let source = String::from_utf8_lossy(&resp.body).to_string();
194                    self.on_fetch_succeeded_dynamic_style(node_id, source);
195                }
196                FetchKind::Image { source } => {
197                    self.on_fetch_succeeded_image(source, &resp.body);
198                }
199                FetchKind::Audio { source } => {
200                    self.on_fetch_succeeded_audio(source, &resp.body);
201                }
202                FetchKind::JavaScript { request_id, .. } => {
203                    let redirected = resp.url != url.as_str();
204                    self.on_fetch_succeeded_js(request_id, resp, redirected);
205                }
206            },
207            Err(err) => match kind {
208                FetchKind::Image { .. } | FetchKind::Audio { .. } => {
209                    log::warn!("Media fetch failed without aborting page load: {url}");
210                }
211                FetchKind::Script { index } => {
212                    log::warn!("Classic script fetch failed without aborting page load: {url}");
213                    self.on_fetch_failed_script(index);
214                }
215                FetchKind::DynamicScript { node_id } => {
216                    log::warn!("Dynamic script fetch failed without aborting page load: {url}");
217                    self.on_fetch_failed_dynamic_script(node_id);
218                }
219                FetchKind::DynamicCss { node_id } => {
220                    log::warn!("Dynamic stylesheet fetch failed without aborting page load: {url}");
221                    self.on_fetch_failed_dynamic_style(node_id);
222                }
223                FetchKind::JavaScript { request_id, .. } => {
224                    self.on_fetch_failed_js(request_id, err.to_string());
225                }
226                FetchKind::Html | FetchKind::Css => self.on_fetch_failed(err, url),
227            },
228        }
229    }

239    /// BrowserApp からの HTML fetch 完了を通知
240    pub fn on_fetch_succeeded_html(&mut self, html: String) {
241        let Some(wv) = self.webview.as_mut() else {
242            return;
243        };
244
245        wv.on_html_fetched(html, self.document_url.as_ref().unwrap().clone());
246        self.title = wv.title().cloned();
247        let base_url = wv.base_url().unwrap().clone();
248        log::info!("HTML fetched, base_url={}", base_url);
249        self.base_url = Some(base_url);
250
251        if let TabState::Error(TabError::NetworkError(err), url_opt) = &self.state {
252            let error_message = match url_opt {
253                Some(url) => format!("Failed to load {}: {}", url, err),
254                None => format!("Failed to load page: {}", err),
255            };
256
257            let error_message_element = wv
258                .document_info()
259                .unwrap()
260                .dom
261                .get_elements_by_class_name("error-message");
262            let error_message_element = error_message_element.first().unwrap();
263            let new_child = TreeNode::new(HtmlNodeType::Text(error_message));
264            TreeNode::replace_child(error_message_element, 0, new_child);
265
266            // Update page to show error message
267            // This is a stub implementation for now as you can see in WebView.update_page().
268            wv.update_page();
269        } else {
270            self.state = TabState::Loaded;
271        }
272    }

392    pub fn handle_mouse_input(&mut self, px: f32, py: f32, state: ElementState) -> (bool, bool) {
393        // Hit-test the content area, dispatch the pointer event to custom
394        // nodes, and remember which DOM element the press/release landed on.
395        let Some((_, info)) = self.layout_and_info() else {
396            return (false, false);
397        };
398
399        let path = self.hit_test(px, py);
400        let dom_id = crate::engine::input::hit_dom_id(&path);
401
402        let mut repaint = false;
403
404        match state {
405            ElementState::Pressed => {
406                crate::engine::input::dismiss_open_popups(info, &path);
407
408                let input_target = path.iter().find_map(|hit| {
409                    if let layouter::types::NodeKind::Custom { node, .. } = &hit.info.kind
410                        && node.accepts_text_input()
411                    {
412                        Some(Arc::clone(node))
413                    } else {
414                        None
415                    }
416                });
417
418                let input_focused =
419                    crate::engine::input::focus_text_input(info, input_target.as_ref());
420
421                crate::engine::input::dispatch_pointer(&path, PointerEvent::Down { x: px, y: py });
422
423                if let Some(href) = path.iter().find_map(|hit| {
424                    if let layouter::types::NodeKind::Container { role, .. } = &hit.info.kind
425                        && let layouter::types::ContainerRole::Link { href } = role
426                    {
427                        Some(href.clone())
428                    } else {
429                        None
430                    }
431                }) {
432                    self.move_to(&href);
433                }
434
435                repaint |= input_focused;
436
437                self.pressed_dom_id = dom_id;
438            }
439
440            ElementState::Released => {
441                crate::engine::input::dispatch_pointer(&path, PointerEvent::Up { x: px, y: py });
442
443                let pressed = self.pressed_dom_id.take();
444                if let (Some(pressed), Some(released)) = (pressed, dom_id)
445                    && pressed == released
446                {
447                    repaint |= self.on_js_click(released);
448                }
449            }
450        }
451
452        let (move_repaint, move_focused) = self.handle_pointer_move(px, py);
453        repaint |= move_repaint;
454
455        (repaint, move_focused)
456    }

458    /// Dispatches a pointer move and updates hover state without touching
459    /// click bookkeeping. Unlike [`Tab::handle_mouse_input`] it never
460    /// synthesizes press/release events, so it is safe to call on every
461    /// cursor movement.
462    pub fn handle_pointer_move(&mut self, px: f32, py: f32) -> (bool, bool) {
463        let move_path = self.hit_test(px, py);
464
465        let input_focused =
466            crate::engine::input::dispatch_pointer(&move_path, PointerEvent::Move { x: px, y: py });
467
468        let hover_changed = {
469            let previous = self.hovered.as_ref();
470            crate::engine::input::update_hover(&move_path, previous)
471        };
472
473        let mut repaint = false;
474        if hover_changed {
475            repaint = true;
476            self.hovered = crate::engine::input::hit_custom_node(&move_path).cloned();
477        }
478
479        (repaint, input_focused)
480    }

593    fn navigate_internal(&mut self, url: Url, record_history: bool) {
594        if record_history
595            && self.document_url.as_ref() != Some(&url)
596            && let Some(previous) = self.document_url.clone()
597        {
598            self.history.push(previous);
599        }
600        self.document_url = Some(url);
601        let mut webview = WebView::new(self.system_color_scheme, self.js_policy);
602        webview.navigate();
603        self.webview = Some(webview);
604        self.state = TabState::Loading;
605        self.title = None;
606        self.base_url = None;
607    }

665    pub fn needs_redraw(&self) -> bool {
666        self.webview.as_ref().is_some_and(|wv| wv.needs_redraw())
667    }

669    pub fn clear_redraw_flag(&mut self) {

671            wv.clear_redraw_flag();

740    #[test]
741    fn pointer_move_between_press_and_release_keeps_click_pending() {
742        let mut tab = Tab::default();
743        tab.navigate(url("https://example.test/a"));
744        tab.on_fetch_succeeded_html("<html><body><p>click me</p></body></html>".to_string());
745
746        // Force a relayout, wait for the background layout thread, then draw
747        // again so the applied tree gets its boxes positioned.
748        let mut buf = Vec::new();
749        tab.draw(&mut buf, 800.0, 600.0);
750        for _ in 0..500 {
751            for _ in tab.tick() {}
752            if tab.layout_and_info().is_some() {
753                break;
754            }
755            std::thread::sleep(std::time::Duration::from_millis(2));
756        }
757        tab.draw(&mut buf, 800.0, 600.0);
758
759        // Press on the page, move before releasing: the move must not
760        // complete or cancel the in-flight press.
761        tab.handle_mouse_input(400.0, 12.0, ElementState::Pressed);
762        assert!(tab.has_pending_press(), "press should land on an element");
763
764        tab.handle_pointer_move(401.0, 13.0);
765        assert!(
766            tab.has_pending_press(),
767            "a pointer move must not cancel an in-flight press"
768        );
769
770        // The release completes the click and clears the pending press.
771        tab.handle_mouse_input(401.0, 13.0, ElementState::Released);
772        assert!(!tab.has_pending_press());
773    }
