58/// Tab はブラウザで開かれた 1 つのページを表す構造体です。
59///
60/// 主な責務:
61/// - 現在表示しているページのタイトルの保持
62/// - ページ内容を扱う WebView の保持
63///
64/// WebView が「ページそのもの」の状態を管理するのに対し、
65/// Tab は UI 上のタブとしてのメタ情報（タイトルなど）を管理します。
66pub struct Tab {
67    title: Option<String>,
68    base_url: Option<Url>,
69    document_url: Option<Url>,
70    webview: Option<WebView>,
71
72    system_color_scheme: ColorScheme,
73
74    /// Policy controlling whether page scripts are executed and how
75    /// `<noscript>` contents are parsed.
76    js_policy: JsPolicy,
77
78    state: TabState,
79    /// Previously visited URLs, most recent last. Used by the back button.
80    history: Vec<Url>,
81
82    /// The custom node currently under the pointer, if any.
83    hovered: Option<Arc<dyn CustomNode>>,
84    /// The DOM node under the pointer when the left button was pressed.
85    ///
86    /// Used to detect a completed click (press and release on the same node),
87    /// which is forwarded to the page's JS `onclick` handler.
88    pressed_dom_id: Option<u32>,
89}

131    /// The tab's current page origin, derived from the loaded document URL.
132    ///
133    /// Opaque until a document has been fetched, so pages backed by internal
134    /// schemes (e.g. the DevTools page) are treated as non-network origins.
135    fn page_origin(&self) -> Origin {
136        self.document_url
137            .as_ref()
138            .map(Origin::from_url)
139            .unwrap_or_else(Origin::opaque)
140    }
141
142    /// Whether a `fetch()`/`XMLHttpRequest` response may be read by the given
143    /// initiator.
144    ///
145    /// Internal (opaque) initiators may always read responses; web origins are
146    /// restricted to same-origin responses and cross-origin responses that opt
147    /// in via `Access-Control-Allow-Origin`. Responses targeted at internal
148    /// schemes are exempt here because the resource loader already prevented
149    /// web origins from ever receiving them.
150    fn may_read_fetch_response(
151        &self,
152        initiator: &Origin,
153        url: &Url,
154        headers: &[(String, String)],
155    ) -> bool {
156        if !initiator.is_network() {
157            return true;
158        }
159        match url.scheme() {
160            "http" | "https" => {
161                let response_origin = Origin::from_url_string(url.as_str());
162                initiator.same_origin(&response_origin) || headers_allow_cors(headers, initiator)
163            }
164            _ => true,
165        }
166    }

168    /// Tab 内の状態を 1 ステップ進める
169    ///
170    /// - WebView.tick() を呼び出す
171    /// - 発生した Task を BrowserApp に返す
172    pub fn tick(&mut self) -> Vec<TabTask> {
173        let mut tasks = Vec::new();
174        let page_origin = self.page_origin();
175        let Some(wv) = self.webview.as_mut() else {
176            return tasks;
177        };
178
179        for task in wv.tick() {
180            match task {
181                WebViewTask::Fetch { url, kind } => {
182                    log::info!("Fetch requested in Tab: url={}", url);
183                    tasks.push(TabTask::Fetch {
184                        url,
185                        kind,
186                        origin: page_origin.clone(),
187                    });
188                }
189                WebViewTask::AskTabHtml => {
190                    tasks.push(TabTask::Fetch {
191                        url: self.document_url.as_ref().unwrap().clone(),
192                        kind: FetchKind::Html,
193                        origin: page_origin.clone(),
194                    });
195                }
196                WebViewTask::DevToolsRequest { id, method, params } => {
197                    tasks.push(TabTask::DevToolsRequest { id, method, params });
198                }
199            }
200        }
201
202        if wv.needs_redraw() {
203            tasks.push(TabTask::NeedsRedraw);
204        }
205
206        tasks
207    }

209    /// Delivers the result of a resource fetch to this tab.
210    ///
211    /// This is shared by regular browser tabs and the DevTools tab embedded in
212    /// the browser chrome so both follow the same loading and error handling
213    /// path.
214    pub(crate) fn deliver_fetch(
215        &mut self,
216        kind: FetchKind,
217        url: Url,
218        response: Result<BrowserResponse, BrowserNetworkError>,
219    ) {
220        match response {
221            Ok(resp) => match kind {
222                FetchKind::Html => {
223                    let html = String::from_utf8_lossy(&resp.body).to_string();
224                    self.on_fetch_succeeded_html(html);
225                }
226                FetchKind::Css => {
227                    let css = String::from_utf8_lossy(&resp.body).to_string();
228                    self.on_fetch_succeeded_css_from(css, &url);
229                }
230                FetchKind::Script { index } => {
231                    let source = String::from_utf8_lossy(&resp.body).to_string();
232                    self.on_fetch_succeeded_script(index, source);
233                }
234                FetchKind::DynamicScript { node_id } => {
235                    let source = String::from_utf8_lossy(&resp.body).to_string();
236                    self.on_fetch_succeeded_dynamic_script(node_id, source);
237                }
238                FetchKind::DynamicCss { node_id } => {
239                    let source = String::from_utf8_lossy(&resp.body).to_string();
240                    self.on_fetch_succeeded_dynamic_style(node_id, source);
241                }
242                FetchKind::Image { source } => {
243                    self.on_fetch_succeeded_image(source, &resp.body);
244                }
245                FetchKind::Audio { source } => {
246                    self.on_fetch_succeeded_audio(source, &resp.body);
247                }
248                FetchKind::JavaScript { request_id, .. } => {
249                    let initiator = self.page_origin();
250                    if self.may_read_fetch_response(&initiator, &url, &resp.headers) {
251                        let redirected = resp.url != url.as_str();
252                        self.on_fetch_succeeded_js(request_id, resp, redirected);
253                    } else {
254                        log::warn!(
255                            "Blocked CORS read of {url} from {}",
256                            initiator.ascii_serialization()
257                        );
258                        self.on_fetch_failed_js(
259                            request_id,
260                            "Cross-origin response blocked by CORS policy".to_string(),
261                        );
262                    }
263                }
264            },
265            Err(err) => match kind {
266                FetchKind::Image { .. } | FetchKind::Audio { .. } => {
267                    log::warn!("Media fetch failed without aborting page load: {url}");
268                }
269                FetchKind::Script { index } => {
270                    log::warn!("Classic script fetch failed without aborting page load: {url}");
271                    self.on_fetch_failed_script(index);
272                }
273                FetchKind::DynamicScript { node_id } => {
274                    log::warn!("Dynamic script fetch failed without aborting page load: {url}");
275                    self.on_fetch_failed_dynamic_script(node_id);
276                }
277                FetchKind::DynamicCss { node_id } => {
278                    log::warn!("Dynamic stylesheet fetch failed without aborting page load: {url}");
279                    self.on_fetch_failed_dynamic_style(node_id);
280                }
281                FetchKind::JavaScript { request_id, .. } => {
282                    self.on_fetch_failed_js(request_id, err.to_string());
283                }
284                FetchKind::Html | FetchKind::Css => self.on_fetch_failed(err, url),
285            },
286        }
287    }

297    /// BrowserApp からの HTML fetch 完了を通知
298    pub fn on_fetch_succeeded_html(&mut self, html: String) {
299        let Some(wv) = self.webview.as_mut() else {
300            return;
301        };
302
303        wv.on_html_fetched(html, self.document_url.as_ref().unwrap().clone());
304        self.title = wv.title().cloned();
305        let base_url = wv.base_url().unwrap().clone();
306        log::info!("HTML fetched, base_url={}", base_url);
307        self.base_url = Some(base_url);
308
309        if let TabState::Error(TabError::NetworkError(err), url_opt) = &self.state {
310            let error_message = match url_opt {
311                Some(url) => format!("Failed to load {}: {}", url, err),
312                None => format!("Failed to load page: {}", err),
313            };
314
315            let error_message_element = wv
316                .document_info()
317                .unwrap()
318                .dom
319                .get_elements_by_class_name("error-message");
320            let error_message_element = error_message_element.first().unwrap();
321            let new_child = TreeNode::new(HtmlNodeType::Text(error_message));
322            TreeNode::replace_child(error_message_element, 0, new_child);
323
324            // Update page to show error message
325            // This is a stub implementation for now as you can see in WebView.update_page().
326            wv.update_page();
327        } else {
328            self.state = TabState::Loaded;
329        }
330    }

450    pub fn handle_mouse_input(&mut self, px: f32, py: f32, state: ElementState) -> (bool, bool) {
451        // Hit-test the content area, dispatch the pointer event to custom
452        // nodes, and remember which DOM element the press/release landed on.
453        let Some((_, info)) = self.layout_and_info() else {
454            return (false, false);
455        };
456
457        let path = self.hit_test(px, py);
458        let dom_id = crate::engine::input::hit_dom_id(&path);
459
460        let mut repaint = false;
461
462        match state {
463            ElementState::Pressed => {
464                crate::engine::input::dismiss_open_popups(info, &path);
465
466                let input_target = path.iter().find_map(|hit| {
467                    if let layouter::types::NodeKind::Custom { node, .. } = &hit.info.kind
468                        && node.accepts_text_input()
469                    {
470                        Some(Arc::clone(node))
471                    } else {
472                        None
473                    }
474                });
475
476                let input_focused =
477                    crate::engine::input::focus_text_input(info, input_target.as_ref());
478
479                crate::engine::input::dispatch_pointer(&path, PointerEvent::Down { x: px, y: py });
480
481                if let Some(href) = path.iter().find_map(|hit| {
482                    if let layouter::types::NodeKind::Container { role, .. } = &hit.info.kind
483                        && let layouter::types::ContainerRole::Link { href } = role
484                    {
485                        Some(href.clone())
486                    } else {
487                        None
488                    }
489                }) {
490                    self.move_to(&href);
491                }
492
493                repaint |= input_focused;
494
495                self.pressed_dom_id = dom_id;
496            }
497
498            ElementState::Released => {
499                crate::engine::input::dispatch_pointer(&path, PointerEvent::Up { x: px, y: py });
500
501                let pressed = self.pressed_dom_id.take();
502                if let (Some(pressed), Some(released)) = (pressed, dom_id)
503                    && pressed == released
504                {
505                    repaint |= self.on_js_click(released);
506                }
507            }
508        }
509
510        let (move_repaint, move_focused) = self.handle_pointer_move(px, py);
511        repaint |= move_repaint;
512
513        (repaint, move_focused)
514    }

516    /// Dispatches a pointer move and updates hover state without touching
517    /// click bookkeeping. Unlike [`Tab::handle_mouse_input`] it never
518    /// synthesizes press/release events, so it is safe to call on every
519    /// cursor movement.
520    pub fn handle_pointer_move(&mut self, px: f32, py: f32) -> (bool, bool) {
521        let move_path = self.hit_test(px, py);
522
523        let input_focused =
524            crate::engine::input::dispatch_pointer(&move_path, PointerEvent::Move { x: px, y: py });
525
526        let hover_changed = {
527            let previous = self.hovered.as_ref();
528            crate::engine::input::update_hover(&move_path, previous)
529        };
530
531        let mut repaint = false;
532        if hover_changed {
533            repaint = true;
534            self.hovered = crate::engine::input::hit_custom_node(&move_path).cloned();
535        }
536
537        (repaint, input_focused)
538    }

650    fn navigate_internal(&mut self, url: Url, record_history: bool) {
651        if record_history
652            && self.document_url.as_ref() != Some(&url)
653            && let Some(previous) = self.document_url.clone()
654        {
655            self.history.push(previous);
656        }
657        self.document_url = Some(url);
658        let mut webview = WebView::new(self.system_color_scheme, self.js_policy);
659        webview.navigate();
660        self.webview = Some(webview);
661        self.state = TabState::Loading;
662        self.title = None;
663        self.base_url = None;
664    }

740/// Whether the response headers allow `initiator` to read the response.
741///
742/// Credentials are not tracked, so a wildcard `*` grants access exactly as an
743/// explicit origin would.
744fn headers_allow_cors(headers: &[(String, String)], initiator: &Origin) -> bool {
745    let serialized = initiator.ascii_serialization();
746    headers
747        .iter()
748        .find(|(name, _)| name.eq_ignore_ascii_case("access-control-allow-origin"))
749        .is_some_and(|(_, value)| value == "*" || value == &serialized)
750}
