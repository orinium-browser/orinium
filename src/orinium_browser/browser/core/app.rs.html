1//! Browser core: application entry and lifecycle manager.
2//!
3//! Responsibilities:
4//! - Manage the window collection and each window's [`BrowserUi`].
5//! - Forward winit window events to the owning window's UI.
6//! - Coordinate network/resource loading and route responses to the owning window.
7//!
8//! Tab state, input handling, and rendering are delegated to [`BrowserUi`] /
9//! [`BrowserRenderer`]; this type stays a thin orchestrator.
10//!
11//! Processing flow (high-level):
12//! 1. Initialize platform components (system window, GPU renderer, network core).
13//! 2. Create and register `BrowserUi` instances and navigate to initial URLs.
14//! 3. Enter event loop: forward events -> delegate to `BrowserUi` -> route fetches.
15//!
16//! Example (for contributors / local testing):
17//! ```no_run
18//! use orinium_browser::browser::{BrowserApp, BrowserUi, Tab};
19//!
20//! let mut tab = Tab::default();
21//! tab.navigate("resource:///test/test.html".parse().unwrap());
22//! let mut app = BrowserApp::default();
23//! app.set_default_ui(BrowserUi::with_tab(tab));
24//! app.run().unwrap();
25//! ```
26//!
27//! Developer notes:
28//! - For parsing and layout details see `engine::html`, `engine::css`, and `engine::layouter`.
29//! - For platform integration see `platform::{network, renderer, system}`.
30//! - Keep public API small and document invariants for Tab lifecycle and fetch handling.

50pub struct PendingFetches {
51    /// Maps (id) to (window_id, tab_id, FetchKind, Url)
52    /// Id is used to track pending fetch requests.
53    map: HashMap<usize, (WindowId, TabId, FetchKind, Url)>,
54    counter: usize,
55}
56
57impl PendingFetches {
58    pub fn new() -> Self {
59        Self {
60            map: HashMap::new(),
61            counter: 0,
62        }
63    }
64
65    /// URLとFetchKindを受け取り、一意IDを生成して登録
66    pub fn insert(
67        &mut self,
68        window_id: WindowId,
69        tab_id: TabId,
70        kind: FetchKind,
71        url: Url,
72    ) -> usize {
73        self.counter += 1;
74
75        let id = self.generate_id(&url);
76
77        self.map.insert(id, (window_id, tab_id, kind, url));
78        id
79    }
80
81    fn generate_id(&self, url: &Url) -> usize {
82        // URLをハッシュ化
83        let mut hasher = DefaultHasher::new();
84        url.hash(&mut hasher);
85        let url_hash = hasher.finish() as usize;
86
87        // 現在時刻ナノ秒
88        let now = SystemTime::now()
89            .duration_since(UNIX_EPOCH)
90            .expect("Time went backwards")
91            .as_nanos() as usize;
92
93        // ナノ秒 XOR カウンタ XOR URLハッシュ
94        now ^ self.counter ^ url_hash
95    }
96
97    pub fn remove(&mut self, id: usize) -> Option<(WindowId, TabId, FetchKind, Url)> {
98        self.map.remove(&id)
99    }
100}

102/// Main browser application struct.
103///
104/// Responsibilities:
105/// - Manage the window collection and per-window [`BrowserUi`] instances.
106/// - Forward winit window events to each window's UI.
107/// - Coordinate resource loading and route fetched results to the owning window.
108///
109/// Tab state, input handling, and rendering are delegated to [`BrowserUi`] /
110/// [`BrowserRenderer`].
111///
112/// Typical lifecycle:
113/// 1. Construct `BrowserApp::new(...)`, which wires platform components (network, system).
114/// 2. Create `Tab` objects, wrap them in a `BrowserUi`, and call `set_default_ui`.
115/// 3. Call `run()` to start the event loop. Each loop iteration:
116///    - Forward winit events to the owning window's `BrowserUi`.
117///    - Tick the UI, forward fetch requests to the network, and route responses back.
118///
119/// Example usage:
120/// ```no_run
121/// use orinium_browser::browser::{BrowserApp, BrowserUi, Tab};
122///
123/// let mut tab = Tab::default();
124/// tab.navigate("resource:///test/test.html".parse().unwrap());
125/// let mut app = BrowserApp::default();
126/// app.set_default_ui(BrowserUi::with_tab(tab));
127/// app.run().unwrap();
128/// ```
129pub struct BrowserApp {
130    /// Maps each window to its UI (tabs, input state, renderer).
131    windows: HashMap<WindowId, BrowserUi>,
132    /// Default window size used when opening a new window.
133    default_window_size: (u32, u32),
134    /// Default window title used when opening a new window.
135    default_window_title: String,
136    network: BrowserResourceLoader,
137    pending_fetches: PendingFetches,
138    /// UI used when the first window opens (if set before `run()`).
139    default_ui: Option<BrowserUi>,
140}
141
142impl Default for BrowserApp {
143    fn default() -> Self {
144        Self::new((1280, 800), "Orinium Browser".to_string()).unwrap()
145    }
146}

148impl BrowserApp {
149    /// Starts the main browser event loop asynchronously.
150    /// Returns an error if no default UI was set via `set_default_ui`.
151    pub fn run(self) -> Result<()> {
152        if self.default_ui.is_none() {
153            anyhow::bail!("set_default_ui must be called before run()");
154        }
155        run_with_winit_backend(self)
156    }
157
158    /// Creates a new browser instance with the given default window size and title.
159    /// Windows are registered later via `open_window`.
160    pub fn new(
161        default_window_size: (u32, u32),
162        default_window_title: String,
163    ) -> Result<Self, io::Error> {
164        let network = BrowserResourceLoader::new(Some(Rc::new(NetworkCore::new()?)));
165
166        Ok(Self {
167            windows: HashMap::new(),
168            default_window_size,
169            default_window_title,
170            network,
171            pending_fetches: PendingFetches::new(),
172            default_ui: None,
173        })
174    }
175
176    /// Registers a new window with the given id, size, title, scale factor, and associated UI.
177    pub fn open_window(
178        &mut self,
179        window_id: WindowId,
180        window_size: (u32, u32),
181        window_title: String,
182        scale_factor: f64,
183        mut root_ui: BrowserUi,
184    ) {
185        root_ui.set_window(window_size, scale_factor, window_title);
186        self.windows.insert(window_id, root_ui);
187    }
188
189    /// Removes a window's state when the window is closed.
190    pub fn close_window(&mut self, window_id: WindowId) {
191        self.windows.remove(&window_id);
192    }
193
194    /// Returns the default window size for opening new windows.
195    pub fn default_window_size(&self) -> (f32, f32) {
196        (
197            self.default_window_size.0 as f32,
198            self.default_window_size.1 as f32,
199        )
200    }
201
202    /// Returns the default window title for opening new windows.
203    pub fn default_window_title(&self) -> String {
204        self.default_window_title.clone()
205    }
206
207    /// Sets the UI to use when the first window opens.
208    /// Must be called before `run()`.
209    pub fn set_default_ui(&mut self, ui: BrowserUi) {
210        self.default_ui = Some(ui);
211    }
212
213    /// Takes the default UI, or returns `None` if not set.
214    pub fn take_default_ui(&mut self) -> Option<BrowserUi> {
215        self.default_ui.take()
216    }
217
218    /// Handles a `winit` window event for the given window and returns a `BrowserCommand`.
219    pub fn handle_window_event(
220        &mut self,
221        window_id: WindowId,
222        event: WindowEvent,
223        gpu: &mut GpuRenderer,
224    ) -> BrowserCommand {
225        let browser_cmd = match self.windows.get_mut(&window_id) {
226            Some(ui) => ui.handle_window_event(event, gpu),
227            None => BrowserCommand::None,
228        };
229        let cmd_from_tick = self.tick(window_id);
230        match browser_cmd {
231            BrowserCommand::None => {
232                if matches!(cmd_from_tick, BrowserCommand::RequestRedraw) {
233                    self.redraw(window_id, gpu);
234                }
235                cmd_from_tick
236            }
237            BrowserCommand::RenameWindowTitle => {
238                if matches!(cmd_from_tick, BrowserCommand::RequestRedraw) {
239                    // tick() が追加の処理を要求 → RequestRedraw に昇格させる。
240                    // RequestRedraw のハンドラはタイトル設定も行うので情報は失われない。
241                    self.redraw(window_id, gpu);
242                    BrowserCommand::RequestRedraw
243                } else {
244                    browser_cmd
245                }
246            }
247            _ => {
248                if matches!(cmd_from_tick, BrowserCommand::RequestRedraw) {
249                    self.redraw(window_id, gpu);
250                }
251                browser_cmd
252            }
253        }
254    }
255
256    /// Rebuilds the render tree and sends draw commands to the GPU for the given window.
257    pub fn redraw(&mut self, window_id: WindowId, gpu: &mut GpuRenderer) {
258        let Some(ui) = self.windows.get_mut(&window_id) else {
259            return;
260        };
261        ui.redraw(gpu);
262    }
263
264    /// Applies the current draw commands for the given window to the GPU renderer.
265    pub fn apply_draw_commands(&self, window_id: WindowId, gpu: &mut GpuRenderer) {
266        if let Some(ui) = self.windows.get(&window_id) {
267            ui.apply_draw_commands(gpu);
268        }
269    }
270
271    /// Advances background page work for a window between OS events.
272    ///
273    /// This keeps animated custom controls, such as an active audio timer,
274    /// repainting even when the user is not moving the pointer.
275    pub(crate) fn poll_window(&mut self, window_id: WindowId) -> bool {
276        matches!(self.tick(window_id), BrowserCommand::RequestRedraw)
277    }
278
279    /// Returns the current window size for the given window as `(width, height)` in floating-point pixels.
280    pub fn window_size(&self, window_id: WindowId) -> (f32, f32) {
281        match self.windows.get(&window_id) {
282            Some(ui) => {
283                let (width, height) = ui.window_size();
284                (width as f32, height as f32)
285            }
286            None => (
287                self.default_window_size.0 as f32,
288                self.default_window_size.1 as f32,
289            ),
290        }
291    }
292
293    /// Returns the window title for the given window.
294    pub fn window_title(&self, window_id: WindowId) -> String {
295        match self.windows.get(&window_id) {
296            Some(ui) => ui.window_title(),
297            None => self.default_window_title.clone(),
298        }
299    }
300
301    /// Ticks the given window's UI and forwards its fetch requests to the network.
302    fn tick(&mut self, window_id: WindowId) -> BrowserCommand {
303        self.handle_network_messages();
304
305        let Some(ui) = self.windows.get_mut(&window_id) else {
306            return BrowserCommand::None;
307        };
308        let outcome = ui.tick();
309
310        for fetch in outcome.fetches {
311            let url = fetch.request.url;
312            let kind = fetch.request.kind;
313            log::info!("Fetch requested in App: url={}", url);
314            let request = match &kind {
315                FetchKind::JavaScript {
316                    method,
317                    headers,
318                    body,
319                    ..
320                } => NetworkRequest {
321                    url: url.to_string(),
322                    method: method.clone(),
323                    headers: headers.clone(),
324                    body: body.clone(),
325                },
326                _ => NetworkRequest::get(url.to_string()),
327            };
328            let id = self
329                .pending_fetches
330                .insert(window_id, fetch.tab_id, kind, url);
331            self.network.fetch_request_async(request, id);
332        }
333
334        if outcome.needs_redraw {
335            BrowserCommand::RequestRedraw
336        } else {
337            BrowserCommand::None
338        }
339    }
340
341    fn handle_network_messages(&mut self) {
342        let messages = self.network.try_receive();
343
344        for msg in messages {
345            log::info!("Network message received in App for fetch_id={}", msg.id);
346
347            // pending_fetches から fetch 情報を取得
348            let Some((window_id, tab_id, kind, url)) = self.pending_fetches.remove(msg.id) else {
349                log::warn!("No pending fetch found for fetch_id={}", msg.id);
350                continue;
351            };
352
353            // 該当ウィンドウの UI へ配送
354            let Some(ui) = self.windows.get_mut(&window_id) else {
355                log::warn!("There is no window called id={:?}", window_id);
356                continue;
357            };
358            ui.deliver_fetch(&tab_id, kind, url, msg.response);
359        }
360    }
361}

363fn run_with_winit_backend(app: BrowserApp) -> Result<()> {
364    configure_winit_backend_for_wslg();
365    if env::var_os("ORINIUM_FORCE_X11").is_some() {
366        configure_winit_backend_forced_x11();
367    }
368
369    run_event_loop(app)
370}
371
372fn run_event_loop(app: BrowserApp) -> Result<()> {
373    let event_loop = winit::event_loop::EventLoop::new()?;
374    event_loop.set_control_flow(winit::event_loop::ControlFlow::Poll);
375    let mut app = App::new(app);
376    event_loop.run_app(&mut app)?;
377    Ok(())
378}

380fn configure_winit_backend_forced_x11() {
381    let current = env::var("WINIT_UNIX_BACKEND").ok();
382    let should_force_x11 = !matches!(current.as_deref(), Some("x11"));
383
384    if should_force_x11 {
385        unsafe {
386            env::set_var("WINIT_UNIX_BACKEND", "x11");
387            env::remove_var("WAYLAND_DISPLAY");
388        }
389        log::info!("Forcing X11 (WINIT_UNIX_BACKEND=x11, WAYLAND_DISPLAY cleared)");
390    }
391}
392
393fn configure_winit_backend_for_wslg() {
394    let is_wsl = env::var_os("WSL_DISTRO_NAME").is_some() || env::var_os("WSL_INTEROP").is_some();
395    if !is_wsl {
396        return;
397    }
398
399    // On WSLg, Wayland is often unstable; default to X11 unless explicitly requested.
400    if env::var_os("ORINIUM_PREFER_WAYLAND").is_some() {
401        return;
402    }
403
404    let current = env::var("WINIT_UNIX_BACKEND").ok();
405    let should_force_x11 = !matches!(current.as_deref(), Some("x11"));
406
407    if should_force_x11 {
408        unsafe {
409            env::set_var("WINIT_UNIX_BACKEND", "x11");
410            env::remove_var("WAYLAND_DISPLAY");
411        }
412        log::info!("WSLg detected: defaulting to X11 backend for stability");
413    }
