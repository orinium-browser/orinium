1//! Browser core: application entry and lifecycle manager.
2//!
3//! Responsibilities:
4//! - Manage the window collection and each window's [`BrowserUi`].
5//! - Forward winit window events to the owning window's UI.
6//! - Coordinate network/resource loading and route responses to the owning window.
7//!
8//! Tab state, input handling, and rendering are delegated to [`BrowserUi`] /
9//! [`BrowserRenderer`]; this type stays a thin orchestrator.
10//!
11//! Processing flow (high-level):
12//! 1. Initialize platform components (system window, GPU renderer, network core).
13//! 2. Create and register `BrowserUi` instances and navigate to initial URLs.
14//! 3. Enter event loop: forward events -> delegate to `BrowserUi` -> route fetches.
15//!
16//! Example (for contributors / local testing):
17//! ```no_run
18//! use orinium_browser::browser::{BrowserApp, BrowserUi, Tab};
19//!
20//! let mut tab = Tab::default();
21//! tab.navigate("resource:///test/test.html".parse().unwrap());
22//! let mut app = BrowserApp::default();
23//! app.set_default_ui(BrowserUi::with_tab(tab));
24//! app.run().unwrap();
25//! ```
26//!
27//! Developer notes:
28//! - For parsing and layout details see `engine::html`, `engine::css`, and `engine::layouter`.
29//! - For platform integration see `platform::{network, renderer, system}`.
30//! - Keep public API small and document invariants for Tab lifecycle and fetch handling.

51pub struct PendingFetches {
52    /// Maps (id) to (window_id, tab_id, FetchKind, Url)
53    /// Id is used to track pending fetch requests.
54    map: HashMap<usize, (WindowId, TabId, FetchKind, Url, u64)>,
55    counter: usize,
56}
57
58impl PendingFetches {
59    pub fn new() -> Self {
60        Self {
61            map: HashMap::new(),
62            counter: 0,
63        }
64    }
65
66    /// URLとFetchKindを受け取り、一意IDを生成して登録
67    pub fn insert(
68        &mut self,
69        window_id: WindowId,
70        tab_id: TabId,
71        kind: FetchKind,
72        url: Url,
73        generation: u64,
74    ) -> usize {
75        self.counter += 1;
76
77        let id = self.generate_id(&url);
78
79        self.map.insert(id, (window_id, tab_id, kind, url, generation));
80        id
81    }
82
83    fn generate_id(&self, url: &Url) -> usize {
84        // URLをハッシュ化
85        let mut hasher = DefaultHasher::new();
86        url.hash(&mut hasher);
87        let url_hash = hasher.finish() as usize;
88
89        // 現在時刻ナノ秒
90        let now = SystemTime::now()
91            .duration_since(UNIX_EPOCH)
92            .expect("Time went backwards")
93            .as_nanos() as usize;
94
95        // ナノ秒 XOR カウンタ XOR URLハッシュ
96        now ^ self.counter ^ url_hash
97    }
98
99    pub fn remove(&mut self, id: usize) -> Option<(WindowId, TabId, FetchKind, Url, u64)> {
100        self.map.remove(&id)
101    }
102}

104/// Main browser application struct.
105///
106/// Responsibilities:
107/// - Manage the window collection and per-window [`BrowserUi`] instances.
108/// - Forward winit window events to each window's UI.
109/// - Coordinate resource loading and route fetched results to the owning window.
110///
111/// Tab state, input handling, and rendering are delegated to [`BrowserUi`] /
112/// [`BrowserRenderer`].
113///
114/// Typical lifecycle:
115/// 1. Construct `BrowserApp::new(...)`, which wires platform components (network, system).
116/// 2. Create `Tab` objects, wrap them in a `BrowserUi`, and call `set_default_ui`.
117/// 3. Call `run()` to start the event loop. Each loop iteration:
118///    - Forward winit events to the owning window's `BrowserUi`.
119///    - Tick the UI, forward fetch requests to the network, and route responses back.
120///
121/// Example usage:
122/// ```no_run
123/// use orinium_browser::browser::{BrowserApp, BrowserUi, Tab};
124///
125/// let mut tab = Tab::default();
126/// tab.navigate("resource:///test/test.html".parse().unwrap());
127/// let mut app = BrowserApp::default();
128/// app.set_default_ui(BrowserUi::with_tab(tab));
129/// app.run().unwrap();
130/// ```
131pub struct BrowserApp {
132    /// Maps each window to its UI (tabs, input state, renderer).
133    windows: HashMap<WindowId, BrowserUi>,
134    /// Default window size used when opening a new window.
135    default_window_size: (u32, u32),
136    /// Default window title used when opening a new window.
137    default_window_title: String,
138    network: BrowserResourceLoader,
139    pending_fetches: PendingFetches,
140    /// UI used when the first window opens (if set before `run()`).
141    default_ui: Option<BrowserUi>,
142}
143
144impl Default for BrowserApp {
145    fn default() -> Self {
146        Self::new((1280, 800), "Orinium Browser".to_string()).unwrap()
147    }
148}

154        if self.default_ui.is_none() {

215    /// Handles a `winit` window event for the given window and returns a `BrowserCommand`.
216    pub fn handle_window_event(
217        &mut self,
218        window_id: WindowId,
219        event: WindowEvent,
220        sink: &mut dyn DrawSink,
221    ) -> BrowserCommand {
222        let browser_cmd = match self.windows.get_mut(&window_id) {
223            Some(ui) => ui.handle_window_event(event, sink),
224            None => BrowserCommand::None,
225        };
226        let cmd_from_tick = self.tick(window_id);
227        match browser_cmd {
228            BrowserCommand::None => {
229                if matches!(cmd_from_tick, BrowserCommand::RequestRedraw) {
230                    self.redraw(window_id, sink);
231                }
232                cmd_from_tick
233            }
234            BrowserCommand::RenameWindowTitle => {
235                if matches!(cmd_from_tick, BrowserCommand::RequestRedraw) {
236                    // tick() が追加の処理を要求 → RequestRedraw に昇格させる。
237                    // RequestRedraw のハンドラはタイトル設定も行うので情報は失われない。
238                    self.redraw(window_id, sink);
239                    BrowserCommand::RequestRedraw
240                } else {
241                    browser_cmd
242                }
243            }
244            _ => {
245                if matches!(cmd_from_tick, BrowserCommand::RequestRedraw) {
246                    self.redraw(window_id, sink);
247                }
248                browser_cmd
249            }
250        }
251    }

253    /// Rebuilds the render tree and sends draw commands to the GPU for the given window.
254    pub fn redraw(&mut self, window_id: WindowId, sink: &mut dyn DrawSink) {
255        let Some(ui) = self.windows.get_mut(&window_id) else {
256            return;
257        };
258        ui.redraw(sink);
259    }

263        if let Some(ui) = self.windows.get(&window_id) {
264            ui.apply_draw_commands(sink);
265        }

268    /// Advances background page work for a window between OS events.
269    ///
270    /// This keeps animated custom controls, such as an active audio timer,
271    /// repainting even when the user is not moving the pointer.
272    pub(crate) fn poll_window(&mut self, window_id: WindowId) -> bool {

276    /// Returns the current window size for the given window as `(width, height)` in floating-point pixels.

278        match self.windows.get(&window_id) {
279            Some(ui) => {

281                (width as f32, height as f32)

283            None => (

298    /// Ticks the given window's UI and forwards its fetch requests to the network.
299    fn tick(&mut self, window_id: WindowId) -> BrowserCommand {
300        self.handle_network_messages();
301
302        let Some(ui) = self.windows.get_mut(&window_id) else {
303            return BrowserCommand::None;
304        };
305        let outcome = ui.tick();
306
307        for fetch in outcome.fetches {
308            let url = fetch.request.url;
309            let kind = fetch.request.kind;
310            let initiator = fetch.request.origin;
311            log::info!("Fetch requested in App: url={}", url);
312            let mut request = match &kind {
313                FetchKind::JavaScript {
314                    method,
315                    headers,
316                    body,
317                    ..
318                } => NetworkRequest {
319                    url: url.to_string(),
320                    method: method.clone(),
321                    headers: headers.clone(),
322                    body: body.clone(),
323                },
324                _ => NetworkRequest::get(url.to_string()),
325            };
326            // Add browser-controlled Origin / Referer headers and strip any the
327            // page script supplied, so internal resource URLs never leak to
328            // external servers.
329            apply_fetch_headers(&mut request, &initiator, &kind);
330            let id = self.pending_fetches.insert(
331                window_id,
332                fetch.tab_id,
333                kind,
334                url,
335                fetch.request.generation,
336            );
337            // The initiator gates access to internal schemes inside the loader.
338            // We also track the originating document generation so stale results
339            // from an earlier navigation are dropped when they arrive.
340            self.network.fetch_request_async(request, id, &initiator);
341        }
342
343        if outcome.needs_redraw {
344            BrowserCommand::RequestRedraw
345        } else {
346            BrowserCommand::None
347        }
348    }
349
350    fn handle_network_messages(&mut self) {
351        let messages = self.network.try_receive();
352
353        for msg in messages {
354            log::info!("Network message received in App for fetch_id={}", msg.id);
355
356            // pending_fetches から fetch 情報を取得
357            let Some((window_id, tab_id, kind, url, generation)) = self.pending_fetches.remove(msg.id) else {
358                log::warn!("No pending fetch found for fetch_id={}", msg.id);
359                continue;
360            };
361
362            let Some(ui) = self.windows.get_mut(&window_id) else {
363                log::warn!("There is no window called id={:?}", window_id);
364                continue;
365            };
366            if tab_id.0 != 0 {
367                let Some(tab) = ui.tab(&tab_id) else {
368                    log::warn!("Tab was closed before fetch completed: tab_id={}", tab_id);
369                    continue;
370                };
371                let current_generation = tab.document_generation();
372                if current_generation != generation {
373                    log::info!(
374                        "Dropping stale fetch result for tab_id={} generation={} (current={})",
375                        tab_id,
376                        generation,
377                        current_generation
378                    );
379                    continue;
380                }
381            }
382            ui.deliver_fetch(&tab_id, kind, url, msg.response);
383        }
384    }

443/// Applies the browser-controlled `Origin` / `Referer` request headers.
444///
445/// - Any `Origin` / `Referer` supplied by page scripts is stripped.
446/// - An `Origin` header is sent for CORS-mode requests (`fetch`/`XMLHttpRequest`)
447///   and for non-GET requests; the value is the initiator's serialized origin
448///   (`"null"` for opaque/internal pages), as required by the Fetch standard.
449/// - A `Referer` origin-only header is sent for network origins only; opaque
450///   pages never send one, so bundled `resource:` URLs cannot be leaked.
451fn apply_fetch_headers(request: &mut NetworkRequest, initiator: &Origin, kind: &FetchKind) {
452    request.headers.retain(|(name, _)| {
453        let name = name.to_ascii_lowercase();
454        name != "origin" && name != "referer"
455    });
456
457    if !request.url.starts_with("http://") && !request.url.starts_with("https://") {
458        return;
459    }
460
461    let is_cors_request = matches!(kind, FetchKind::JavaScript { .. }) || request.method != "GET";
462    if is_cors_request {
463        request
464            .headers
465            .push(("Origin".to_string(), initiator.ascii_serialization()));
466    }
467    if initiator.is_network() {
468        request
469            .headers
470            .push(("Referer".to_string(), initiator.ascii_serialization()));
471    }
472}

474fn run_with_winit_backend(app: BrowserApp) -> Result<()> {
475    configure_winit_backend_for_wslg();
476    if env::var_os("ORINIUM_FORCE_X11").is_some() {
477        configure_winit_backend_forced_x11();
478    }
479
480    run_event_loop(app)
481}
482
483fn run_event_loop(app: BrowserApp) -> Result<()> {
484    let event_loop = winit::event_loop::EventLoop::new()?;
485    event_loop.set_control_flow(winit::event_loop::ControlFlow::Poll);
486    let mut app = App::new(Box::new(app));
487    event_loop.run_app(&mut app)?;
488    Ok(())
489}
490
491fn configure_winit_backend_forced_x11() {
492    let current = env::var("WINIT_UNIX_BACKEND").ok();
493    let should_force_x11 = !matches!(current.as_deref(), Some("x11"));
494
495    if should_force_x11 {
496        unsafe {
497            env::set_var("WINIT_UNIX_BACKEND", "x11");
498            env::remove_var("WAYLAND_DISPLAY");
499        }
500        log::info!("Forcing X11 (WINIT_UNIX_BACKEND=x11, WAYLAND_DISPLAY cleared)");
501    }
502}
503
504fn configure_winit_backend_for_wslg() {
505    let is_wsl = env::var_os("WSL_DISTRO_NAME").is_some() || env::var_os("WSL_INTEROP").is_some();
506    if !is_wsl {
507        return;
508    }
509
510    // On WSLg, Wayland is often unstable; default to X11 unless explicitly requested.
511    if env::var_os("ORINIUM_PREFER_WAYLAND").is_some() {
512        return;
513    }
514
515    let current = env::var("WINIT_UNIX_BACKEND").ok();
516    let should_force_x11 = !matches!(current.as_deref(), Some("x11"));
517
518    if should_force_x11 {
519        unsafe {
520            env::set_var("WINIT_UNIX_BACKEND", "x11");
521            env::remove_var("WAYLAND_DISPLAY");
522        }
523        log::info!("WSLg detected: defaulting to X11 backend for stability");
524    }
