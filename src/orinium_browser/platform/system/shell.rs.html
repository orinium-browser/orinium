1//! The contract between the browser core and the OS shell.
2//!
3//! The browser core drives the browser; [`App`](super::App) owns the `winit` event loop,
4//! the OS windows and the GPU surface. Neither side may depend on the other's
5//! internals, so the seam is declared *here*, in the layer that owns the OS,
6//! and implemented by the browser core.
7//!
8//! Everything crossing this seam is a value owned by `platform`:
9//!
10//! - [`WindowGeometry`] instead of a per-window UI object
11//! - [`ShellCommand`] instead of a browser-internal command enum
12//! - plain `winit` / GPU handles instead of engine state
13//!
14//! That way the event loop can hold any [`BrowserHost`] — including one
15//! living in another process — without naming a single browser type.
16use winit::event::WindowEvent;
17use winit::window::WindowId;
18
19use crate::platform::renderer::draw_sink::DrawSink;
20
21/// An instruction for the OS shell, returned after handling an event.
22#[derive(Debug, Clone, PartialEq)]
23pub enum ShellCommand {
24    /// Nothing to do.
25    None,
26    /// Close the window; the shell exits once no window remains.
27    Exit,
28    /// Repaint the window.
29    RequestRedraw,
30    /// Re-read the window title from the host and apply it.
31    RenameWindowTitle,
32    /// Create an additional window.
33    OpenNewWindow,
34    /// Enable or disable OS IME input near the last click position.
35    SetImeAllowed { allowed: bool, position: (f64, f64) },
36}
37
38/// The properties the shell needs to realise a window.
39#[derive(Debug, Clone)]
40pub struct WindowGeometry {
41    /// Initial inner size, in physical pixels.
42    pub size: (u32, u32),
43    /// Initial title.
44    pub title: String,
45    /// Display scale factor at creation time.
46    pub scale_factor: f64,
47}
48
49/// The browser side of the shell contract.
50///
51/// The shell calls these methods; the browser core implements them. Every
52/// method takes and returns `platform`-owned values only.
53pub trait BrowserHost {
54    /// The size to give a newly opened window.
55    fn default_window_size(&self) -> (u32, u32);
56
57    /// The title to give a newly opened window.
58    fn default_window_title(&self) -> String;
59
60    /// Registers a window with the host.
61    ///
62    /// The host decides what the window shows: the first call installs the UI
63    /// it was given before the loop started, and any later call opens an
64    /// empty window.
65    fn open_window(&mut self, window_id: WindowId, geometry: WindowGeometry);
66
67    /// Whether a UI is still waiting to be installed in the first window.
68    ///
69    /// The shell checks this before creating the first window so it can report
70    /// a misconfigured startup instead of silently opening a blank window.
71    fn has_pending_default_window(&self) -> bool;
72
73    /// Forgets a window that the shell closed.
74    fn close_window(&mut self, window_id: WindowId);
75
76    /// The current title for a window.
77    fn window_title(&self, window_id: WindowId) -> String;
78
79    /// Handles one OS event for a window and reports what the shell should do.
80    fn handle_window_event(
81        &mut self,
82        window_id: WindowId,
83        event: WindowEvent,
84        sink: &mut dyn DrawSink,
85    ) -> ShellCommand;
86
87    /// Uploads the window's pending draw commands to the sink, without
88    /// presenting.
89    fn apply_draw_commands(&mut self, window_id: WindowId, sink: &mut dyn DrawSink);
90
91    /// Advances background page work for a window, returning whether it needs
92    /// a repaint.
93    fn poll_window(&mut self, window_id: WindowId) -> bool;
94}
