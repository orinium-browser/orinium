6/// (fontdb ID, glyph_id, font_size_bits)
7type GlyphKey = (fontdb::ID, u32, u32);
8
9/// (layer_index, alloc_id, rectangle_packed, actual_width, actual_height)
10type GlyphCacheValue = (u32, etagere::AllocId, etagere::Rectangle, u32, u32);
11
12/// (layer_index, rectangle, alpha_mask, actual_width, actual_height)
13type PendingUpload = (u32, etagere::Rectangle, Vec<u8>, u32, u32);
14
15/// Normalized UV rect: (layer, u, v, width, height)
16pub type GlyphUVRect = (u32, f32, f32, f32, f32);

18/// A packed glyph atlas managed as a wgpu 2D array texture.
19///
20/// Each layer holds a single row of shelf-packed glyph bitmaps.
21/// When a layer runs out of space, a new layer is added (up to
22/// MAX_LAYERS).  When all layers are full, the atlas is grown by
23/// doubling the texture size and re-packing every existing glyph.
24pub struct GlyphAtlas {
25    texture: wgpu::Texture,
26    view: wgpu::TextureView,
27    size: u32,
28    layers: u32,
29    allocators: Vec<BucketedAtlasAllocator>,
30    glyph_map: HashMap<GlyphKey, GlyphCacheValue>,
31    pending_uploads: Vec<PendingUpload>,
32}
33
34const MAX_LAYERS: u32 = 16;
35const INITIAL_SIZE: u32 = 1024;
36
37impl GlyphAtlas {
38    pub fn new(device: &wgpu::Device) -> Self {
39        let size = INITIAL_SIZE;
40        let layers = 1;
41        let texture = device.create_texture(&wgpu::TextureDescriptor {
42            label: Some("Glyph Atlas"),
43            size: wgpu::Extent3d {
44                width: size,
45                height: size,
46                depth_or_array_layers: layers,
47            },
48            mip_level_count: 1,
49            sample_count: 1,
50            dimension: wgpu::TextureDimension::D2,
51            format: wgpu::TextureFormat::R8Unorm,
52            usage: wgpu::TextureUsages::TEXTURE_BINDING | wgpu::TextureUsages::COPY_DST,
53            view_formats: &[],
54        });
55        let view = texture.create_view(&wgpu::TextureViewDescriptor {
56            label: Some("Glyph Atlas View"),
57            dimension: Some(wgpu::TextureViewDimension::D2Array),
58            ..Default::default()
59        });
60        let allocator = BucketedAtlasAllocator::new(size2(size as i32, size as i32));
61
62        Self {
63            texture,
64            view,
65            size,
66            layers,
67            allocators: vec![allocator],
68            glyph_map: HashMap::new(),
69            pending_uploads: Vec::new(),
70        }
71    }
72
73    pub fn texture_view(&self) -> &wgpu::TextureView {
74        &self.view
75    }

77    /// Returns cached (layer, u, v, w, h) if the glyph is already in the atlas.
78    pub fn lookup(&self, font_key: FontKey, glyph_id: u32, font_size: f32) -> Option<GlyphUVRect> {
79        let key = (font_key.0, glyph_id, font_size.to_bits());
80        let (layer, _alloc_id, rect, aw, ah) = self.glyph_map.get(&key)?;
81        let (u, v) = (
82            rect.min.x as f32 / self.size as f32,
83            rect.min.y as f32 / self.size as f32,
84        );
85        let (w, h) = (*aw as f32 / self.size as f32, *ah as f32 / self.size as f32);
86        Some((*layer, u, v, w, h))
87    }

89    pub fn upload(
90        &mut self,
91        device: &wgpu::Device,
92        queue: &wgpu::Queue,
93        font_key: FontKey,
94        glyph_id: u32,
95        font_size: f32,
96        alpha_mask: &[u8],
97        mask_width: u32,
98        mask_height: u32,
99    ) -> GlyphUVRect {
100        if mask_width == 0 || mask_height == 0 {
101            return (0, 0.0, 0.0, 0.0, 0.0);
102        }
103
104        let key = (font_key.0, glyph_id, font_size.to_bits());
105
106        if let Some((layer, _alloc_id, rect, aw, ah)) = self.glyph_map.get(&key) {
107            let (u, v) = (
108                rect.min.x as f32 / self.size as f32,
109                rect.min.y as f32 / self.size as f32,
110            );
111            let (w, h) = (*aw as f32 / self.size as f32, *ah as f32 / self.size as f32);
112            return (*layer, u, v, w, h);
113        }
114
115        let item_w = mask_width.max(1) as i32;
116        let item_h = mask_height.max(1) as i32;
117
118        for (layer_idx, alloc) in self.allocators.iter_mut().enumerate() {
119            if let Some(allocation) = alloc.allocate(size2(item_w, item_h)) {
120                let rect = allocation.rectangle;
121                let alloc_id = allocation.id;
122                self.glyph_map.insert(
123                    key,
124                    (layer_idx as u32, alloc_id, rect, mask_width, mask_height),
125                );
126                self.pending_uploads.push((
127                    layer_idx as u32,
128                    rect,
129                    alpha_mask.to_vec(),
130                    mask_width,
131                    mask_height,
132                ));
133                let (u, v) = (
134                    rect.min.x as f32 / self.size as f32,
135                    rect.min.y as f32 / self.size as f32,
136                );
137                let (w, h) = (
138                    mask_width as f32 / self.size as f32,
139                    mask_height as f32 / self.size as f32,
140                );
141                return (layer_idx as u32, u, v, w, h);
142            }
143        }
144
145        if self.layers < MAX_LAYERS {
146            let new_layer = self.layers;
147            self.layers += 1;
148            let mut alloc = BucketedAtlasAllocator::new(size2(self.size as i32, self.size as i32));
149            if let Some(allocation) = alloc.allocate(size2(item_w, item_h)) {
150                let rect = allocation.rectangle;
151                let alloc_id = allocation.id;
152                self.allocators.push(alloc);
153                self.glyph_map
154                    .insert(key, (new_layer, alloc_id, rect, mask_width, mask_height));
155                self.pending_uploads.push((
156                    new_layer,
157                    rect,
158                    alpha_mask.to_vec(),
159                    mask_width,
160                    mask_height,
161                ));
162
163                let texture = device.create_texture(&wgpu::TextureDescriptor {
164                    label: Some("Glyph Atlas"),
165                    size: wgpu::Extent3d {
166                        width: self.size,
167                        height: self.size,
168                        depth_or_array_layers: self.layers,
169                    },
170                    mip_level_count: 1,
171                    sample_count: 1,
172                    dimension: wgpu::TextureDimension::D2,
173                    format: wgpu::TextureFormat::R8Unorm,
174                    usage: wgpu::TextureUsages::TEXTURE_BINDING | wgpu::TextureUsages::COPY_DST,
175                    view_formats: &[],
176                });
177                let mut encoder =
178                    device.create_command_encoder(&wgpu::CommandEncoderDescriptor::default());
179                for old_layer in 0..new_layer {
180                    encoder.copy_texture_to_texture(
181                        wgpu::TexelCopyTextureInfo {
182                            texture: &self.texture,
183                            mip_level: 0,
184                            origin: wgpu::Origin3d {
185                                x: 0,
186                                y: 0,
187                                z: old_layer,
188                            },
189                            aspect: wgpu::TextureAspect::All,
190                        },
191                        wgpu::TexelCopyTextureInfo {
192                            texture: &texture,
193                            mip_level: 0,
194                            origin: wgpu::Origin3d {
195                                x: 0,
196                                y: 0,
197                                z: old_layer,
198                            },
199                            aspect: wgpu::TextureAspect::All,
200                        },
201                        wgpu::Extent3d {
202                            width: self.size,
203                            height: self.size,
204                            depth_or_array_layers: 1,
205                        },
206                    );
207                }
208                queue.submit(std::iter::once(encoder.finish()));
209
210                self.texture = texture;
211                self.view = self.texture.create_view(&wgpu::TextureViewDescriptor {
212                    label: Some("Glyph Atlas View"),
213                    dimension: Some(wgpu::TextureViewDimension::D2Array),
214                    ..Default::default()
215                });
216
217                self.flush_uploads(queue);
218
219                let (u, v) = (
220                    rect.min.x as f32 / self.size as f32,
221                    rect.min.y as f32 / self.size as f32,
222                );
223                let (w, h) = (
224                    mask_width as f32 / self.size as f32,
225                    mask_height as f32 / self.size as f32,
226                );
227                return (new_layer, u, v, w, h);
228            }
229        }
230
231        log::warn!(target:"GlyphAtlas", "glyph atlas full ({} layers used)", self.layers);
232        (0, 0.0, 0.0, 0.0, 0.0)
233    }

235    pub fn flush_uploads(&mut self, queue: &wgpu::Queue) {
236        for (layer, rect, data, actual_w, actual_h) in self.pending_uploads.drain(..) {
237            let width = actual_w.max(1);
238            let height = actual_h.max(1);
239
240            let bytes_per_pixel = 1u32;
241            let stride = width * bytes_per_pixel;
242            let align = wgpu::COPY_BYTES_PER_ROW_ALIGNMENT;
243            let padded_stride = ((stride + align - 1) / align) * align;
244
245            let layout = wgpu::TexelCopyBufferLayout {
246                offset: 0,
247                bytes_per_row: Some(padded_stride),
248                rows_per_image: Some(height),
249            };
250
251            if padded_stride == stride {
252                queue.write_texture(
253                    wgpu::TexelCopyTextureInfo {
254                        texture: &self.texture,
255                        mip_level: 0,
256                        origin: wgpu::Origin3d {
257                            x: rect.min.x as u32,
258                            y: rect.min.y as u32,
259                            z: layer,
260                        },
261                        aspect: wgpu::TextureAspect::All,
262                    },
263                    &data,
264                    layout,
265                    wgpu::Extent3d {
266                        width,
267                        height,
268                        depth_or_array_layers: 1,
269                    },
270                );
271            } else {
272                let mut padded = Vec::with_capacity((padded_stride * height) as usize);
273                for row in data.chunks(stride as usize) {
274                    padded.extend_from_slice(row);
275                    let pad = (padded_stride - stride) as usize;
276                    if pad > 0 {
277                        padded.extend(std::iter::repeat(0u8).take(pad));
278                    }
279                }
280                queue.write_texture(
281                    wgpu::TexelCopyTextureInfo {
282                        texture: &self.texture,
283                        mip_level: 0,
284                        origin: wgpu::Origin3d {
285                            x: rect.min.x as u32,
286                            y: rect.min.y as u32,
287                            z: layer,
288                        },
289                        aspect: wgpu::TextureAspect::All,
290                    },
291                    &padded,
292                    layout,
293                    wgpu::Extent3d {
294                        width,
295                        height,
296                        depth_or_array_layers: 1,
297                    },
298                );
299            }
300        }
301    }
