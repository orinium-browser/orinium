1use std::sync::{LazyLock, Mutex};
2
3use orinium_text::FontSystem;
4
5static GLOBAL_FONT_SYSTEM: LazyLock<Mutex<FontSystem>> = LazyLock::new(|| {
6    let font_sys = if let Ok(p) = std::env::var("ORINIUM_FONT") {
7        let source = orinium_text::fontdb::Source::File(p.into());
8        FontSystem::new_with_fonts(vec![source])
9    } else {
10        FontSystem::new()
11    };
12    Mutex::new(font_sys)
13});
14
15pub fn with_global_font_system<T>(f: impl FnOnce(&mut FontSystem) -> T) -> T {
16    let mut fs = GLOBAL_FONT_SYSTEM
17        .lock()
18        .expect("GLOBAL_FONT_SYSTEM lock failed");
19    f(&mut fs)
20}
21
22pub fn global_font_system_ready() -> bool {
23    GLOBAL_FONT_SYSTEM
24        .lock()
25        .map(|fs| !fs.db.is_empty())
26        .unwrap_or(false)
27}
28
29#[allow(dead_code)]
30pub fn load_global_font_data(data: Vec<u8>) -> Vec<orinium_text::FontKey> {
31    let mut fs = GLOBAL_FONT_SYSTEM
32        .lock()
33        .expect("GLOBAL_FONT_SYSTEM lock failed");
34    fs.load_font_data(data)
35}

37pub fn load_global_font_path(path: &str) -> Option<Vec<orinium_text::FontKey>> {
38    let bytes = std::fs::read(path).ok()?;
39    let mut fs = GLOBAL_FONT_SYSTEM
40        .lock()
41        .expect("GLOBAL_FONT_SYSTEM lock failed");
42    Some(fs.load_font_data(bytes))
