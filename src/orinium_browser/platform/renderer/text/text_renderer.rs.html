30const SUBPIXEL_PHASES_X: i32 = 4;
31const BEARING_CACHE_CAPACITY: usize = 32_768;

33fn quantize_subpixel_x(x: f32) -> (f32, u8) {
34    let quantized = (x * SUBPIXEL_PHASES_X as f32).round() as i32;
35    (
36        quantized.div_euclid(SUBPIXEL_PHASES_X) as f32,
37        quantized.rem_euclid(SUBPIXEL_PHASES_X) as u8,
38    )
39}

41/// Pre-clip test for a glyph using layout metrics alone, before any atlas
42/// lookup. `glyph.x` already includes the x-bearing and `width`/`height` are
43/// ink sizes, so the ink rect is known without rasterization. A small slack
44/// covers subpixel quantization and hinting differences between the layout
45/// metrics and the rasterized mask.
46fn glyph_fully_outside_clip(
47    section: &TextSection,
48    glyph_x: f32,
49    glyph_y: f32,
50    glyph_width: f32,
51    glyph_height: f32,
52) -> bool {
53    const CLIP_SLACK_PX: f32 = 4.0;
54    let base_x = section.screen_position.0;
55    let base_y = section.screen_position.1;
56    let clip_l = section.clip_origin.0;
57    let clip_t = section.clip_origin.1;
58    let clip_r = section.clip_origin.0 + section.bounds.0;
59    let clip_b = section.clip_origin.1 + section.bounds.1;
60    let ink_l = base_x + glyph_x - CLIP_SLACK_PX;
61    let ink_t = base_y + glyph_y - CLIP_SLACK_PX;
62    let ink_r = ink_l + glyph_width + 2.0 * CLIP_SLACK_PX;
63    let ink_b = ink_t + glyph_height + 2.0 * CLIP_SLACK_PX;
64    ink_r < clip_l || ink_l > clip_r || ink_b < clip_t || ink_t > clip_b
65}

67fn rasterize_glyph(
68    scale_context: &mut ScaleContext,
69    font_system: &mut FontSystem,
70    font_key: FontKey,
71    glyph_id: u32,
72    font_size: f32,
73    font_weight: u16,
74    phase_x: u8,
75) -> Option<RasterizedMask> {
76    let face_index = font_system.db.face(font_key.0)?.index as usize;
77    let data = font_system.get_font_data(font_key)?;
78    let font = FontRef::from_index(data.as_slice(), face_index)?;
79    let mut scaler = scale_context
80        .builder(font)
81        .size(font_size)
82        .variations([("wght", font_weight as f32)])
83        .hint(true)
84        .build();
85    let mut render = Render::new(&[Source::Outline]);
86    render
87        .format(Format::Alpha)
88        .offset(Vector::new(phase_x as f32 / SUBPIXEL_PHASES_X as f32, 0.0));
89    let image = render.render(&mut scaler, glyph_id as u16)?;
90    Some(RasterizedMask {
91        width: image.placement.width,
92        height: image.placement.height,
93        left: image.placement.left,
94        top: image.placement.top,
95        data: image.data,
96    })
97}

99fn unhinted_bearings(
100    font_system: &mut FontSystem,
101    font_key: FontKey,
102    glyph_id: u32,
103    font_size: f32,
104) -> Option<(f32, f32)> {
105    let face_index = font_system.db.face(font_key.0)?.index;
106    let data = font_system.get_font_data(font_key)?;
107    let face = ttf_parser::Face::parse(data.as_slice(), face_index).ok()?;
108    let bbox = face.glyph_bounding_box(ttf_parser::GlyphId(glyph_id as u16))?;
109    let scale = font_size / face.units_per_em() as f32;
110    Some((bbox.x_min as f32 * scale, bbox.y_max as f32 * scale))
111}

156fn pack_size(w: f32, h: f32) -> u32 {
157    let qw = (w * 64.0).round() as u16;
158    let qh = (h * 64.0).round() as u16;
159    ((qw as u32) << 16) | (qh as u32)

162fn pack_uv_off(u: f32, v: f32) -> u32 {
163    let qu = (u * 2048.0).round() as i16;
164    let qv = (v * 2048.0).round() as i16;
165    ((qu as u32) << 16) | (qv as u16 as u32)
166}

182pub fn build_family_list<'a>(families: &'a [String]) -> Vec<fontdb::Family<'a>> {
183    if families.is_empty() {
184        return vec![fontdb::Family::SansSerif, fontdb::Family::Serif];
185    }
186    let mut list: Vec<fontdb::Family<'a>> = families
187        .iter()
188        .map(|f| {
189            let lower = f.to_ascii_lowercase();
190            match lower.as_str() {
191                "serif" => fontdb::Family::Serif,
192                "sans-serif" => fontdb::Family::SansSerif,
193                "monospace" => fontdb::Family::Monospace,
194                "cursive" => fontdb::Family::Cursive,
195                "fantasy" => fontdb::Family::Fantasy,
196                _ => fontdb::Family::Name(f.as_str()),
197            }
198        })
199        .collect();
200    list.push(fontdb::Family::SansSerif);
201    list.push(fontdb::Family::Serif);
202    list.push(fontdb::Family::Monospace);
203    list
204}

238fn sections_hash(sections: &[TextSection]) -> u64 {
239    let mut h: u64 = 0;
240    for s in sections {
241        h = h.wrapping_mul(6364136223846793005);
242        h ^= s.screen_position.0.to_bits() as u64;
243        h ^= s.screen_position.1.to_bits() as u64;
244        h ^= s.clip_origin.0.to_bits() as u64;
245        h ^= s.clip_origin.1.to_bits() as u64;
246        h ^= s.bounds.0.to_bits() as u64;
247        h ^= s.bounds.1.to_bits() as u64;
248        h ^= s.font_weight as u64;
249        h ^= Arc::as_ptr(&s.layout) as u64;
250    }
251    h
252}

498    pub fn create_buffer_for_text(
499        &mut self,
500        text: &str,
501        style: TextStyle,
502        mut flow_style: TextFlowStyle,
503    ) -> Arc<TextLayout> {
504        flow_style.font_size = quantize_font_size(flow_style.font_size);
505
506        let font_size_bits = flow_style.font_size.to_bits();
507        let color = ((style.color.0 as u32) << 24)
508            | ((style.color.1 as u32) << 16)
509            | ((style.color.2 as u32) << 8)
510            | (style.color.3 as u32);
511        let font_weight = style.font_weight.0;
512        let font_style = style.font_style;
513        let line_height = CachedLineHeight::from_line_height(flow_style.line_height);
514        let font_families = style.font_families.clone();
515
516        for i in 0..self.layout_cache.len() {
517            let cached = &self.layout_cache[i];
518
519            if cached.text == text
520                && cached.font_size_bits == font_size_bits
521                && cached.color == color
522                && cached.font_weight == font_weight
523                && cached.font_style == font_style
524                && cached.line_height == line_height
525                && cached.font_families == font_families
526            {
527                let entry = self.layout_cache.remove(i);
528                let layout = Arc::clone(&entry.layout);
529                self.layout_cache.insert(0, entry);
530                return layout;
531            }
532        }
533
534        let layout = Arc::new(self.create_buffer_for_text_inner(text, &style, &flow_style));
535
536        self.layout_cache.insert(
537            0,
538            CachedLayout {
539                text: text.to_string(),
540                font_size_bits,
541                color,
542                font_weight,
543                font_style,
544                line_height,
545                font_families,
546                layout: Arc::clone(&layout),
547            },
548        );
549
550        if self.layout_cache.len() > 128 {
551            self.layout_cache.pop();
552        }
553
554        layout
555    }

557    fn create_buffer_for_text_inner(
558        &mut self,
559        text: &str,
560        style: &TextStyle,
561        flow_style: &TextFlowStyle,
562    ) -> TextLayout {
563        perf_scope!(total);
564
565        let font_size = flow_style.font_size;
566        let line_height_ratio = match flow_style.line_height {
567            LineHeight::Normal => 1.2,
568            LineHeight::Number(n) => n,
569            LineHeight::Px(px) => px / font_size,
570        };
571
572        let font_families = build_family_list(&style.font_families);
573
574        let ori_style = OriTextStyle {
575            font_size,
576            color: OriColor(style.color.0, style.color.1, style.color.2, style.color.3),
577            font_weight: OriFontWeight(style.font_weight.0),
578            font_style: match style.font_style {
579                FontStyle::Normal => OriFontStyle::Normal,
580                FontStyle::Italic => OriFontStyle::Italic,
581                FontStyle::Oblique => OriFontStyle::Oblique,
582            },
583            line_height: line_height_ratio,
584            bidi_mode: orinium_text::BidiMode::Auto,
585            font_families,
586            exact_fonts: Vec::new(),
587            variant: orinium_text::FontVariant::Normal,
588        };
589
590        perf_scope!(shape);
591        let shaped = global_font::with_global_font_system(|fs| {
592            self.layouter.shape_text(fs, text, &ori_style)
593        });
594        #[cfg(any(feature = "profile", debug_assertions))]
595        let shape_time = shape.elapsed();
596
597        perf_scope!(lines);
598        let line_ranges: Vec<(usize, usize)> = text
599            .split('\n')
600            .scan(0usize, |offset, line| {
601                let start = *offset;
602                *offset += line.len() + 1;
603                let end = start + line.len();
604                Some((start, end))
605            })
606            .collect();
607
608        let layout = global_font::with_global_font_system(|fs| {
609            self.layouter
610                .layout_lines(fs, &shaped, &line_ranges, &ori_style)
611        });
612        #[cfg(any(feature = "profile", debug_assertions))]
613        let lines_time = lines.elapsed();
614
615        profile_log!(
616            target: "TextRenderer",
617            log::Level::Info,
618            "  create_buffer: text={:?} len={} font_size={}  shape={:?}  layout={:?}  total={:?}",
619            crate::profile::text_preview(text),
620            text.len(),
621            font_size,
622            shape_time,
623            lines_time,
624            total.elapsed(),
625        );
626
627        layout
628    }

643        sections: &[TextSection],
644    ) -> anyhow::Result<()> {
645        perf_scope!(total);
646
647        let s_hash = sections_hash(sections);
648        if !self.atlas_dirty
649            && s_hash == self.prev_sections_hash
650            && sections.len() == self.prev_sections_count
651        {
652            self.instances.clear();
653            log::trace!(target:"TextRenderer", "queue: fast-path ({} sections, {} instances)", sections.len(), self.num_instances);
654            return Ok(());
655        }
656        self.prev_sections_hash = s_hash;
657        self.prev_sections_count = sections.len();
658
659        self.instances.clear();
660        self.section_ranges.clear();
661
662        #[cfg(any(feature = "profile", debug_assertions))]
663        let mut glyph_count = 0u32;
664        #[cfg(any(feature = "profile", debug_assertions))]
665        let mut culled_count = 0u32;
666        #[cfg(any(feature = "profile", debug_assertions))]
667        let mut atlas_miss_count = 0u32;
668        #[cfg(any(feature = "profile", debug_assertions))]
669        let mut atlas_lookup_time = std::time::Duration::ZERO;
670        #[cfg(any(feature = "profile", debug_assertions))]
671        let mut atlas_rasterize_time = std::time::Duration::ZERO;
672
673        global_font::with_global_font_system(|fs| {
674            for section in sections {
675                let range_start = self.instances.len() as u32;
676                let layout = &section.layout;
677
678                let base_x = section.screen_position.0;
679                let base_y = section.screen_position.1;
680
681                let clip_l = section.clip_origin.0;
682                let clip_t = section.clip_origin.1;
683                let clip_r = section.clip_origin.0 + section.bounds.0;
684                let clip_b = section.clip_origin.1 + section.bounds.1;
685
686                for line in &layout.lines {
687                    for glyph in &line.glyphs {
688                        // Authors commonly use `font-size: 0` to remove the
689                        // whitespace between inline-blocks. Swash does not
690                        // define useful raster output for a zero-sized scaler
691                        // and some fonts return enormous bogus masks.
692                        if !rasterizable_font_size(glyph.font_size) {
693                            continue;
694                        }
695                        if glyph_fully_outside_clip(
696                            section,
697                            glyph.x,
698                            glyph.y,
699                            glyph.width,
700                            glyph.height,
701                        ) {
702                            #[cfg(any(feature = "profile", debug_assertions))]
703                            {
704                                culled_count += 1;
705                            }
706                            continue;
707                        }
708
709                        let Some(font_key) = glyph.font_key else {
710                            continue;
711                        };
712
713                        let bearing_key = (
714                            font_key.0,
715                            glyph.glyph_id,
716                            glyph.font_size.to_bits(),
717                            section.font_weight,
718                        );
719                        let (bearing_x, bearing_y) = if let Some(bearings) =
720                            self.bearing_cache.get(&bearing_key)
721                        {
722                            *bearings
723                        } else {
724                            let bearings =
725                                unhinted_bearings(fs, font_key, glyph.glyph_id, glyph.font_size)
726                                    .unwrap_or((0.0, 0.0));
727                            self.bearing_cache.put(bearing_key, bearings);
728                            bearings
729                        };
730
731                        let glyph_origin_x = base_x + glyph.x - bearing_x;
732                        let glyph_baseline_y = (base_y + glyph.y + bearing_y).round();
733                        let (origin_x, phase_x) = quantize_subpixel_x(glyph_origin_x);
734
735                        perf_scope!(atlas_lap);
736                        let glyph_key: GlyphKey = (
737                            font_key.0,
738                            glyph.glyph_id,
739                            glyph.font_size.to_bits(),
740                            section.font_weight,
741                            phase_x,
742                        );
743                        let atlas_entry = match self.atlas.lookup(glyph_key) {
744                            Some(entry) => {
745                                #[cfg(any(feature = "profile", debug_assertions))]
746                                {
747                                    atlas_lookup_time += atlas_lap.elapsed();
748                                }
749                                entry
750                            }
751                            None => {
752                                #[cfg(any(feature = "profile", debug_assertions))]
753                                {
754                                    atlas_miss_count += 1;
755                                }
756                                perf_scope!(rasterize_lap);
757                                if let Some(mask) = rasterize_glyph(
758                                    &mut self.scale_context,
759                                    fs,
760                                    font_key,
761                                    glyph.glyph_id,
762                                    glyph.font_size,
763                                    section.font_weight,
764                                    phase_x,
765                                ) {
766                                    #[cfg(any(feature = "profile", debug_assertions))]
767                                    {
768                                        atlas_rasterize_time += rasterize_lap.elapsed();
769                                    }
770                                    if mask.width == 0 || mask.height == 0 {
771                                        continue;
772                                    }
773                                    self.atlas_dirty = true;
774                                    #[cfg(any(feature = "profile", debug_assertions))]
775                                    {
776                                        atlas_lookup_time += atlas_lap.elapsed();
777                                    }
778                                    self.atlas.upload(device, queue, glyph_key, &mask)
779                                } else {
780                                    continue;
781                                }
782                            }
783                        };
784                        #[cfg(any(feature = "profile", debug_assertions))]
785                        {
786                            glyph_count += 1;
787                        }
788
789                        let gx = origin_x + atlas_entry.left as f32;
790                        let gy = glyph_baseline_y - atlas_entry.top as f32;
791                        let gw = atlas_entry.pixel_width as f32;
792                        let gh = atlas_entry.pixel_height as f32;
793
794                        let vis_l = gx.max(clip_l);
795                        let vis_t = gy.max(clip_t);
796                        let vis_r = (gx + gw).min(clip_r);
797                        let vis_b = (gy + gh).min(clip_b);
798
799                        if vis_l >= vis_r || vis_t >= vis_b {
800                            continue;
801                        }
802
803                        let u0 = atlas_entry.u + (vis_l - gx) / gw * atlas_entry.uv_width;
804                        let u1 = atlas_entry.u + (vis_r - gx) / gw * atlas_entry.uv_width;
805                        let v0 = atlas_entry.v + (vis_t - gy) / gh * atlas_entry.uv_height;
806                        let v1 = atlas_entry.v + (vis_b - gy) / gh * atlas_entry.uv_height;
807
808                        let inst = GlyphInstance {
809                            pos: pack_position(vis_l, vis_t),
810                            size: pack_size(vis_r - vis_l, vis_b - vis_t),
811                            uv_off: pack_uv_off(u0, v0),
812                            uv_size: pack_uv_size(u1 - u0, v1 - v0),
813                            layer: pack_layer(atlas_entry.layer),
814                            color: pack_color(&glyph.color),
815                        };
816                        self.instances.push(inst);
817                    }
818                }
819                self.section_ranges
820                    .push((range_start, self.instances.len() as u32 - range_start));
821            }
822        });
823
824        perf_scope!(flush);
825        self.atlas.flush_uploads(queue);
826        #[cfg(any(feature = "profile", debug_assertions))]
827        let flush_time = flush.elapsed();
828
829        self.num_instances = self.instances.len() as u32;
830
831        perf_scope!(buf);
832        if self.num_instances > 0 {
833            let needed = self.instances.len();
834            let instance_bytes: &[u8] = bytemuck::cast_slice(&self.instances);
835            if needed > self.instance_capacity {
836                let new_cap = (needed * 2).max(64);
837                let buf_size = (new_cap as u64) * size_of::<GlyphInstance>() as u64;
838                let buffer = device.create_buffer(&wgpu::BufferDescriptor {
839                    label: Some("Text Instance Buffer"),
840                    size: buf_size,
841                    usage: wgpu::BufferUsages::VERTEX | wgpu::BufferUsages::COPY_DST,
842                    mapped_at_creation: false,
843                });
844                queue.write_buffer(&buffer, 0, instance_bytes);
845                self.instance_buffer = Some(buffer);
846                self.instance_capacity = new_cap;
847            } else {
848                queue.write_buffer(self.instance_buffer.as_ref().unwrap(), 0, instance_bytes);
849            }
850        } else {
851            self.instance_buffer = None;
852            self.instance_capacity = 0;
853        }
854        #[cfg(any(feature = "profile", debug_assertions))]
855        let buf_time = buf.elapsed();
856
857        if self.atlas_dirty {
858            let bind_group = device.create_bind_group(&wgpu::BindGroupDescriptor {
859                label: Some("Text Bind Group"),
860                layout: &self.bind_group_layout,
861                entries: &[
862                    wgpu::BindGroupEntry {
863                        binding: 0,
864                        resource: wgpu::BindingResource::Buffer(wgpu::BufferBinding {
865                            buffer: &self.uniform_buffer,
866                            offset: 0,
867                            size: None,
868                        }),
869                    },
870                    wgpu::BindGroupEntry {
871                        binding: 1,
872                        resource: wgpu::BindingResource::TextureView(self.atlas.texture_view()),
873                    },
874                    wgpu::BindGroupEntry {
875                        binding: 2,
876                        resource: wgpu::BindingResource::Sampler(&self.sampler),
877                    },
878                ],
879            });
880            self.bind_group = Some(bind_group);
881            self.atlas_dirty = false;
882        }
883
884        profile_log!(
885            target: "TextRenderer",
886            log::Level::Info,
887            "queue: {} sections, {} glyphs ({} culled, {} atlas misses), lookup={:?} rasterize={:?} flush={:?} buf={:?} total={:?}",
888            sections.len(),
889            glyph_count,
890            culled_count,
891            atlas_miss_count,
892            atlas_lookup_time,
893            atlas_rasterize_time,
894            flush_time,
895            buf_time,
896            total.elapsed(),
897        );
898
899        Ok(())
900    }
901
902    /// Returns the `(start, count)` instance range for the section at `index`,
903    /// if it was queued this frame.
904    pub fn section_range(&self, index: usize) -> Option<(u32, u32)> {
905        self.section_ranges.get(index).copied()
906    }
907
908    /// Draws instances `start..start + count` with the glyph pipeline.
909    pub fn draw_range<'a>(&mut self, rpass: &mut wgpu::RenderPass<'a>, start: u32, count: u32) {
910        let Some(ref bind_group) = self.bind_group else {
911            return;
912        };
913        if count == 0 {
914            return;
915        }
916
917        rpass.set_pipeline(&self.pipeline);
918        rpass.set_bind_group(0, bind_group, &[]);
919        rpass.set_vertex_buffer(0, self.quad_vertex_buffer.slice(..));
920        if let Some(ref ib) = self.instance_buffer {
921            rpass.set_vertex_buffer(1, ib.slice(..));
922        }
923        rpass.set_index_buffer(self.quad_index_buffer.slice(..), wgpu::IndexFormat::Uint32);
924        rpass.draw_indexed(0..6, 0, start..start + count);
925    }
926}
