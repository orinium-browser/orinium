47    /// システムフォントから初期化する
48    pub fn new_from_device(
49        device: &wgpu::Device,
50        queue: &wgpu::Queue,
51        format: wgpu::TextureFormat,
52    ) -> anyhow::Result<Self> {
53        if let Ok(p) = std::env::var("ORINIUM_FONT")
54            && let Ok(bytes) = std::fs::read(&p)
55        {
56            return Self::new_from_bytes(device, queue, format, bytes);
57        }
58
59        for p in crate::platform::font::system_font_candidates()? {
60            if let Ok(bytes) = std::fs::read(p) {
61                return Self::new_from_bytes(device, queue, format, bytes);
62            }
63        }
64
65        anyhow::bail!("no system font found");
66    }

69    pub fn new_with_fontsys(
70        device: &wgpu::Device,
71        _queue: &wgpu::Queue,
72        format: wgpu::TextureFormat,
73        font_sys: FontSystem,
74    ) -> anyhow::Result<Self> {
75        let atlas = GlyphAtlas::new(device);
76
77        let sampler = device.create_sampler(&wgpu::SamplerDescriptor {
78            label: Some("Glyph Atlas Sampler"),
79            address_mode_u: wgpu::AddressMode::ClampToEdge,
80            address_mode_v: wgpu::AddressMode::ClampToEdge,
81            address_mode_w: wgpu::AddressMode::ClampToEdge,
82            mag_filter: wgpu::FilterMode::Linear,
83            min_filter: wgpu::FilterMode::Linear,
84            mipmap_filter: wgpu::MipmapFilterMode::Nearest,
85            ..Default::default()
86        });
87
88        let bind_group_layout = device.create_bind_group_layout(&wgpu::BindGroupLayoutDescriptor {
89            label: Some("Text Bind Group Layout"),
90            entries: &[
91                wgpu::BindGroupLayoutEntry {
92                    binding: 0,
93                    visibility: wgpu::ShaderStages::FRAGMENT,
94                    ty: wgpu::BindingType::Texture {
95                        sample_type: wgpu::TextureSampleType::Float { filterable: true },
96                        view_dimension: wgpu::TextureViewDimension::D2Array,
97                        multisampled: false,
98                    },
99                    count: None,
100                },
101                wgpu::BindGroupLayoutEntry {
102                    binding: 1,
103                    visibility: wgpu::ShaderStages::FRAGMENT,
104                    ty: wgpu::BindingType::Sampler(wgpu::SamplerBindingType::Filtering),
105                    count: None,
106                },
107            ],
108        });
109
110        let shader = device.create_shader_module(wgpu::ShaderModuleDescriptor {
111            label: Some("Text Shader"),
112            source: wgpu::ShaderSource::Wgsl(include_str!("../shader/text.wgsl").into()),
113        });
114
115        let pipeline_layout = device.create_pipeline_layout(&wgpu::PipelineLayoutDescriptor {
116            label: Some("Text Pipeline Layout"),
117            bind_group_layouts: &[Some(&bind_group_layout)],
118            immediate_size: 0,
119        });
120
121        let pipeline = device.create_render_pipeline(&wgpu::RenderPipelineDescriptor {
122            label: Some("Text Render Pipeline"),
123            layout: Some(&pipeline_layout),
124            cache: None,
125            vertex: wgpu::VertexState {
126                module: &shader,
127                entry_point: Some("vs_main"),
128                buffers: &[Some(wgpu::VertexBufferLayout {
129                    array_stride: size_of::<GlyphVertex>() as wgpu::BufferAddress,
130                    step_mode: wgpu::VertexStepMode::Vertex,
131                    attributes: &[
132                        wgpu::VertexAttribute {
133                            format: wgpu::VertexFormat::Float32x2,
134                            offset: 0,
135                            shader_location: 0,
136                        },
137                        wgpu::VertexAttribute {
138                            format: wgpu::VertexFormat::Float32x2,
139                            offset: size_of::<[f32; 2]>() as wgpu::BufferAddress,
140                            shader_location: 1,
141                        },
142                        wgpu::VertexAttribute {
143                            format: wgpu::VertexFormat::Float32,
144                            offset: (size_of::<[f32; 2]>() * 2) as wgpu::BufferAddress,
145                            shader_location: 2,
146                        },
147                        wgpu::VertexAttribute {
148                            format: wgpu::VertexFormat::Float32x4,
149                            offset: (size_of::<[f32; 2]>() * 2 + size_of::<f32>())
150                                as wgpu::BufferAddress,
151                            shader_location: 3,
152                        },
153                    ],
154                })],
155                compilation_options: wgpu::PipelineCompilationOptions::default(),
156            },
157            fragment: Some(wgpu::FragmentState {
158                module: &shader,
159                entry_point: Some("fs_main"),
160                targets: &[Some(wgpu::ColorTargetState {
161                    format,
162                    blend: Some(wgpu::BlendState::ALPHA_BLENDING),
163                    write_mask: wgpu::ColorWrites::ALL,
164                })],
165                compilation_options: wgpu::PipelineCompilationOptions::default(),
166            }),
167            primitive: wgpu::PrimitiveState {
168                topology: wgpu::PrimitiveTopology::TriangleList,
169                strip_index_format: None,
170                front_face: wgpu::FrontFace::Ccw,
171                cull_mode: None,
172                polygon_mode: wgpu::PolygonMode::Fill,
173                unclipped_depth: false,
174                conservative: false,
175            },
176            depth_stencil: None,

178                count: 1,
179                mask: !0,
180                alpha_to_coverage_enabled: false,
181            },
182            multiview_mask: None,
183        });
184
185        let layouter = TextLayouter::new();
186
187        Ok(Self {
188            font_sys,
189            layouter,
190            atlas,
191            pipeline,
192            bind_group_layout,
193            bind_group: None,
194            sampler,
195            vertex_buffer: None,
196            index_buffer: None,
197            num_indices: 0,
198            vertices: Vec::new(),

202        })
203    }

216    /// Create an orinium_text `TextLayout` for the given text using the internal `FontSystem`.
217    pub fn create_buffer_for_text(
218        &mut self,
219        text: &str,
220        style: TextStyle,
221    ) -> orinium_text::TextLayout {
222        let font_size = style.font_size;
223        let line_height_ratio = match style.line_height {
224            LineHeight::Normal => 1.2,
225            LineHeight::Number(n) => n,
226            LineHeight::Px(px) => px / font_size,
227        };
228
229        let ori_style = OriTextStyle {
230            font_size,
231            color: OriColor(style.color.0, style.color.1, style.color.2, style.color.3),
232            font_weight: OriFontWeight(style.font_weight.0),
233            font_style: match style.font_style {
234                FontStyle::Normal => OriFontStyle::Normal,
235                FontStyle::Italic => OriFontStyle::Italic,
236                FontStyle::Oblique => OriFontStyle::Oblique,
237            },
238            line_height: line_height_ratio,
239            bidi_mode: orinium_text::BidiMode::Auto,
240            font_families: vec![fontdb::Family::SansSerif],
241            exact_fonts: self.font_sys.font_keys(),
242        };
243
244        let shaped = self
245            .layouter
246            .shape_text(&mut self.font_sys, text, &ori_style);
247
248        let line_ranges: Vec<(usize, usize)> = text
249            .split('\n')
250            .scan(0usize, |offset, line| {
251                let start = *offset;
252                *offset += line.len() + 1;
253                let end = start + line.len();
254                Some((start, end))
255            })
256            .collect();
257
258        self.layouter
259            .layout_lines(&mut self.font_sys, &shaped, &line_ranges, &ori_style)
260    }

267    /// 指定されたセクション群をギリフォン用の TextArea に変換して Atlas に転送する
268    pub fn queue(
269        &mut self,
270        device: &wgpu::Device,
271        queue: &wgpu::Queue,
272        sections: &[TextSection],
273    ) -> anyhow::Result<()> {
274        self.vertices.clear();
275        self.indices.clear();
276
277        let ndc_x = |sx: f32| (sx / self.screen_width) * 2.0 - 1.0;
278        let ndc_y = |sy: f32| -((sy / self.screen_height) * 2.0 - 1.0);
279
280        for section in sections {
281            let layout = &section.layout;
282
283            let base_x = section.screen_position.0;
284            let base_y = section.screen_position.1;
285
286            let clip_l = section.clip_origin.0;
287            let clip_t = section.clip_origin.1;
288            let clip_r = section.clip_origin.0 + section.bounds.0;
289            let clip_b = section.clip_origin.1 + section.bounds.1;
290
291            for line in &layout.lines {
292                for glyph in &line.glyphs {
293                    let Some(font_key) = glyph.font_key else {
294                        continue;
295                    };
296
297                    let gx = base_x + glyph.x;
298                    let gy = base_y + glyph.y;
299                    let gw = glyph.width;
300                    let gh = glyph.height;
301
302                    // Compute visible portion against clip rect
303                    let vis_l = gx.max(clip_l);
304                    let vis_t = gy.max(clip_t);
305                    let vis_r = (gx + gw).min(clip_r);
306                    let vis_b = (gy + gh).min(clip_b);
307
308                    if vis_l >= vis_r || vis_t >= vis_b {
309                        continue;
310                    }
311
312                    let (layer, u, v, uw, uh) =
313                        match self.atlas.lookup(font_key, glyph.glyph_id, glyph.font_size) {
314                            Some(uv) => uv,
315                            None => {
316                                if let Some((metrics, alpha_mask)) =
317                                    self.font_sys.get_or_rasterize_with_bitmap(
318                                        font_key,
319                                        glyph.glyph_id,
320                                        glyph.font_size,
321                                    )
322                                {
323                                    if metrics.width == 0 || metrics.height == 0 {
324                                        continue;
325                                    }
326                                    let mask_w = metrics.width;
327                                    let mask_h = metrics.height;
328                                    self.atlas.upload(
329                                        device,
330                                        queue,
331                                        font_key,
332                                        glyph.glyph_id,
333                                        glyph.font_size,
334                                        &alpha_mask,
335                                        mask_w,
336                                        mask_h,
337                                    )
338                                } else {
339                                    continue;
340                                }
341                            }
342                        };
343
344                    // Convert visible rect to NDC
345                    let quad_x1 = ndc_x(vis_l);
346                    let quad_y1 = ndc_y(vis_t);
347                    let quad_x2 = ndc_x(vis_r);
348                    let quad_y2 = ndc_y(vis_b);
349
350                    // NDC Y is flipped: upper-left becomes min, lower-right becomes max
351                    let (qy1, qy2) = if quad_y1 < quad_y2 {
352                        (quad_y1, quad_y2)
353                    } else {
354                        (quad_y2, quad_y1)
355                    };
356
357                    // Compute UVs for the visible portion
358                    let u0 = u + (vis_l - gx) / gw * uw;
359                    let u1 = u + (vis_r - gx) / gw * uw;
360                    let v0 = v + (vis_b - gy) / gh * uh;
361                    let v1 = v + (vis_t - gy) / gh * uh;
362
363                    let idx = self.vertices.len() as u32;
364                    let layer_f = layer as f32;
365
366                    let color_arr = crate::engine::layouter::types::Color(
367                        glyph.color.0,
368                        glyph.color.1,
369                        glyph.color.2,
370                        glyph.color.3,
371                    )
372                    .to_linear_f32_array();
373
374                    self.vertices.extend_from_slice(&[
375                        GlyphVertex {
376                            position: [quad_x1, qy1],
377                            tex_coord: [u0, v0],
378                            layer: layer_f,
379                            color: color_arr,
380                        },
381                        GlyphVertex {
382                            position: [quad_x2, qy1],
383                            tex_coord: [u1, v0],
384                            layer: layer_f,
385                            color: color_arr,
386                        },
387                        GlyphVertex {
388                            position: [quad_x2, qy2],
389                            tex_coord: [u1, v1],
390                            layer: layer_f,
391                            color: color_arr,
392                        },
393                        GlyphVertex {
394                            position: [quad_x1, qy2],
395                            tex_coord: [u0, v1],
396                            layer: layer_f,
397                            color: color_arr,
398                        },
399                    ]);
400
401                    self.indices
402                        .extend_from_slice(&[idx, idx + 1, idx + 2, idx, idx + 2, idx + 3]);
403                }
404            }
405        }
406
407        self.atlas.flush_uploads(queue);
408
409        let bind_group = device.create_bind_group(&wgpu::BindGroupDescriptor {
410            label: Some("Text Bind Group"),
411            layout: &self.bind_group_layout,
412            entries: &[
413                wgpu::BindGroupEntry {
414                    binding: 0,
415                    resource: wgpu::BindingResource::TextureView(self.atlas.texture_view()),
416                },
417                wgpu::BindGroupEntry {
418                    binding: 1,
419                    resource: wgpu::BindingResource::Sampler(&self.sampler),
420                },
421            ],
422        });
423        self.bind_group = Some(bind_group);
424
425        self.num_indices = self.indices.len() as u32;
426        if !self.vertices.is_empty() {
427            self.vertex_buffer = Some(device.create_buffer_init(
428                &wgpu::util::BufferInitDescriptor {
429                    label: Some("Text Vertex Buffer"),
430                    contents: bytemuck::cast_slice(&self.vertices),
431                    usage: wgpu::BufferUsages::VERTEX,
432                },
433            ));
434        } else {
435            self.vertex_buffer = None;
436        }
437        if !self.indices.is_empty() {
438            self.index_buffer = Some(device.create_buffer_init(
439                &wgpu::util::BufferInitDescriptor {
440                    label: Some("Text Index Buffer"),
441                    contents: bytemuck::cast_slice(&self.indices),
442                    usage: wgpu::BufferUsages::INDEX,
443                },
444            ));
445        } else {
446            self.index_buffer = None;
447        }
448
449        Ok(())
450    }

452    /// テキストをレンダリングする
453    pub fn draw<'a>(&mut self, rpass: &mut wgpu::RenderPass<'a>) {
454        let Some(ref bind_group) = self.bind_group else {
455            return;
456        };
457        let Some(ref vertex_buffer) = self.vertex_buffer else {
458            return;
459        };
460        let Some(ref index_buffer) = self.index_buffer else {
461            return;
462        };
463        if self.num_indices == 0 {
464            return;
465        }
466
467        rpass.set_pipeline(&self.pipeline);
468        rpass.set_bind_group(0, bind_group, &[]);
469        rpass.set_vertex_buffer(0, vertex_buffer.slice(..));
470        rpass.set_index_buffer(index_buffer.slice(..), wgpu::IndexFormat::Uint32);
471        rpass.draw_indexed(0..self.num_indices, 0, 0..1);
472    }
